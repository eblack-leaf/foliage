1//! The files `init` writes.
2//!
3//! Every Gradle and Java file here is a parameterised copy of `rust-mobile/android-activity`'s own
4//! `agdk-mainloop` example -- the crate that implements winit's `android-game-activity` backend --
5//! verified against `androidx.games:games-activity` rather than invented. Only the values that have
6//! to vary per app are placeholders; everything else is copied as it stands.
7//!
8//! Placeholders are `{{NAME}}` and are substituted by [`fill`], rather than the text being built up
9//! by `format!`. A Gradle build file is mostly braces, and the escaping that would take is exactly
10//! the noise that makes a template hard to check against its upstream.
11
12/// Substitutes `{{NAME}}` placeholders.
13fn fill(template: &str, values: &[(&str, &str)]) -> String {
14    let mut out = template.to_string();
15    for (name, value) in values {
16        out = out.replace(&format!("{{{{{name}}}}}"), value);
17    }
18    out
19}

135
136import android.Manifest;
137import android.app.Activity;
138import android.content.Context;
139import android.content.pm.PackageManager;
140import android.location.LocationListener;
141import android.location.LocationManager;

145/**
146 * Where the device is, handed to foliage as each fix arrives: from every provider the device has
147 * on, so a tablet with no GPS is still placed by its network. Which of two fixes is the better is
148 * foliage's to say.
149 *
150 * Written by `foliage-android init` for an app that asks for a location permission.
151 */
152public final class Locator implements LocationListener {
153
154    /** The request the permission is asked for under. */
155    public static final int ASKED = 0x10ca;
156
157    /** How often a provider is asked for a fix, in milliseconds. */
158    private static final long EVERY_MS = 1000;
159
160    /** foliage's: one fix. An accuracy below zero is none, and a bearing that is NaN is none. */
161    private static native void fix(double lat, double lon, float accuracy, float bearing, long time);
162
163    private final Activity activity;
164    private final LocationManager manager;
165    private boolean asked = false;
166    private boolean listening = false;
167
168    public Locator(Activity activity) {
169        this.activity = activity;
170        this.manager = (LocationManager) activity.getSystemService(Context.LOCATION_SERVICE);
171    }
172
173    private boolean allowed(String permission) {
174        return activity.checkSelfPermission(permission) == PackageManager.PERMISSION_GRANTED;
175    }
176
177    /** Listens, or asks for the permission to -- once a run, so a refusal is not asked again. */
178    public void start() {
179        if (manager == null || listening) {
180            return;
181        }
182        if (!allowed(Manifest.permission.ACCESS_FINE_LOCATION)
183                && !allowed(Manifest.permission.ACCESS_COARSE_LOCATION)) {
184            if (!asked) {
185                asked = true;
186                activity.requestPermissions(new String[] {
187                        Manifest.permission.ACCESS_FINE_LOCATION,
188                        Manifest.permission.ACCESS_COARSE_LOCATION,
189                }, ASKED);
190            }
191            return;
192        }
193        for (String provider : manager.getProviders(true)) {
194            if (LocationManager.PASSIVE_PROVIDER.equals(provider)) {
195                continue;
196            }
197            try {
198                manager.requestLocationUpdates(provider, EVERY_MS, 0f, this,
199                        Looper.getMainLooper());
200                listening = true;
201                android.location.Location last = manager.getLastKnownLocation(provider);
202                if (last != null) {
203                    onLocationChanged(last);
204                }
205            } catch (SecurityException | IllegalArgumentException e) {
206                // Taken away between the check and the request, or gone: nothing to listen to.
207            }
208        }
209    }
210
211    public void stop() {
212        if (manager != null && listening) {
213            manager.removeUpdates(this);
214            listening = false;
215        }
216    }
217
218    @Override
219    public void onLocationChanged(android.location.Location at) {
220        fix(at.getLatitude(), at.getLongitude(),
221                at.hasAccuracy() ? at.getAccuracy() : -1f,
222                at.hasBearing() ? at.getBearing() : Float.NaN,
223                at.getTime());
224    }
225
226    // Abstract before API 30, so a device older than that calls them and they have to be here.
227    @Override
228    public void onStatusChanged(String provider, int status, Bundle extras) {}
229
230    @Override
231    public void onProviderEnabled(String provider) {}
232
233    @Override
234    public void onProviderDisabled(String provider) {}
235}
236"#;

461public class MainActivity extends GameActivity {

464        // Must match both the entry crate's cdylib output name and the `android.app.lib_name`
465        // meta-data value in AndroidManifest.xml -- all three have to agree, or the JVM cannot
466        // find `android_main` to call into.
467        System.loadLibrary("{{LIB_NAME}}");
468    }

470    /** How long after a bar is found covering the window it is hidden again. */
471    private static final long REHIDE_MS = 300;
472
473    private final Runnable rehide = this::hideSystemUI;
474
475    private void hideSystemUI() {
476        getWindow().getAttributes().layoutInDisplayCutoutMode
477                = WindowManager.LayoutParams.LAYOUT_IN_DISPLAY_CUTOUT_MODE_ALWAYS;
478        View decorView = getWindow().getDecorView();
479        WindowInsetsControllerCompat controller = new WindowInsetsControllerCompat(getWindow(),

481        controller.hide(WindowInsetsCompat.Type.systemBars());
482        controller.hide(WindowInsetsCompat.Type.displayCutout());
483        controller.setSystemBarsBehavior(
484                WindowInsetsControllerCompat.BEHAVIOR_SHOW_TRANSIENT_BARS_BY_SWIPE);
485    }

488    protected void onCreate(Bundle savedInstanceState) {
489        WindowCompat.setDecorFitsSystemWindows(getWindow(), false);
490        hideSystemUI();
491        super.onCreate(savedInstanceState);{{LOCATE_CREATE}}
492    }
493
494    protected void onResume() {
495        super.onResume();
496        hideSystemUI();{{LOCATE_RESUME}}
497    }{{LOCATE_MEMBERS}}
498
499    // The soft keyboard brings the navigation bar up with it and leaves it there when it goes,
500    // over the foot of the surface, until something hides it again. Nothing about insets is said
501    // when it goes (measured on Android 10: no insets are applied, and GameActivity still thinks
502    // the keyboard is up), but the window is laid out again. So on every layout: if something
503    // covers the foot of the window, and less than a keyboard would, it is a bar, and the bars are
504    // hidden again -- a moment later, once the keyboard has finished going: asked while it is still
505    // closing, the bars stay.
506    @Override
507    public void onGlobalLayout() {
508        super.onGlobalLayout();
509        View decor = getWindow().getDecorView();
510        Rect visible = new Rect();
511        decor.getWindowVisibleDisplayFrame(visible);
512        int covered = decor.getHeight() - visible.bottom;

516        }
517    }

519    // And whenever the window has focus back: a dialog or the notification shade shows the bars
520    // the same way.

529    /** How soon after one editor action another is the same press of the key. */
530    private static final long ENTER_MS = 100;

532    private long entered = Long.MIN_VALUE / 2;
533
534    // The soft keyboard's enter key is turned into an editor action by the text input's own
535    // connection, and the key itself goes no further -- and winit reads the action and drops it. So
536    // the action is handed on as the key it was, which reaches the app like a hardware enter does.
537    // Once a press: the connection makes an action of the key going down and of it coming up, a
538    // few milliseconds apart.
539    @Override

541        super.onEditorAction(action);
542        long now = SystemClock.uptimeMillis();
543        if (now - entered < ENTER_MS) {
544            return;
545        }
546        entered = now;
547        onKeyDown(KeyEvent.KEYCODE_ENTER,
548                new KeyEvent(now, now, KeyEvent.ACTION_DOWN, KeyEvent.KEYCODE_ENTER, 0));
549        onKeyUp(KeyEvent.KEYCODE_ENTER,
550                new KeyEvent(now, now, KeyEvent.ACTION_UP, KeyEvent.KEYCODE_ENTER, 0));
551    }

553    // A soft keyboard's control key is never sent as a key of its own: it rides on the letter, as
554    // meta state, and winit reads no meta state. So a key that carries control without being it is
555    // given a control key going down before it and coming up after, which is what a hardware
556    // keyboard would have sent -- and Ctrl+A selects rather than typing an "a".
557    @Override
558    public boolean onKeyDown(int keyCode, KeyEvent event) {
559        if (!chorded(event)) {
560            return super.onKeyDown(keyCode, event);
561        }
562        super.onKeyDown(KeyEvent.KEYCODE_CTRL_LEFT, control(event, KeyEvent.ACTION_DOWN));
563        boolean handled = super.onKeyDown(keyCode, event);
564        super.onKeyUp(KeyEvent.KEYCODE_CTRL_LEFT, control(event, KeyEvent.ACTION_UP));
565        return handled;
566    }
567
568    private static boolean chorded(KeyEvent event) {
569        int code = event.getKeyCode();
570        return event.isCtrlPressed()
571                && code != KeyEvent.KEYCODE_CTRL_LEFT
572                && code != KeyEvent.KEYCODE_CTRL_RIGHT;
573    }
574
575    private static KeyEvent control(KeyEvent event, int action) {
576        return new KeyEvent(event.getDownTime(), event.getEventTime(), action,
577                KeyEvent.KEYCODE_CTRL_LEFT, 0, event.getMetaState(), event.getDeviceId(), 0,
578                event.getFlags(), event.getSource());
579    }

621const README: &str = r#"# {{APP_NAME}} -- Android
622
623Generated by `foliage-android init`, **including this file**. A rerun overwrites everything in this
624directory, so keep notes of your own somewhere else. Settings live in `foliage-android.toml` at the
625repo root -- edit there and rerun `init`, rather than editing these files.
626
627This Gradle project exists because `androidx.games:games-activity` -- Google's `GameActivity`, which
628winit's `android-game-activity` backend targets -- is a Java/Kotlin AAR, and there is no way to link
629one without Gradle. Unavoidable rather than a workaround.
630
631## Building
632
633From the repo root:
634
635```sh
636foliage-android setup    # the SDK, NDK and rust targets, into the `sdk` directory
637foliage-android doctor   # what is installed, and what is not
638foliage-android build    # every configured ABI, then the APK
639foliage-android run      # the above, installed and launched on a connected device
640```
641
642`build` compiles each ABI into `app/src/main/jniLibs/<abi>/` and then assembles. Doing it by hand is
643what the tool exists to avoid: `jniLibs/` is an *input* directory to Gradle rather than an output it
644tracks, so whatever `.so` is sitting there gets packaged. A Rust build that failed, was skipped, or
645wrote somewhere else still assembles successfully -- shipping the *previous* library. The luckier
646outcome is an `UnsatisfiedLinkError` at launch; the unluckier one is old code running quietly.
647
648## SDK levels
649
650Three numbers do three different jobs, and only one gates hardware:
651
652- **`compile-sdk`** -- which API the compiler sees. A build-time download
653  (`platforms/android-{{COMPILE_SDK}}`), with no bearing on what the APK runs on.
654- **`min-sdk`** -- the device gate. Anything below refuses to install the APK.
655- **`target-sdk`** -- which platform behaviours the app opts into. Affects runtime behaviour on
656  newer devices; not an install gate.
657
658Lowering `compile-sdk` does not widen device support -- `min-sdk` is the knob for that. Whichever
659`compile-sdk` is set, install the matching `platforms/android-<n>` or Gradle will not find it.
660
661## Emulator
662
663Useful for exercising API levels you do not own hardware for.
664
665```sh
666android sdk install emulator system-images/android-{{COMPILE_SDK}}/google_apis/{{EMULATOR_ABI}}
667android emulator create --list-profiles      # what you can pick from
668android emulator create medium_phone         # profile is positional, and required
669android emulator start medium_phone
670```
671
672On Linux the emulator needs `/dev/kvm`; `test -r /dev/kvm && test -w /dev/kvm && echo ok` settles
673whether you have it, and `sudo usermod -aG kvm $USER` plus a re-login is the fix. Group membership
674is not the whole story -- some distributions grant access through an ACL -- so test rather than
675reading `groups`.
676
677**Force the GPU, or everything is software-rendered.** New AVDs default to `hw.gpu.mode=auto`, and
678`auto` readily decides against a capable host GPU and falls back to SwiftShader -- CPU rendering,
679sluggish enough that the launcher feels broken, not just your app. Set `hw.gpu.mode=host` in
680`~/.android/avd/<name>.avd/config.ini`. To check which you got:
681`adb shell dumpsys SurfaceFlinger | grep -i gles` names either your real GPU or `SwiftShader`.
682
683**Build the emulator's own ABI.** Current x86_64 system images translate ARM binaries, so an
684`arm64-v8a`-only APK does run on an x86_64 emulator -- routed through `libndk_translation.so`, slow
685enough to make frame timings meaningless and to trip ANR dialogs. `init` adds this machine's ABI to
686`foliage-android.toml` for that reason; both live in `jniLibs/` at once and Android prefers a native
687ABI over a translated one, so one APK serves emulator and phone.
688
689AVDs live in `~/.android/avd/`, **not** in the SDK directory -- deleting the SDK leaves them behind.

691## Release signing
692
693`foliage-android build` needs nothing extra: AGP generates and uses a debug keystore.
694`--release` also runs without one, but produces an **unsigned** APK no device will install. For a
695real signed release:
696
6971. Generate a keystore once. Keep the file and passwords somewhere safe outside the repo -- losing
698   this keystore means never shipping an update to the same `applicationId` under the same identity
699   again.
700
701   ```sh
702   keytool -genkeypair -v -keystore ~/keys/{{LIB_NAME}}-release.keystore \
703     -alias {{LIB_NAME}} -keyalg RSA -keysize 2048 -validity 10000
704   ```
705
7062. Create `{{PROJECT}}/keystore.properties`, and gitignore it:
707
708   ```properties
709   storeFile=/home/you/keys/{{LIB_NAME}}-release.keystore
710   storePassword=<the password you set above>
711   keyAlias={{LIB_NAME}}
712   keyPassword=<same password, unless you set a separate key password>
713   ```
714
7153. That is all -- `app/build.gradle` reads that file whenever it is present.

717## Troubleshooting
718
719- **App installs but crashes immediately.** Clear the log, launch, and read the abort message:
720
721  ```sh
722  adb logcat -c
723  adb shell am start -n {{APP_NAME}}/.MainActivity
724  adb logcat -d | grep -B5 -A40 "Fatal signal"
725  ```
726
727  The tombstone's `Abort message:` line names the cause. `System.loadLibrary` failing means
728  `jniLibs/` has nothing for the device's ABI -- add it to `abis` in `foliage-android.toml`.
729- **`RegisterNatives failed` / `NoSuchMethodError` on a GameActivity method.** The AAR and the
730  GameActivity C++ compiled into the library are different versions. `foliage-android doctor`
731  compares the two and names the mismatch. This aborts before any Rust runs, so it is not a bug in
732  your app. Prefer `cargo update -p android-activity` over pinning an older AAR: matching them at
733  the older version also stops the crash, but a GameActivity years behind the running platform can
734  deliver no touch input at all -- the app launches, draws correctly, and ignores every tap, with
735  nothing in the log.
736- **Gradle disagrees about where the SDK is.** Check `local.properties`, which Gradle writes with
737  the path it resolved, against what `foliage-android doctor` reports. Nothing in this toolchain
738  reads `ANDROID_HOME`: the SDK is the `sdk` path in `foliage-android.toml` and nowhere else, and
739  the variables Gradle and `cargo ndk` need are set on those processes rather than in your shell.
740- **A download died partway.** `setup` installs one package at a time and retries each, so what
741  already arrived is kept -- rerun it and it resumes rather than starting over.

744
745`foliage-android run` uses `adb`, which needs USB debugging. Without it, transfer the APK however
