3use std::io::Cursor;
4
5use image::{ExtendedColorType, ImageEncoder, codecs::png::PngEncoder};
6
7/// Renders a baked field at an on-screen size, as a PNG.
8///
9/// Reproduces the shader's own sampling -- a bilinear fetch, the median of the three colour
10/// channels, and the screen-pixel spread that turns a distance into coverage -- so a field's
11/// quality can be judged without running an application. The mark is drawn dark on a light
12/// ground, which is what makes an artifact read as a crack rather than as a highlight.
13///
14/// # Errors
15///
16/// If `field` is not a square RGBA field, or the PNG cannot be encoded.
17pub fn preview(field: &[u8], size: u32, range: f32) -> Result<Vec<u8>, String> {
18    let side = (field.len() as f64 / 4.0).sqrt() as usize;
19    if side == 0 || side * side * 4 != field.len() {
20        return Err(format!(
21            "not a square RGBA field: {} bytes is not a whole number of texels squared",
22            field.len()
23        ));
24    }
25
26    // The shader's own conversion: how many screen pixels the spread covers at this size.
27    let spread = (size as f32 / side as f32 * range).max(1.0);
28
29    let sample = |x: f32, y: f32, channel: usize| -> f32 {
30        // A bilinear fetch at a texture coordinate in `0.0..=1.0`, clamped to the edge.
31        let (fx, fy) = (x * side as f32 - 0.5, y * side as f32 - 0.5);
32        let (x0, y0) = (fx.floor(), fy.floor());
33        let (ax, ay) = (fx - x0, fy - y0);
34        let clamp = |v: f32| (v.max(0.0) as usize).min(side - 1);
35        let (left, right) = (clamp(x0), clamp(x0 + 1.0));
36        let (top, bottom) = (clamp(y0), clamp(y0 + 1.0));
37        let at = |x: usize, y: usize| field[(y * side + x) * 4 + channel] as f32 / 255.0;
38        let upper = at(left, top) * (1.0 - ax) + at(right, top) * ax;
39        let lower = at(left, bottom) * (1.0 - ax) + at(right, bottom) * ax;
40        upper * (1.0 - ay) + lower * ay
41    };
42
43    let mut rendered = image::GrayImage::new(size, size);
44    for y in 0..size {
45        for x in 0..size {
46            let (tx, ty) = (
47                (x as f32 + 0.5) / size as f32,
48                (y as f32 + 0.5) / size as f32,
49            );
50            let (r, g, b) = (sample(tx, ty, 0), sample(tx, ty, 1), sample(tx, ty, 2));
51            let median = r.max(g.min(b)).min(r.min(g).max(b));
52            let coverage = ((median - 0.5) * spread + 0.5).clamp(0.0, 1.0);
53            rendered.put_pixel(x, y, image::Luma([255 - (coverage * 255.0).round() as u8]));
54        }
55    }
56
57    let mut png = Vec::new();
58    PngEncoder::new(Cursor::new(&mut png))
59        .write_image(&rendered, size, size, ExtendedColorType::L8)
60        .map_err(|e| format!("encoding the preview: {e}"))?;
61    Ok(png)
62}

64#[cfg(test)]
65mod tests {
66    use super::preview;
67
68    #[test]
69    fn a_field_that_is_not_square_is_refused() {
70        assert!(preview(&[255; 7], 32, 3.0).is_err());
71        assert!(preview(&[], 32, 3.0).is_err());
72    }
73
74    /// A field that is entirely inside renders as a fully covered square, which is what says the
75    /// median and the coverage step run the way the shader's do.
76    #[test]
77    fn a_field_that_is_wholly_inside_renders_covered() {
78        let png = preview(&[255; 4 * 4 * 4], 16, 3.0).expect("preview");
79        assert_eq!(&png[1..4], b"PNG");
80    }
