6/// Emits the module that registers `stems` as a `Marks` set called `marks`.
7///
8/// A struct with one `Field` per mark, and the `Marks` impl that fills it. The fields are read
9/// with `include_bytes!`, so the module belongs in the directory they were written to.
10///
11/// Every mark is reached by the name it was given rather than by an index, which is what lets a
12/// set be regenerated after a mark is added or removed without moving a callsite.
13///
14/// # Errors
15///
16/// If `marks` does not reduce to a usable type name.
17pub fn generate(stems: &[String], spec: Bake, marks: &str) -> Result<String, String> {
18    let name = type_name(marks)?;
19    let mut out = String::new();
20
21    out.push_str("// AUTO-GENERATED by foliage-icons -- do not edit by hand.\n");
22    out.push_str(&format!(
23        "// Regenerate with:\n\
24         //     foliage-icons bake --svg <svg-dir> --out <this-dir> --marks {name} \\\n\
25         //         --side {} --range {}\n\n",
26        spec.side, spec.range
27    ));
28
29    out.push_str("use foliage::{Field, Grove, Marks};\n\n");
30    out.push_str("/// How every field beside this module was baked.\n");
31    out.push_str("///\n");
32    out.push_str(
33        "/// Both describe the bake rather than any one mark, which is why a set states them\n\
34         /// once rather than per field.\n",
35    );
36    out.push_str(&format!("const SIDE: u32 = {};\n", spec.side));
37    out.push_str(&format!("const RANGE: f32 = {:?};\n\n", spec.range));
38
39    out.push_str(&format!("/// Every mark `{name}` holds.\n"));
40    out.push_str(&format!("pub struct {name} {{\n"));
41    for stem in stems {
42        out.push_str(&format!("    /// The `{stem}` mark.\n"));
43        out.push_str(&format!("    pub {}: Field,\n", naming::field(stem)));
44    }
45    out.push_str("}\n\n");
46
47    out.push_str(&format!("impl Marks for {name} {{\n"));
48    out.push_str("    fn register(grove: &mut Grove) -> Self {\n");
49    out.push_str("        Self {\n");
50    for stem in stems {
51        out.push_str(&format!(
52            "            {}: grove.icon(include_bytes!(\"{stem}.icon\"), SIDE, RANGE),\n",
53            naming::field(stem)
54        ));
55    }
56    out.push_str("        }\n    }\n}\n");
57    Ok(out)
58}

60/// `marks` as a `PascalCase` type name.
61fn type_name(marks: &str) -> Result<String, String> {
62    let name: String = naming::snake_case(marks)
63        .split('_')
64        .filter(|word| !word.is_empty())
65        .map(|word| {
66            let mut characters = word.chars();
67            match characters.next() {
68                Some(first) => first.to_uppercase().collect::<String>() + characters.as_str(),
69                None => String::new(),
70            }
71        })
72        .collect();
73    match name.starts_with(|c: char| c.is_alphabetic()) {
74        true => Ok(name),
75        false => Err(format!("`{marks}` is not a usable type name")),
76    }
77}

79#[cfg(test)]
80mod tests {
81    use super::{generate, type_name};
82    use crate::msdf::Bake;
83
84    fn stems() -> Vec<String> {
85        vec!["arrow-up".to_string(), "check".to_string()]
86    }
87
88    #[test]
89    fn a_set_name_is_taken_however_it_is_written() {
90        assert_eq!(type_name("Icons").as_deref(), Ok("Icons"));
91        assert_eq!(type_name("app_icons").as_deref(), Ok("AppIcons"));
92        assert_eq!(type_name("app-icons").as_deref(), Ok("AppIcons"));
93        assert!(type_name("2icons").is_err());
94        assert!(type_name("--").is_err());
95    }
96
97    /// The generated module is what a dependent compiles, so every mark has to arrive as a
98    /// documented field and as the line that registers it.
99    #[test]
100    fn every_mark_is_a_field_and_a_registration() {
101        let generated = generate(&stems(), Bake::new(48, 3.0), "Icons").expect("generate");
102        assert!(generated.contains("pub struct Icons {"));
103        assert!(generated.contains("/// The `arrow-up` mark.\n    pub arrow_up: Field,"));
104        assert!(generated.contains("pub check: Field,"));
105        assert!(generated.contains("impl Marks for Icons {"));
106        assert!(generated.contains(
107            "arrow_up: grove.icon(include_bytes!(\"arrow-up.icon\"), SIDE, RANGE),"
108        ));
109        assert!(generated.contains("const SIDE: u32 = 48;"));
110        assert!(generated.contains("const RANGE: f32 = 3.0;"));
111    }
112
113    /// A range that is not a whole number still has to emit as an `f32` literal rather than as
114    /// an integer one, which would not compile.
115    #[test]
116    fn a_fractional_range_emits_as_a_float() {
117        let generated = generate(&stems(), Bake::new(64, 2.5), "Icons").expect("generate");
118        assert!(generated.contains("const RANGE: f32 = 2.5;"));
119    }
120
121    /// A mark named for a keyword is ordinary input, and the set has to compile with it in.
122    #[test]
123    fn a_keyword_mark_is_written_raw() {
124        let stems = vec!["move".to_string()];
125        let generated = generate(&stems, Bake::new(48, 3.0), "Icons").expect("generate");
126        assert!(generated.contains("pub r#move: Field,"));
127        assert!(generated.contains("r#move: grove.icon(include_bytes!(\"move.icon\")"));
128    }
