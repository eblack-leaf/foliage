1//! File stems into Rust identifiers.
2
3/// Rust keywords that cannot be written as raw identifiers, so a name colliding with one is
4/// suffixed instead.
5const UNRAWABLE: [&str; 4] = ["crate", "self", "Self", "super"];
6
7/// Every other reserved word, which `r#` makes usable as-is.
8const KEYWORDS: [&str; 47] = [
9    "abstract", "as", "async", "await", "become", "box", "break", "const", "continue", "do",
10    "dyn", "else", "enum", "extern", "false", "final", "fn", "for", "gen", "if", "impl", "in",
11    "let", "loop", "macro", "match", "mod", "move", "mut", "override", "priv", "pub", "ref",
12    "return", "static", "struct", "trait", "true", "try", "type", "typeof", "unsafe", "unsized",
13    "use", "virtual", "where", "while",
14];
15
16/// A file stem as a `snake_case` word: `arrow-up` and `ArrowUp` both become `arrow_up`.
17///
18/// Anything that is not alphanumeric separates, and a case change inside a word does too, so a
19/// set generated from either naming convention reads the same.
20pub fn snake_case(stem: &str) -> String {
21    let mut out = String::with_capacity(stem.len());
22    let mut previous: Option<char> = None;
23    for character in stem.chars() {
24        if !character.is_alphanumeric() {
25            if !out.ends_with('_') && !out.is_empty() {
26                out.push('_');
27            }
28            previous = None;
29            continue;
30        }
31        // A lower-to-upper step is a word boundary the separators did not state.
32        if character.is_uppercase()
33            && previous.is_some_and(|p| p.is_lowercase() || p.is_numeric())
34            && !out.ends_with('_')
35        {
36            out.push('_');
37        }
38        out.extend(character.to_lowercase());
39        previous = Some(character);
40    }
41    out.trim_matches('_').to_string()
42}
43
44/// A file stem as a struct field: [`snake_case`], made usable where it would not be.
45///
46/// A name that collides with a keyword is written raw, except for the four that cannot be, which
47/// are suffixed. A name opening with a digit is prefixed, and one that reduces to nothing keeps
48/// the set compiling rather than emitting an empty field.
49pub fn field(stem: &str) -> String {
50    let name = snake_case(stem);
51    if name.is_empty() {
52        return "_unnamed".to_string();
53    }
54    if UNRAWABLE.contains(&name.as_str()) {
55        return format!("{name}_");
56    }
57    if KEYWORDS.contains(&name.as_str()) {
58        return format!("r#{name}");
59    }
60    if name.starts_with(|c: char| c.is_numeric()) {
61        return format!("_{name}");
62    }
63    name
64}

66#[cfg(test)]
67mod tests {
68    use super::{field, snake_case};
69
70    #[test]
71    fn separators_and_case_changes_both_break_words() {
72        assert_eq!(snake_case("arrow-up"), "arrow_up");
73        assert_eq!(snake_case("book_open"), "book_open");
74        assert_eq!(snake_case("ArrowUp"), "arrow_up");
75        assert_eq!(snake_case("chevrons.left"), "chevrons_left");
76        assert_eq!(snake_case("box"), "box");
77        assert_eq!(snake_case("x"), "x");
78    }
79
80    /// A run of separators is one boundary, and the edges of a name are not boundaries at all.
81    #[test]
82    fn runs_and_edges_do_not_leave_bare_underscores() {
83        assert_eq!(snake_case("--arrow--up--"), "arrow_up");
84        assert_eq!(snake_case("_leading"), "leading");
85    }
86
87    /// An icon is named by whoever drew it, so a stem that is a keyword or opens with a digit is
88    /// ordinary input rather than a mistake -- and every one of them has to compile.
89    #[test]
90    fn a_name_that_cannot_be_written_plainly_is_made_usable() {
91        assert_eq!(field("move"), "r#move");
92        assert_eq!(field("type"), "r#type");
93        assert_eq!(field("self"), "self_");
94        assert_eq!(field("crate"), "crate_");
95        assert_eq!(field("2x"), "_2x");
96        assert_eq!(field("---"), "_unnamed");
97        assert_eq!(field("arrow-up"), "arrow_up");
98    }
