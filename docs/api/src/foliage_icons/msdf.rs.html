1//! SVG geometry into a multi-channel signed distance field.
2//!
3//! RGB is the multi-channel field, where a median across the three reconstructs a corner that a
4//! single channel would round off; A is a plain signed distance. Icon sets are commonly
5//! stroke-only (`fill="none" stroke=...`), so strokes are outlined into fill contours before
6//! anything is generated -- otherwise they would bake empty.
7//!
8//! All geometry is boolean-unioned into non-overlapping contours first. Generation assumes a
9//! clean outline: an edge buried inside an overlap -- a stroke crossing another, a cap meeting a
10//! line -- still contributes a coloured distance and drags the median below the edge along the
11//! seam, which draws as a crack at every intersection.
12

26/// Flattening tolerance, as a fraction of the mark's extent.
27///
28/// Paired with [`CORNER_SIN_ALPHA`]: a chord this close to the curve it came from turns by less
29/// than the corner threshold at any feature the field can resolve, so a vertex left by
30/// flattening stays smooth while a genuine corner still splits colours.
31const FLATTEN_TOLERANCE: f64 = 5e-5;
32
33/// Sine of the sharpest turn that is not taken for a corner while colouring edges, about 8.6
34/// degrees.
35const CORNER_SIN_ALPHA: f64 = 0.15;
36
37/// How deep a curve is subdivided before its chord is accepted regardless of tolerance.
38const MAX_FLATTEN_DEPTH: u32 = 12;
39
40/// How a field is baked: the two numbers a mark is registered with.
41///
42/// Both describe the bake rather than the artwork, and foliage is told them at registration
43/// because a field that was fetched cannot be asked.
44#[derive(Copy, Clone, Debug, PartialEq)]
45pub struct Bake {
46    /// Texels per side of the square field.
47    ///
48    /// One field serves every on-screen size, so this is bake-generously territory: 32 to 64
49    /// keeps a small mark crisp without making the file large.
50    pub side: u32,
51    /// How many texels the distance spread covers.
52    ///
53    /// What turns a sampled distance into an edge one screen pixel wide at whatever size the
54    /// mark is drawn. Two to four balances sharpness at small sizes against smoothness at large.
55    pub range: f32,
56}
57
58impl Bake {
59    /// A square field `side` texels across, with a `range`-texel spread.
60    pub fn new(side: u32, range: f32) -> Self {
61        Self { side, range }
62    }
63}

65/// One baked field, and what it was baked with.
66#[derive(Clone, Debug)]
67pub struct Baked {
68    /// How it was baked.
69    pub spec: Bake,
70    /// `side` by `side` texels of RGBA, row-major -- exactly what foliage registers.
71    pub rgba: Vec<u8>,
72}

74/// Bakes one SVG into a square field.
75///
76/// The artwork is fitted to the field's interior, leaving a margin of `range` texels on each
77/// edge so the spread is not clipped at the mark's own boundary. Bounds are taken from the
78/// unioned geometry rather than from the `viewBox`, so a stroke that overshoots still fits.
79///
80/// # Errors
81///
82/// If the SVG cannot be parsed, or holds no fill or stroke to bake.
83pub fn bake(svg: &[u8], spec: Bake) -> Result<Baked, String> {
84    let range = spec.range as f64;
85    let tree = usvg::Tree::from_data(svg, &usvg::Options::default()).map_err(|e| e.to_string())?;
86
87    // Every fill and stroke-outline, in absolute user space, split by fill rule.
88    let mut nonzero: Vec<SkPath> = Vec::new();
89    let mut evenodd: Vec<SkPath> = Vec::new();
90    collect(tree.root(), &mut nonzero, &mut evenodd);
91    if nonzero.is_empty() && evenodd.is_empty() {
92        return Err("no fill or stroke geometry to bake".to_string());
93    }
94
95    // The control-point extent is accurate enough to pick a flattening tolerance from.
96    let (mut min_x, mut min_y, mut max_x, mut max_y) = (f32::MAX, f32::MAX, f32::MIN, f32::MIN);
97    for path in nonzero.iter().chain(&evenodd) {
98        let bounds = path.bounds();
99        min_x = min_x.min(bounds.left());
100        min_y = min_y.min(bounds.top());
101        max_x = max_x.max(bounds.right());
102        max_y = max_y.max(bounds.bottom());
103    }
104    let rough = ((max_x - min_x).max(max_y - min_y) as f64).max(0.0);
105    let tolerance = (rough * FLATTEN_TOLERANCE).max(1e-9);
106
107    // Flatten to polylines, normalise even-odd fills to non-overlapping nonzero contours, then
108    // union everything into one outline with holes wound against their outers.
109    let mut contours: Vec<Vec<[f64; 2]>> = Vec::new();
110    for path in &nonzero {
111        flatten(path, tolerance, &mut contours);
112    }
113    for path in &evenodd {
114        let mut alone: Vec<Vec<[f64; 2]>> = Vec::new();
115        flatten(path, tolerance, &mut alone);
116        for shape in alone.simplify_shape(UnionFillRule::EvenOdd) {
117            contours.extend(shape);
118        }
119    }
120    let unioned = contours.simplify_shape(UnionFillRule::NonZero);
121    if unioned.iter().flatten().next().is_none() {
122        return Err("geometry produced no contours".to_string());
123    }
124
125    let (mut min_x, mut min_y, mut max_x, mut max_y) = (f64::MAX, f64::MAX, f64::MIN, f64::MIN);
126    for [x, y] in unioned.iter().flatten().flatten() {
127        min_x = min_x.min(*x);
128        min_y = min_y.min(*y);
129        max_x = max_x.max(*x);
130        max_y = max_y.max(*y);
131    }
132    let (width, height) = (max_x - min_x, max_y - min_y);
133
134    // A uniform scale fitting the longer side into the interior, then centred.
135    let usable = spec.side as f64 - 2.0 * range;
136    let extent = width.max(height);
137    let scale = match extent > 0.0 {
138        true => usable / extent,
139        false => 1.0,
140    };
141    let x = range + (usable - width * scale) * 0.5 - min_x * scale;
142    let y = range + (usable - height * scale) * 0.5 - min_y * scale;
143    let transformation =
144        nalgebra::convert::<_, Affine2<f64>>(Similarity2::new(Vector2::new(x, y), 0.0, scale));
145
146    // The generator wants control points already in field texels, so the shape is transformed
147    // before it is coloured.
148    let mut shape: Shape<Contour> = Shape::default();
149    for contour in unioned.iter().flatten() {
150        if contour.len() < 3 {
151            continue;
152        }
153        let mut built = Contour::default();
154        for pair in contour.windows(2) {
155            built.segments.push(Segment::line(point(pair[0]), point(pair[1])));
156        }
157        let last = *contour.last().expect("a contour of three or more");
158        built
159            .segments
160            .push(Segment::line(point(last), point(contour[0])));
161        shape.contours.push(built);
162    }
163    if shape.contours.is_empty() {
164        return Err("geometry produced no contours".to_string());
165    }
166    shape.transform(&transformation);
167
168    // Generated as a float field because the error-correction pass reads one.
169    let coloured = Shape::edge_coloring_simple(shape, CORNER_SIN_ALPHA, 69_441_337_420);
170    let prepared = coloured.prepare();
171    let mut field = Rgba32FImage::new(spec.side, spec.side);
172    generate_mtsdf(&prepared, range, &mut field);
173    correct_sign_mtsdf(&mut field, &prepared, FillRule::Nonzero);
174    // Predicts and repairs the median clashes that otherwise draw as dark dots at corners and
175    // stroke caps, which the raw generate leaves in.
176    correct_error_mtsdf(
177        &mut field,
178        &coloured,
179        &prepared,
180        range,
181        &ErrorCorrectionConfig::default(),
182    );
183
184    let rgba = field
185        .into_raw()
186        .into_iter()
187        .map(|channel| (channel.clamp(0.0, 1.0) * 255.0).round() as u8)
188        .collect();
189    Ok(Baked { spec, rgba })
190}

192/// Walks the tree, pushing each path's fill geometry and its stroke outline in absolute user
193/// coordinates.
194///
195/// Even-odd fills are kept apart so they can be normalised under their own rule before the
196/// nonzero union.
197fn collect(group: &usvg::Group, nonzero: &mut Vec<SkPath>, evenodd: &mut Vec<SkPath>) {
198    for node in group.children() {
199        match node {
200            usvg::Node::Group(group) => collect(group, nonzero, evenodd),
201            usvg::Node::Path(path) => {
202                let transform = path.abs_transform();
203                let data = path.data();
204                if let Some(fill) = path.fill()
205                    && let Some(placed) = data.clone().transform(transform)
206                {
207                    match fill.rule() {
208                        usvg::FillRule::NonZero => nonzero.push(placed),
209                        usvg::FillRule::EvenOdd => evenodd.push(placed),
210                    }
211                }
212                if let Some(stroke) = path.stroke()
213                    && let Some(outline) = data.stroke(&stroke.to_tiny_skia(), 1.0)
214                    && let Some(placed) = outline.transform(transform)
215                {
216                    nonzero.push(placed);
217                }
218            }
219            _ => {}
220        }
221    }
222}

224/// Flattens a path into polyline contours within `tolerance` of the curves they came from.
225fn flatten(path: &SkPath, tolerance: f64, out: &mut Vec<Vec<[f64; 2]>>) {
226    let mut contour: Vec<[f64; 2]> = Vec::new();
227    let close = |contour: &mut Vec<[f64; 2]>, out: &mut Vec<Vec<[f64; 2]>>| {
228        match contour.len() >= 3 {
229            true => out.push(std::mem::take(contour)),
230            false => contour.clear(),
231        }
232    };
233    for segment in path.segments() {
234        match segment {
235            PathSegment::MoveTo(to) => {
236                close(&mut contour, out);
237                contour.push(at(to));
238            }
239            PathSegment::LineTo(to) => contour.push(at(to)),
240            PathSegment::QuadTo(control, to) => {
241                let from = *contour.last().unwrap_or(&at(to));
242                quad(from, at(control), at(to), tolerance, 0, &mut contour);
243            }
244            PathSegment::CubicTo(first, second, to) => {
245                let from = *contour.last().unwrap_or(&at(to));
246                cubic(
247                    from,
248                    at(first),
249                    at(second),
250                    at(to),
251                    tolerance,
252                    0,
253                    &mut contour,
254                );
255            }
256            PathSegment::Close => close(&mut contour, out),
257        }
258    }
259    if contour.len() >= 3 {
260        out.push(contour);
261    }
262}

264/// Subdivides a quadratic until its chord is within `tolerance`.
265fn quad(from: [f64; 2], control: [f64; 2], to: [f64; 2], tolerance: f64, depth: u32, out: &mut Vec<[f64; 2]>) {
266    if depth >= MAX_FLATTEN_DEPTH || chord(control, from, to) <= tolerance {
267        out.push(to);
268        return;
269    }
270    let first = mid(from, control);
271    let second = mid(control, to);
272    let middle = mid(first, second);
273    quad(from, first, middle, tolerance, depth + 1, out);
274    quad(middle, second, to, tolerance, depth + 1, out);
275}

277/// Subdivides a cubic until both its control points are within `tolerance` of its chord.
278fn cubic(
279    from: [f64; 2],
280    first: [f64; 2],
281    second: [f64; 2],
282    to: [f64; 2],
283    tolerance: f64,
284    depth: u32,
285    out: &mut Vec<[f64; 2]>,
286) {
287    if depth >= MAX_FLATTEN_DEPTH
288        || (chord(first, from, to) <= tolerance && chord(second, from, to) <= tolerance)
289    {
290        out.push(to);
291        return;
292    }
293    let (a, b, c) = (mid(from, first), mid(first, second), mid(second, to));
294    let (d, e) = (mid(a, b), mid(b, c));
295    let middle = mid(d, e);
296    cubic(from, a, d, middle, tolerance, depth + 1, out);
297    cubic(middle, e, c, to, tolerance, depth + 1, out);
298}

300/// The midpoint of two points.
301fn mid(a: [f64; 2], b: [f64; 2]) -> [f64; 2] {
302    [(a[0] + b[0]) * 0.5, (a[1] + b[1]) * 0.5]
303}
304
305/// The distance from a point to the segment between two others.
306fn chord(point: [f64; 2], from: [f64; 2], to: [f64; 2]) -> f64 {
307    let (dx, dy) = (to[0] - from[0], to[1] - from[1]);
308    let length = dx * dx + dy * dy;
309    let (px, py) = (point[0] - from[0], point[1] - from[1]);
310    if length <= f64::EPSILON {
311        return (px * px + py * py).sqrt();
312    }
313    let along = ((px * dx + py * dy) / length).clamp(0.0, 1.0);
314    let (ex, ey) = (px - along * dx, py - along * dy);
315    (ex * ex + ey * ey).sqrt()
316}

318/// A path point as a pair.
319fn at(point: SkPoint) -> [f64; 2] {
320    [point.x as f64, point.y as f64]
321}
322
323/// A pair as a generator point.
324fn point(pair: [f64; 2]) -> Point {
325    Point::new(pair[0], pair[1])
326}

332    /// The median of a texel's three colour channels, which is what the shader reconstructs.
333    fn median(texel: &[u8]) -> u8 {
334        let mut channels = [texel[0], texel[1], texel[2]];
335        channels.sort_unstable();
336        channels[1]
337    }

339    /// An icon set is commonly stroke-only, so a stroke has to be outlined into a fill before
340    /// anything is generated. Dropped strokes bake an all-outside field, which draws nothing.
341    #[test]
342    fn a_stroke_only_svg_bakes_a_field_with_an_inside() {
343        let svg = br#"<svg xmlns="http://www.w3.org/2000/svg" width="24" height="24"
344            viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2"
345            stroke-linecap="round" stroke-linejoin="round">
346            <rect x="4" y="4" width="16" height="16"></rect></svg>"#;
347        let baked = bake(svg, Bake::new(48, 3.0)).expect("bake");
348        assert_eq!(baked.rgba.len(), 48 * 48 * 4);
349
350        let (mut inside, mut outside) = (0, 0);
351        for texel in baked.rgba.chunks_exact(4) {
352            match median(texel) {
353                m if m > 140 => inside += 1,
354                m if m < 115 => outside += 1,
355                _ => {}
356            }
357        }
358        assert!(inside > 0, "no inside texels -- the stroke was not outlined");
359        assert!(outside > 0, "no outside texels -- the field is degenerate");
360    }

362    /// Two crossing strokes union into one outline, so no interior edge survives the
363    /// intersection. Overlapping contours fed straight to the generator leave a chasm -- medians
364    /// near the edge value -- through the middle of every crossing.
365    #[test]
366    fn crossing_strokes_have_no_chasm_at_the_intersection() {
367        let svg = br#"<svg xmlns="http://www.w3.org/2000/svg" width="24" height="24"
368            viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="4"
369            stroke-linecap="round" stroke-linejoin="round">
370            <line x1="4" y1="4" x2="20" y2="20"></line>
371            <line x1="20" y1="4" x2="4" y2="20"></line></svg>"#;
372        let baked = bake(svg, Bake::new(48, 3.0)).expect("bake");
373
374        // The crossing lands at the field's centre and is wholly inside the union.
375        for dy in -2i32..=2 {
376            for dx in -2i32..=2 {
377                let (x, y) = ((24 + dx) as usize, (24 + dy) as usize);
378                let texel = &baked.rgba[(y * 48 + x) * 4..];
379                assert!(
380                    median(texel) > 150,
381                    "texel ({x},{y}) median {} -- chasm at the intersection",
382                    median(texel)
383                );
384            }
385        }
386    }
