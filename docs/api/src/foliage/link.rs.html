1//! A URL handed to the host.
2//!
3//! Two things an app can ask for that the engine has no way to do itself: go somewhere, and save
4//! something. Both are the host's, and both are nameable on every target because where an app's
5//! links point is a fact about the app rather than about what it was built for.
6//!
7//! Handed over by `photosynthesize` and left in the engine's hands under the headless suite -- the
8//! seam [`Wake`](crate::queue::Wake) sits on. A test therefore opens no browser on whoever is
9//! running it.
10
11use tracing::debug;
12
13/// Whether a URL reaches the host at all.
14///
15/// One flag rather than a handle, because neither of these holds anything: a navigation is a line
16/// of JavaScript or a process the desktop starts, and there is nothing either keeps between calls.
17#[derive(Default)]
18pub(crate) struct Links {
19    attached: bool,
20}
21
22impl Links {
23    /// Lets URLs through, once, at boot.
24    pub(crate) fn attach(&mut self) {
25        self.attached = true;
26    }
27
28    /// Goes to `url`.
29    ///
30    /// On the web this replaces the page. Off it the desktop is asked to open the URL, which is
31    /// what "go there" means where there is no page to replace.
32    pub(crate) fn navigate(&self, url: &str) {
33        debug!(url, "navigating");
34        if !self.attached {
35            return;
36        }
37        #[cfg(not(target_family = "wasm"))]
38        if let Err(reason) = open::that_detached(url) {
39            tracing::warn!(url, %reason, "navigation refused");
40        }
41        #[cfg(target_family = "wasm")]
42        if let Some(window) = web_sys::window()
43            && window.location().set_href(url).is_err()
44        {
45            tracing::warn!(url, "navigation refused");
46        }
47    }
48
49    /// Asks the host to save what is at `url`.
50    ///
51    /// The web's, and only the web's: a browser is what turns a URL into a file in a person's
52    /// downloads, and off it a program that wants a file on disk has one to write it with. Off the
53    /// web this is traced and nothing else -- the surface is here on every target so that an app
54    /// naming what it offers names it once.
55    pub(crate) fn download(&self, url: &str) {
56        debug!(url, "downloading");
57        if !self.attached {
58            return;
59        }
60        #[cfg(target_family = "wasm")]
61        if let Some(anchor) = anchor(url) {
62            // Clicked rather than navigated to: `download` is an attribute of the link and not of
63            // the address, so there is no form of `location` that carries it.
64            anchor.click();
65            anchor.remove();
66        }
67    }
68}
69
70/// An anchor carrying `url`, in the page and ready to be clicked.
71///
72/// Built and taken out again around one click, because it is not something the page has -- it is
73/// how the one verb the browser offers is spelled.
74#[cfg(target_family = "wasm")]
75fn anchor(url: &str) -> Option<web_sys::HtmlElement> {
76    use wasm_bindgen::JsCast;
77
78    let document = web_sys::window()?.document()?;
79    let anchor = document
80        .create_element("a")
81        .ok()?
82        .dyn_into::<web_sys::HtmlElement>()
83        .ok()?;
84    anchor.set_attribute("href", url).ok()?;
85    anchor.set_attribute("download", "").ok()?;
86    anchor.set_attribute("style", "display:none").ok()?;
87    document.body()?.append_child(&anchor).ok()?;
88    Some(anchor)
89}
