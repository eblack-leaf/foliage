1//! Ginkgo -- the GPU.
2//!
3//! The device, the surface, and everything sized against it. Every renderer in [`Ash`](crate::ash)
4//! asks this for what it needs to build a pipeline and hands it back a draw, so there is one place
5//! that knows what a device is and one place that talks to a surface.
6//!
7//! # The scale factor stops here
8//!
9//! The engine is written in logical pixels throughout. The surface is configured in physical ones,
10//! and the display's ratio between them is applied at exactly two places, both of them below this
11//! line: the surface and its depth attachment are sized in physical pixels, and the projection is
12//! built from the logical area. Between them the rasteriser does the conversion, so no instance,
13//! no radius and no section is ever carried in device pixels.
14

54impl Ginkgo {
55    /// Acquires a device for `window` and configures a surface on it.
56    ///
57    /// Asynchronous because the web's is: a browser resolves an adapter and a device through
58    /// promises, and there is no thread there to block. Native awaits it to completion at the
59    /// callsite instead.
60    pub(crate) async fn acquire(window: Arc<Window>, area: Area, scale: f32) -> Self {
61        let instance = wgpu::Instance::new(InstanceDescriptor {
62            backends: Backends::VULKAN | Backends::METAL | Backends::DX12 | Backends::GL,
63            ..InstanceDescriptor::new_without_display_handle()
64        });
65        let surface = instance.create_surface(window).expect("surface");
66        let adapter = instance
67            .request_adapter(&RequestAdapterOptions {
68                power_preference: PowerPreference::default(),
69                force_fallback_adapter: false,
70                compatible_surface: Some(&surface),
71                apply_limit_buckets: false,
72            })
73            .await
74            .expect("adapter");
75        info!(adapter = adapter.get_info().name, backend = ?adapter.get_info().backend, "adapter");
76        // Web and Android run on downlevel devices, and asking for more than one of those has is a
77        // failure at device request rather than at the draw that needed it.
78        let limits = if cfg!(any(target_family = "wasm", target_os = "android")) {
79            Limits::downlevel_webgl2_defaults()
80        } else {
81            Limits::default()
82        };
83        let (device, queue) = adapter
84            .request_device(&DeviceDescriptor {
85                label: Some("foliage"),
86                required_features: Features::default(),
87                required_limits: limits.using_resolution(adapter.limits()),
88                ..DeviceDescriptor::default()
89            })
90            .await
91            .expect("device");
92        // A `Color` holds sRGB channels, which is the form a scheme is stated in and the form a
93        // shader reads. Taking the surface's non-sRGB view writes those channels through unchanged;
94        // the sRGB view would encode them a second time.
95        let format = surface
96            .get_capabilities(&adapter)
97            .formats
98            .first()
99            .expect("surface format")
100            .remove_srgb_suffix();
101        info!(?format, "surface");
102        let configuration = configuration(format, area, scale);
103        surface.configure(&device, &configuration);
104        let depth = Depth::new(&device, &configuration);
105        let viewport = Viewport::new(&device, area);
106        let layout = device.create_bind_group_layout(&BindGroupLayoutDescriptor {
107            label: Some("viewport-layout"),
108            entries: &[BindGroupLayoutEntry {
109                binding: 0,
110                visibility: ShaderStages::VERTEX,
111                ty: BindingType::Buffer {
112                    ty: BufferBindingType::Uniform,
113                    has_dynamic_offset: false,
114                    min_binding_size: None,
115                },
116                count: None,
117            }],
118        });
119        let binding = device.create_bind_group(&BindGroupDescriptor {
120            label: Some("viewport-binding"),
121            layout: &layout,
122            entries: &[BindGroupEntry {
123                binding: 0,
124                resource: viewport.binding(),
125            }],
126        });
127        Self {
128            device,
129            queue,
130            surface,
131            configuration,
132            depth,
133            viewport,
134            layout,
135            binding,
136            scale,
137        }
138    }

140    /// Resizes the surface and everything measured against it.
141    ///
142    /// The projection is rebuilt from the logical area and the attachments from the physical one,
143    /// which is the whole of what the scale factor is for.
144    pub(crate) fn resize(&mut self, area: Area, scale: f32) {
145        self.scale = scale;
146        self.configuration = configuration(self.configuration.format, area, scale);
147        self.surface.configure(&self.device, &self.configuration);
148        self.depth = Depth::new(&self.device, &self.configuration);
149        self.viewport.resize(&self.queue, area);
150        debug!(
151            width = self.configuration.width,
152            height = self.configuration.height,
153            scale,
154            "surface configured"
155        );
156    }

158    /// The scissor rect a logical clip becomes, in the surface's own pixels.
159    ///
160    /// Expanded outward to whole pixels rather than truncated: a fractional scale factor makes these
161    /// rects fractional, and taking the floor of both edges would shave the far side off every
162    /// clipped region. Clamped to the surface, because a scissor outside the attachment is not a
163    /// smaller region -- it is invalid.
164    pub(crate) fn scissor(&self, clip: Section) -> (u32, u32, u32, u32) {
165        let width = self.configuration.width as f32;
166        let height = self.configuration.height as f32;
167        let left = (clip.left() * self.scale).floor().clamp(0.0, width);
168        let top = (clip.top() * self.scale).floor().clamp(0.0, height);
169        let right = (clip.right() * self.scale).ceil().clamp(left, width);
170        let bottom = (clip.bottom() * self.scale).ceil().clamp(top, height);
171        (
172            left as u32,
173            top as u32,
174            (right - left) as u32,
175            (bottom - top) as u32,
176        )
177    }

222    /// The depth state every pipeline shares: nearer wins, and every draw writes.
223    ///
224    /// `LessEqual` rather than `Less` so that a redraw of the same content leaves the same result.

226        Some(DepthStencilState {

235    /// Draws one frame, clearing to `clear` and handing the pass to `draw`.
236    ///
237    /// A surface that cannot be acquired is skipped rather than retried in place: it means the
238    /// window is gone, occluded, or being resized, and the next paint has everything it needs to
239    /// produce the same image again. Nothing is lost by not painting, because what is drawn is
240    /// held by the renderers rather than consumed by the draw.
241    pub(crate) fn draw(&self, clear: Color, draw: impl FnOnce(&mut RenderPass<'_>)) {
242        let Some(frame) = self.acquire_frame() else {
243            return;
244        };
245        let view = frame.texture.create_view(&TextureViewDescriptor::default());
246        let mut encoder = self
247            .device
248            .create_command_encoder(&CommandEncoderDescriptor {
249                label: Some("draw"),
250            });
251        {
252            let mut pass = encoder.begin_render_pass(&RenderPassDescriptor {
253                label: Some("draw"),
254                color_attachments: &[Some(RenderPassColorAttachment {
255                    view: &view,
256                    depth_slice: None,
257                    resolve_target: None,
258                    ops: Operations {
259                        load: LoadOp::Clear(wgpu::Color {
260                            r: clear.red as f64,
261                            g: clear.green as f64,
262                            b: clear.blue as f64,
263                            a: clear.alpha as f64,
264                        }),
265                        store: StoreOp::Store,
266                    },
267                })],
268                depth_stencil_attachment: Some(self.depth.attachment()),
269                timestamp_writes: None,
270                occlusion_query_set: None,
271                multiview_mask: None,
272            });
273            pass.set_bind_group(0, &self.binding, &[]);
274            draw(&mut pass);
275        }
276        self.queue.submit(Some(encoder.finish()));
277        self.queue.present(frame);
278    }

280    /// The surface's next texture, reconfiguring once if the surface went stale under us.
281    fn acquire_frame(&self) -> Option<wgpu::SurfaceTexture> {
282        match self.surface.get_current_texture() {
283            CurrentSurfaceTexture::Success(frame) | CurrentSurfaceTexture::Suboptimal(frame) => {
284                Some(frame)
285            }
286            CurrentSurfaceTexture::Outdated | CurrentSurfaceTexture::Lost => {
287                debug!("surface reconfigured");
288                self.surface.configure(&self.device, &self.configuration);
289                match self.surface.get_current_texture() {
290                    CurrentSurfaceTexture::Success(frame)
291                    | CurrentSurfaceTexture::Suboptimal(frame) => Some(frame),
292                    _ => None,
293                }
294            }
295            CurrentSurfaceTexture::Timeout | CurrentSurfaceTexture::Occluded => None,
296            CurrentSurfaceTexture::Validation => panic!("surface validation"),
297        }
298    }

301/// The surface as the platform holds it: physical pixels, and at least one of each.
302fn configuration(format: TextureFormat, area: Area, scale: f32) -> SurfaceConfiguration {
303    SurfaceConfiguration {
304        usage: TextureUsages::RENDER_ATTACHMENT,
305        format,
306        width: (area.width * scale).round().max(1.0) as u32,
307        height: (area.height * scale).round().max(1.0) as u32,
308        present_mode: PresentMode::Fifo,
309        desired_maximum_frame_latency: 2,
310        alpha_mode: CompositeAlphaMode::Auto,
311        color_space: SurfaceColorSpace::Auto,
312        view_formats: vec![format],
313    }
