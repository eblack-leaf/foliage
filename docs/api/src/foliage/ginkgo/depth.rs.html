1//! The depth attachment, and the range an element's rank is drawn at.
2
3use wgpu::{
4    Extent3d, LoadOp, Operations, RenderPassDepthStencilAttachment, StoreOp, SurfaceConfiguration,
5    TextureDescriptor, TextureDimension, TextureFormat, TextureUsages, TextureView,
6    TextureViewDescriptor,
7};
8
9/// The depth buffer, sized with the surface.
10pub(crate) struct Depth {
11    view: TextureView,
12}
13
14impl Depth {
15    pub(crate) const FORMAT: TextureFormat = TextureFormat::Depth32Float;
16
17    /// Where an element's rank is placed.
18    ///
19    /// Open at both ends: nothing is drawn at exactly the near or the far plane, so the range
20    /// always has room on either side of what is in it. `0.0` is nearest the viewer, which is the
21    /// opposite sense to [`ResolvedElevation`](crate::elevation::ResolvedElevation) -- the
22    /// front-most element takes the smallest depth.
23    pub(crate) const NEAR: f32 = 0.0;
24    pub(crate) const FAR: f32 = 1.0;
25
26    pub(crate) fn new(device: &wgpu::Device, configuration: &SurfaceConfiguration) -> Self {
27        let texture = device.create_texture(&TextureDescriptor {
28            label: Some("depth"),
29            size: Extent3d {
30                width: configuration.width,
31                height: configuration.height,
32                depth_or_array_layers: 1,
33            },
34            mip_level_count: 1,
35            sample_count: 1,
36            dimension: TextureDimension::D2,
37            format: Self::FORMAT,
38            usage: TextureUsages::RENDER_ATTACHMENT,
39            view_formats: &[Self::FORMAT],
40        });
41        Self {
42            view: texture.create_view(&TextureViewDescriptor::default()),
43        }
44    }
45
46    /// The depth `index` of `total` sits at, front-most last.
47    ///
48    /// Evenly spaced across the range and strictly inside it, so two elements never share a depth
49    /// and the ordering the resolver produced is the ordering the depth test enforces. Taken from
50    /// the position in the sorted order rather than from the rank itself, because a rank is a pair
51    /// of an accumulated elevation and an allocation counter and has no meaningful magnitude to
52    /// scale -- only an order, which is exactly what a position is.
53    ///
54    /// The range therefore holds any number of elements: it subdivides rather than fills, so there
55    /// is no budget to exhaust and no value an element can be given that leaves no room beside it.
56    /// The one real ceiling is the attachment's precision -- the step is `1 / (total + 1)`, and
57    /// [`FORMAT`](Depth::FORMAT) separates two depths until that falls under an `f32` interval,
58    /// which is on the order of eight million elements. Assigning from the rank's own magnitude is
59    /// what would need a budget, and then a rebalance whenever a gap in it ran out.
60    pub(crate) fn of(index: usize, total: usize) -> f32 {
61        let step = (index + 1) as f32 / (total + 1) as f32;
62        Self::FAR - step * (Self::FAR - Self::NEAR)
63    }
64
65    pub(crate) fn attachment(&self) -> RenderPassDepthStencilAttachment<'_> {
66        RenderPassDepthStencilAttachment {
67            view: &self.view,
68            depth_ops: Some(Operations {
69                load: LoadOp::Clear(Self::FAR),
70                store: StoreOp::Store,
71            }),
72            stencil_ops: None,
73        }
74    }
75}
