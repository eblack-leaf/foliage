1//! The projection every renderer draws through.
2
3use wgpu::util::{BufferInitDescriptor, DeviceExt};
4use wgpu::{Buffer, BufferUsages, Device, Queue};
5
6use crate::coordinate::Area;
7
8/// The matrix taking logical pixels to clip space, held in the one uniform every pipeline binds.
9///
10/// Built from the *logical* area, so an instance is written in the same units an app stated it in
11/// and the display's scale factor is applied by the rasteriser mapping clip space onto a surface
12/// that was configured in physical pixels. Depth passes through untouched: a renderer hands over a
13/// value already inside [`Depth`](crate::ginkgo::depth::Depth)'s range.
14pub(crate) struct Viewport {
15    buffer: Buffer,
16}
17
18impl Viewport {
19    pub(crate) fn new(device: &Device, area: Area) -> Self {
20        Self {
21            buffer: device.create_buffer_init(&BufferInitDescriptor {
22                label: Some("viewport"),
23                contents: bytemuck::cast_slice(&projection(area)),
24                usage: BufferUsages::UNIFORM | BufferUsages::COPY_DST,
25            }),
26        }
27    }
28
29    pub(crate) fn resize(&mut self, queue: &Queue, area: Area) {
30        queue.write_buffer(&self.buffer, 0, bytemuck::cast_slice(&projection(area)));
31    }
32
33    pub(crate) fn binding(&self) -> wgpu::BindingResource<'_> {
34        self.buffer.as_entire_binding()
35    }
36}
37
38/// Column-major, as WGSL reads a `mat4x4<f32>`.
39///
40/// `x` runs left to right across the area and `y` top to bottom, which is the sense
41/// [`Position`](crate::Position) is written in, so the projection is what flips it rather than
42/// every value that passes through it.
43fn projection(area: Area) -> [[f32; 4]; 4] {
44    [
45        [2.0 / area.width, 0.0, 0.0, 0.0],
46        [0.0, -2.0 / area.height, 0.0, 0.0],
47        [0.0, 0.0, 1.0, 0.0],
48        [-1.0, 1.0, 0.0, 1.0],
49    ]
50}
