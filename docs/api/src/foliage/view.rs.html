1//! Scrolling regions -- what a gesture hands itself to, and where it goes afterwards.
2//!
3//! An element scrolls because it said so. A grid divides an element's box for its children and says
4//! nothing about scrolling, and an axis that was not declared does not scroll, has no extent, and
5//! cannot be moved: it is not a scrolling axis with a range of zero, which is a different and
6//! simpler thing to reason about.
7//!
8//! Three values, each written by one pass:
9//!
10//! | | Pass | Holds |
11//! |---|---|---|
12//! | [`Extent`] | R3, bottom-up | how far the content reaches from the region's own near edges |
13//! | [`Offset`] | R4, top-down | how far the region has been moved, clamped to what it can reach |
14//! | [`Clipped`] | R5, top-down | the box a scrolling ancestor leaves visible |
15//!
16//! # One unit: logical pixels
17//!
18//! Offset is logical pixels, read and written. [`ScrollTo`] still accepts other framings, and each
19//! of them **names its unit at the call site** -- which is what was missing when a fraction was the
20//! only thing that could be written and everything else in the system was in pixels.
21//!
22//! # Coast is a view behaviour
23//!
24//! A drag released with speed keeps the region moving. The gesture supplies its initial velocity
25//! and is then done; the decay, the clamp against the extent, and whether reaching an end while
26//! coasting chains outward or absorbs are all here. It is deliberately not an
27//! [`Aspen`](crate::aspen) tween: there is no target and no duration, and an integration that runs
28//! until it settles is a different shape from an interpolation between two known endpoints.
29

42/// Which axes an element scrolls, and what each of them does at its end.
43///
44/// Absent, the element does not scroll, and a drag inside it goes on outward to something that
45/// does.
46#[derive(Component, Copy, Clone, Debug, PartialEq, Eq)]
47pub(crate) struct Scrolls(pub(crate) Scroll);
48
49/// What a region declares about scrolling: the axes it moves on, and which of them absorb.
50///
51/// The two are one value rather than two flags, because a boundary policy on an axis the region
52/// does not scroll describes nothing, and two declarations that can disagree about which axes are
53/// in play is exactly the failure the per-axis split exists to avoid.
54///
55/// ```no_run
56/// # use foliage::{Axes, Place, Scroll, Stem};
57/// // Reaching the end hands the gesture outward to the next scrolling ancestor.
58/// Stem::new().scrolls(Axes::Vertical);
59/// // The region absorbs it instead. Nothing outside moves.
60/// Stem::new().scrolls(Scroll::new(Axes::Vertical).contain(Axes::Vertical));
61/// ```
62#[derive(Copy, Clone, Debug, PartialEq, Eq)]
63pub struct Scroll {
64    axes: Axes,
65    contains: Option<Axes>,
66}
67
68impl Scroll {
69    /// A region scrolling `axes`, each of them **chaining**.
70    ///
71    /// Reaching the end of a chaining axis hands the gesture outward to the next scrolling
72    /// ancestor, which is what lets a drag inside a list keep moving the page once the list is
73    /// done. It is the default because it is what a reader expects of ordinary content.
74    pub fn new(axes: Axes) -> Self {
75        Self {
76            axes,
77            contains: None,
78        }
79    }
80
81    /// The axes named **contain** rather than chain: the region absorbs a gesture that reaches
82    /// their end, and nothing outside it moves.
83    ///
84    /// For a region that owns its gesture outright -- a map, an editor, a pane inside a fixed
85    /// shell -- where reaching the bottom and having the whole page lurch is a bug every time.
86    ///
87    /// Naming an axis the region does not scroll says nothing, because there is no gesture on that
88    /// axis for it to absorb.
89    pub fn contain(mut self, axes: Axes) -> Self {
90        self.contains = Some(axes);
91        self
92    }
93
94    /// Whether this region scrolls `axis` at all.
95    pub(crate) fn covers(self, axis: Axis) -> bool {
96        self.axes.covers(axis)
97    }
98
99    /// Whether reaching the end of `axis` stops here rather than passing outward.
100    pub(crate) fn absorbs(self, axis: Axis) -> bool {
101        self.covers(axis) && self.contains.is_some_and(|axes| axes.covers(axis))
102    }
103}
104
105impl From<Axes> for Scroll {
106    fn from(axes: Axes) -> Self {
107        Self::new(axes)
108    }
109}

111/// Where a region is asked to move to, with its unit named at the call site.
112///
113/// ```no_run
114/// # use foliage::{Grove, Grow, Leaf, ScrollTo};
115/// # fn f(grove: &mut Grove, column: Leaf, section: Leaf) {
116/// grove.scroll(column, ScrollTo::px(240.0));
117/// grove.scroll(column, ScrollTo::fraction(0.5));
118/// grove.scroll(column, ScrollTo::end());
119/// grove.scroll(column, ScrollTo::show(section));
120/// # }
121/// ```
122///
123/// Every form lands in the same place: a number of logical pixels from the content's origin,
124/// clamped to what the region can reach. Reading the offset back afterwards returns those pixels.
125///
126/// # Which axis it moves
127///
128/// Every framing but one is stated **relative to the region's own range**, so it means the same
129/// thing on either axis and needs no axis named: the end is the end of each, half way is half of
130/// each, and bringing a descendant into view is one place in two dimensions.
131///
132/// [`px`](ScrollTo::px) is the exception, and it is the only absolute distance here: two hundred
133/// pixels down and two hundred pixels across are unrelated distances that happen to share a number.
134/// So on a region scrolling both axes it names one with [`on`](ScrollTo::on), and an op that does
135/// not is dropped rather than moving two axes by a coincidence.
136#[derive(Copy, Clone, Debug, PartialEq)]
137pub struct ScrollTo {
138    to: Destination,
139    /// Which axes it moves. Absent, read off the region.
140    axes: Option<Axes>,
141}
142
143/// The framings a destination can be stated in.
144#[derive(Copy, Clone, Debug, PartialEq)]
145enum Destination {
146    Px(f32),
147    Fraction(f32),
148    Start,
149    End,
150    Show(Leaf),
151}
152
153impl ScrollTo {
154    /// That many logical pixels from the content's origin. The unit everything else is in.
155    pub fn px(offset: f32) -> Self {
156        Self::to(Destination::Px(offset))
157    }
158
159    /// That fraction of the region's range, where `1.0` is as far as it goes.
160    pub fn fraction(of: f32) -> Self {
161        Self::to(Destination::Fraction(of))
162    }
163
164    /// Back to the content's origin.
165    pub fn start() -> Self {
166        Self::to(Destination::Start)
167    }
168
169    /// As far as the region reaches.
170    pub fn end() -> Self {
171        Self::to(Destination::End)
172    }
173
174    /// The least distance that brings a descendant into view, and no further.
175    ///
176    /// Jumping to a section is common enough that every app otherwise computes it by hand. A
177    /// descendant already in view moves the region nowhere.
178    ///
179    /// Dropped, like any op naming something it does not apply to, if the element named is not
180    /// grown under the region.
181    pub fn show(leaf: Leaf) -> Self {
182        Self::to(Destination::Show(leaf))
183    }
184
185    /// Which axes this moves, where the region scrolls more than one.
186    ///
187    /// Naming an axis the region does not scroll leaves nothing for the destination to move, and
188    /// the op that carried it is dropped rather than doing nothing quietly.
189    pub fn on(mut self, axes: Axes) -> Self {
190        self.axes = Some(axes);
191        self
192    }
193
194    fn to(to: Destination) -> Self {
195        Self { to, axes: None }
196    }
197
198    /// The element this destination is stated against, if it is stated against one.
199    pub(crate) fn names(&self) -> Option<Leaf> {
200        match self.to {
201            Destination::Show(leaf) => Some(leaf),
202            _ => None,
203        }
204    }
205
206    /// Which axes of a region scrolling `scrolls` this destination actually moves.
207    ///
208    /// `None` is an op with nothing to do, and it covers both ways of getting there: an absolute
209    /// distance that named no axis on a region scrolling two, and a named axis the region does not
210    /// scroll. Neither is refused quietly -- the drain names it where the write was made.
211    pub(crate) fn over(&self, scrolls: Scroll) -> Option<Axes> {
212        let named = match self.axes {
213            Some(axes) => axes,
214            // Stated against the range, so it says the same thing on either axis.
215            None if self.relative() => Axes::Both,
216            None => match (
217                scrolls.covers(Axis::Horizontal),
218                scrolls.covers(Axis::Vertical),
219            ) {
220                (true, false) => Axes::Horizontal,
221                (false, true) => Axes::Vertical,
222                // Both, and no way to tell which of two unrelated distances was meant.
223                _ => return None,
224            },
225        };
226        named.shared(scrolls.axes)
227    }
228
229    /// Whether the destination is stated against the region rather than in absolute pixels.
230    fn relative(&self) -> bool {
231        !matches!(self.to, Destination::Px(_))
232    }
233
234    /// Where on `axis` this destination puts a region that currently sits at `at` and can travel
235    /// `span`.
236    ///
237    /// `shown` is the box of the element [`show`](ScrollTo::show) named, in the same layout
238    /// coordinates as `region` -- which is what makes the arithmetic a subtraction rather than a
239    /// walk.
240    fn landing(&self, axis: Axis, at: f32, span: f32, region: Section, shown: Section) -> f32 {
241        let landing = match self.to {
242            Destination::Px(offset) => offset,
243            Destination::Fraction(of) => of.clamp(0.0, 1.0) * span,
244            Destination::Start => 0.0,
245            Destination::End => span,
246            Destination::Show(_) => {
247                let (near, far, seen) = match axis {
248                    Axis::Horizontal => (
249                        shown.left() - region.left(),
250                        shown.right() - region.left(),
251                        region.width(),
252                    ),
253                    Axis::Vertical => (
254                        shown.top() - region.top(),
255                        shown.bottom() - region.top(),
256                        region.height(),
257                    ),
258                };
259                // The least movement that brings it in, and none at all if it is already there.
260                if near < at {
261                    near
262                } else if far > at + seen {
263                    far - seen
264                } else {
265                    at
266                }
267            }
268        };
269        landing.clamp(0.0, span)
270    }
271}

273/// How a released drag coasts.
274///
275/// A **global tuning value**, like every other number behind how the engine feels. Set it once,
276/// before the engine runs:
277///
278/// ```no_run
279/// # use core::time::Duration;
280/// # use foliage::{Foliage, Momentum};
281/// let mut foliage = Foliage::new();
282/// foliage.tune(Momentum {
283///     half_life: Duration::from_millis(350),
284///     minimum: 40.0,
285/// });
286/// ```
287#[derive(Copy, Clone, Debug, PartialEq)]
288pub struct Momentum {
289    /// How long a coast takes to lose half its speed.
290    ///
291    /// Stated as a half-life because that is the one form of it a person can read and predict: a
292    /// coast at half speed after this long, a quarter after twice as long, and travelling in total
293    /// a little under its release speed times this. The decay is continuous, so the ground a fling
294    /// covers does not depend on how often frames happen to run.
295    ///
296    /// Zero is no coast at all: a release stops the region where it was.
297    pub half_life: Duration,
298    /// The speed, in logical pixels per second, below which a release starts no coast and a
299    /// running one stops.
300    pub minimum: f32,
301}
302
303impl Default for Momentum {
304    fn default() -> Self {
305        Self {
306            half_life: Duration::from_millis(350),
307            minimum: 40.0,
308        }
309    }
310}

312/// How far a region has been moved from its content's origin, in logical pixels.
313///
314/// One unit throughout, read and written the same way. Positive is content moved toward the near
315/// edge -- what a drag away from that edge produces.
316#[derive(Component, Copy, Clone, Debug, Default, PartialEq)]
317pub(crate) struct Offset(pub(crate) Position);
318
319/// How far a region's content reaches, measured outward from the region's own near edges.
320///
321/// Derived from where the children landed, never from what is currently drawn: a child scrolled out
322/// of sight is the content the extent exists to describe, so culling has no say in it. Never
323/// smaller than the region's own box, so an empty region has a range of zero rather than a negative
324/// one.
325#[derive(Component, Copy, Clone, Debug, Default, PartialEq)]
326pub(crate) struct Extent(pub(crate) Area);
327
328/// How far out of the regions above it a floating element reaches.
329///
330/// Stated rather than defaulted, because both answers are right somewhere and neither is right
331/// everywhere. A menu on a row of a table that has its own scrollbar, inside a settings panel that
332/// has another, wants out of both. A tooltip on a file in a list inside a sidebar wants out of the
333/// list and *not* out of the sidebar, or it paints across the page beside it. Nothing about the
334/// two pictures differs to the engine, so the callsite says which.
335#[derive(Copy, Clone, Debug, PartialEq, Eq)]
336pub enum Escape {
337    /// Out of the region it is grown in, and no further: whatever clips that region clips this too.
338    Region,
339    /// Out of every region above it, held only by the surface.
340    Surface,
341    /// Out of every region up to the one named, which still holds it.
342    ///
343    /// The general case the other two are shorthands for, and the answer where neither of them is:
344    /// three regions deep, leaving two and staying inside the third. It names the element rather
345    /// than counting regions, because a count is a fact about the tree's current shape and would
346    /// quietly mean something else the moment a wrapper was added.
347    ///
348    /// An element that is not one of this one's ancestors holds nothing, so it falls back to
349    /// [`Region`](Escape::Region) rather than escaping further than it was told to.
350    Within(Leaf),
351}
352
353/// The element sits over its region rather than in it.
354///
355/// The trunk decides what takes an element down, what it stacks among, and what clips it; an
356/// [`anchor`](crate::Place::anchored) decides where it sits. That split is what lets an element be
357/// positioned by one element while living under another -- and it is also what leaves an element
358/// positioned *outside* its trunk cut off at the trunk's edge, which is the whole point of having
359/// put it out there.
360///
361/// This is how an element says it is not part of the region it is grown in. Two consequences, and
362/// they are one question asked twice, so they get one answer:
363///
364/// - it is **not clipped** by that region -- [`Escape`] says whether anything further out still is
365/// - it **contributes nothing** to that region's extent, because it is not content
366///
367/// The extent half takes no [`Escape`], and needs none: a region contributes its own box and never
368/// its content to whatever contains it, so there is no second region for an overlay to be excluded
369/// from. The choice is about clipping alone.
370///
371/// What it keeps is the region's offset: it travels with the content it is anchored to, which is
372/// what makes a menu follow the row that opened it. That is the whole difference from
373/// [`pinned`](crate::Place::pinned), which keeps the clip and escapes the movement.
374///
375/// Both are relative to the **nearest** scrolling ancestor: the region the element is grown in is
376/// the one it is not part of.
377#[derive(Component, Copy, Clone, Debug, PartialEq, Eq)]
378pub(crate) struct Floats(pub(crate) Escape);
379
380/// The element does not travel with its region's content.
381///
382/// *Moving with the content* and *counting toward extent* are the same question, so they get one
383/// answer: a pinned element receives no offset from its nearest scrolling ancestor in R4 and
384/// contributes nothing to that region's extent in R3. Two separate flags could drift out of
385/// agreement with each other; one cannot.
386#[derive(Component, Copy, Clone, Debug, Default, PartialEq, Eq)]
387pub(crate) struct Pinned;
388
389/// The box a scrolling ancestor leaves visible: the intersection of every one of them.
390///
391/// A rect, and only a rect. Whether an element is *culled* is a decision extraction makes by
392/// comparing this against the element's box, and is never recorded on the element -- which is what
393/// keeps culling out of anything that reads state, extent first among them.
394#[derive(Component, Copy, Clone, Debug, PartialEq)]
395pub(crate) struct Clipped(pub(crate) Section);
396
397impl Clipped {
398    /// Nothing clipped: what an element with no scrolling ancestor carries.
399    pub(crate) fn unbounded() -> Self {
400        Self(Section::from_edges(
401            f32::MIN / 2.0,
402            f32::MIN / 2.0,
403            f32::MAX / 2.0,
404            f32::MAX / 2.0,
405        ))
406    }
407}

415/// Every region still moving from a release, and how fast.
416///
417/// Held beside the tree rather than on the elements, for the same reason the running tweens are:
418/// what is coasting is a small set and almost never the tree, so the loop can ask whether anything
419/// is (F9) without a pass over anything.
420///
421/// Keyed per axis, because the two axes reach their ends independently and each hands outward on
422/// its own.
423#[derive(Default)]
424pub(crate) struct Coasting {
425    running: HashMap<(Leaf, Axis), Coast>,
426}
427
428/// One region still moving.
429#[derive(Copy, Clone, Debug)]
430struct Coast {
431    /// Its speed, in logical pixels of *offset* per second.
432    velocity: f32,
433    /// Whether a frame's delta has been charged to it yet.
434    ///
435    /// A coast begins at the dispatch of the frame that released it, and that frame's delta is the
436    /// interval *ending* there -- the interval the drag was still being made in, and which the drag
437    /// has already been paid for. Charging it as well would move the region twice for one stretch
438    /// of time. The same rule a tween created at the drain follows, for the same reason.
439    charged: bool,
440}
441
442impl Coasting {
443    /// Sets a region coasting, replacing whatever it was doing on that axis.
444    fn start(&mut self, leaf: Leaf, axis: Axis, velocity: f32) {
445        self.running.insert(
446            (leaf, axis),
447            Coast {
448                velocity,
449                charged: false,
450            },
451        );
452    }
453
454    /// Puts one back, as it now stands.
455    fn resume(&mut self, leaf: Leaf, axis: Axis, coast: Coast) {
456        self.running.insert((leaf, axis), coast);
457    }
458
459    /// Stops a region on one axis, and says whether it was moving.
460    pub(crate) fn halt(&mut self, leaf: Leaf, axis: Axis) -> bool {
461        self.running.remove(&(leaf, axis)).is_some()
462    }
463
464    /// Stops a region on both axes, which is what taking hold of it does, and says whether it
465    /// caught anything -- a hand that stopped something has spent its press on stopping it.
466    pub(crate) fn stop(&mut self, leaf: Leaf) -> bool {
467        Axis::BOTH
468            .into_iter()
469            .fold(false, |caught, axis| self.halt(leaf, axis) || caught)
470    }
471
472    /// Drops every coast on elements that have gone.
473    pub(crate) fn wither(&mut self, gone: &[Leaf]) {
474        if self.running.is_empty() {
475            return;
476        }
477        for leaf in gone {
478            self.stop(*leaf);
479        }
480    }
481
482    /// Whether nothing is coasting, which is one of the clauses in whether the loop owes a frame.
483    pub(crate) fn idle(&self) -> bool {
484        self.running.is_empty()
485    }
486
487    /// How many coasts are running. Reported to the trace.
488    pub(crate) fn len(&self) -> usize {
489        self.running.len()
490    }
491
492    /// Takes every running coast, leaving nothing. Whatever is still moving is put back.
493    fn take(&mut self) -> Vec<(Leaf, Axis, Coast)> {
494        self.running
495            .drain()
496            .map(|((leaf, axis), coast)| (leaf, axis, coast))
497            .collect()
498    }
499}

501/// How far a region can be moved on one axis: what its content reaches, less the box it is seen
502/// through.
503pub(crate) fn range(extent: Area, own: Area, axis: Axis) -> f32 {
504    (extent.along(axis) - own.along(axis)).max(0.0)
505}
506
507/// How much of `delta` a region sitting at `offset` can still take.
508///
509/// The whole of what "can no longer consume" means, and the reason it needs no flag: a region at its
510/// end returns zero for a delta going further out and the full delta for one coming back, so the
511/// answer is about where the region currently sits rather than about what it was told at spawn.
512pub(crate) fn consumable(offset: f32, range: f32, delta: f32) -> f32 {
513    (offset + delta).clamp(0.0, range) - offset
514}

516/// How far a coast travelling at `velocity` gets in `elapsed` seconds, and the speed it leaves,
517/// given a `half_life` in seconds.
518///
519/// The decay is continuous, so the distance covered is the integral of the speed rather than one
520/// frame's rectangle. That is what makes a fling reach the same place at thirty frames a second as
521/// at a hundred and twenty, which a per-frame multiply does not.
522pub(crate) fn coasted(velocity: f32, half_life: f32, elapsed: f32) -> (f32, f32) {
523    // No half-life at all is a coast that stops dead. No *time* is not: a frame that took no time
524    // costs a coast neither distance nor speed, or a run of them would settle one that never moved.
525    if half_life <= 0.0 {
526        return (0.0, 0.0);
527    }
528    if elapsed <= 0.0 {
529        return (0.0, velocity);
530    }
531    let remaining = 0.5f32.powf(elapsed / half_life);
532    let travelled = velocity * half_life * (1.0 - remaining) / core::f32::consts::LN_2;
533    (travelled, velocity * remaining)
534}

536/// How far through its range a region sits on one axis, as the `0.0..=1.0` a scrollbar reads.
537///
538/// Derived rather than held. A region with nowhere to go reads zero, which is the honest answer:
539/// there is no progress through a range of nothing.
540pub(crate) fn progress(offset: Position, extent: Area, own: Area, axis: Axis) -> f32 {
541    let range = range(extent, own, axis);
542    match range > 0.0 {
543        true => (offset.along(axis) / range).clamp(0.0, 1.0),
544        false => 0.0,
545    }
546}

548/// Hands a coast its initial velocity, in logical pixels of offset per second.
549///
550/// The whole of what a gesture has to do with one: everything after this -- the decay, the clamp,
551/// and whether an end chains outward or absorbs -- is the region's.
552pub(crate) fn launch(grove: &mut Grove, leaf: Leaf, axis: Axis, velocity: f32) {
553    if grove.momentum.half_life.is_zero() || velocity.abs() < grove.momentum.minimum {
554        return;
555    }
556    debug!(leaf = leaf.id(), ?axis, velocity, "coasting");
557    grove.coasting.start(leaf, axis, velocity);
558}

560/// The offset each region is asked for this frame, before R4 clamps it and carries it down.
561///
562/// Three ways to ask, all of them writes to one value: a coast still running from a release, a
563/// [`scroll`](crate::Grow::scroll) written this frame, and a
564/// [`Motion::Scroll`](crate::Motion::Scroll) part way through. They never contend, because each of
565/// them ends the others where it starts (F8).
566///
567/// All three are answered here rather than where they were written, because all three need the
568/// extent -- and the extent is R3's, one pass earlier in this same frame. Asking to scroll to the
569/// end therefore lands at the end of the content as it is *now*, in the frame the content changed,
570/// with no settling frame.
571pub(crate) fn asked(grove: &mut Grove, boxes: &HashMap<Leaf, Section>) {
572    coast(grove, boxes);
573    sought(grove, boxes);
574    animated(grove, boxes);
575}

577/// Every running coast, advanced by this frame's share of its decay.
578fn coast(grove: &mut Grove, boxes: &HashMap<Leaf, Section>) {
579    if grove.coasting.idle() {
580        return;
581    }
582    let elapsed = grove.clock.delta().as_secs_f32();
583    let momentum = grove.momentum;
584    for (leaf, axis, mut coast) in grove.coasting.take() {
585        if !coast.charged {
586            coast.charged = true;
587            grove.coasting.resume(leaf, axis, coast);
588            continue;
589        }
590        let section = boxes.get(&leaf).copied().unwrap_or_default();
591        let span = range(grove.tree.extent(leaf), section.area, axis);
592        let offset = grove.tree.offset(leaf);
593        let (travelled, speed) = coasted(coast.velocity, momentum.half_life.as_secs_f32(), elapsed);
594        let at = offset.along(axis);
595        let wanted = at + travelled;
596        let landed = wanted.clamp(0.0, span);
597        if landed != at {
598            grove.tree.set_offset(leaf, offset.set(axis, landed));
599        }
600        // An end reached while coasting chains outward or absorbs exactly as a drag would, because
601        // it is the same question asked of the same region at the same place in its own extent.
602        if landed != wanted {
603            match handed(grove, leaf, axis) {
604                Some(next) => {
605                    debug!(from = leaf.id(), to = next.id(), "coast handed outward");
606                    // Already mid-flight: this frame's time is spent, and the region it came from
607                    // is what spent it.
608                    grove.coasting.resume(
609                        next,
610                        axis,
611                        Coast {
612                            velocity: speed,
613                            charged: true,
614                        },
615                    );
616                }
617                None => debug!(leaf = leaf.id(), "coast absorbed"),
618            }
619            continue;
620        }
621        if speed.abs() < momentum.minimum {
622            debug!(leaf = leaf.id(), "coast settled");
623            continue;
624        }
625        grove.coasting.resume(
626            leaf,
627            axis,
628            Coast {
629                velocity: speed,
630                charged: true,
631            },
632        );
633    }
634}
635
636/// The region a coast passes to when `leaf` can move no further, or `None` where it absorbs.
637fn handed(grove: &Grove, leaf: Leaf, axis: Axis) -> Option<Leaf> {
638    let chain = interaction::chain(grove, leaf);
639    // A coasting region scrolls, so it is the innermost link of its own chain -- unless it stopped
640    // being one under the coast, in which case there is nothing to hand anything to.
641    if chain.first() != Some(&leaf) {
642        return None;
643    }
644    interaction::outward(grove, &chain, 0, axis).map(|index| chain[index])
645}

647/// The one-shot destinations written this frame.
648fn sought(grove: &mut Grove, boxes: &HashMap<Leaf, Section>) {
649    for (leaf, to) in core::mem::take(&mut grove.sought) {
650        if let Some(landed) = destination(grove, boxes, leaf, to) {
651            grove.tree.set_offset(leaf, landed);
652            debug!(leaf = leaf.id(), x = landed.x, y = landed.y, "scrolled");
653        }
654    }
655}
656
657/// The destinations a motion is part way to.
658///
659/// Both ends are answered every frame, in this frame's context: the target re-resolves against the
660/// extent R3 just measured, so a motion toward the end of a list that grew under it still lands on
661/// the end. What it left is a number of pixels and is carried as one, because the offset is a
662/// resolved value rather than a declaration that could be resolved again.
663fn animated(grove: &mut Grove, boxes: &HashMap<Leaf, Section>) {
664    for (leaf, from, to, at) in grove.aspen.scrolling() {
665        let Some(target) = destination(grove, boxes, leaf, to) else {
666            continue;
667        };
668        grove.tree.set_offset(
669            leaf,
670            Position::new(blend(from.x, target.x, at), blend(from.y, target.y, at)),
671        );
672    }
673}

675/// Where `to` puts `leaf`, on each axis the destination speaks for.
676///
677/// `None` where there is no such axis, which is the same refusal the drain already named, read one
678/// pass later.
679fn destination(
680    grove: &Grove,
681    boxes: &HashMap<Leaf, Section>,
682    leaf: Leaf,
683    to: ScrollTo,
684) -> Option<Position> {
685    let moves = to.over(grove.tree.scrolls(leaf)?)?;
686    let section = boxes.get(&leaf).copied().unwrap_or_default();
687    let extent = grove.tree.extent(leaf);
688    let offset = grove.tree.offset(leaf);
689    let shown = to
690        .names()
691        .and_then(|named| boxes.get(&named).copied())
692        .unwrap_or_default();
693    let mut landed = offset;
694    for axis in Axis::BOTH {
695        if !moves.covers(axis) {
696            continue;
697        }
698        let span = range(extent, section.area, axis);
699        landed = landed.set(
700            axis,
701            to.landing(axis, offset.along(axis), span, section, shown),
702        );
703    }
704    Some(landed)
705}
