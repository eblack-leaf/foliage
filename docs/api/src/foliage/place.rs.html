16/// Where the caller was standing. Carried from the call that wrote a placement to the drain that
17/// applies it, so a refusal names the write rather than the pass that noticed it.
18pub(crate) type Caller = &'static core::panic::Location<'static>;

20/// What a seed carries about where it will sit. Anything left unsaid takes its default.
21///
22/// [`location`](Placement::location) and [`traced`](Placement::traced) are the two ways to say it,
23/// and an element states one of them: a box is a rectangle the grammar resolves, and a trace is the
24/// same grammar read as vertices. Which of the two a seed offers is a type -- [`Boxed`] against
25/// [`Line::between`](crate::Line::between) -- so nothing can state both and no pass has to decide
26/// which was meant.
27#[derive(Clone, Debug, Default)]
28pub(crate) struct Placement {
29    pub(crate) location: Option<Location>,
30    /// The two ends of a stroked element, in place of a box.
31    pub(crate) traced: Option<Traced>,
32    /// How thick a stroked element is drawn. Placement input rather than decoration: it is what a
33    /// trace's box is inflated by, so a rule with two ends on one line still has a box.
34    pub(crate) stroke: Option<Stroke>,
35    pub(crate) grid: Option<Grid>,
36    pub(crate) anchor: Option<Anchored>,
37    pub(crate) elevation: Option<Elevation>,
38    /// Absent until the element names a font or a size, because an element with neither has no
39    /// character cell and reads zero for everything measured in one.
40    pub(crate) typeface: Option<Typeface>,
41    pub(crate) manner: Manner,
42}
43
44/// What a seed carries about how it behaves: what it does with a gesture, whether it scrolls, and
45/// which of the ways to be off it starts in.
46///
47/// Everything here is one element's own declaration. None of it predicts what a gesture will turn
48/// out to mean, and none of it is derived from anything else -- which is the whole test for
49/// belonging here rather than being worked out at the time.
50#[derive(Clone, Debug, Default)]
51pub(crate) struct Manner {
52    pub(crate) gestures: Gestures,
53    pub(crate) scrolls: Option<Scrolls>,
54    /// Declared [`pinned`](Place::pinned): does not travel with its region's content.
55    pub(crate) pinned: bool,
56    /// Declared [`floats`](Place::floats): sits over its region rather than in it, and how far out
57    /// of the regions above it that reaches.
58    pub(crate) floats: Option<Escape>,
59    pub(crate) focusing: Focusing,
60    pub(crate) visible: Visible,
61    pub(crate) opacity: Opacity,
62}

85/// Stating where an element will sit, and how it behaves.
86///
87/// Implemented by every [`Seed`](crate::Seed), so this reads identically whatever is being grown.
88/// Sealed: it can be called, never implemented.
89#[allow(private_bounds)]
90pub trait Place: Places + Sized {
91    /// How this element's box is divided for the elements grown under it.
92    ///
93    /// Undeclared, it is a single column and a single row.
94    fn grid(mut self, grid: Grid) -> Self {
95        self.placement().grid = Some(grid);
96        self
97    }
98
99    /// How far in front of its trunk the element sits, from [`Elevation::up`] or
100    /// [`Elevation::down`].
101    ///
102    /// Undeclared, it sits at its trunk's own elevation, which leaves it just in front of it.
103    ///
104    /// On every element, not only on one that draws: what is grown under an element accumulates
105    /// from its elevation, so a wrapper that carried none would flatten the subtree beneath it.
106    fn elevate(mut self, elevation: Elevation) -> Self {
107        self.placement().elevation = Some(elevation);
108        self
109    }
110
111    /// The one other element this one's placement may read through [`anchor()`](crate::anchor).
112    ///
113    /// It carries every reading a trunk does, so an element grown away from what it describes --
114    /// to clear a stack, or a clip -- goes on addressing that element's grid, box, font and measure
115    /// in the same words.
116    ///
117    /// # Panics
118    ///
119    /// If the anchor would close a cycle. A ↔ B is a contradiction rather than an ordering problem:
120    /// an element whose placement cannot resolve has no box, so there is no state for it to fall
121    /// back to. It is refused when the op is applied, naming both elements and the write that did
122    /// it, which leaves the tree acyclic by construction.
123    #[track_caller]
124    fn anchored(mut self, to: Leaf) -> Self {
125        self.placement().anchor = Some(Anchored {
126            to,
127            at: core::panic::Location::caller(),
128        });
129        self
130    }
131
132    /// Which registered font the element composes in.
133    ///
134    /// On every element, not only on one that draws glyphs: a font and a size are what give a
135    /// character cell its size, and a cell is what [`letters`](crate::Source::letters) and a
136    /// letter-pitched track are measured in. An element that names neither has no cell and reads
137    /// zero for both.
138    ///
139    /// Undeclared, it is the bundled font.
140    fn font(mut self, font: Font) -> Self {
141        self.placement().typeface.get_or_insert_default().font = font;
142        self
143    }
144
145    /// How large the element's characters are, per breakpoint.
146    ///
147    /// Undeclared, it is [`FontSize::DEFAULT`].
148    fn font_size(mut self, size: FontSize) -> Self {
149        self.placement().typeface.get_or_insert_default().size = size;
150        self
151    }
152
153    /// The element receives gestures.
154    ///
155    /// A gesture goes to the top of the box stack whatever is there; this decides whether the
156    /// element at the top does anything with it. An element at the top that did not say this
157    /// **eats** the gesture, which is what a backdrop, a sheet backing and a menu's padding are,
158    /// and none of them declares anything to be it.
159    ///
160    /// It is also what makes an element reachable by focus, because the set that asked to receive
161    /// input is the set a keyboard should be able to reach.
162    fn interactive(mut self) -> Self {
163        self.placement().manner.gestures.receives = true;
164        self
165    }
166
167    /// A gesture over the element goes to whatever is beneath it.
168    ///
169    /// What a composite marks its own decoration with. A label drawn over a button, a highlight, a
170    /// gradient across a card -- each of them is above its target at those pixels, and each would
171    /// otherwise be what a press lands on.
172    ///
173    /// It does not take the element out of the box stack, only out of what may be the top of it:
174    /// the drag that follows a press still finds the region containing it, so scrolling works over
175    /// decoration exactly as it does over anything else.
176    ///
177    /// Not what a backdrop, a sheet backing or a menu's padding is. Those are solid and eat the
178    /// gesture, which is what an element at the top that did not declare
179    /// [`interactive`](Place::interactive) already does.
180    fn intangible(mut self) -> Self {
181        self.placement().manner.gestures.intangible = true;
182        self
183    }
184
185    /// Which drags this element takes.
186    ///
187    /// The one thing about a gesture the engine cannot work out for itself: a slider taking drags
188    /// along its own axis is information only the app has.
189    ///
190    /// Undeclared, an element **takes no drags** -- so it holds a gesture only until that gesture
191    /// becomes a drag and then yields, which is what makes a button inside a scrolling list behave
192    /// on touch. Press it and it holds; drag and the list scrolls; release without moving and it
193    /// gets a tap.
194    fn drags(mut self, axes: Axes) -> Self {
195        self.placement().manner.gestures.drags = Some(axes);
196        self
197    }
198
199    /// Hits are tested against the ellipse inscribed in the element's box rather than the box.
200    ///
201    /// For a round control, so it does not take presses in the square corners it does not draw.
202    fn round_hit_area(mut self) -> Self {
203        self.placement().manner.gestures.shape = Shape::Round;
204        self
205    }
206
207    /// The element scrolls, on the axes named.
208    ///
209    /// Dividing an element's box with a [`grid`](Place::grid) says nothing about scrolling: an
210    /// element scrolls because it said so, and for no other reason. An axis that was not named does
211    /// not scroll and has no extent -- it is not a scrolling axis with a range of zero.
212    ///
213    /// A drag anywhere inside it scrolls it, whether or not what the drag landed on is a target,
214    /// and a region that reaches its end hands the drag outward to the next one containing it --
215    /// unless it said it [`contain`](Scroll::contain)s that axis, which is the one knob.
216    ///
217    /// ```no_run
218    /// # use foliage::{Axes, Place, Scroll, Stem};
219    /// Stem::new().scrolls(Axes::Vertical);
220    /// Stem::new().scrolls(Scroll::new(Axes::Both).contain(Axes::Vertical));
221    /// ```
222    fn scrolls(mut self, scroll: impl Into<Scroll>) -> Self {
223        self.placement().manner.scrolls = Some(Scrolls(scroll.into()));
224        self
225    }
226
227    /// The element does not travel with its region's content.
228    ///
229    /// A header that stays at the top while content slides under it, a button pinned to a corner.
230    /// *Moving with the content* and *counting toward extent* are the same question, so this is one
231    /// declaration rather than two that could drift out of agreement: a pinned element receives no
232    /// offset from its nearest scrolling ancestor, and contributes nothing to that region's extent.
233    ///
234    /// It keeps its place in the tree, and with it the clipping, the opacity product and the
235    /// disable cascade that parenting it outside the view and anchoring back would have cost.
236    fn pinned(mut self) -> Self {
237        self.placement().manner.pinned = true;
238        self
239    }
240
241    /// The element sits over the region it is grown in rather than in it.
242    ///
243    /// A menu that opens past the edge of the list it belongs to, a tooltip beside a row. What an
244    /// [`anchor`](Place::anchored) buys is a position taken from one element while living under
245    /// another; without this, an element positioned *outside* its trunk is still cut off at the
246    /// trunk's edge, which is the whole point of having put it out there.
247    ///
248    /// Two consequences, from one declaration because they are one question: it is **not clipped**
249    /// by that region, and it **contributes nothing** to that region's extent. An overlay is not
250    /// content, so it neither hides behind the edge nor invents room to scroll to.
251    ///
252    /// It still travels with the region's content, which is what keeps a menu against the row that
253    /// opened it. That is the whole difference from [`pinned`](Place::pinned): this one escapes the
254    /// clip and keeps the movement, and that one escapes the movement and keeps the clip.
255    ///
256    /// How far the clip escape reaches is [`Escape`]'s, and is stated rather than defaulted because
257    /// no one answer is right everywhere. The extent half takes no such argument and needs none: a
258    /// region contributes its own box and never its content to whatever contains it, so there is no
259    /// second region for an overlay to be excluded from.
260    ///
261    /// ```no_run
262    /// # use foliage::{Escape, Leaf, Panel, Place};
263    /// # fn f(sheet: Leaf) {
264    /// Panel::new().floats(Escape::Region);        // out of the list it is in
265    /// Panel::new().floats(Escape::Surface);       // out of everything
266    /// Panel::new().floats(Escape::Within(sheet)); // out of everything up to the sheet
267    /// # }
268    /// ```
269    fn floats(mut self, escape: Escape) -> Self {
270        self.placement().manner.floats = Some(escape);
271        self
272    }
273
274    /// Where the element sits in focus order, relative to the elements around it.
275    ///
276    /// Focus order is reading order, derived. This pulls one element earlier (negative) or later
277    /// (positive) where a layout's meaning differs from its geometry; everything sharing a value
278    /// keeps reading order among themselves, so stating one moves one element and renumbers
279    /// nothing.
280    fn focus_order(mut self, order: i32) -> Self {
281        self.placement().manner.focusing.order = order;
282        self
283    }
284
285    /// Focus cycles inside this element while it is in there.
286    ///
287    /// What a drawer or a dialog declares. Without it, stepping through an overlay walks off into
288    /// the page behind it.
289    fn focus_scope(mut self) -> Self {
290        self.placement().manner.focusing.scope = true;
291        self
292    }
293
294    /// Whether the element is drawn at all.
295    ///
296    /// The real hide: skipped by drawing, out of the box stack, and contributing nothing to a
297    /// containing region's extent, while keeping its state and its [`Leaf`]. App intent only --
298    /// content scrolled out of sight is not hidden, and still counts.
299    fn visible(mut self, visible: bool) -> Self {
300        self.placement().manner.visible = Visible(visible);
301        self
302    }
303
304    /// How opaque the element is, in `0.0..=1.0`, multiplied through everything grown under it.
305    ///
306    /// Fully transparent is **not there**: it is out of the box stack and receives nothing, which
307    /// closes the case of an element faded out that went on taking presses. Anything above zero is
308    /// there and takes them normally.
309    fn opacity(mut self, opacity: f32) -> Self {
310        self.placement().manner.opacity = Opacity::new(opacity);
311        self
312    }
313}

317/// A seed placed by a box.
318///
319/// Everything the engine draws as a rectangle -- a [`Stem`](crate::Stem), a
320/// [`Panel`](crate::Panel), a [`Text`](crate::Text), an [`Icon`](crate::Icon), an
321/// [`Image`](crate::Image), a [`Polygon`](crate::Polygon) -- and nothing else.
322///
323/// It is a separate trait from [`Place`] rather than a method on it because a
324/// [`Line`](crate::Line) has no box to state: it is two ends, said with
325/// [`between`](crate::Line::between). An `at` it could be handed and would ignore is the kind of
326/// surface that has to be remembered rather than read, so it is not there to hand it.
327///
328/// Sealed: it can be called, never implemented.
329#[allow(private_bounds)]
330pub trait Boxed: Places + Sized {
331    /// Where the element sits.
332    ///
333    /// An element that says nothing fills its parent.
334    fn at(mut self, location: Location) -> Self {
335        self.placement().location = Some(location);
336        self
337    }
