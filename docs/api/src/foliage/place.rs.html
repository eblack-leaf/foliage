15/// Where the caller was standing. Carried from the call that wrote a placement to the drain that
16/// applies it, so a refusal names the write rather than the pass that noticed it.
17pub(crate) type Caller = &'static core::panic::Location<'static>;

19/// What a seed carries about where it will sit. Anything left unsaid takes its default.
20///
21/// [`location`](Placement::location) and [`traced`](Placement::traced) are the two ways to say it,
22/// and an element states one of them: a box is a rectangle the grammar resolves, and a trace is the
23/// same grammar read as vertices. Which of the two a seed offers is a type -- [`Boxed`] against
24/// [`Line::trace`](crate::Line::trace) -- so nothing can state both and no pass has to decide
25/// which was meant.
26#[derive(Clone, Debug, Default)]
27pub(crate) struct Placement {
28    pub(crate) location: Option<Location>,
29    /// The two ends of a stroked element, per breakpoint, in place of a box.
30    pub(crate) traced: Option<Trace>,
31    /// How thick a stroked element is drawn. Placement input rather than decoration: it is what a
32    /// trace's box is inflated by, so a rule with two ends on one line still has a box.
33    pub(crate) stroke: Option<Stroke>,
34    pub(crate) grid: Option<Grid>,
35    pub(crate) anchor: Option<Anchored>,
36    pub(crate) elevation: Option<Elevation>,
37    /// Absent until the element names a font or a size, because an element with neither has no
38    /// character cell and reads zero for everything measured in one.
39    pub(crate) typeface: Option<Typeface>,
40    pub(crate) manner: Manner,
41    /// How it is turned, scaled and slid where it is drawn.
42    pub(crate) pose: crate::pose::Pose,
43}
44
45impl Placement {
46    /// What a mark starts from: nothing stated, and a gesture over it passing through.
47    pub(crate) fn mark() -> Self {
48        Self {
49            manner: Manner {
50                gestures: Gestures::mark(),
51                ..Manner::default()
52            },
53            ..Self::default()
54        }
55    }
56}

58/// What a seed carries about how it behaves: what it does with a gesture, whether it scrolls, and
59/// which of the ways to be off it starts in.
60///
61/// Everything here is one element's own declaration. None of it predicts what a gesture will turn
62/// out to mean, and none of it is derived from anything else -- which is the whole test for
63/// belonging here rather than being worked out at the time.
64#[derive(Clone, Debug, Default)]
65pub(crate) struct Manner {
66    pub(crate) gestures: Gestures,
67    pub(crate) scrolls: Option<Scrolls>,
68    /// Declared [`pinned`](Place::pinned): does not travel with its region's content.
69    pub(crate) pinned: bool,
70    /// Declared [`floats`](Place::floats): sits over its region rather than in it, and how far out
71    /// of the regions above it that reaches.
72    pub(crate) floats: Option<Escape>,
73    pub(crate) focusing: Focusing,
74    pub(crate) visible: Visible,
75    pub(crate) opacity: Opacity,
76    /// Declared [`together`](Place::together): drawn with what is under it as one picture while
77    /// it fades.
78    pub(crate) together: bool,
79}

102/// Stating where an element will sit, and how it behaves.
103///
104/// Implemented by every [`Seed`](crate::Seed), so this reads identically whatever is being grown.
105/// Sealed: it can be called, never implemented.
106#[allow(private_bounds)]
107pub trait Place: Places + Sized {
108    /// How this element's box is divided for the elements grown under it.
109    ///
110    /// Undeclared, it is a single column and a single row.
111    fn grid(mut self, grid: Grid) -> Self {
112        self.placement().grid = Some(grid);
113        self
114    }
115
116    /// How far in front of its trunk the element sits, from [`Elevation::up`] or
117    /// [`Elevation::down`].
118    ///
119    /// Undeclared, it sits at its trunk's own elevation, which leaves it just in front of it.
120    ///
121    /// On every element, not only on one that draws: what is grown under an element accumulates
122    /// from its elevation, so a wrapper that carried none would flatten the subtree beneath it.
123    fn elevate(mut self, elevation: Elevation) -> Self {
124        self.placement().elevation = Some(elevation);
125        self
126    }
127
128    /// The one other element this one's placement may read through [`anchor()`](crate::anchor).
129    ///
130    /// It carries every reading a trunk does, so an element grown away from what it describes --
131    /// to clear a stack, or a clip -- goes on addressing that element's grid, box, font and measure
132    /// in the same words.
133    ///
134    /// # Panics
135    ///
136    /// If the anchor would close a cycle. A ↔ B is a contradiction rather than an ordering problem:
137    /// an element whose placement cannot resolve has no box, so there is no state for it to fall
138    /// back to. It is refused when the op is applied, naming both elements and the write that did
139    /// it, which leaves the tree acyclic by construction.
140    #[track_caller]
141    fn anchored(mut self, to: Leaf) -> Self {
142        self.placement().anchor = Some(Anchored {
143            to,
144            at: core::panic::Location::caller(),
145        });
146        self
147    }
148
149    /// Which registered font the element composes in.
150    ///
151    /// On every element, not only on one that draws glyphs: a font and a size are what give a
152    /// character cell its size, and a cell is what [`letters`](crate::Source::letters) and a
153    /// letter-pitched track are measured in. An element that names neither has no cell and reads
154    /// zero for both.
155    ///
156    /// Undeclared, it is the bundled font.
157    fn font(mut self, font: Font) -> Self {
158        self.placement().typeface.get_or_insert_default().font = font;
159        self
160    }
161
162    /// How large the element's characters are, per breakpoint.
163    ///
164    /// Undeclared, it is [`FontSize::DEFAULT`].
165    fn font_size(mut self, size: FontSize) -> Self {
166        self.placement().typeface.get_or_insert_default().size = size;
167        self
168    }
169
170    /// The element receives gestures.
171    ///
172    /// A gesture goes to the top of the box stack whatever is there; this decides whether the
173    /// element at the top does anything with it. An element at the top that did not say this
174    /// **eats** the gesture, which is what a backdrop, a sheet backing and a menu's padding are,
175    /// and none of them declares anything to be it.
176    ///
177    /// It is also what makes an element reachable by focus, because the set that asked to receive
178    /// input is the set a keyboard should be able to reach.
179    ///
180    /// On a mark it overrides the passing through a mark starts with. What takes one out of reach
181    /// is [`visible`](Place::visible), [`disable`](crate::Grow::disable),
182    /// [`intangible`](Place::intangible), or a fade to nothing -- on it or on anything it is grown
183    /// under. A press target that shows nothing is a [`Stem`](crate::Stem), or a panel in a clear
184    /// color, at full opacity.
185    fn interactive(mut self) -> Self {
186        self.placement().manner.gestures.receives = true;
187        self
188    }
189
190    /// A gesture over the element goes to whatever is beneath it.
191    ///
192    /// What a composite marks its own decoration with. A highlight, a gradient across a card, the
193    /// ground of a row -- each of them is above its target at those pixels, and each would
194    /// otherwise be what a press lands on. A mark -- [`Text`](crate::Text),
195    /// [`Icon`](crate::Icon), [`Polygon`](crate::Polygon), [`Line`](crate::Line) -- already
196    /// passes gestures through unless it says [`tangible`](Place::tangible), so it needs nothing.
197    ///
198    /// It does not take the element out of the box stack, only out of what may be the top of it:
199    /// the drag that follows a press still finds the region containing it, so scrolling works over
200    /// decoration exactly as it does over anything else.
201    ///
202    /// Not what a backdrop, a sheet backing or a menu's padding is. Those are solid and eat the
203    /// gesture, which is what an element at the top that did not declare
204    /// [`interactive`](Place::interactive) already does.
205    fn intangible(mut self) -> Self {
206        self.placement().manner.gestures.intangible = true;
207        self
208    }
209
210    /// A gesture over the element stops on it: the opposite of
211    /// [`intangible`](Place::intangible).
212    ///
213    /// What every surface -- a panel, an image, a stem -- already is, and what a mark is not until
214    /// it says so. For a mark that should block what is beneath it without receiving anything
215    /// itself; one that receives says [`interactive`](Place::interactive), which is tangible
216    /// already.
217    fn tangible(mut self) -> Self {
218        let gestures = &mut self.placement().manner.gestures;
219        gestures.intangible = false;
220        gestures.passes = false;
221        self
222    }
223
224    /// Which drags this element takes.
225    ///
226    /// The one thing about a gesture the engine cannot work out for itself: a slider taking drags
227    /// along its own axis is information only the app has.
228    ///
229    /// Undeclared, an element **takes no drags** -- so it holds a gesture only until that gesture
230    /// becomes a drag and then yields, which is what makes a button inside a scrolling list behave
231    /// on touch. Press it and it holds; drag and the list scrolls; release without moving and it
232    /// gets a tap.
233    fn drags(mut self, axes: Axes) -> Self {
234        self.placement().manner.gestures.drags = Some(axes);
235        self
236    }
237
238    /// The element takes the wheel.
239    ///
240    /// A notch over it is reported to it as [`wheeled`](crate::Pollen::wheeled) and scrolls
241    /// nothing, as a drag it [`drags`](Place::drags) on scrolls nothing. What a notch means is the
242    /// app's: a map zooms on it, a dial turns.
243    ///
244    /// Undeclared, a notch scrolls the region under it. Only an element that is
245    /// [`interactive`](Place::interactive) takes it, as only one of those takes a drag.
246    fn wheels(mut self) -> Self {
247        self.placement().manner.gestures.wheels = true;
248        self
249    }
250
251    /// Hits are tested against the ellipse inscribed in the element's box rather than the box.
252    ///
253    /// For a round control, so it does not take presses in the square corners it does not draw.
254    fn round_hit_area(mut self) -> Self {
255        self.placement().manner.gestures.shape = Shape::Round;
256        self
257    }
258
259    /// The element scrolls, on the axes named.
260    ///
261    /// Dividing an element's box with a [`grid`](Place::grid) says nothing about scrolling: an
262    /// element scrolls because it said so, and for no other reason. An axis that was not named does
263    /// not scroll and has no extent -- it is not a scrolling axis with a range of zero.
264    ///
265    /// A drag anywhere inside it scrolls it, whether or not what the drag landed on is a target,
266    /// and a region that reaches its end hands the drag outward to the next one containing it --
267    /// unless it said it [`contain`](Scroll::contain)s that axis, which is the one knob.
268    ///
269    /// ```no_run
270    /// # use foliage::{Axes, Place, Scroll, Stem};
271    /// Stem::new().scrolls(Axes::Vertical);
272    /// Stem::new().scrolls(Scroll::new(Axes::Both).contain(Axes::Vertical));
273    /// ```
274    fn scrolls(mut self, scroll: impl Into<Scroll>) -> Self {
275        self.placement().manner.scrolls = Some(Scrolls(scroll.into()));
276        self
277    }
278
279    /// The element does not travel with its region's content.
280    ///
281    /// A header that stays at the top while content slides under it, a button pinned to a corner.
282    /// *Moving with the content* and *counting toward extent* are the same question, so this is one
283    /// declaration rather than two that could drift out of agreement: a pinned element receives no
284    /// offset from its nearest scrolling ancestor, and contributes nothing to that region's extent.
285    ///
286    /// It keeps its place in the tree, and with it the clipping, the opacity product and the
287    /// disable cascade that parenting it outside the view and anchoring back would have cost.
288    fn pinned(mut self) -> Self {
289        self.placement().manner.pinned = true;
290        self
291    }
292
293    /// The element sits over the region it is grown in rather than in it.
294    ///
295    /// A menu that opens past the edge of the list it belongs to, a tooltip beside a row. What an
296    /// [`anchor`](Place::anchored) buys is a position taken from one element while living under
297    /// another; without this, an element positioned *outside* its trunk is still cut off at the
298    /// trunk's edge, which is the whole point of having put it out there.
299    ///
300    /// Two consequences, from one declaration because they are one question: it is **not clipped**
301    /// by that region, and it **contributes nothing** to that region's extent. An overlay is not
302    /// content, so it neither hides behind the edge nor invents room to scroll to.
303    ///
304    /// It still travels with the region's content, which is what keeps a menu against the row that
305    /// opened it. That is the whole difference from [`pinned`](Place::pinned): this one escapes the
306    /// clip and keeps the movement, and that one escapes the movement and keeps the clip.
307    ///
308    /// How far the clip escape reaches is [`Escape`]'s, and is stated rather than defaulted because
309    /// no one answer is right everywhere. The extent half takes no such argument and needs none: a
310    /// region contributes its own box and never its content to whatever contains it, so there is no
311    /// second region for an overlay to be excluded from.
312    ///
313    /// ```no_run
314    /// # use foliage::{Escape, Leaf, Panel, Place};
315    /// # fn f(sheet: Leaf) {
316    /// Panel::new().floats(Escape::Region);        // out of the list it is in
317    /// Panel::new().floats(Escape::Surface);       // out of everything
318    /// Panel::new().floats(Escape::Within(sheet)); // out of everything up to the sheet
319    /// # }
320    /// ```
321    fn floats(mut self, escape: Escape) -> Self {
322        self.placement().manner.floats = Some(escape);
323        self
324    }
325
326    /// Where the element sits in focus order, relative to the elements around it.
327    ///
328    /// Focus order is reading order, derived. This pulls one element earlier (negative) or later
329    /// (positive) where a layout's meaning differs from its geometry; everything sharing a value
330    /// keeps reading order among themselves, so stating one moves one element and renumbers
331    /// nothing.
332    fn focus_order(mut self, order: i32) -> Self {
333        self.placement().manner.focusing.order = order;
334        self
335    }
336
337    /// Focus cycles inside this element while it is in there.
338    ///
339    /// What a drawer or a dialog declares. Without it, stepping through an overlay walks off into
340    /// the page behind it.
341    fn focus_scope(mut self) -> Self {
342        self.placement().manner.focusing.scope = true;
343        self
344    }
345
346    /// Whether the element is drawn at all.
347    ///
348    /// The real hide: skipped by drawing, out of the box stack, and contributing nothing to a
349    /// containing region's extent, while keeping its state and its [`Leaf`]. App intent only --
350    /// content scrolled out of sight is not hidden, and still counts.
351    fn visible(mut self, visible: bool) -> Self {
352        self.placement().manner.visible = Visible(visible);
353        self
354    }
355
356    /// How opaque the element is, in `0.0..=1.0`, multiplied through everything grown under it.
357    ///
358    /// Each element under it is faded on its own, so where two of them overlap, the one behind
359    /// shows through the one in front. Declare [`together`](Place::together) on the element to fade
360    /// it and everything under it as one picture instead.
361    ///
362    /// Fully transparent draws nothing and takes nothing: the element and everything under it
363    /// are out of the box stack, wherever the fade was declared, so a fade to nothing is enough to
364    /// be rid of something. A press target that shows nothing -- the ground of a row that only
365    /// fades in while it is pointed at -- is a [`Stem`](crate::Stem), or a panel in a clear color,
366    /// at full opacity, with what fades in grown under it.
367    fn opacity(mut self, opacity: f32) -> Self {
368        self.placement().manner.opacity = Opacity::new(opacity);
369        self
370    }
371
372    /// Draws the element and everything grown under it as one picture whenever they are drawn at
373    /// less than full opacity, so a fade fades the whole of it at once.
374    ///
375    /// Without it, opacity is applied to each element on its own. A card fading out shows its
376    /// ground through its text, and a button's face through its label. With it, the card is drawn
377    /// first at full strength, and the picture is what fades.
378    ///
379    /// While it fades, it is drawn at its own place in the stack, and nothing grown elsewhere sits
380    /// between its parts. That is how a CSS stacking context behaves. At full opacity nothing is
381    /// different, and it costs nothing. While it fades it costs one picture the size of the window,
382    /// so keep this to what fades as a whole: a card, a sheet, a page coming in. A `together`
383    /// inside another one that is fading is drawn as part of the outer picture.
384    fn together(mut self) -> Self {
385        self.placement().manner.together = true;
386        self
387    }
388
389    /// Turns the element where it is drawn, clockwise by `radians` about its middle -- and
390    /// everything grown under it with it.
391    ///
392    /// Draw-time, as a [`shift`](crate::Grow::shift) is: where the layout put it does not change,
393    /// so nothing placed against it moves and nothing is laid out again. What is drawn and what is
394    /// pressed turn; [`Drawn`](crate::Vein::Drawn) still reads the upright box. Written later with
395    /// [`turn`](crate::Grow::turn), and moved with [`Motion::Turn`](crate::Motion::Turn).
396    fn turn(mut self, radians: f32) -> Self {
397        self.placement().pose.turn = radians;
398        self
399    }
400
401    /// Draws the element `scale` times its size about its middle, and everything grown under it
402    /// with it: a card pressed down to `0.97`, a mark that swells while it is lit.
403    ///
404    /// Draw-time, as [`turn`](Place::turn) is. Written later with [`scale`](crate::Grow::scale), and
405    /// moved with [`Motion::Scale`](crate::Motion::Scale).
406    fn scale(mut self, scale: f32) -> Self {
407        self.placement().pose.scale = scale;
408        self
409    }
410
411    /// Draws the element `by` its own size from where the layout put it, and everything grown under
412    /// it with it: `(1.0, 0.0)` is one of its own widths to the right.
413    ///
414    /// A [`shift`](crate::Grow::shift) measured in the element rather than in pixels, so a sheet
415    /// slides out by exactly its width without asking how wide it came out. The two add together.
416    /// Written later with [`slide`](crate::Grow::slide), and moved with
417    /// [`Motion::Slide`](crate::Motion::Slide).
418    fn slide(mut self, by: crate::Position) -> Self {
419        self.placement().pose.slide = by;
420        self
421    }
422}

426/// A seed placed by a box.
427///
428/// Everything the engine draws as a rectangle -- a [`Stem`](crate::Stem), a
429/// [`Panel`](crate::Panel), a [`Text`](crate::Text), an [`Icon`](crate::Icon), an
430/// [`Image`](crate::Image), a [`Polygon`](crate::Polygon) -- and nothing else.
431///
432/// It is a separate trait from [`Place`] rather than a method on it because a
433/// [`Line`](crate::Line) has no box to state: it is two ends, said with
434/// [`between`](crate::Line::between) or, per breakpoint, with [`trace`](crate::Line::trace). An
435/// `at` it could be handed and would ignore is the kind of surface that has to be remembered rather
436/// than read, so it is not there to hand it.
437///
438/// Sealed: it can be called, never implemented.
439#[allow(private_bounds)]
440pub trait Boxed: Places + Sized {
441    /// Where the element sits.
442    ///
443    /// An element that says nothing fills its parent.
444    fn at(mut self, location: Location) -> Self {
445        self.placement().location = Some(location);
446        self
447    }
