1//! Where bytes come from, and what happens when they arrive.
2//!
3//! Everything foliage is given is bytes: a font, an icon's field, a picture. What is here is the
4//! road from a file or a URL to them, and it is deliberately the *only* difference between the two
5//! cases -- [`font`](crate::Grove::font), [`icon`](crate::Grove::icon) and
6//! [`image`](crate::Grove::image) each take what they are given or where to go and get it, and hand
7//! back the same name either way. An app that fetches writes the same line as one that bundles.
8//!
9//! An arrival is an **op**, pushed onto the one queue from wherever it finished. That is what keeps
10//! it on F1's footing: ordered by when it arrived like every other change, drained where every other
11//! change is drained, and never written into a frame from the side.
12

14use crate::op::Op;

18/// Where bytes an app does not already hold are read from.
19///
20/// A path exists only where there is a filesystem, so naming one on the web is a compile error
21/// rather than a warning at runtime. A URL is nameable everywhere, and answered on the web.
22///
23/// A URL is used exactly as given. Composing an origin or a base is the app's, because where its
24/// assets are hosted is a fact about its deployment and not about the engine.
25#[derive(Clone, Debug)]
26pub enum Origin {
27    /// A file, at a path this program can read.
28    #[cfg(not(target_family = "wasm"))]
29    Path(std::path::PathBuf),
30    /// A URL, as the platform resolves it.
31    Url(String),
32}
33
34impl Origin {
35    /// A file to read.
36    #[cfg(not(target_family = "wasm"))]
37    pub fn path(path: impl Into<std::path::PathBuf>) -> Self {
38        Self::Path(path.into())
39    }
40
41    /// A URL to fetch.
42    ///
43    /// Always fetched on the web, where the browser is the client. Off it, fetched under the
44    /// `origin-url` feature, which brings an http client and a TLS stack with it; **without that
45    /// feature a URL off the web is accepted and reported as
46    /// [`missing`](crate::Pollen::missing)**, so an app that bundles its assets pays for neither.
47    ///
48    /// The name is here on every target either way, so that a program stating its assets' URLs
49    /// states them once.
50    pub fn url(url: impl Into<String>) -> Self {
51        Self::Url(url.into())
52    }
53}

55/// What a registration is given: bytes an app holds, or an [`Origin`] to read them from.
56///
57/// Never named at a callsite -- every verb that takes one takes `impl Into<Bytes>`, so
58/// `include_bytes!(..)` and `Origin::path(..)` are both simply passed.
59#[derive(Clone, Debug)]
60pub struct Bytes(pub(crate) Supply);
61
62#[derive(Clone, Debug)]
63pub(crate) enum Supply {
64    /// Here now. Registered in the call that was handed them.
65    Held(Vec<u8>),
66    /// Somewhere else. Registered in the frame they arrive in.
67    At(Origin),
68}
69
70impl From<Vec<u8>> for Bytes {
71    fn from(bytes: Vec<u8>) -> Self {
72        Self(Supply::Held(bytes))
73    }
74}
75
76impl From<&[u8]> for Bytes {
77    fn from(bytes: &[u8]) -> Self {
78        Self(Supply::Held(bytes.to_vec()))
79    }
80}
81
82impl From<&Vec<u8>> for Bytes {
83    fn from(bytes: &Vec<u8>) -> Self {
84        Self(Supply::Held(bytes.clone()))
85    }
86}
87
88/// What `include_bytes!` produces, which is the bundled case and so the one that has to be free.
89impl<const N: usize> From<&[u8; N]> for Bytes {
90    fn from(bytes: &[u8; N]) -> Self {
91        Self(Supply::Held(bytes.to_vec()))
92    }
93}
94
95impl From<Origin> for Bytes {
96    fn from(origin: Origin) -> Self {
97        Self(Supply::At(origin))
98    }
99}

101/// What a retrieval fills once its bytes land.
102///
103/// Carried on the arrival op, because the name was handed out before anything was read and the
104/// drain has to know what it is filling.
105#[derive(Copy, Clone, Debug)]
106pub(crate) enum Destination {
107    Face(Font),
108    /// A mark, with the two numbers only the app knows about the field it is fetching.
109    Mark(crate::icon::Field, u32, f32),
110    Picture(Plate),
111}
112
113/// Something named before its bytes existed, which can now be asked whether they arrived.
114///
115/// One question over three handles: a [`Font`], a [`Field`](crate::Field) and a [`Plate`] are filled
116/// by the same road, so [`loaded`](crate::Pollen::loaded) and [`missing`](crate::Pollen::missing)
117/// ask about them the same way and an app that waits on several waits on them alike.
118#[derive(Copy, Clone, PartialEq, Eq, Hash, Debug)]
119pub enum Arrival {
120    /// A typeface.
121    Face(Font),
122    /// A mark.
123    Mark(crate::icon::Field),
124    /// A picture.
125    Picture(Plate),
126}
127
128impl From<Font> for Arrival {
129    fn from(font: Font) -> Self {
130        Self::Face(font)
131    }
132}
133
134impl From<crate::icon::Field> for Arrival {
135    fn from(field: crate::icon::Field) -> Self {
136        Self::Mark(field)
137    }
138}
139
140impl From<Plate> for Arrival {
141    fn from(plate: Plate) -> Self {
142        Self::Picture(plate)
143    }
144}
145
146impl From<Destination> for Arrival {
147    fn from(destination: Destination) -> Self {
148        match destination {
149            Destination::Face(font) => Self::Face(font),
150            Destination::Mark(field, _, _) => Self::Mark(field),
151            Destination::Picture(plate) => Self::Picture(plate),
152        }
153    }
154}

156/// What a retrieval finished with: the bytes, or why there are none.
157pub(crate) type Retrieved = Result<Vec<u8>, String>;
158
159/// Starts a retrieval, whichever road the target has.
160///
161/// The name has already been handed to the app by the time this runs, which is the whole point: what
162/// is waited on is nameable before it arrives, so everything that draws it is written now.
163pub(crate) fn retrieve(queue: &Queue, wake: &Wake, destination: Destination, origin: Origin) {
164    let (queue, wake) = (queue.clone(), wake.clone());
165    match origin {
166        #[cfg(not(target_family = "wasm"))]
167        Origin::Path(path) => {
168            // Off the calling thread even for a local file. A read is not instant because it is
169            // local -- a font is megabytes -- and the thread that called this is the one drawing.
170            std::thread::spawn(move || {
171                let bytes = std::fs::read(&path).map_err(|failed| failed.to_string());
172                arrived(&queue, &wake, destination, bytes);
173            });
174        }
175        // On a thread for the reason a path is, and doubly: the client is blocking, and a network
176        // round trip is longer than any read from disk.
177        #[cfg(all(not(target_family = "wasm"), feature = "origin-url"))]
178        Origin::Url(url) => {
179            std::thread::spawn(move || {
180                let bytes = get(&url);
181                arrived(&queue, &wake, destination, bytes);
182            });
183        }
184        // No client was built in. The URL is answered rather than rejected at the callsite, because
185        // which targets fetch is a fact about how the program was built and not about the asset.
186        #[cfg(all(not(target_family = "wasm"), not(feature = "origin-url")))]
187        Origin::Url(url) => {
188            arrived(
189                &queue,
190                &wake,
191                destination,
192                Err(format!("a URL off the web needs `origin-url`: {url}")),
193            );
194        }
195        #[cfg(target_family = "wasm")]
196        Origin::Url(url) => {
197            wasm_bindgen_futures::spawn_local(async move {
198                let bytes = fetch(&url).await;
199                arrived(&queue, &wake, destination, bytes);
200            });
201        }
202    }
203}

205/// Queues an arrival and asks for the frame that will drain it.
206///
207/// The whole of what a retrieval does when it finishes, so every one of them lands the same way
208/// whatever read the bytes.
209fn arrived(queue: &Queue, wake: &Wake, destination: Destination, bytes: Retrieved) {
210    queue.push(Op::Arrived { destination, bytes });
211    wake.rouse();
212}

214/// The largest body a fetch off the web will hold.
215///
216/// A read into a `Vec` needs a ceiling or a server decides how much memory this program uses. The
217/// number is far above what an asset is -- a typeface is megabytes and a picture rarely more -- so
218/// what it stops is a body that was never going to be one of them, reported like any other failure.
219#[cfg(all(not(target_family = "wasm"), feature = "origin-url"))]
220const CEILING: u64 = 64 * 1024 * 1024;
221
222/// The fetch itself, blocking, on the thread [`retrieve`] spawned for it.
223///
224/// A status outside the 2xx range is a failure here without being asked, which is what makes this
225/// the same shape as the web's: the bytes did not arrive, however far the request got.
226#[cfg(all(not(target_family = "wasm"), feature = "origin-url"))]
227fn get(url: &str) -> Retrieved {
228    ureq::get(url)
229        .call()
230        .map_err(|failed| failed.to_string())?
231        .body_mut()
232        .with_config()
233        .limit(CEILING)
234        .read_to_vec()
235        .map_err(|failed| failed.to_string())
236}

238/// The fetch itself, as a sequence of promises.
239///
240/// Every failure along it is one failure to an app -- the bytes did not arrive -- so each is named
241/// for the trace and none is distinguished in what is reported.
242#[cfg(target_family = "wasm")]
243async fn fetch(url: &str) -> Retrieved {
244    use wasm_bindgen::JsCast;
245    use wasm_bindgen_futures::JsFuture;
246
247    let window = web_sys::window().ok_or("no window")?;
248    let response = JsFuture::from(window.fetch_with_str(url))
249        .await
250        .map_err(|_| "the fetch failed".to_string())?
251        .dyn_into::<web_sys::Response>()
252        .map_err(|_| "not a response".to_string())?;
253    if !response.ok() {
254        return Err(format!("answered {}", response.status()));
255    }
256    let body = JsFuture::from(response.array_buffer().map_err(|_| "no body".to_string())?)
257        .await
258        .map_err(|_| "the body failed".to_string())?;
259    Ok(js_sys::Uint8Array::new(&body).to_vec())
260}
