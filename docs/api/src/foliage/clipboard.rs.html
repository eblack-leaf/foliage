1//! The clipboard -- what is put on it, and what comes back off it.
2//!
3//! A write is answered where it is made. A **read is not**: on the web it is a promise, and on a
4//! desktop it is a round trip to whichever program owns the selection. So a read is a request, and
5//! what it finds is an **op** -- pushed onto the one queue from wherever it finished, ordered by
6//! when it arrived like every other change, and drained where every other change is drained. That is
7//! the road [`asset`](crate::asset) already takes, and taking it here is what makes a paste mean the
8//! same thing on both targets rather than landing in this frame on one and the next on the other.
9//!
10//! The platform's own clipboard is opened by `photosynthesize` and left shut under the headless
11//! suite -- the seam [`Wake`] already sits on. A test therefore reads and writes the engine's own
12//! mirror and never touches the clipboard of whoever is running it.
13
14use tracing::debug;
15
16use crate::leaf::Leaf;
17use crate::op::Op;
18use crate::queue::{Queue, Wake};

20/// The system clipboard, and what this program last put on it.
21///
22/// The mirror is not a cache. It is the answer where the platform has none to give -- a desktop with
23/// no display server, a browser that refuses to be read without a gesture it did not get -- so a
24/// copy inside an app still round-trips to a paste inside it, whatever the host allows.
25#[derive(Default)]
26pub(crate) struct Clipboard {
27    /// What this program last wrote. Read back when the platform will not answer.
28    mirror: String,
29    /// The platform's own. Absent until [`attach`](Clipboard::attach), and absent for good under the
30    /// headless suite.
31    system: Option<System>,
32}
33
34/// The platform's own clipboard, once it has been opened.
35///
36/// Native holds its context for the life of the program because on X11 the program that wrote the
37/// selection is the one that serves it -- a context dropped after a write takes what it wrote with
38/// it. The web holds nothing: `navigator.clipboard` is reached by name each time.
39#[cfg(not(target_family = "wasm"))]
40struct System(arboard::Clipboard);
41
42#[cfg(target_family = "wasm")]
43struct System;
44
45impl Clipboard {
46    /// Opens the platform's own clipboard, once, at boot.
47    ///
48    /// A desktop with nothing to open one against is not an error: the mirror answers, and the
49    /// reason is traced once rather than at every copy.
50    pub(crate) fn attach(&mut self) {
51        #[cfg(not(target_family = "wasm"))]
52        match arboard::Clipboard::new() {
53            Ok(clipboard) => self.system = Some(System(clipboard)),
54            Err(reason) => tracing::warn!(%reason, "no system clipboard"),
55        }
56        #[cfg(target_family = "wasm")]
57        {
58            self.system = Some(System);
59        }
60    }
61
62    /// Puts `text` on the clipboard.
63    ///
64    /// The mirror is written whether or not the platform took it, because what an app copied is
65    /// what it can paste back even where the host says no.
66    pub(crate) fn write(&mut self, text: String) {
67        match &mut self.system {
68            #[cfg(not(target_family = "wasm"))]
69            Some(System(clipboard)) => {
70                if let Err(reason) = clipboard.set_text(&text) {
71                    tracing::warn!(%reason, "clipboard write refused");
72                }
73            }
74            #[cfg(target_family = "wasm")]
75            Some(System) => {
76                if let Some(window) = web_sys::window() {
77                    let _ = window.navigator().clipboard().write_text(&text);
78                }
79            }
80            None => {}
81        }
82        debug!(characters = text.chars().count(), "copied");
83        self.mirror = text;
84    }
85
86    /// Asks the platform what is on the clipboard, for whoever wants it.
87    ///
88    /// `into` is the field that asked to be pasted into, or `None` where the app asked for itself.
89    /// It is carried through to the arrival because the name was handed out before anything was
90    /// read, exactly as an asset's [`Destination`](crate::asset::Destination) is.
91    ///
92    /// Whatever the platform says, an answer is always pushed. There is one thing an app or a field
93    /// does about an empty clipboard and about one it was not allowed to read, so the two are one
94    /// outcome and the reason is traced rather than reported.
95    pub(crate) fn read(&mut self, queue: &Queue, wake: &Wake, into: Option<Leaf>) {
96        debug!(into = into.map(|leaf| leaf.id()), "pasting");
97        match &mut self.system {
98            #[cfg(not(target_family = "wasm"))]
99            Some(System(clipboard)) => {
100                // On the frame's own thread, unlike a file, because the context that serves what
101                // this program wrote has to outlive the write and so cannot be moved onto a thread
102                // per read. A selection round trip is bounded and human-triggered; a font is
103                // neither.
104                let text = match clipboard.get_text() {
105                    Ok(text) => text,
106                    Err(reason) => {
107                        debug!(%reason, "clipboard read refused");
108                        self.mirror.clone()
109                    }
110                };
111                answered(queue, wake, into, text);
112            }
113            #[cfg(target_family = "wasm")]
114            Some(System) => {
115                let (queue, wake, mirror) = (queue.clone(), wake.clone(), self.mirror.clone());
116                wasm_bindgen_futures::spawn_local(async move {
117                    let text = read_text().await.unwrap_or(mirror);
118                    answered(&queue, &wake, into, text);
119                });
120            }
121            // Nothing was ever attached, which is the headless suite. The mirror is the whole
122            // clipboard there, and it answers at once so a test reads a paste in the next frame the
123            // way a platform's would.
124            None => answered(queue, wake, into, self.mirror.clone()),
125        }
126    }
127}
128
129/// Queues what the clipboard held and asks for the frame that will drain it.
130///
131/// The whole of what a read does when it finishes, so every one of them lands the same way whatever
132/// answered it.
133fn answered(queue: &Queue, wake: &Wake, into: Option<Leaf>, text: String) {
134    queue.push(Op::Pasted { into, text });
135    wake.rouse();
136}

138/// The read itself, as a promise.
139///
140/// Refused far more often than it is answered -- it is permission-gated, and outside a user gesture
141/// most browsers say no. Each failure is one failure to an app, so none is distinguished.
142#[cfg(target_family = "wasm")]
143async fn read_text() -> Option<String> {
144    use wasm_bindgen_futures::JsFuture;
145
146    let window = web_sys::window()?;
147    let text = JsFuture::from(window.navigator().clipboard().read_text())
148        .await
149        .ok()?;
150    text.as_string()
151}
