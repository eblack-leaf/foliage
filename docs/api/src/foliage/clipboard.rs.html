1//! The clipboard -- what is put on it, and what comes back off it.
2//!
3//! A write is answered where it is made. A **read is not**: on the web it is a promise, and on a
4//! desktop it is a round trip to whichever program owns the selection. So a read is a request, and
5//! what it finds is an **op** -- pushed onto the one queue from wherever it finished, ordered by
6//! when it arrived like every other change, and drained where every other change is drained. That is
7//! the road [`asset`](crate::asset) already takes, and taking it here is what makes a paste mean the
8//! same thing on both targets rather than landing in this frame on one and the next on the other.
9//!
10//! Asking is not the only way text arrives. A host that announces a paste -- a browser, which hands
11//! the text to the gesture that pasted it and will not serve the same read afterwards -- is heard as
12//! [`Input::Pasted`](crate::interaction::input::Input::Pasted) and finishes on the very same op. So
13//! nothing downstream knows which of the two happened, and this module is only ever the asking half.
14//!
15//! The platform's own clipboard is opened by `photosynthesize` and left shut under the headless
16//! suite -- the seam [`Wake`] already sits on. A test therefore reads and writes the engine's own
17//! mirror and never touches the clipboard of whoever is running it.

25/// The system clipboard, and what this program last put on it.
26///
27/// The mirror is not a cache. It is the answer where the platform has none to give -- a desktop with
28/// no display server, a browser that refuses to be read without a gesture it did not get -- so a
29/// copy inside an app still round-trips to a paste inside it, whatever the host allows.
30#[derive(Default)]
31pub(crate) struct Clipboard {
32    /// What this program last wrote. Read back when the platform will not answer.
33    mirror: String,
34    /// The platform's own. Absent until [`attach`](Clipboard::attach), and absent for good under the
35    /// headless suite.
36    ///
37    /// Absent as a *field* on Android, which is the difference between a host that might have one
38    /// and a host that cannot: reaching its `ClipboardManager` means calling into the JVM, which
39    /// nothing else in the engine does. There is nothing to hold, so there is no slot to hold it.
40    ///
41    // TODO: Android's clipboard. `ClipboardManager` is reached from the `AndroidApp` the keyboard
42    // already takes at boot, through its VM and activity pointers -- so the handle is in the
43    // program and what is missing is the JNI to spend it: a `jni` dependency, the unsafe calls
44    // under it, and a slot on this struct that Android is no longer cut out of. Until then the
45    // mirror answers and a copy round-trips inside the app and no further.
46    #[cfg(not(target_os = "android"))]
47    system: Option<System>,
48}
49
50/// The platform's own clipboard, once it has been opened.
51///
52/// Native holds its context for the life of the program because on X11 the program that wrote the
53/// selection is the one that serves it -- a context dropped after a write takes what it wrote with
54/// it. The web holds nothing: `navigator.clipboard` is reached by name each time.
55#[cfg(all(not(target_family = "wasm"), not(target_os = "android")))]
56struct System(arboard::Clipboard);
57
58#[cfg(target_family = "wasm")]
59struct System;
60
61impl Clipboard {
62    /// Opens the platform's own clipboard, once, at boot.
63    ///
64    /// A desktop with nothing to open one against is not an error: the mirror answers, and the
65    /// reason is traced once rather than at every copy. Neither is Android, which never has one.
66    pub(crate) fn attach(&mut self) {
67        #[cfg(all(not(target_family = "wasm"), not(target_os = "android")))]
68        match arboard::Clipboard::new() {
69            Ok(clipboard) => self.system = Some(System(clipboard)),
70            Err(reason) => tracing::warn!(%reason, "no system clipboard"),
71        }
72        #[cfg(target_family = "wasm")]
73        {
74            self.system = Some(System);
75        }
76    }
77
78    /// Puts `text` on the clipboard.
79    ///
80    /// The mirror is written whether or not the platform took it, because what an app copied is
81    /// what it can paste back even where the host says no.
82    pub(crate) fn write(&mut self, text: String) {
83        #[cfg(not(target_os = "android"))]
84        match &mut self.system {
85            #[cfg(all(not(target_family = "wasm"), not(target_os = "android")))]
86            Some(System(clipboard)) => {
87                if let Err(reason) = clipboard.set_text(&text) {
88                    tracing::warn!(%reason, "clipboard write refused");
89                }
90            }
91            #[cfg(target_family = "wasm")]
92            Some(System) => {
93                if let Some(window) = web_sys::window() {
94                    let _ = window.navigator().clipboard().write_text(&text);
95                }
96            }
97            None => {}
98        }
99        debug!(characters = text.chars().count(), "copied");
100        self.mirror = text;
101    }
102
103    /// Asks the platform what is on the clipboard, for whoever wants it.
104    ///
105    /// `into` is the field that asked to be pasted into, or `None` where the app asked for itself.
106    /// It is carried through to the arrival because the name was handed out before anything was
107    /// read, exactly as an asset's [`Destination`](crate::asset::Destination) is.
108    ///
109    /// Whatever the platform says, an answer is always pushed. There is one thing an app or a field
110    /// does about an empty clipboard and about one it was not allowed to read, so the two are one
111    /// outcome and the reason is traced rather than reported.
112    pub(crate) fn read(&mut self, queue: &Queue, wake: &Wake, into: Option<Leaf>) {
113        debug!(into = into.map(|leaf| leaf.id()), "pasting");
114        // Android has no system half to ask, so the mirror answers on the same frame the headless
115        // suite's does.
116        #[cfg(target_os = "android")]
117        answered(queue, wake, into, self.mirror.clone());
118        #[cfg(not(target_os = "android"))]
119        match &mut self.system {
120            #[cfg(all(not(target_family = "wasm"), not(target_os = "android")))]
121            Some(System(clipboard)) => {
122                // On the frame's own thread, unlike a file, because the context that serves what
123                // this program wrote has to outlive the write and so cannot be moved onto a thread
124                // per read. A selection round trip is bounded and human-triggered; a font is
125                // neither.
126                let text = match clipboard.get_text() {
127                    Ok(text) => text,
128                    Err(reason) => {
129                        debug!(%reason, "clipboard read refused");
130                        self.mirror.clone()
131                    }
132                };
133                answered(queue, wake, into, text);
134            }
135            #[cfg(target_family = "wasm")]
136            Some(System) => {
137                let (queue, wake, mirror) = (queue.clone(), wake.clone(), self.mirror.clone());
138                wasm_bindgen_futures::spawn_local(async move {
139                    let text = read_text().await.unwrap_or(mirror);
140                    answered(&queue, &wake, into, text);
141                });
142            }
143            // Nothing was ever attached: the headless suite, or a desktop that had no clipboard to
144            // open. The mirror is the whole clipboard there, and it answers at once so a paste is
145            // read in the next frame the way a platform's would be.
146            None => answered(queue, wake, into, self.mirror.clone()),
147        }
148    }
149}

151/// Queues what the clipboard held and asks for the frame that will drain it.
152///
153/// The whole of what a read does when it finishes, so every one of them lands the same way whatever
154/// answered it.
155fn answered(queue: &Queue, wake: &Wake, into: Option<Leaf>, text: String) {
156    queue.push(Op::Pasted { into, text });
157    wake.rouse();
158}

160/// The read itself, as a promise.
161///
162/// Refused far more often than it is answered -- it is permission-gated, and outside a user gesture
163/// most browsers say no. Each failure is one failure to an app, so none is distinguished.
164///
165/// Only ever the app asking for itself. `Ctrl+V` in a field is a gesture, and a gesture is answered
166/// by the `paste` event the keyboard's hidden input hears, which never reaches here.
167#[cfg(target_family = "wasm")]
168async fn read_text() -> Option<String> {
169    use wasm_bindgen_futures::JsFuture;
170
171    let window = web_sys::window()?;
172    let text = JsFuture::from(window.navigator().clipboard().read_text())
173        .await
174        .ok()?;
175    text.as_string()
176}
