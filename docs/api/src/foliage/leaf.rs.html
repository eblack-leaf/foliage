4/// An opaque name for one element.
5///
6/// Allocated the moment you ask for it and usable immediately -- as a parent, as the target of a
7/// write -- even though the element itself comes into existence when the frame's ops are drained.
8///
9/// A `Leaf` naming something that has since been pruned is inert rather than dangerous: every op
10/// targeting it is dropped and every tap of it reads absent. Nothing panics, and **a name is never
11/// reused**, so a stale one cannot come to address whatever grew after it.
12#[derive(Copy, Clone, PartialEq, Eq, Hash, PartialOrd, Ord, Debug)]
13pub struct Leaf(pub(crate) Entity);
14
15impl Leaf {
16    /// A stable number for this element, for logging or as a map key. Not an address -- there is
17    /// nothing to be done with it but tell two elements apart.
18    pub fn id(&self) -> u64 {
19        self.0.to_bits()
20    }
21}
22
23/// What a [`Leaf`] names right now.
24///
25/// The three states are ordered and terminal at the end: `Planted` -> `Live` -> `Withered`.
26#[derive(Copy, Clone, PartialEq, Eq, Debug)]
27pub enum Presence {
28    /// Named, and the op that grows it has not been drained yet -- the normal state for the
29    /// remainder of the frame that planted it. Taps read absent until it lands.
30    ///
31    /// Also what a name reads as when the op that would have grown it was dropped because its
32    /// parent had withered by the time the drain reached it.
33    Planted,
34    /// Live in the tree.
35    Live,
36    /// Pruned, or taken down with an ancestor. Terminal: growing again means a new `Leaf`.
37    Withered,
38}

40/// Marks an element the app grew itself, as opposed to one foliage spawned underneath it.
41#[derive(Component, Copy, Clone, Default)]
42pub(crate) struct Grown;
43
44/// The order the name was allocated in.
45///
46/// Taken when the [`Leaf`] is handed out rather than when the element is grown, so it is the order
47/// `plant` and `branch` were called in and not the order the drain reached them. Monotonic for the
48/// life of the run and never reused, which is what lets it settle the elevation tie-break totally
49/// and survive a prune of anything allocated earlier.
50#[derive(Component, Copy, Clone, Debug, Default, PartialEq, Eq, PartialOrd, Ord)]
51pub(crate) struct Growth(pub(crate) u64);
