2use std::hash::{BuildHasherDefault, Hash, Hasher};

4/// An opaque name for one element.
5///
6/// Allocated the moment you ask for it and usable immediately -- as a parent, as the target of a
7/// write -- even though the element itself comes into existence when the frame's ops are drained.
8///
9/// A `Leaf` naming something that has since been pruned is inert rather than dangerous: every op
10/// targeting it is dropped and every tap of it reads absent. Nothing panics, and **a name is never
11/// reused**, so a stale one cannot come to address whatever grew after it.
12///
13/// Inside, it is an index and a generation. The index is where the element is held, and goes back
14/// to be named again when the element withers; the generation is how many times it has, so a name
15/// that came before is never the name that came after.
16#[derive(Copy, Clone, PartialEq, Eq, PartialOrd, Ord, Debug)]
17pub struct Leaf {
18    pub(crate) index: u32,
19    pub(crate) generation: u32,
20}
21
22impl Hash for Leaf {
23    fn hash<H: Hasher>(&self, state: &mut H) {
24        state.write_u64(self.id());
25    }
26}

28/// A set of elements, hashed as the one number an element's name is.
29///
30/// A name is a generational index and hashes as one, so what the engine keys on elements is hashed
31/// in a multiply rather than by a general-purpose hash built to resist inputs chosen against it --
32/// which a name, handed out by the engine itself, never is.
33pub(crate) type Leaves = HashSet<Leaf, BuildHasherDefault<Fibonacci>>;
34
35/// A hash that leaves a name's index where it is in the low bits and spreads it into the high ones,
36/// in one multiply: names numbered together land together in a table, and what a table compares
37/// first still differs between them.
38///
39/// For a set emptied every frame, which is what [`Leaves`] are. A map held across frames is hashed
40/// by [`Spread`] instead.
41#[derive(Default)]
42pub(crate) struct Fibonacci {
43    hash: u64,
44}
45
46impl Hasher for Fibonacci {
47    fn finish(&self) -> u64 {
48        self.hash
49    }
50
51    fn write_u64(&mut self, value: u64) {
52        // The high half is the reciprocal of the golden ratio, which is what spreads the index
53        // upward; the low half passes the index through.
54        self.hash = value.wrapping_mul(0x9e37_79b9_0000_0001);
55    }
56
57    fn write(&mut self, bytes: &[u8]) {
58        for &byte in bytes {
59            self.write_u64(self.hash.rotate_left(8) ^ u64::from(byte));
60        }
61    }
62}

64/// A map keyed on anything that hashes as one name: an element, or what one is held under.
65///
66/// Hashed by [`Spread`] rather than as [`Leaves`] are. A name's index is dense -- elements grown
67/// together are numbered together -- and hashed as itself it puts what was grown together in one
68/// unbroken run of the table. That is what a set emptied every frame wants. A map held across
69/// frames, whose entries come and go by the hundred thousand as a map's streets do, does not:
70/// taking an entry out of a full run leaves a marker rather than a gap, and every lookup for a
71/// name not there walks the markers until it finds a gap, further each time entries come and go.
72pub(crate) type Named<K, V> = HashMap<K, V, BuildHasherDefault<Spread>>;
73
74/// A hash that mixes every bit of a name into every bit of the hash, so names numbered together
75/// land apart: the finaliser of SplitMix64, two multiplies.
76#[derive(Default)]
77pub(crate) struct Spread {
78    hash: u64,
79}
80
81impl Hasher for Spread {
82    fn finish(&self) -> u64 {
83        let mut x = self.hash;
84        x ^= x >> 30;
85        x = x.wrapping_mul(0xbf58_476d_1ce4_e5b9);
86        x ^= x >> 27;
87        x = x.wrapping_mul(0x94d0_49bb_1331_11eb);
88        x ^ (x >> 31)
89    }
90
91    fn write_u64(&mut self, value: u64) {
92        self.hash = self.hash.rotate_left(29) ^ value;
93    }
94
95    fn write_u32(&mut self, value: u32) {
96        self.write_u64(u64::from(value));
97    }
98
99    fn write(&mut self, bytes: &[u8]) {
100        for &byte in bytes {
101            self.write_u64(u64::from(byte));
102        }
103    }
104}

106impl Leaf {
107    /// A stable number for this element, for logging or as a map key. Not an address -- there is
108    /// nothing to be done with it but tell two elements apart.
109    pub fn id(&self) -> u64 {
110        (u64::from(self.generation) << 32) | u64::from(self.index)
111    }
112
113    /// Where the element is held, which is what everything kept per element is indexed by.
114    pub(crate) fn slot(self) -> usize {
115        self.index as usize
116    }
117}

119/// What a [`Leaf`] names right now.
120///
121/// The three states are ordered and terminal at the end: `Planted` -> `Live` -> `Withered`.
122#[derive(Copy, Clone, PartialEq, Eq, Debug)]
123pub enum Presence {
124    /// Named, and the op that grows it has not been drained yet -- the normal state for the
125    /// remainder of the frame that planted it. Taps read absent until it lands.
126    ///
127    /// Also what a name reads as when the op that would have grown it was dropped because its
128    /// parent had withered by the time the drain reached it.
129    Planted,
130    /// Live in the tree.
131    Live,
132    /// Pruned, or taken down with an ancestor. Terminal: growing again means a new `Leaf`.
133    Withered,
134}
135
136/// The order the name was allocated in.
137///
138/// Taken when the [`Leaf`] is handed out rather than when the element is grown, so it is the order
139/// `plant` and `branch` were called in and not the order the drain reached them. Monotonic for the
140/// life of the run and never reused, which is what lets it settle the elevation tie-break totally
141/// and survive a prune of anything allocated earlier.
142#[derive(Copy, Clone, Debug, Default, PartialEq, Eq, PartialOrd, Ord)]
143pub(crate) struct Growth(pub(crate) u64);
