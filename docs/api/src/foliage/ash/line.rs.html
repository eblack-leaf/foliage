1//! Turning a stroke into the form the GPU draws it from.
2
3use bytemuck::{Pod, Zeroable};
4
5use crate::color::Color;
6use crate::line::{Cap, LineInstance};
7
8/// One stroke, in the form the GPU takes it: the segment itself, and how thick and how capped.
9///
10/// The **segment** rather than four corners, because the vertex stage places the quad and the
11/// fragment stage measures distance to the segment -- so corners worked out on the CPU would be
12/// carried only to be turned back into the segment they came from. What is derived here is what the
13/// density decides: an axis-aligned stroke put on whole device pixels, and how wide the edge's
14/// coverage ramp is.
15#[repr(C)]
16#[derive(Copy, Clone, Debug, PartialEq, Pod, Zeroable)]
17pub(crate) struct LineQuad {
18    /// The two ends: the near one, then the far one.
19    pub(crate) segment: [f32; 4],
20    pub(crate) color: Color,
21    /// Half the stroke's weight, which is the radius its field is offset by.
22    pub(crate) half: f32,
23    /// Whether the ends are round, as a number the shader can branch on.
24    pub(crate) round: f32,
25    /// Half the coverage ramp -- see [`feather`].
26    pub(crate) feather: f32,
27    /// How it and everything it is drawn inside of are turned and scaled where it is drawn.
28    pub(crate) pose: crate::pose::Affine,
29}
30
31impl LineQuad {
32    /// What `instance` draws as, on a display of this density.
33    pub(crate) fn new(instance: LineInstance, scale: f32) -> Self {
34        let round = match instance.cap {
35            Cap::Round => 1.0,
36            Cap::Butt => 0.0,
37        };
38        // Snapped where it runs along an axis on the surface itself: a turn takes it off the axis
39        // and a scale off the pixel grid, and either way there is no edge left to put on a whole
40        // pixel -- the feather is what draws it cleanly then, as at any other angle.
41        let along = instance.from.x == instance.to.x || instance.from.y == instance.to.y;
42        let (from, to, half) = match along && instance.pose.upright() {
43            true => snapped(instance, scale),
44            false => (instance.from, instance.to, instance.weight / 2.0),
45        };
46        Self {
47            segment: [from.x, from.y, to.x, to.y],
48            color: instance.color,
49            half,
50            round,
51            // One device pixel as drawn, which under a scale is a different length of the stroke's
52            // own field.
53            feather: feather(half, scale * instance.pose.magnification()),
54            pose: instance.pose,
55        }
56    }
57}
58
59/// Half the stroke's coverage ramp, in the logical pixels its field is measured in.
60///
61/// The ramp spans exactly one device pixel, centred on the true edge. That width is the one at which
62/// a linear ramp sampled at pixel centres sums to the coverage the shape actually has: wider, and
63/// what a column of pixels sums to depends on where the centreline falls between them, which draws
64/// as a stroke thinning and thickening along its own length as it drifts; narrower, and the edge
65/// steps.
66///
67/// Stated here rather than derived in the shader because the density is the whole of the answer and
68/// the density is known here. A screen-space derivative cannot state it. `fwidth` is `|dpdx| +
69/// |dpdy|`, which over a distance field is `|cos| + |sin|` of the gradient's direction and so
70/// reports between one and one and a half device pixels depending on the angle the stroke runs at;
71/// it is taken per 2x2 quad, so whatever it reports is quantised across pairs of pixels; and it is
72/// discontinuous along a butt cap, where the field's `max` changes which term is nearest.
73///
74/// Capped at half the weight, so a stroke thinner than its own ramp still reaches full coverage
75/// along its centreline at any angle.
76fn feather(half: f32, scale: f32) -> f32 {
77    (0.5 / scale).min(half)
78}
79
80/// An axis-aligned stroke, with its edges on whole device pixels.
81///
82/// The same crisp alignment every other primitive gets from its own field, which a stroke has no
83/// field to get. A rule whose centreline lands on a pixel boundary is split evenly across two rows,
84/// so a one-pixel rule draws as two half-lit ones: the right amount of ink, twice the width, and
85/// dimmer than it was asked to be.
86///
87/// The **thickness** is snapped and the centreline is then placed from it, rather than the other way
88/// round: a whole number of device pixels of ink, sitting on the boundary, is the thing that reads
89/// crisply. Snapping the centreline instead would put an even weight half a pixel out.
90///
91/// Axis-aligned only. At any other angle a stroke has no edge that *can* be snapped -- moving it
92/// changes both its width and its angle -- and the shader's feather is what makes those read
93/// cleanly instead.
94fn snapped(
95    instance: LineInstance,
96    scale: f32,
97) -> (
98    crate::coordinate::Position,
99    crate::coordinate::Position,
100    f32,
101) {
102    let whole = |value: f32| (value * scale).round() / scale;
103    // At least one device pixel of ink, whatever was asked for.
104    let thickness = ((instance.weight * scale).round().max(1.0)) / scale;
105    let half = thickness / 2.0;
106    let (mut from, mut to) = (instance.from, instance.to);
107    if from.y == to.y {
108        let edge = whole(from.y - half);
109        from.y = edge + half;
110        to.y = edge + half;
111        from.x = whole(from.x);
112        to.x = whole(to.x);
113    } else {
114        let edge = whole(from.x - half);
115        from.x = edge + half;
116        to.x = edge + half;
117        from.y = whole(from.y);
118        to.y = whole(to.y);
119    }
120    (from, to, half)
121}

123#[cfg(test)]
124mod tests {
125    use super::*;
126    use crate::coordinate::Position;
127    use crate::pose::Affine;
128
129    fn stroke(from: (f32, f32), to: (f32, f32), pose: Affine) -> LineInstance {
130        LineInstance {
131            from: Position::new(from.0, from.1),
132            to: Position::new(to.0, to.1),
133            color: Color::rgb(1.0, 1.0, 1.0),
134            weight: 4.0,
135            cap: Cap::Butt,
136            pose,
137        }
138    }
139
140    /// The ramp is one device pixel as drawn: scaled up twice, a pixel is half as much of the
141    /// stroke's own field, and scaled down, twice as much -- capped at half the weight as ever.
142    #[test]
143    fn a_scaled_stroke_feathers_one_pixel_as_drawn() {
144        let at = |scale: f32| {
145            let pose = Affine::about(Position::new(50.0, 50.0), 0.3, scale);
146            LineQuad::new(stroke((0.0, 0.0), (100.0, 40.0), pose), 1.0).feather
147        };
148        assert_eq!(at(1.0), 0.5);
149        assert!((at(2.0) - 0.25).abs() < 1e-5, "{}", at(2.0));
150        assert!((at(0.5) - 1.0).abs() < 1e-5, "{}", at(0.5));
151    }
152
153    /// Along an axis and upright, a stroke is put on whole pixels. Turned or scaled, it has no
154    /// pixel edge left to be put on, and is drawn where it was asked to be.
155    #[test]
156    fn only_an_upright_stroke_is_snapped() {
157        let upright = LineQuad::new(stroke((0.0, 10.3), (100.0, 10.3), Affine::IDENTITY), 1.0);
158        assert_ne!(upright.segment[1], 10.3);
159        let scaled = Affine::about(Position::new(50.0, 50.0), 0.0, 1.5);
160        let posed = LineQuad::new(stroke((0.0, 10.3), (100.0, 10.3), scaled), 1.0);
161        assert_eq!(posed.segment[1], 10.3);
162        assert_eq!(posed.half, 2.0);
163    }
