1//! Turning a stroke into the form the GPU draws it from.
2
3use bytemuck::{Pod, Zeroable};
4
5use crate::color::Color;
6use crate::line::{Cap, LineInstance};
7
8/// One stroke, in the form the GPU takes it: the segment itself, and how thick and how capped.
9///
10/// The **segment** rather than four corners, because the vertex stage places the quad and the
11/// fragment stage measures distance to the segment -- so corners worked out on the CPU would be
12/// carried only to be turned back into the segment they came from. What is derived here is what the
13/// density decides: an axis-aligned stroke put on whole device pixels, and how wide the edge's
14/// coverage ramp is.
15#[repr(C)]
16#[derive(Copy, Clone, Debug, PartialEq, Pod, Zeroable)]
17pub(crate) struct LineQuad {
18    /// The two ends: the near one, then the far one.
19    pub(crate) segment: [f32; 4],
20    pub(crate) color: Color,
21    /// Half the stroke's weight, which is the radius its field is offset by.
22    pub(crate) half: f32,
23    /// Whether the ends are round, as a number the shader can branch on.
24    pub(crate) round: f32,
25    /// Half the coverage ramp -- see [`feather`].
26    pub(crate) feather: f32,
27}
28
29impl LineQuad {
30    /// What `instance` draws as, on a display of this density.
31    pub(crate) fn new(instance: LineInstance, scale: f32) -> Self {
32        let round = match instance.cap {
33            Cap::Round => 1.0,
34            Cap::Butt => 0.0,
35        };
36        let (from, to, half) =
37            match instance.from.x == instance.to.x || instance.from.y == instance.to.y {
38                true => snapped(instance, scale),
39                false => (instance.from, instance.to, instance.weight / 2.0),
40            };
41        Self {
42            segment: [from.x, from.y, to.x, to.y],
43            color: instance.color,
44            half,
45            round,
46            feather: feather(half, scale),
47        }
48    }
49}
50
51/// Half the stroke's coverage ramp, in the logical pixels its field is measured in.
52///
53/// The ramp spans exactly one device pixel, centred on the true edge. That width is the one at which
54/// a linear ramp sampled at pixel centres sums to the coverage the shape actually has: wider, and
55/// what a column of pixels sums to depends on where the centreline falls between them, which draws
56/// as a stroke thinning and thickening along its own length as it drifts; narrower, and the edge
57/// steps.
58///
59/// Stated here rather than derived in the shader because the density is the whole of the answer and
60/// the density is known here. A screen-space derivative cannot state it. `fwidth` is `|dpdx| +
61/// |dpdy|`, which over a distance field is `|cos| + |sin|` of the gradient's direction and so
62/// reports between one and one and a half device pixels depending on the angle the stroke runs at;
63/// it is taken per 2x2 quad, so whatever it reports is quantised across pairs of pixels; and it is
64/// discontinuous along a butt cap, where the field's `max` changes which term is nearest.
65///
66/// Capped at half the weight, so a stroke thinner than its own ramp still reaches full coverage
67/// along its centreline at any angle.
68fn feather(half: f32, scale: f32) -> f32 {
69    (0.5 / scale).min(half)
70}
71
72/// An axis-aligned stroke, with its edges on whole device pixels.
73///
74/// The same crisp alignment every other primitive gets from its own field, which a stroke has no
75/// field to get. A rule whose centreline lands on a pixel boundary is split evenly across two rows,
76/// so a one-pixel rule draws as two half-lit ones: the right amount of ink, twice the width, and
77/// dimmer than it was asked to be.
78///
79/// The **thickness** is snapped and the centreline is then placed from it, rather than the other way
80/// round: a whole number of device pixels of ink, sitting on the boundary, is the thing that reads
81/// crisply. Snapping the centreline instead would put an even weight half a pixel out.
82///
83/// Axis-aligned only. At any other angle a stroke has no edge that *can* be snapped -- moving it
84/// changes both its width and its angle -- and the shader's feather is what makes those read
85/// cleanly instead.
86fn snapped(
87    instance: LineInstance,
88    scale: f32,
89) -> (
90    crate::coordinate::Position,
91    crate::coordinate::Position,
92    f32,
93) {
94    let whole = |value: f32| (value * scale).round() / scale;
95    // At least one device pixel of ink, whatever was asked for.
96    let thickness = ((instance.weight * scale).round().max(1.0)) / scale;
97    let half = thickness / 2.0;
98    let (mut from, mut to) = (instance.from, instance.to);
99    if from.y == to.y {
100        let edge = whole(from.y - half);
101        from.y = edge + half;
102        to.y = edge + half;
103        from.x = whole(from.x);
104        to.x = whole(to.x);
105    } else {
106        let edge = whole(from.x - half);
107        from.x = edge + half;
108        to.x = edge + half;
109        from.y = whole(from.y);
110        to.y = whole(to.y);
111    }
112    (from, to, half)
113}
