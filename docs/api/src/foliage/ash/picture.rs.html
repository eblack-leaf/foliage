1//! The pictures the backend is holding: one texture each.
2//!
3//! Not a sheet, and that is the decision rather than an omission. A picture is arbitrarily large,
4//! arrives whenever a decode finishes, and there may be any number of them -- so packing them onto
5//! a shared texture would need eviction, and evicting one orphans every instance already pointing at
6//! it. A texture each costs a binding change between spans that draw different pictures, which is
7//! what the stack's spans already cut on.
8
9use std::collections::HashMap;
10
11use bytemuck::{Pod, Zeroable};
12use tracing::debug;
13use wgpu::{
14    BindGroup, BindGroupLayout, Extent3d, Origin3d, Sampler, TexelCopyBufferLayout,
15    TexelCopyTextureInfo, TextureDescriptor, TextureDimension, TextureFormat, TextureUsages,
16    TextureViewDescriptor,
17};
18
19use crate::ash::quad::{bind, sampler, sampling};
20use crate::coordinate::Section;
21use crate::ginkgo::Ginkgo;
22use crate::image::Plate;
23
24/// One picture, in the form the GPU takes it.
25///
26/// `#[repr(C)]` over the types the coordinate module guarantees the layout of. Which texture to bind
27/// is not here: it is the span's, because a binding is not per-instance state.
28#[repr(C)]
29#[derive(Copy, Clone, Debug, PartialEq, Pod, Zeroable)]
30pub(crate) struct PictureQuad {
31    pub(crate) section: Section,
32    pub(crate) crop: [f32; 4],
33    pub(crate) radii: [f32; 4],
34    pub(crate) opacity: f32,
35}
36
37/// Every picture the backend has uploaded.
38pub(crate) struct Pictures {
39    layout: BindGroupLayout,
40    sampler: Sampler,
41    held: HashMap<Plate, BindGroup>,
42}
43
44impl Pictures {
45    pub(crate) fn new(ginkgo: &Ginkgo) -> Self {
46        let device = ginkgo.device();
47        Self {
48            layout: sampling(device, "picture"),
49            sampler: sampler(device, "picture"),
50            held: HashMap::new(),
51        }
52    }
53
54    /// Uploads one picture, replacing whatever was held under that name.
55    ///
56    /// Called for every plate the drain loaded, whether or not it was held before, which is what
57    /// makes writing the same name again -- a re-fetch at a higher resolution -- reach every element
58    /// drawing it.
59    pub(crate) fn upload(
60        &mut self,
61        ginkgo: &Ginkgo,
62        plate: Plate,
63        pixels: &[u8],
64        size: (u32, u32),
65    ) {
66        let device = ginkgo.device();
67        let (width, height) = size;
68        if width == 0 || height == 0 {
69            return;
70        }
71        let texture = device.create_texture(&TextureDescriptor {
72            label: Some("picture"),
73            size: Extent3d {
74                width,
75                height,
76                depth_or_array_layers: 1,
77            },
78            mip_level_count: 1,
79            sample_count: 1,
80            dimension: TextureDimension::D2,
81            format: TextureFormat::Rgba8Unorm,
82            usage: TextureUsages::TEXTURE_BINDING | TextureUsages::COPY_DST,
83            view_formats: &[],
84        });
85        ginkgo.queue().write_texture(
86            TexelCopyTextureInfo {
87                texture: &texture,
88                mip_level: 0,
89                origin: Origin3d::ZERO,
90                aspect: wgpu::TextureAspect::All,
91            },
92            pixels,
93            TexelCopyBufferLayout {
94                offset: 0,
95                bytes_per_row: Some(width * 4),
96                rows_per_image: Some(height),
97            },
98            Extent3d {
99                width,
100                height,
101                depth_or_array_layers: 1,
102            },
103        );
104        let view = texture.create_view(&TextureViewDescriptor::default());
105        self.held.insert(
106            plate,
107            bind(
108                device,
109                &self.layout,
110                &view,
111                &self.sampler,
112                "picture-binding",
113            ),
114        );
115        debug!(plate = plate.0, width, height, "picture uploaded");
116    }
117
118    /// The binding for one picture, or `None` if its pixels have not been uploaded.
119    pub(crate) fn binding(&self, plate: Plate) -> Option<&BindGroup> {
120        self.held.get(&plate)
121    }
122
123    /// The layout a pipeline sampling a picture declares at group 1.
124    pub(crate) fn layout(&self) -> &BindGroupLayout {
125        &self.layout
126    }
127}
