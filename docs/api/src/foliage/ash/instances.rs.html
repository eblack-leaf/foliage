13/// One renderer's instance buffers, kept in rank order.
14///
15/// Generic over the instance, because what a renderer sends is the renderer's own business. This
16/// owns the upload and knows nothing about what is being uploaded.
17///
18/// # Two buffers
19///
20/// The renderer's own data is one, and the depth each instance is drawn at is the other. They are
21/// separate because they change for different reasons: a depth comes from an instance's *position*
22/// in the stack, so it is untouched while the stack is, and a value written to an instance already
23/// in it does not disturb anyone else's. Holding depth inside the instance would make every
24/// recolour a candidate for a full rewrite.
25///
26/// # What this does not decide
27///
28/// Where its instances sit among *everyone's* is [`Ash`](crate::ash::Ash)'s: the depths are written
29/// from there, and so is the order the draws go in. This keeps its slots sorted by rank so that a
30/// walk of the whole stack meets them in slot order, and reports the rank and clip of each so that
31/// walk has something to sort and cut on.
32pub(crate) struct Instances<I: Pod> {
33    held: Named<Key, Held<I>>,
34    /// The renderer's data, in slot order.
35    data: Vec<I>,
36    /// One depth per slot, written by `Ash` from the whole stack's order.
37    depths: Vec<f32>,
38    /// One rank per slot, which is what that order is built from.
39    ranks: Vec<ResolvedElevation>,
40    /// One clip per slot. CPU-side only: it is what the draw is cut on, and is never uploaded.
41    clips: Vec<Section>,
42    /// What each slot binds, for the one renderer whose instances do not all bind the same thing.
43    /// CPU-side only, and zero everywhere except pictures -- a binding is not per-instance state,
44    /// it is what a span is cut on.
45    groups: Vec<u32>,
46    buffer: Buffer,
47    depth: Buffer,
48    capacity: u32,
49    label: &'static str,
50    /// Whether the order itself changed, which moves every slot after the change.
51    resort: bool,
52    /// Whether anything the stack is built from changed: which slots there are, or what one is
53    /// clipped to.
54    disturbed: bool,
55    /// Slots whose value changed while the order did not.
56    touched: Vec<u32>,
57}
58
59/// One instance, and where it currently sits.
60struct Held<I> {
61    instance: I,
62    rank: ResolvedElevation,
63    clip: Section,
64    group: u32,
65    slot: u32,
66}
67
68impl<I: Pod> Instances<I> {
69    /// Room for `capacity` instances, grown as it fills.
70    pub(crate) fn new(device: &Device, label: &'static str, capacity: u32) -> Self {
71        let capacity = capacity.max(1);
72        Self {
73            held: Named::default(),
74            data: Vec::new(),
75            depths: Vec::new(),
76            ranks: Vec::new(),
77            clips: Vec::new(),
78            groups: Vec::new(),
79            buffer: buffer(device, label, size_of::<I>() as u32 * capacity),
80            depth: buffer(device, label, size_of::<f32>() as u32 * capacity),
81            capacity,
82            label,
83            resort: false,
84            disturbed: false,
85            touched: Vec::new(),
86        }
87    }
88
89    /// Takes one instance from a batch: the value, the rank it is to be drawn at, and the clip it is
90    /// drawn under.
91    ///
92    /// A rank that is unchanged leaves the order alone, so the ordinary write -- a move, a recolour,
93    /// a rounding -- costs one slot rather than a re-sort. A clip that changed while the order did
94    /// not costs the stack being cut again, and no upload at all.
95    pub(crate) fn write(
96        &mut self,
97        key: Key,
98        rank: ResolvedElevation,
99        clip: Section,
100        group: u32,
101        instance: I,
102    ) {
103        match self.held.get_mut(&key) {
104            Some(held) => {
105                held.instance = instance;
106                // Both are what the stack is cut on rather than what is uploaded, so either one
107                // changing costs the walk again and no upload of its own.
108                let recut = held.clip != clip || held.group != group;
109                held.clip = clip;
110                held.group = group;
111                if held.rank == rank {
112                    let slot = held.slot as usize;
113                    self.data[slot] = instance;
114                    self.touched.push(slot as u32);
115                    if recut {
116                        self.clips[slot] = clip;
117                        self.groups[slot] = group;
118                        self.disturbed = true;
119                    }
120                } else {
121                    held.rank = rank;
122                    self.resort = true;
123                }
124            }
125            None => {
126                self.held.insert(
127                    key,
128                    Held {
129                        instance,
130                        rank,
131                        clip,
132                        group,
133                        slot: 0,
134                    },
135                );
136                self.resort = true;
137            }
138        }
139    }
140
141    /// Drops one instance from what is held.
142    pub(crate) fn withdraw(&mut self, key: Key) {
143        if self.held.remove(&key).is_some() {
144            self.resort = true;
145        }
146    }
147
148    /// Puts what changed onto the GPU.
149    ///
150    /// A batch that added, removed, or moved anything through the stack rewrites the instance
151    /// buffer, because a slot is a position in one order and every position after the change has
152    /// moved. A batch that only rewrote values already in the order writes those slots and nothing
153    /// else -- as runs of neighbouring slots, one write each, since a write costs something of its
154    /// own whatever it carries and a frame that refilled part of a page touched a stretch of it. A
155    /// frame with no batch at all does nothing here.
156    pub(crate) fn flush(&mut self, device: &Device, queue: &Queue) {
157        if self.resort {
158            self.reorder(device, queue);
159            return;
160        }
161        let stride = size_of::<I>() as BufferAddress;
162        for run in runs(&mut self.touched) {
163            queue.write_buffer(
164                &self.buffer,
165                run.start as BufferAddress * stride,
166                bytemuck::cast_slice(&self.data[run]),
167            );
168        }
169        self.touched.clear();
170    }
171
172    /// Rebuilds the order back to front and rewrites the instance buffer.
173    fn reorder(&mut self, device: &Device, queue: &Queue) {
174        self.resort = false;
175        self.disturbed = true;
176        self.touched.clear();
177        let mut order = self
178            .held
179            .iter()
180            .map(|(key, held)| (held.rank, *key))
181            .collect::<Vec<_>>();
182        // A rank orders back to front and is total -- its allocation counter separates two elements
183        // that accumulated to the same elevation -- so this is one sort with no tie left in it, and
184        // two identical runs order identically.
185        order.sort_unstable();
186        let total = order.len();
187        self.data.clear();
188        self.ranks.clear();
189        self.clips.clear();
190        self.groups.clear();
191        self.depths.clear();
192        self.depths.resize(total, 0.0);
193        for (slot, (rank, key)) in order.into_iter().enumerate() {
194            let held = self.held.get_mut(&key).expect("held");
195            held.slot = slot as u32;
196            self.data.push(held.instance);
197            self.ranks.push(rank);
198            self.clips.push(held.clip);
199            self.groups.push(held.group);
200        }
201        if total as u32 > self.capacity {
202            self.capacity = (total as u32).next_power_of_two();
203            self.buffer = buffer(device, self.label, size_of::<I>() as u32 * self.capacity);
204            self.depth = buffer(device, self.label, size_of::<f32>() as u32 * self.capacity);
205        }
206        if total == 0 {
207            return;
208        }
209        queue.write_buffer(&self.buffer, 0, bytemuck::cast_slice(&self.data));
210    }
211
212    /// Whether the stack has to be walked again: a slot appeared, went, moved, or changed what it
213    /// is clipped to. Cleared by the walk.
214    pub(crate) fn disturbed(&mut self) -> bool {
215        core::mem::take(&mut self.disturbed)
216    }
217
218    /// The rank of each slot, in slot order.
219    pub(crate) fn ranks(&self) -> &[ResolvedElevation] {
220        &self.ranks
221    }
222
223    /// What slot `slot` is clipped to.
224    pub(crate) fn clip(&self, slot: u32) -> Section {
225        self.clips[slot as usize]
226    }
227
228    /// What slot `slot` binds. Zero for every renderer that binds one thing or nothing.
229    pub(crate) fn group(&self, slot: u32) -> u32 {
230        self.groups[slot as usize]
231    }
232
233    /// Where in the whole stack slot `slot` sits, as a depth.
234    pub(crate) fn set_depth(&mut self, slot: u32, depth: f32) {
235        self.depths[slot as usize] = depth;
236    }
237
238    /// Puts the depths on the GPU, once the whole stack has been walked and every slot has one.
239    pub(crate) fn flush_depths(&self, queue: &Queue) {
240        if self.depths.is_empty() {
241            return;
242        }
243        queue.write_buffer(&self.depth, 0, bytemuck::cast_slice(&self.depths));
244    }
245
246    pub(crate) fn data(&self) -> BufferSlice<'_> {
247        self.buffer.slice(..)
248    }
249
250    pub(crate) fn depths(&self) -> BufferSlice<'_> {
251        self.depth.slice(..)
252    }
253}

255/// How many untouched slots a write reaches across rather than being cut in two. Rewriting a slot
256/// with what it already holds is harmless, and cheaper than a write of its own.
257const BRIDGE: u32 = 16;
258
259/// The touched slots as runs to write, in order: neighbours, and slots no further apart than
260/// [`BRIDGE`], are one run.
261pub(crate) fn runs(touched: &mut [u32]) -> impl Iterator<Item = Range<usize>> + '_ {
262    touched.sort_unstable();
263    let mut at = 0;
264    core::iter::from_fn(move || {
265        let start = *touched.get(at)?;
266        let mut end = start;
267        while let Some(&next) = touched.get(at) {
268            if next > end + BRIDGE {
269                break;
270            }
271            end = end.max(next);
272            at += 1;
273        }
274        Some(start as usize..end as usize + 1)
275    })
276}

278fn buffer(device: &Device, label: &'static str, size: u32) -> Buffer {
279    device.create_buffer(&BufferDescriptor {
280        label: Some(label),
281        size: size as BufferAddress,
282        usage: BufferUsages::VERTEX | BufferUsages::COPY_DST,
283        mapped_at_creation: false,
284    })
