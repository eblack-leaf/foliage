12/// One renderer's instance buffers, kept in rank order.
13///
14/// Generic over the instance, because what a renderer sends is the renderer's own business. This
15/// owns the upload and knows nothing about what is being uploaded.
16///
17/// # Two buffers
18///
19/// The renderer's own data is one, and the depth each instance is drawn at is the other. They are
20/// separate because they change for different reasons: a depth comes from an instance's *position*
21/// in the stack, so it is untouched while the stack is, and a value written to an instance already
22/// in it does not disturb anyone else's. Holding depth inside the instance would make every
23/// recolour a candidate for a full rewrite.
24///
25/// # What this does not decide
26///
27/// Where its instances sit among *everyone's* is [`Ash`](crate::ash::Ash)'s: the depths are written
28/// from there, and so is the order the draws go in. This keeps its slots sorted by rank so that a
29/// walk of the whole stack meets them in slot order, and reports the rank and clip of each so that
30/// walk has something to sort and cut on.
31pub(crate) struct Instances<I: Pod> {
32    slots: Slots<I>,
33    buffer: Buffer,
34    depth: Buffer,
35    capacity: u32,
36    label: &'static str,
37}
38
39impl<I: Pod> Instances<I> {
40    /// Room for `capacity` instances, grown as it fills.
41    pub(crate) fn new(device: &Device, label: &'static str, capacity: u32) -> Self {
42        let capacity = capacity.max(1);
43        Self {
44            slots: Slots::default(),
45            buffer: buffer(device, label, size_of::<I>() as u32 * capacity),
46            depth: buffer(device, label, size_of::<f32>() as u32 * capacity),
47            capacity,
48            label,
49        }
50    }
51
52    /// Takes one instance from a batch. See [`Slots::write`].
53    pub(crate) fn write(
54        &mut self,
55        key: Key,
56        rank: ResolvedElevation,
57        clip: Section,
58        group: u32,
59        instance: I,
60    ) {
61        self.slots.write(key, rank, clip, group, instance);
62    }
63
64    /// Lets go of everything held, so the next batch states every instance there is.
65    ///
66    /// The GPU's buffers are kept for their room. What is in them is rewritten in full by the next
67    /// flush, because the order has to be built again from whatever the batch holds.
68    pub(crate) fn forget(&mut self) {
69        self.slots.forget();
70    }
71
72    /// Drops one instance from what is held.
73    pub(crate) fn withdraw(&mut self, key: Key) {
74        self.slots.withdraw(key);
75    }
76
77    /// Puts what changed onto the GPU.
78    ///
79    /// A batch that added, removed, or moved anything through the stack rewrites the instance
80    /// buffer, because a slot is a position in one order and every position after the change has
81    /// moved. A batch that only rewrote values already in the order writes those slots and nothing
82    /// else -- as runs of neighbouring slots, one write each, since a write costs something of its
83    /// own whatever it carries and a frame that refilled part of a page touched a stretch of it --
84    /// or, where it rewrote most of them, the whole buffer at once, which is what the runs would
85    /// come to after sorting every slot to find them. A frame with no batch at all does nothing
86    /// here.
87    pub(crate) fn flush(&mut self, device: &Device, queue: &Queue) {
88        let whole = match self.slots.resort {
89            true => {
90                self.slots.reorder();
91                let total = self.slots.data.len() as u32;
92                if total > self.capacity {
93                    self.capacity = total.next_power_of_two();
94                    self.buffer = buffer(device, self.label, size_of::<I>() as u32 * self.capacity);
95                    self.depth =
96                        buffer(device, self.label, size_of::<f32>() as u32 * self.capacity);
97                }
98                true
99            }
100            false => self.slots.touched.len() * 2 >= self.slots.data.len(),
101        };
102        if whole {
103            self.slots.touched.clear();
104            if !self.slots.data.is_empty() {
105                queue.write_buffer(&self.buffer, 0, bytemuck::cast_slice(&self.slots.data));
106            }
107            return;
108        }
109        let stride = size_of::<I>() as BufferAddress;
110        for run in runs(&mut self.slots.touched) {
111            queue.write_buffer(
112                &self.buffer,
113                run.start as BufferAddress * stride,
114                bytemuck::cast_slice(&self.slots.data[run]),
115            );
116        }
117        self.slots.touched.clear();
118    }
119
120    /// Whether anything the stack is built from changed since this was last asked.
121    pub(crate) fn disturbed(&mut self) -> bool {
122        core::mem::take(&mut self.slots.disturbed)
123    }
124
125    /// The rank of each slot, in slot order.
126    pub(crate) fn ranks(&self) -> &[ResolvedElevation] {
127        &self.slots.ranks
128    }
129
130    /// What slot `slot` is clipped to.
131    pub(crate) fn clip(&self, slot: u32) -> Section {
132        self.slots.clips[slot as usize]
133    }
134
135    /// What slot `slot` binds. Zero for every renderer that binds one thing or nothing.
136    pub(crate) fn group(&self, slot: u32) -> u32 {
137        self.slots.groups[slot as usize]
138    }
139
140    /// Where in the whole stack slot `slot` sits, as a depth.
141    pub(crate) fn set_depth(&mut self, slot: u32, depth: f32) {
142        self.slots.depths[slot as usize] = depth;
143    }
144
145    /// Puts the depths on the GPU, once the whole stack has been walked and every slot has one.
146    pub(crate) fn flush_depths(&self, queue: &Queue) {
147        if self.slots.depths.is_empty() {
148            return;
149        }
150        queue.write_buffer(&self.depth, 0, bytemuck::cast_slice(&self.slots.depths));
151    }
152
153    pub(crate) fn data(&self) -> BufferSlice<'_> {
154        self.buffer.slice(..)
155    }
156
157    pub(crate) fn depths(&self) -> BufferSlice<'_> {
158        self.depth.slice(..)
159    }
160}

162/// What one renderer holds, in slot order, apart from the buffers it is put on: every instance,
163/// where it sits, and what changed since the last flush.
164///
165/// Kept sorted by rank, then by name. An order is only ever repaired, never sorted whole: what
166/// stays keeps its place relative to the rest, so it is already in order, and only what arrived or
167/// changed rank is sorted before the two are merged. A frame that adds a hundred thousand instances
168/// to as many already held sorts the hundred thousand.
169struct Slots<I> {
170    held: Table<Held<I>>,
171    /// Which instance each slot holds.
172    keys: Vec<Key>,
173    /// The renderer's data, in slot order.
174    data: Vec<I>,
175    /// One depth per slot, written by `Ash` from the whole stack's order.
176    depths: Vec<f32>,
177    /// One rank per slot, which is what that order is built from.
178    ranks: Vec<ResolvedElevation>,
179    /// One clip per slot. CPU-side only: it is what the draw is cut on, and is never uploaded.
180    clips: Vec<Section>,
181    /// What each slot binds, for the one renderer whose instances do not all bind the same thing.
182    /// CPU-side only, and zero everywhere except pictures -- a binding is not per-instance state,
183    /// it is what a span is cut on.
184    groups: Vec<u32>,
185    /// Slots whose instance went, or moved to another rank, since the order was last made.
186    vacated: Vec<u32>,
187    /// What has no slot yet: arrived, or moved to another rank. Placed by the next order.
188    arrived: Vec<Key>,
189    /// Whether the order itself changed, which moves every slot after the change.
190    resort: bool,
191    /// Whether anything the stack is built from changed: which slots there are, or what one is
192    /// clipped to.
193    disturbed: bool,
194    /// Slots whose value changed while the order did not.
195    touched: Vec<u32>,
196}
197
198/// One instance, and where it currently sits: [`UNPLACED`] until an order places it.
199struct Held<I> {
200    instance: I,
201    rank: ResolvedElevation,
202    clip: Section,
203    group: u32,
204    slot: u32,
205}
206
207/// The slot of an instance no order has placed yet.
208const UNPLACED: u32 = u32::MAX;
209
210impl<I> Default for Slots<I> {
211    fn default() -> Self {
212        Self {
213            held: Table::default(),
214            keys: Vec::new(),
215            data: Vec::new(),
216            depths: Vec::new(),
217            ranks: Vec::new(),
218            clips: Vec::new(),
219            groups: Vec::new(),
220            vacated: Vec::new(),
221            arrived: Vec::new(),
222            resort: false,
223            disturbed: false,
224            touched: Vec::new(),
225        }
226    }
227}

229impl<I: Copy> Slots<I> {
230    /// Takes one instance from a batch: the value, the rank it is to be drawn at, and the clip it is
231    /// drawn under.
232    ///
233    /// A rank that is unchanged leaves the order alone, so the ordinary write -- a move, a recolour,
234    /// a rounding -- costs one slot rather than a new order. A clip that changed while the order
235    /// did not costs the stack being cut again, and no upload at all.
236    fn write(&mut self, key: Key, rank: ResolvedElevation, clip: Section, group: u32, instance: I) {
237        match self.held.get_mut(key) {
238            Some(held) => {
239                held.instance = instance;
240                // Both are what the stack is cut on rather than what is uploaded, so either one
241                // changing costs the walk again and no upload of its own.
242                let recut = held.clip != clip || held.group != group;
243                held.clip = clip;
244                held.group = group;
245                if held.slot == UNPLACED {
246                    // Waiting to be placed, which takes everything from what is held.
247                    held.rank = rank;
248                } else if held.rank == rank {
249                    let slot = held.slot as usize;
250                    self.data[slot] = instance;
251                    self.touched.push(slot as u32);
252                    if recut {
253                        self.clips[slot] = clip;
254                        self.groups[slot] = group;
255                        self.disturbed = true;
256                    }
257                } else {
258                    held.rank = rank;
259                    self.vacated.push(held.slot);
260                    held.slot = UNPLACED;
261                    self.arrived.push(key);
262                    self.resort = true;
263                }
264            }
265            None => {
266                let held = Held {
267                    instance,
268                    rank,
269                    clip,
270                    group,
271                    slot: UNPLACED,
272                };
273                // A name withered and another grown into its index in the same batch, with the
274                // write ahead of the withdrawal: the one it displaced goes now, as it would have.
275                if let Some((_, displaced)) = self.held.insert(key, held)
276                    && displaced.slot != UNPLACED
277                {
278                    self.vacated.push(displaced.slot);
279                }
280                self.arrived.push(key);
281                self.resort = true;
282            }
283        }
284    }
285
286    fn forget(&mut self) {
287        self.held.clear();
288        self.keys.clear();
289        self.data.clear();
290        self.ranks.clear();
291        self.clips.clear();
292        self.groups.clear();
293        self.depths.clear();
294        self.vacated.clear();
295        self.arrived.clear();
296        self.touched.clear();
297        self.resort = true;
298    }
299
300    fn withdraw(&mut self, key: Key) {
301        if let Some(held) = self.held.remove(key) {
302            if held.slot != UNPLACED {
303                self.vacated.push(held.slot);
304            }
305            self.resort = true;
306        }
307    }
308
309    /// Makes the order again: what stays, in the order it was, merged with what arrived, sorted.
310    /// Every slot an instance moved to is written back to what is held for it, and only those.
311    fn reorder(&mut self) {
312        self.resort = false;
313        self.disturbed = true;
314        self.touched.clear();
315        let mut vacant = vec![false; self.keys.len()];
316        for &slot in &self.vacated {
317            vacant[slot as usize] = true;
318        }
319        self.vacated.clear();
320        // An instance can arrive twice in one frame -- written, withdrawn and written again -- and
321        // one that arrived can go again before it is placed. What is held says which are still
322        // waiting, and at what rank.
323        let mut arrived: Vec<(ResolvedElevation, Key)> = self
324            .arrived
325            .drain(..)
326            .filter_map(|key| {
327                let held = self.held.get(key)?;
328                (held.slot == UNPLACED).then_some((held.rank, key))
329            })
330            .collect();
331        arrived.sort_unstable();
332        arrived.dedup();
333        let total = vacant.iter().filter(|vacant| !**vacant).count() + arrived.len();
334        let mut keys = Vec::with_capacity(total);
335        let mut data = Vec::with_capacity(total);
336        let mut ranks = Vec::with_capacity(total);
337        let mut clips = Vec::with_capacity(total);
338        let mut groups = Vec::with_capacity(total);
339        let mut stays = (0..self.keys.len())
340            .filter(|slot| !vacant[*slot])
341            .peekable();
342        let mut arriving = arrived.into_iter().peekable();
343        loop {
344            let from_stays = match (stays.peek(), arriving.peek()) {
345                (Some(&slot), Some(&(rank, key))) => {
346                    (self.ranks[slot], self.keys[slot]) < (rank, key)
347                }
348                (Some(_), None) => true,
349                (None, Some(_)) => false,
350                (None, None) => break,
351            };
352            let at = keys.len() as u32;
353            if from_stays {
354                let Some(slot) = stays.next() else { break };
355                let key = self.keys[slot];
356                if slot as u32 != at
357                    && let Some(held) = self.held.get_mut(key)
358                {
359                    held.slot = at;
360                }
361                keys.push(key);
362                data.push(self.data[slot]);
363                ranks.push(self.ranks[slot]);
364                clips.push(self.clips[slot]);
365                groups.push(self.groups[slot]);
366            } else {
367                let Some((rank, key)) = arriving.next() else {
368                    break;
369                };
370                let Some(held) = self.held.get_mut(key) else {
371                    continue;
372                };
373                held.slot = at;
374                keys.push(key);
375                data.push(held.instance);
376                ranks.push(rank);
377                clips.push(held.clip);
378                groups.push(held.group);
379            }
380        }
381        self.keys = keys;
382        self.data = data;
383        self.ranks = ranks;
384        self.clips = clips;
385        self.groups = groups;
386        self.depths.clear();
387        self.depths.resize(total, 0.0);
388    }
389}

391/// How many untouched slots a write reaches across rather than being cut in two. Rewriting a slot
392/// with what it already holds is harmless, and cheaper than a write of its own.
393const BRIDGE: u32 = 16;
394
395/// The touched slots as runs to write, in order: neighbours, and slots no further apart than
396/// [`BRIDGE`], are one run.
397pub(crate) fn runs(touched: &mut [u32]) -> impl Iterator<Item = Range<usize>> + '_ {
398    touched.sort_unstable();
399    let mut at = 0;
400    core::iter::from_fn(move || {
401        let start = *touched.get(at)?;
402        let mut end = start;
403        while let Some(&next) = touched.get(at) {
404            if next > end + BRIDGE {
405                break;
406            }
407            end = end.max(next);
408            at += 1;
409        }
410        Some(start as usize..end as usize + 1)
411    })
412}

414fn buffer(device: &Device, label: &'static str, size: u32) -> Buffer {
415    device.create_buffer(&BufferDescriptor {
416        label: Some(label),
417        size: size as BufferAddress,
418        usage: BufferUsages::VERTEX | BufferUsages::COPY_DST,
419        mapped_at_creation: false,
420    })
421}

436    /// A name withered and another grown into its index, in one batch with the write first, leaves
437    /// the new one held and the old one's slot let go -- and the withdrawal that follows names
438    /// something no longer held.
439    #[test]
440    fn a_name_grown_into_an_index_displaces_what_withered_from_it() {
441        let mut slots: Slots<u32> = Slots::default();
442        for n in 0..4 {
443            slots.write(named(n), rank(1, n as u64), Section::default(), 0, n);
444        }
445        slots.reorder();
446        let withered = named(2);
447        let grown = Key(withered.0 + (1 << 32));
448        slots.write(grown, rank(1, 9), Section::default(), 0, 9);
449        slots.withdraw(withered);
450        slots.reorder();
451        assert_eq!(slots.keys, vec![named(0), named(1), named(3), grown]);
452        assert_eq!(slots.data, vec![0, 1, 3, 9]);
453        assert_eq!(slots.held.len(), 4);
454        assert!(slots.held.get(withered).is_none());
455    }

457    /// Every slot in rank-then-name order, every held instance where its slot says, and holding
458    /// what was last written to it.
459    fn consistent(slots: &Slots<u32>, expected: &[(Key, ResolvedElevation, u32)]) {
460        let mut sorted = expected.to_vec();
461        sorted.sort_by_key(|(key, rank, _)| (*rank, *key));
462        let got: Vec<(Key, ResolvedElevation, u32)> = (0..slots.keys.len())
463            .map(|slot| (slots.keys[slot], slots.ranks[slot], slots.data[slot]))
464            .collect();
465        assert_eq!(got, sorted);
466        for (slot, key) in slots.keys.iter().enumerate() {
467            assert_eq!(
468                slots.held.get(*key).map(|held| held.slot),
469                Some(slot as u32)
470            );
471        }
472        assert_eq!(slots.held.len(), expected.len());
473        assert_eq!(slots.depths.len(), expected.len());
474    }

476    #[test]
477    fn a_repaired_order_is_the_order_sorting_everything_makes() {
478        let mut slots: Slots<u32> = Slots::default();
479        let mut expected: Vec<(Key, ResolvedElevation, u32)> = Vec::new();
480        let mut seed = 7u64;
481        let mut next = move |below: u64| {
482            seed = seed
483                .wrapping_mul(6_364_136_223_846_793_005)
484                .wrapping_add(1_442_695_040_888_963_407);
485            (seed >> 33) % below
486        };
487        for frame in 0..60 {
488            for _ in 0..40 {
489                let key = named(next(120) as u32);
490                let value = (frame * 1000) as u32 + next(1000) as u32;
491                match next(5) {
492                    // Gone.
493                    0 => {
494                        slots.withdraw(key);
495                        expected.retain(|(held, _, _)| *held != key);
496                    }
497                    // Written, at a rank that changes now and then.
498                    _ => {
499                        let at = rank(next(4) as i32, key.0);
500                        slots.write(key, at, Section::default(), 0, value);
501                        match expected.iter_mut().find(|(held, _, _)| *held == key) {
502                            Some(held) => *held = (key, at, value),
503                            None => expected.push((key, at, value)),
504                        }
505                    }
506                }
507            }
508            if slots.resort {
509                slots.reorder();
510            }
511            consistent(&slots, &expected);
512        }
513    }
