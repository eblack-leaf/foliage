1//! The atlas: every glyph the backend has rasterised, on one texture.
2//!
3//! # A cut is a glyph at a density
4//!
5//! What is rasterised is a character of a face at a size *for a display*, because the bitmap is made
6//! in device pixels and the display says how many of those one logical pixel is. So the density is
7//! part of the key rather than something applied afterwards: the same character at the same size on
8//! a different screen is a different cut, and asking for it makes one.
9//!
10//! Everything the atlas hands back about *where* a glyph goes is in logical pixels. The device
11//! pixels stop at the texture, which is the same line [`Ginkgo`](crate::ginkgo) draws.
12//!
13//! # What is on it is what the frame draws
14//!
15//! A cut is packed the first time a run asks for it and stays, because the ordinary frame draws
16//! what the last one did and should find every cut it needs already there. Nothing is taken off one
17//! at a time. When a cut finds no room, the sheet is cut again from the runs the frame draws and
18//! nothing else -- so whatever the frame stopped drawing gives its room back at once: words that
19//! were rewritten, a size no longer in use. Only when the frame's own cuts do not fit is the sheet
20//! full, and that is reported. A density that changed takes every cut off at once, since none of
21//! them is drawn at the density that is left.
22//!
23//! [`Cuts`] is the bookkeeping that decides all of it, apart from the texture it is about, so what
24//! happens to a cut is decided the same way with no device at all.
25

42/// How wide and tall the one texture is, in device pixels.
43///
44/// A fixed sheet rather than one that grows: growing means re-cutting every glyph on it, and a
45/// monospaced app addresses a small alphabet at a handful of sizes. Four megabytes of coverage holds
46/// thousands of cuts. Running out cuts it again from what the frame draws, and only what the frame
47/// draws not fitting is reported -- see [`Cuts`].
48const SIDE: u32 = 2048;
49
50/// A pixel of clearance around every cut, so that filtering a glyph at its edge cannot reach into
51/// the one packed beside it.
52const GUTTER: u32 = 1;
53
54/// One rasterised character: a face, at a size, for a display of a given density.
55#[derive(Copy, Clone, Debug, PartialEq, Eq, Hash)]
56pub(crate) struct Cut {
57    pub(crate) font: Font,
58    pub(crate) size: u32,
59    /// Device pixels per logical one, in thousandths. An integer because a key has to compare
60    /// exactly, and a float that came from a window manager does not.
61    pub(crate) density: u32,
62    pub(crate) character: char,
63}
64
65impl Cut {
66    pub(crate) fn new(font: Font, size: u32, scale: f32, character: char) -> Self {
67        Self {
68            font,
69            size,
70            density: (scale * 1000.0).round().max(1.0) as u32,
71            character,
72        }
73    }
74
75    /// The density as the rasteriser takes it.
76    fn scale(self) -> f32 {
77        self.density as f32 / 1000.0
78    }
79}

81/// Where a cut ended up: the ink's box inside a character cell, and its rect on the sheet.
82#[derive(Copy, Clone, Debug, Default, PartialEq)]
83pub(crate) struct Placed {
84    /// Where the ink's top-left corner sits inside its cell, in logical pixels.
85    pub(crate) offset: Position,
86    /// How large the ink is, in logical pixels.
87    pub(crate) area: Area,
88    /// The ink's rect on the sheet, normalised: origin then size.
89    pub(crate) uv: [f32; 4],
90}

92/// The one sheet, and what is on it.
93pub(crate) struct Atlas {
94    texture: Texture,
95    layout: BindGroupLayout,
96    binding: BindGroup,
97    cuts: Cuts,
98}
99
100impl Atlas {
101    pub(crate) fn new(ginkgo: &Ginkgo) -> Self {
102        let device = ginkgo.device();
103        let texture = device.create_texture(&TextureDescriptor {
104            label: Some("atlas"),
105            size: Extent3d {
106                width: SIDE,
107                height: SIDE,
108                depth_or_array_layers: 1,
109            },
110            mip_level_count: 1,
111            sample_count: 1,
112            dimension: TextureDimension::D2,
113            // Coverage, one channel. A glyph has an alpha and no colour of its own -- what it is
114            // filled with is the element's, and is carried on the instance.
115            format: TextureFormat::R8Unorm,
116            usage: TextureUsages::TEXTURE_BINDING | TextureUsages::COPY_DST,
117            view_formats: &[],
118        });
119        let view = texture.create_view(&TextureViewDescriptor::default());
120        let sampler = device.create_sampler(&SamplerDescriptor {
121            label: Some("atlas"),
122            address_mode_u: AddressMode::ClampToEdge,
123            address_mode_v: AddressMode::ClampToEdge,
124            address_mode_w: AddressMode::ClampToEdge,
125            mag_filter: FilterMode::Linear,
126            min_filter: FilterMode::Linear,
127            ..SamplerDescriptor::default()
128        });
129        let layout = device.create_bind_group_layout(&BindGroupLayoutDescriptor {
130            label: Some("atlas-layout"),
131            entries: &[
132                BindGroupLayoutEntry {
133                    binding: 0,
134                    visibility: ShaderStages::FRAGMENT,
135                    ty: BindingType::Texture {
136                        sample_type: TextureSampleType::Float { filterable: true },
137                        view_dimension: TextureViewDimension::D2,
138                        multisampled: false,
139                    },
140                    count: None,
141                },
142                BindGroupLayoutEntry {
143                    binding: 1,
144                    visibility: ShaderStages::FRAGMENT,
145                    ty: BindingType::Sampler(SamplerBindingType::Filtering),
146                    count: None,
147                },
148            ],
149        });
150        let binding = device.create_bind_group(&BindGroupDescriptor {
151            label: Some("atlas-binding"),
152            layout: &layout,
153            entries: &[
154                BindGroupEntry {
155                    binding: 0,
156                    resource: BindingResource::TextureView(&view),
157                },
158                BindGroupEntry {
159                    binding: 1,
160                    resource: BindingResource::Sampler(&sampler),
161                },
162            ],
163        });
164        Self {
165            texture,
166            layout,
167            binding,
168            cuts: Cuts::new(SIDE),
169        }
170    }
171
172    /// Where `cut` is on the sheet, cutting it if this is the first time it has been asked for.
173    ///
174    /// A cut that found no room is held as nothing for now, and [`overflowed`](Atlas::overflowed)
175    /// says so: cutting the sheet again is the caller's, because the caller is what knows which runs
176    /// the frame draws.
177    pub(crate) fn place(&mut self, fonts: &Fonts, ginkgo: &Ginkgo, cut: Cut) -> Placed {
178        let (placed, packed) = self.cuts.place(fonts, cut);
179        if let Some(Packed { x, y, ink }) = packed {
180            ginkgo.queue().write_texture(
181                TexelCopyTextureInfo {
182                    texture: &self.texture,
183                    mip_level: 0,
184                    origin: Origin3d { x, y, z: 0 },
185                    aspect: TextureAspect::All,
186                },
187                &ink.coverage,
188                TexelCopyBufferLayout {
189                    offset: 0,
190                    bytes_per_row: Some(ink.width),
191                    rows_per_image: Some(ink.height),
192                },
193                Extent3d {
194                    width: ink.width,
195                    height: ink.height,
196                    depth_or_array_layers: 1,
197                },
198            );
199        }
200        placed
201    }
202
203    /// Whether a cut has found no room since the sheet was last cut again.
204    pub(crate) fn overflowed(&self) -> bool {
205        self.cuts.overflowed()
206    }
207
208    /// Takes every cut off, for the sheet to be cut again from what is asked for next.
209    ///
210    /// The texture keeps what was written to it until a new cut is written over it, and nothing
211    /// points at what is left in between.
212    pub(crate) fn clear(&mut self) {
213        self.cuts.clear();
214    }
215
216    /// Says the sheet has been cut again from everything the frame draws. See [`Cuts::settle`].
217    pub(crate) fn settle(&mut self) {
218        self.cuts.settle();
219    }
220
221    /// Takes every cut off and forgets that the sheet was ever full, for one cut at a new density.
222    pub(crate) fn forget(&mut self) {
223        self.cuts.forget();
224    }
225
226    /// The layout a pipeline sampling the sheet declares at group 1.
227    pub(crate) fn layout(&self) -> &BindGroupLayout {
228        &self.layout
229    }
230
231    /// The sheet, as a pass binds it.
232    pub(crate) fn binding(&self) -> &BindGroup {
233        &self.binding
234    }
235}

237/// What is on the sheet and where, where the next cut goes, and whether one has run out of room:
238/// the atlas, less the texture.
239pub(crate) struct Cuts {
240    held: HashMap<Cut, Placed>,
241    shelves: Shelves,
242    /// Whether a cut has found no room since the sheet was last cut again.
243    overflowed: bool,
244    /// Whether, the last time the sheet was cut again from what the frame draws, that did not fit
245    /// either. Kept so the report is made once rather than every time it is found again.
246    full: bool,
247}
248
249/// A cut just packed: where on the sheet, and the ink to write there.
250pub(crate) struct Packed {
251    pub(crate) x: u32,
252    pub(crate) y: u32,
253    pub(crate) ink: Ink,
254}
255
256impl Cuts {
257    /// An empty sheet `side` texels across and down.
258    pub(crate) fn new(side: u32) -> Self {
259        Self {
260            held: HashMap::new(),
261            shelves: Shelves::new(side),
262            overflowed: false,
263            full: false,
264        }
265    }
266
267    /// Where `cut` is on the sheet, rasterising and packing it if this is the first time it has been
268    /// asked for -- and then, beside it, the ink and where to write it.
269    ///
270    /// A character with no outline -- and there are more of them than a space -- is held as nothing
271    /// rather than not held: asking again must not rasterise again. So is one that found no room,
272    /// for as long as the sheet stands; that one also marks the sheet
273    /// [`overflowed`](Cuts::overflowed), and is asked for again once the sheet is cut again.
274    pub(crate) fn place(&mut self, fonts: &Fonts, cut: Cut) -> (Placed, Option<Packed>) {
275        if let Some(placed) = self.held.get(&cut) {
276            return (*placed, None);
277        }
278        let ink = fonts.rasterize(cut.font, cut.size, cut.scale(), cut.character);
279        debug!(
280            character = ?cut.character,
281            size = cut.size,
282            width = ink.width,
283            height = ink.height,
284            "glyph cut"
285        );
286        let (placed, packed) = match ink.width == 0 || ink.height == 0 {
287            true => (
288                Placed {
289                    offset: ink.offset,
290                    ..Placed::default()
291                },
292                None,
293            ),
294            false => match self.shelves.pack(ink.width, ink.height) {
295                Some((x, y)) => {
296                    let side = self.shelves.side as f32;
297                    let scale = cut.scale();
298                    let placed = Placed {
299                        offset: ink.offset,
300                        area: Area::new(ink.width as f32 / scale, ink.height as f32 / scale),
301                        uv: [
302                            x as f32 / side,
303                            y as f32 / side,
304                            ink.width as f32 / side,
305                            ink.height as f32 / side,
306                        ],
307                    };
308                    (placed, Some(Packed { x, y, ink }))
309                }
310                None => {
311                    self.overflowed = true;
312                    (Placed::default(), None)
313                }
314            },
315        };
316        self.held.insert(cut, placed);
317        (placed, packed)
318    }
319
320    /// Whether a cut has found no room since the sheet was last cut again.
321    pub(crate) fn overflowed(&self) -> bool {
322        self.overflowed
323    }
324
325    /// Whether the sheet was found full: what the frame draws, cut again on its own, did not fit.
326    #[cfg(test)]
327    pub(crate) fn full(&self) -> bool {
328        self.full
329    }
330
331    /// Takes every cut off, for the sheet to be cut again from what is asked for next.
332    pub(crate) fn clear(&mut self) {
333        self.held.clear();
334        self.shelves.clear();
335        self.overflowed = false;
336    }
337
338    /// Says the sheet has been cut again from everything the frame draws, and nothing else.
339    ///
340    /// A cut that found no room then is one the frame's own cuts leave none for, and no cutting
341    /// again could make room for it. So that is the sheet full, and it is reported -- once, until a
342    /// time the frame's cuts do fit. What found no room stays held as nothing, and is asked for
343    /// again when something new runs out of room and the sheet is cut again for it.
344    pub(crate) fn settle(&mut self) {
345        match (self.overflowed, self.full) {
346            (true, false) => {
347                self.full = true;
348                error!(
349                    side = self.shelves.side,
350                    held = self.held.len(),
351                    "the glyph atlas is full: what this frame draws does not fit, and the \
352                     characters that found no room will not draw"
353                );
354            }
355            (true, true) => {}
356            (false, _) => self.full = false,
357        }
358        self.overflowed = false;
359    }
360
361    /// Takes every cut off and forgets the sheet was ever full: a sheet cut at a new density is a new
362    /// sheet.
363    pub(crate) fn forget(&mut self) {
364        self.clear();
365        self.full = false;
366    }
367}

369/// Where the next bitmap goes on a square sheet: shelves.
370///
371/// A row is filled left to right and the next one starts below the tallest bitmap in it. Glyphs of
372/// one size are all much of a height, so the waste is a line of it per size rather than per glyph,
373/// and a shelf packer needs no bookkeeping to be undone when what is on it is only ever taken off
374/// all at once.
375pub(crate) struct Shelves {
376    /// How wide and tall the sheet is.
377    side: u32,
378    /// The shelf being filled: where the next bitmap goes on it, and how tall it is.
379    x: u32,
380    y: u32,
381    shelf: u32,
382}
383
384impl Shelves {
385    pub(crate) fn new(side: u32) -> Self {
386        Self {
387            side,
388            x: 0,
389            y: 0,
390            shelf: 0,
391        }
392    }
393
394    /// Where a bitmap of this size goes, or `None` if there is no room left for it.
395    pub(crate) fn pack(&mut self, width: u32, height: u32) -> Option<(u32, u32)> {
396        if width > self.side || height > self.side {
397            return None;
398        }
399        if self.x + width > self.side {
400            self.y += self.shelf + GUTTER;
401            self.x = 0;
402            self.shelf = 0;
403        }
404        if self.y + height > self.side {
405            return None;
406        }
407        let at = (self.x, self.y);
408        self.x += width + GUTTER;
409        self.shelf = self.shelf.max(height);
410        Some(at)
411    }
412
413    /// Empties the sheet, so the next bitmap goes at its top-left corner again.
414    pub(crate) fn clear(&mut self) {
415        self.x = 0;
416        self.y = 0;
417        self.shelf = 0;
418    }
