1//! The text renderer: one entry in the stack per run, and one quad per glyph under it.
2//!
3//! # A run is one thing to the stack and many things here
4//!
5//! Every other renderer draws one instance per element, so its slots and its instances are the same
6//! list. A run is not: it is **one** entry in the one stack -- at one rank, under one clip, at one
7//! depth -- holding as many quads as it has characters. So this keeps two numberings. Slots are
8//! runs, sorted by rank, and are what [`Ash`](crate::ash::Ash) walks and cuts spans on. Glyphs are
9//! this renderer's own, laid out in slot order with each run's contiguous, so a span of runs is a
10//! range of glyphs and one draw covers it.
11//!
12//! That is what keeps the shared stack the size of the tree rather than the size of the text on it,
13//! and what keeps R6's total order over elements from having to say anything about characters.
14

35/// How many glyphs there is room for before the buffers are grown.
36const CAPACITY: u32 = 256;
37
38/// One glyph, in the form the GPU takes it.
39///
40/// Built here rather than in extraction, because two of its three parts are the backend's own: where
41/// the ink sits inside its cell and where it is on the sheet are both answers only the rasteriser
42/// has. What extraction supplies is the cell and the character.
43#[repr(C)]
44#[derive(Copy, Clone, Debug, PartialEq, Pod, Zeroable)]
45struct GlyphInstance {
46    /// The ink's box on the surface, in logical pixels.
47    section: Section,
48    /// What the run is filled with. Carried per glyph because that is what a vertex buffer is, and
49    /// because a per-character tint is a fill over part of the run's own index space.
50    color: Color,
51    /// The ink's rect on the sheet, normalised: origin then size.
52    sheet: [f32; 4],
53}
54
55/// The pipeline, the sheet, and every run the backend is holding.
56pub(crate) struct Texts {
57    pipeline: RenderPipeline,
58    corners: Buffer,
59    atlas: Atlas,
60    held: HashMap<Key, Held>,
61    /// Slot order: which run each slot is, and the rank and clip the stack is built from.
62    order: Vec<Key>,
63    ranks: Vec<ResolvedElevation>,
64    clips: Vec<Section>,
65    /// Which glyphs each slot covers.
66    blocks: Vec<Range<u32>>,
67    /// One depth per glyph, written from the run's own place in the stack.
68    depths: Vec<f32>,
69    glyphs: Buffer,
70    depth: Buffer,
71    capacity: u32,
72    total: u32,
73    /// Where one run's glyphs are built before they are taken. Kept for its capacity.
74    scratch: Vec<GlyphInstance>,
75    /// Whether the order itself changed, which moves every slot after the change.
76    resort: bool,
77    /// Whether anything the stack is built from changed: which slots there are, or what one is
78    /// clipped to.
79    disturbed: bool,
80    /// Slots whose glyphs changed while the order did not.
81    touched: Vec<u32>,
82}
83
84/// One run, and where its glyphs currently sit.
85struct Held {
86    glyphs: Vec<GlyphInstance>,
87    rank: ResolvedElevation,
88    clip: Section,
89    slot: u32,
90}

92impl Texts {
93    pub(crate) fn new(ginkgo: &Ginkgo) -> Self {
94        let device = ginkgo.device();
95        let atlas = Atlas::new(ginkgo);
96        let shader = device.create_shader_module(ShaderModuleDescriptor {
97            label: Some("glyph"),
98            source: ShaderSource::Wgsl(include_str!("glyph.wgsl").into()),
99        });
100        let layout = device.create_pipeline_layout(&PipelineLayoutDescriptor {
101            label: Some("glyph"),
102            bind_group_layouts: &[Some(ginkgo.viewport_layout()), Some(atlas.layout())],
103            immediate_size: 0,
104        });
105        let pipeline = device.create_render_pipeline(&RenderPipelineDescriptor {
106            label: Some("glyph"),
107            layout: Some(&layout),
108            vertex: VertexState {
109                module: &shader,
110                entry_point: Some("vertex_entry"),
111                compilation_options: Default::default(),
112                buffers: &[
113                    Some(VertexBufferLayout {
114                        array_stride: size_of::<[f32; 2]>() as u64,
115                        step_mode: VertexStepMode::Vertex,
116                        attributes: &wgpu::vertex_attr_array![0 => Float32x2],
117                    }),
118                    Some(VertexBufferLayout {
119                        array_stride: size_of::<GlyphInstance>() as u64,
120                        step_mode: VertexStepMode::Instance,
121                        attributes: &wgpu::vertex_attr_array![
122                            1 => Float32x4,
123                            2 => Float32x4,
124                            3 => Float32x4,
125                        ],
126                    }),
127                    Some(VertexBufferLayout {
128                        array_stride: size_of::<f32>() as u64,
129                        step_mode: VertexStepMode::Instance,
130                        attributes: &wgpu::vertex_attr_array![4 => Float32],
131                    }),
132                ],
133            },
134            primitive: Ginkgo::triangles(),
135            depth_stencil: Ginkgo::depth_state(),
136            multisample: Ginkgo::samples(),
137            fragment: Some(wgpu::FragmentState {
138                module: &shader,
139                entry_point: Some("fragment_entry"),
140                compilation_options: Default::default(),
141                targets: &ginkgo.blending(),
142            }),
143            multiview_mask: None,
144            cache: None,
145        });
146        Self {
147            pipeline,
148            corners: device.create_buffer_init(&BufferInitDescriptor {
149                label: Some("glyph-corners"),
150                contents: bytemuck::cast_slice(&CORNERS),
151                usage: BufferUsages::VERTEX,
152            }),
153            atlas,
154            held: HashMap::new(),
155            order: Vec::new(),
156            ranks: Vec::new(),
157            clips: Vec::new(),
158            blocks: Vec::new(),
159            depths: Vec::new(),
160            glyphs: buffer(device, size_of::<GlyphInstance>() as u32 * CAPACITY),
161            depth: buffer(device, size_of::<f32>() as u32 * CAPACITY),
162            capacity: CAPACITY,
163            total: 0,
164            scratch: Vec::new(),
165            resort: false,
166            disturbed: false,
167            touched: Vec::new(),
168        }
169    }

171    /// Takes one run: cuts whatever of it has not been cut, and holds the quads it turned out to be.
172    ///
173    /// A run that kept its glyph count and its rank is written where it already sits, which is the
174    /// ordinary case -- a run that moved, scrolled or was refilled has as many characters as it had.
175    /// A run whose length changed moves every run after it, so the order is built again.
176    pub(crate) fn write(&mut self, key: Key, run: &Run, fonts: &Fonts, ginkgo: &Ginkgo) {
177        let mut scratch = core::mem::take(&mut self.scratch);
178        scratch.clear();
179        for glyph in &run.glyphs {
180            let cut = Cut::new(run.font, run.size, ginkgo.scale(), glyph.character);
181            let placed = self.atlas.place(fonts, ginkgo, cut);
182            // A character the face draws nothing for takes no quad. It still advanced the wrap,
183            // which happened above this line and is not undone by there being nothing to draw.
184            if placed.area.width <= 0.0 || placed.area.height <= 0.0 {
185                continue;
186            }
187            scratch.push(GlyphInstance {
188                section: Section::new(
189                    snapped(glyph.cell.position.moved(placed.offset), ginkgo.scale()),
190                    placed.area,
191                ),
192                // The glyph's own, because a tint fills part of a run: what the run declares is
193                // what an untinted character resolved to, and extraction has already answered both.
194                color: glyph.color,
195                sheet: placed.uv,
196            });
197        }
198        match self.held.get_mut(&key) {
199            Some(held) => {
200                let recut = held.clip != run.clip;
201                held.clip = run.clip;
202                let settled = held.rank == run.rank && held.glyphs.len() == scratch.len();
203                held.rank = run.rank;
204                held.glyphs.clear();
205                held.glyphs.extend_from_slice(&scratch);
206                match settled {
207                    true => {
208                        self.touched.push(held.slot);
209                        if recut {
210                            self.clips[held.slot as usize] = run.clip;
211                            self.disturbed = true;
212                        }
213                    }
214                    false => self.resort = true,
215                }
216            }
217            None => {
218                self.held.insert(
219                    key,
220                    Held {
221                        glyphs: scratch.clone(),
222                        rank: run.rank,
223                        clip: run.clip,
224                        slot: 0,
225                    },
226                );
227                self.resort = true;
228            }
229        }
230        self.scratch = scratch;
231    }
232
233    /// Drops one run from what is held.
234    pub(crate) fn withdraw(&mut self, key: Key) {
235        if self.held.remove(&key).is_some() {
236            self.resort = true;
237        }
238    }

240    /// Puts what changed onto the GPU.
241    pub(crate) fn flush(&mut self, device: &Device, queue: &Queue) {
242        if self.resort {
243            self.reorder(device, queue);
244            return;
245        }
246        self.touched.sort_unstable();
247        self.touched.dedup();
248        for slot in self.touched.drain(..) {
249            let block = self.blocks[slot as usize].clone();
250            if block.is_empty() {
251                continue;
252            }
253            let held = &self.held[&self.order[slot as usize]];
254            queue.write_buffer(
255                &self.glyphs,
256                block.start as BufferAddress * size_of::<GlyphInstance>() as BufferAddress,
257                bytemuck::cast_slice(&held.glyphs),
258            );
259        }
260    }
261
262    /// Rebuilds the order back to front and rewrites every glyph.
263    ///
264    /// One write per run rather than one for the whole buffer: the runs are already the unit the
265    /// glyphs are held in, and a flat mirror of them kept only to be uploaded in one call would be
266    /// a second copy of every glyph on the page.
267    fn reorder(&mut self, device: &Device, queue: &Queue) {
268        self.resort = false;
269        self.disturbed = true;
270        self.touched.clear();
271        let mut order = self
272            .held
273            .iter()
274            .map(|(key, held)| (held.rank, *key))
275            .collect::<Vec<_>>();
276        // A rank orders back to front and is total, so this is one sort with no tie left in it.
277        order.sort_unstable();
278        self.order.clear();
279        self.ranks.clear();
280        self.clips.clear();
281        self.blocks.clear();
282        let mut at = 0;
283        for (slot, (rank, key)) in order.into_iter().enumerate() {
284            let held = self.held.get_mut(&key).expect("held");
285            held.slot = slot as u32;
286            let count = held.glyphs.len() as u32;
287            self.order.push(key);
288            self.ranks.push(rank);
289            self.clips.push(held.clip);
290            self.blocks.push(at..at + count);
291            at += count;
292        }
293        self.total = at;
294        self.depths.clear();
295        self.depths.resize(at as usize, 0.0);
296        if at > self.capacity {
297            self.capacity = at.next_power_of_two();
298            self.glyphs = buffer(device, size_of::<GlyphInstance>() as u32 * self.capacity);
299            self.depth = buffer(device, size_of::<f32>() as u32 * self.capacity);
300        }
301        for slot in 0..self.order.len() {
302            let held = &self.held[&self.order[slot]];
303            if held.glyphs.is_empty() {
304                continue;
305            }
306            queue.write_buffer(
307                &self.glyphs,
308                self.blocks[slot].start as BufferAddress
309                    * size_of::<GlyphInstance>() as BufferAddress,
310                bytemuck::cast_slice(&held.glyphs),
311            );
312        }
313    }

330    /// Where in the whole stack the run in `slot` sits, as a depth.
331    ///
332    /// Every glyph of it takes that one depth, because the run is one entry in the stack and its
333    /// glyphs are all at that one place in it.
334    pub(crate) fn set_depth(&mut self, slot: u32, depth: f32) {
335        let block = self.blocks[slot as usize].clone();
336        self.depths[block.start as usize..block.end as usize].fill(depth);
337    }
338
339    /// Puts the depths on the GPU, once every run has one.
340    pub(crate) fn flush_depths(&self, queue: &Queue) {
341        if self.depths.is_empty() {
342            return;
343        }
344        queue.write_buffer(&self.depth, 0, bytemuck::cast_slice(&self.depths));
345    }

347    /// Draws one run of what is held.
348    ///
349    /// The range is a span of the stack and so is in *runs*. The glyphs of a run are contiguous and
350    /// the runs of a span are contiguous, so the glyphs of a span are one range and one draw.
351    pub(crate) fn draw(&self, pass: &mut RenderPass<'_>, span: Range<u32>) {
352        let from = self.blocks[span.start as usize].start;
353        let to = self.blocks[span.end as usize - 1].end;
354        // A span of runs that are all empty -- every character of them a space, or none of them
355        // drawn by the face -- is a draw of nothing.
356        if from == to {
357            return;
358        }
359        pass.set_pipeline(&self.pipeline);
360        pass.set_bind_group(1, self.atlas.binding(), &[]);
361        pass.set_vertex_buffer(0, self.corners.slice(..));
362        pass.set_vertex_buffer(1, self.glyphs.slice(..));
363        pass.set_vertex_buffer(2, self.depth.slice(..));
364        pass.draw(0..CORNERS.len() as u32, from..to);
365    }

367    /// How many glyphs are held. Reported to the trace.
368    pub(crate) fn len(&self) -> u32 {
369        self.total
370    }

373/// `at` moved to the nearest whole device pixel.
374///
375/// A cut is an exact number of device pixels on the sheet, and the quad it is drawn on is that many
376/// wide and tall. Landed on a whole pixel, the two are one to one: every sample falls on a texel
377/// centre and the glyph is the bitmap. Landed on a fraction of one -- and it always is, because a
378/// baseline is a face's own fractional metric and a run's box is wherever the layout put it -- every
379/// sample falls between texels, so the glyph is filtered against the clearance packed around it and
380/// its outermost rows fade. That reads as a glyph trimmed along its edges rather than as a glyph
381/// half a pixel low, which is what it is.
382///
383/// Snapping the whole ink rather than the baseline alone, because both halves of where a glyph
384/// landed are fractional and it is their sum that has to fall on the grid. A line of a run shares
385/// one baseline and one row of cells, so a line snaps as a line and stays level.
386fn snapped(at: Position, scale: f32) -> Position {
387    Position::new(
388        (at.x * scale).round() / scale,
389        (at.y * scale).round() / scale,
390    )
391}

393fn buffer(device: &Device, size: u32) -> Buffer {
394    device.create_buffer(&BufferDescriptor {
395        label: Some("glyph"),
396        size: size as BufferAddress,
397        usage: BufferUsages::VERTEX | BufferUsages::COPY_DST,
398        mapped_at_creation: false,
399    })
