161    /// Reads one property of an element, or `None` if it has withered, has not been grown yet, or
162    /// does not carry that property.
163    pub fn tap(&self, leaf: Leaf, vein: Vein) -> Option<Sap> {
164        if !self.tree.is_live(leaf) {
165            return None;
166        }
167        Some(match vein {
168            Vein::Branches => Sap::Leaves(self.tree.branches(leaf)),
169            Vein::Trunk => Sap::Leaf(self.tree.trunk(leaf)),
170            Vein::Placed => Sap::Section(self.elements.placed(leaf)),
171            Vein::Drawn => Sap::Section(self.elements.drawn(leaf)),
172            Vein::Anchor => Sap::Leaf(self.tree.anchor(leaf)),
173            Vein::Elevation => Sap::Elevation(self.tree.elevation(leaf)),
174            // A field is filled as its value is, since a fill written to it goes to the run.
175            Vein::Color => Sap::Color(match self.tree.parts(leaf) {
176                Some(parts) => self.tree.fill(parts.run)?,
177                None => self.tree.fill(leaf)?,
178            }),
179            Vein::Rounding => Sap::Rounding(self.tree.rounding(leaf)?),
180            Vein::Outline => {
181                let outline = self.tree.outline(leaf)?;
182                Sap::Outline(outline.width, outline.fill)
183            }
184            Vein::Ends => {
185                let stretched = self.elements.stretched(leaf)?;
186                Sap::Ends(stretched.from, stretched.to)
187            }
188            Vein::Weight => Sap::Weight(self.tree.stroke(leaf)?.weight),
189            Vein::Cap => Sap::Cap(self.tree.line_pigment(leaf)?.cap),
190            Vein::Shape => Sap::Shape(self.tree.shape(leaf)?),
191            Vein::Mark => Sap::Mark(self.tree.icon_pigment(leaf)?.field),
192            Vein::Picture => Sap::Picture(self.tree.image_pigment(leaf)?.plate),
193            Vein::Fit => Sap::Fit(self.tree.image_pigment(leaf)?.fit),
194            // A field says what its run says. Every verb and every read is addressed to the field,
195            // so what it is made of is never a name an app has to hold.
196            Vein::Text => Sap::Text(match self.tree.parts(leaf) {
197                Some(parts) => self.tree.lettering(parts.run)?.to_string(),
198                None => self.tree.lettering(leaf)?.to_string(),
199            }),
200            Vein::Selection => {
201                self.tree.parts(leaf)?;
202                Sap::Selection(self.tree.editing(leaf).span())
203            }
204            Vein::Visible => Sap::Visible(self.tree.visible(leaf).0),
205            Vein::Opacity => Sap::Opacity(self.tree.opacity(leaf).0),
206            Vein::Disabled => Sap::Disabled(self.tree.disabled(leaf).0),
207            Vein::Shift => Sap::Position(self.tree.shift(leaf)),
208            // The three a region has and nothing else does. `None` where the element does not
209            // scroll, which is the reading that keeps "an axis that was not declared has no extent"
210            // true of the whole element as well as of one of its axes.
211            Vein::Offset => {
212                self.tree.scrolls(leaf)?;
213                Sap::Position(self.tree.offset(leaf))
214            }
215            Vein::Extent => {
216                self.tree.scrolls(leaf)?;
217                Sap::Area(self.tree.extent(leaf))
218            }
219            Vein::Progress => {
220                self.tree.scrolls(leaf)?;
221                let (offset, extent) = (self.tree.offset(leaf), self.tree.extent(leaf));
222                let own = self.elements.placed(leaf).area;
223                Sap::Progress(Position::new(
224                    progress(offset, extent, own, Axis::Horizontal),
225                    progress(offset, extent, own, Axis::Vertical),
226                ))
227            }
228        })
229    }

231    /// Registers a font and hands back the name elements compose in.
232    ///
233    /// Takes the bytes, or an [`Origin`] to read them from -- and hands back the name either way, at
234    /// once. A face that has yet to arrive is composed in the bundled one until it does, so a page
235    /// laid out in [`letters`](crate::Source::letters) is laid out sensibly from the first frame and
236    /// reflows when the real face lands. [`loaded`](crate::Pollen::loaded) is how an app waits for
237    /// that instead.
238    ///
239    /// ```no_run
240    /// # use foliage::{Grove, Origin};
241    /// # fn f(grove: &mut Grove, bundled: &[u8]) {
242    /// // `bundled` is what `include_bytes!("assets/mono.ttf")` produced.
243    /// let here = grove.font(bundled);
244    /// # #[cfg(not(target_family = "wasm"))]
245    /// let read = grove.font(Origin::path("assets/mono.ttf"));
246    /// # }
247    /// ```
248    ///
249    /// # Panics
250    ///
251    /// If bytes that were given outright are not a monospaced font. Every measurement foliage makes
252    /// is a count of character cells -- [`letters`](crate::Source::letters), a letter-pitched track,
253    /// max-content width, wrapping -- so a proportional font does not degrade, it silently puts
254    /// every column address somewhere it does not belong.
255    ///
256    /// Bytes that were *read* are refused rather than panicked on, and reported as
257    /// [`missing`](crate::Pollen::missing): what a path or a URL turned out to hold is not something
258    /// the program stated.
259    pub fn font(&mut self, bytes: impl Into<Bytes>) -> Font {
260        let font = self.naming.face();
261        match bytes.into().0 {
262            Supply::Held(bytes) => self.fonts.register(font, &bytes),
263            Supply::At(origin) => self.retrieve(Destination::Face(font), origin),
264        }
265        font
266    }

268    /// Registers a mark and hands back the name elements draw it by.
269    ///
270    /// The same registration [`Foliage::icon`](crate::Foliage::icon) is at boot, available at any
271    /// frame -- which is what lets an app that takes root inside the first frame register its marks
272    /// there rather than having to thread handles in from outside.
273    ///
274    /// `field` is a multi-channel signed distance field: `side` by `side` texels of RGBA, row-major.
275    /// `range` is how many texels the baked distance spread covers. Both are stated by the app
276    /// because they are facts about how the field was baked, and a fetched one cannot be asked.
277    ///
278    /// An element drawing a mark that has yet to arrive occupies its box and draws nothing, and
279    /// appears in the frame it lands.
280    ///
281    /// # Panics
282    ///
283    /// If bytes that were given outright are smaller than `side` by `side` texels of RGBA. Bytes
284    /// that were read are refused rather than panicked on, and reported as
285    /// [`missing`](crate::Pollen::missing).
286    pub fn icon(&mut self, field: impl Into<Bytes>, side: u32, range: f32) -> Field {
287        let name = self.naming.mark();
288        match field.into().0 {
289            Supply::Held(bytes) => self.fields.register(name, &bytes, side, range),
290            Supply::At(origin) => self.retrieve(Destination::Mark(name, side, range), origin),
291        }
292        name
293    }

312    /// Registers a picture and hands back the name elements draw it by.
313    ///
314    /// PNG or JPEG, decoded here. The format is read from the bytes and the size is what the decode
315    /// says, so neither is stated: a name and a path can both be wrong about what a file holds, and
316    /// the file cannot be.
317    ///
318    /// Pixels an app made itself are [`pixels`](crate::Grow::pixels), which states a size because
319    /// there is nothing to read one from. An element drawing a picture that has yet to arrive
320    /// occupies its box and draws nothing, and appears in the frame it lands.
321    pub fn image(&mut self, bytes: impl Into<Bytes>) -> Plate {
322        let plate = self.naming.plate();
323        match bytes.into().0 {
324            Supply::Held(bytes) => {
325                if let Err(refused) = self.plates.decoded(plate, &bytes) {
326                    tracing::warn!(plate = plate.0, reason = refused, "asset missing");
327                    self.drift.missing.insert(plate.into());
328                }
329            }
330            Supply::At(origin) => self.retrieve(Destination::Picture(plate), origin),
331        }
332        plate
333    }

406    /// The cursor rests somewhere else, or nowhere: owes a frame if the app asked to be woken for
407    /// any move, or if what it rests on is not what the last frame hovered -- read off the stack the
408    /// last frame drew, which is what the frame it owes will read it from too.
409    pub(crate) fn rest(&mut self, cursor: Option<Position>) {
410        self.cursor = cursor;
411        let rests = cursor.and_then(|at| crate::interaction::reaches(&self.stack, at));
412        self.roused |= self.rouses || rests != self.hovered;
413    }

415    /// Whether a frame is owed on the engine's own account: something is waiting to be drained,
416    /// moved, reported, or noticed.
417    ///
418    /// The loop's question less the clock -- what an app asked for at a time it chose is the
419    /// loop's to wait on, because only the loop has a wall to read it against. The greenhouse asks
420    /// the same question to know when a frame has settled.
421    pub(crate) fn owed(&self) -> bool {
422        // The tree has not been grown yet: `take_root` runs inside the first frame.
423        self.frames == 0
424            || self.again
425            || self.pending_resize.is_some()
426            // Something arrived from off the frame. Usually an op, which the clause below sees on
427            // its own; but a key the hidden input captured is nowhere this can look, and the rouse
428            // is the only word that it is there.
429            || self.wake.pending()
430            || !self.queue.is_empty()
431            || !self.incoming.pending.is_empty()
432            || !self.aspen.idle()
433            // A press that is not moving becomes a hold on its own, with nothing arriving to say
434            // so. Idling under a finger is how that duration would pass unremarked, so the frames
435            // that would notice it are owed until the gesture is past being able to become one.
436            || self.incoming.awaiting_hold()
437            // A coasting region is pending work in its own right: it is not a tween, and there is
438            // nothing on the app's side asking for the frames it needs. So frames keep running
439            // until it settles, and stop afterwards.
440            || !self.coasting.idle()
441            // Steps 4 through 7 emit into the drift and step 3 of the *next* frame is where an app
442            // is handed it, so a report that has been made and not yet delivered owes the frame
443            // that delivers it. Without this the loop could idle holding one.
444            || self.drift.pending()
445            // The cursor moved, and the app asked to be woken for that.
446            || self.roused
447    }

510    /// Asks for another frame, `millis` from now rather than at once.
511    ///
512    /// For motion an app drives itself that does not need every frame the display can show: a
513    /// light drifting across a picture, a clock's second hand, a meter that settles. Asked each
514    /// frame for as long as it is moving, it holds the app to that rate -- and while nothing else is
515    /// owed, the engine sleeps in between rather than drawing frames nothing changed in. A frame
516    /// owed sooner for any other reason runs sooner, and the next one asked for is counted from
517    /// that.
518    ///
519    /// Asked more than once in a frame, the soonest is the one kept. [`again`](Grove::again) is
520    /// this at nothing.
521    pub fn again_in(&mut self, millis: u64) {
522        let after = Duration::from_millis(millis);
523        self.again_in = Some(self.again_in.map_or(after, |asked| asked.min(after)));
524    }
