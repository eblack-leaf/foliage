136    /// Reads one property of an element, or `None` if it has withered, has not been grown yet, or
137    /// does not carry that property.
138    pub fn tap(&self, leaf: Leaf, vein: Vein) -> Option<Sap> {
139        if !self.tree.is_live(leaf) {
140            return None;
141        }
142        Some(match vein {
143            Vein::Branches => Sap::Leaves(self.tree.branches(leaf)),
144            Vein::Trunk => Sap::Leaf(self.tree.trunk(leaf)),
145            Vein::Placed => Sap::Section(self.tree.placed(leaf)),
146            Vein::Drawn => Sap::Section(self.tree.drawn(leaf)),
147            Vein::Anchor => Sap::Leaf(self.tree.anchor(leaf)),
148            Vein::Elevation => Sap::Elevation(self.tree.elevation(leaf)),
149            Vein::Color => Sap::Color(self.tree.fill(leaf)?),
150            Vein::Rounding => Sap::Rounding(self.tree.rounding(leaf)?),
151            Vein::Ends => {
152                let stretched = self.tree.stretched(leaf)?;
153                Sap::Ends(stretched.from, stretched.to)
154            }
155            Vein::Weight => Sap::Weight(self.tree.stroke(leaf)?.weight),
156            Vein::Cap => Sap::Cap(self.tree.line_pigment(leaf)?.cap),
157            Vein::Shape => Sap::Shape(self.tree.shape(leaf)?),
158            Vein::Mark => Sap::Mark(self.tree.icon_pigment(leaf)?.field),
159            Vein::Picture => Sap::Picture(self.tree.image_pigment(leaf)?.plate),
160            Vein::Fit => Sap::Fit(self.tree.image_pigment(leaf)?.fit),
161            // A field says what its run says. Every verb and every read is addressed to the field,
162            // so what it is made of is never a name an app has to hold.
163            Vein::Text => Sap::Text(match self.tree.parts(leaf) {
164                Some(parts) => self.tree.lettering(parts.run)?.to_string(),
165                None => self.tree.lettering(leaf)?.to_string(),
166            }),
167            Vein::Selection => {
168                self.tree.parts(leaf)?;
169                Sap::Selection(self.tree.editing(leaf).span())
170            }
171            Vein::Visible => Sap::Visible(self.tree.visible(leaf).0),
172            Vein::Opacity => Sap::Opacity(self.tree.opacity(leaf).0),
173            Vein::Disabled => Sap::Disabled(self.tree.disabled(leaf).0),
174            // The three a region has and nothing else does. `None` where the element does not
175            // scroll, which is the reading that keeps "an axis that was not declared has no extent"
176            // true of the whole element as well as of one of its axes.
177            Vein::Offset => {
178                self.tree.scrolls(leaf)?;
179                Sap::Position(self.tree.offset(leaf))
180            }
181            Vein::Extent => {
182                self.tree.scrolls(leaf)?;
183                Sap::Area(self.tree.extent(leaf))
184            }
185            Vein::Progress => {
186                self.tree.scrolls(leaf)?;
187                let (offset, extent) = (self.tree.offset(leaf), self.tree.extent(leaf));
188                let own = self.tree.placed(leaf).area;
189                Sap::Progress(Position::new(
190                    progress(offset, extent, own, Axis::Horizontal),
191                    progress(offset, extent, own, Axis::Vertical),
192                ))
193            }
194        })
195    }

197    /// Registers a font and hands back the name elements compose in.
198    ///
199    /// Takes the bytes, or an [`Origin`] to read them from -- and hands back the name either way, at
200    /// once. A face that has yet to arrive is composed in the bundled one until it does, so a page
201    /// laid out in [`letters`](crate::Source::letters) is laid out sensibly from the first frame and
202    /// reflows when the real face lands. [`loaded`](crate::Pollen::loaded) is how an app waits for
203    /// that instead.
204    ///
205    /// ```no_run
206    /// # use foliage::{Grove, Origin};
207    /// # fn f(grove: &mut Grove, bundled: &[u8]) {
208    /// // `bundled` is what `include_bytes!("assets/mono.ttf")` produced.
209    /// let here = grove.font(bundled);
210    /// # #[cfg(not(target_family = "wasm"))]
211    /// let read = grove.font(Origin::path("assets/mono.ttf"));
212    /// # }
213    /// ```
214    ///
215    /// # Panics
216    ///
217    /// If bytes that were given outright are not a monospaced font. Every measurement foliage makes
218    /// is a count of character cells -- [`letters`](crate::Source::letters), a letter-pitched track,
219    /// max-content width, wrapping -- so a proportional font does not degrade, it silently puts
220    /// every column address somewhere it does not belong.
221    ///
222    /// Bytes that were *read* are refused rather than panicked on, and reported as
223    /// [`missing`](crate::Pollen::missing): what a path or a URL turned out to hold is not something
224    /// the program stated.
225    pub fn font(&mut self, bytes: impl Into<Bytes>) -> Font {
226        let font = self.naming.face();
227        match bytes.into().0 {
228            Supply::Held(bytes) => self.fonts.register(font, &bytes),
229            Supply::At(origin) => self.retrieve(Destination::Face(font), origin),
230        }
231        font
232    }

234    /// Registers a mark and hands back the name elements draw it by.
235    ///
236    /// The same registration [`Foliage::icon`](crate::Foliage::icon) is at boot, available at any
237    /// frame -- which is what lets an app that takes root inside the first frame register its marks
238    /// there rather than having to thread handles in from outside.
239    ///
240    /// `field` is a multi-channel signed distance field: `side` by `side` texels of RGBA, row-major.
241    /// `range` is how many texels the baked distance spread covers. Both are stated by the app
242    /// because they are facts about how the field was baked, and a fetched one cannot be asked.
243    ///
244    /// An element drawing a mark that has yet to arrive occupies its box and draws nothing, and
245    /// appears in the frame it lands.
246    ///
247    /// # Panics
248    ///
249    /// If bytes that were given outright are smaller than `side` by `side` texels of RGBA. Bytes
250    /// that were read are refused rather than panicked on, and reported as
251    /// [`missing`](crate::Pollen::missing).
252    pub fn icon(&mut self, field: impl Into<Bytes>, side: u32, range: f32) -> Field {
253        let name = self.naming.mark();
254        match field.into().0 {
255            Supply::Held(bytes) => self.fields.register(name, &bytes, side, range),
256            Supply::At(origin) => self.retrieve(Destination::Mark(name, side, range), origin),
257        }
258        name
259    }

261    /// Registers a picture and hands back the name elements draw it by.
262    ///
263    /// PNG or JPEG, decoded here. The format is read from the bytes and the size is what the decode
264    /// says, so neither is stated: a name and a path can both be wrong about what a file holds, and
265    /// the file cannot be.
266    ///
267    /// Pixels an app made itself are [`pixels`](crate::Grow::pixels), which states a size because
268    /// there is nothing to read one from. An element drawing a picture that has yet to arrive
269    /// occupies its box and draws nothing, and appears in the frame it lands.
270    pub fn image(&mut self, bytes: impl Into<Bytes>) -> Plate {
271        let plate = self.naming.plate();
272        match bytes.into().0 {
273            Supply::Held(bytes) => {
274                if let Err(refused) = self.plates.decoded(plate, &bytes) {
275                    tracing::warn!(plate = plate.0, reason = refused, "asset missing");
276                    self.drift.missing.insert(plate.into());
277                }
278            }
279            Supply::At(origin) => self.retrieve(Destination::Picture(plate), origin),
280        }
281        plate
282    }

289    /// Opens the platform edges, once, at boot.
290    ///
291    /// Called by `photosynthesize` after the [`Wake`] is installed and by nothing else -- so the
292    /// headless suite, which runs frames by hand, reaches none of them. That is what keeps a test
293    /// off the clipboard, the keyboard and the browser of whoever is running it, and it is the same

302    /// A handle on the engine that can be carried off the frame.

304    /// Everything an app writes here it can write from a thread, a promise or a callback through
