1//! Where the device is: the latest fix the platform gave, read by an app whenever it likes.
2//!
3//! On Android the activity `foliage-android init` writes for an app that asks for a location
4//! permission listens to every provider the device has on, and hands each fix here through
5//! [`Java_dev_foliage_Locator_fix`]. Each one that is no worse than the one held -- see
6//! [`better`] -- is kept, and wakes the loop, so a frame runs to read it. Everywhere else nothing
7//! is ever handed in, and [`location`] says there is none.
8
9use std::sync::Mutex;
10#[cfg(not(target_family = "wasm"))]
11use std::sync::OnceLock;
12
13use crate::queue::Wake;
14
15/// Where the device was, as one provider said.
16#[derive(Copy, Clone, Debug, PartialEq)]
17pub struct Fix {
18    /// North of the equator, in degrees.
19    pub lat: f64,
20    /// East of Greenwich, in degrees.
21    pub lon: f64,
22    /// How far off it may be, in metres, as the provider reckons it: the radius it is two thirds
23    /// likely to be within. None where the provider said nothing.
24    pub accuracy: Option<f32>,
25    /// Which way it was moving, in degrees clockwise from north. None where it was not.
26    pub bearing: Option<f32>,
27    /// When it was taken, in milliseconds since the Unix epoch.
28    pub time: i64,
29}
30
31/// How much newer a fix has to be to stand over a more accurate one: a GPS fix is not given up
32/// for a network one every second, but one from a minute ago is not where the device is now.
33const STALE_MS: i64 = 10_000;
34
35/// The fix held, and the wake that says a new one came.
36static HELD: Mutex<Option<Fix>> = Mutex::new(None);
37#[cfg(not(target_family = "wasm"))]
38static WAKE: OnceLock<Wake> = OnceLock::new();
39// On the web a wake is not `Sync` -- see `queue::Rouse` -- so it can't be a static. There is one
40// thread there, so that thread's own is the same thing.
41#[cfg(target_family = "wasm")]
42thread_local! {
43    static WAKE: std::cell::OnceCell<Wake> = const { std::cell::OnceCell::new() };
44}
45
46/// The latest fix the platform gave, or none where it has given none -- every desktop, and an
47/// Android app that asks for no location permission or was refused it.
48pub fn location() -> Option<Fix> {
49    *HELD.lock().unwrap_or_else(|e| e.into_inner())
50}
51
52/// Hands the loop's wake over, so a fix arriving off the frame is answered by one.
53pub(crate) fn attach(wake: &Wake) {
54    #[cfg(not(target_family = "wasm"))]
55    let _ = WAKE.set(wake.clone());
56    #[cfg(target_family = "wasm")]
57    WAKE.with(|held| {
58        let _ = held.set(wake.clone());
59    });
60}
61
62/// Keeps `fix` if it is no worse than what is held, and wakes the loop for it.
63#[cfg_attr(not(target_os = "android"), allow(dead_code))]
64pub(crate) fn hand(fix: Fix) {
65    let mut held = HELD.lock().unwrap_or_else(|e| e.into_inner());
66    if held.is_some_and(|held| !better(fix, held)) {
67        return;
68    }
69    *held = Some(fix);
70    drop(held);
71    #[cfg(not(target_family = "wasm"))]
72    let wake = WAKE.get().cloned();
73    #[cfg(target_family = "wasm")]
74    let wake = WAKE.with(|held| held.get().cloned());
75    if let Some(wake) = wake {
76        wake.rouse();
77    }
78}
79
80/// Whether `new` should stand over `held`: it is at least as accurate, or `held` is old enough not
81/// to be where the device is now. A fix that says no accuracy is taken as the worst.
82fn better(new: Fix, held: Fix) -> bool {
83    let off = |fix: Fix| fix.accuracy.unwrap_or(f32::INFINITY);
84    new.time >= held.time && (off(new) <= off(held) || new.time - held.time >= STALE_MS)
85}
86
87/// The activity's native method: one fix, as Android's `Location` said it. An accuracy below zero
88/// is none, and a bearing that is not a number is none.
89#[cfg(target_os = "android")]
90#[unsafe(no_mangle)]
91pub extern "system" fn Java_dev_foliage_Locator_fix(
92    _env: *mut core::ffi::c_void,
93    _class: *mut core::ffi::c_void,
94    lat: f64,
95    lon: f64,
96    accuracy: f32,
97    bearing: f32,
98    time: i64,
99) {
100    hand(Fix {
101        lat,
102        lon,
103        accuracy: (accuracy >= 0.0).then_some(accuracy),
104        bearing: (!bearing.is_nan()).then_some(bearing),
105        time,
106    });
107}

109#[cfg(test)]
110mod tests {
111    use super::*;
112
113    fn fix(accuracy: Option<f32>, time: i64) -> Fix {
114        Fix {
115            lat: 49.61,
116            lon: 6.13,
117            accuracy,
118            bearing: None,
119            time,
120        }
121    }
122
123    #[test]
124    fn a_worse_fix_does_not_stand_over_a_better_one_until_that_one_is_stale() {
125        let gps = fix(Some(5.0), 1_000);
126        assert!(!better(fix(Some(900.0), 2_000), gps));
127        assert!(better(fix(Some(4.0), 2_000), gps));
128        assert!(better(fix(Some(900.0), 1_000 + STALE_MS), gps));
129        assert!(!better(fix(None, 2_000), gps));
130        assert!(
131            !better(fix(Some(1.0), 500), gps),
132            "an older fix never stands over a newer"
133        );
134    }
