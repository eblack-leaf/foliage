15/// The thinnest stroke that means anything, in logical pixels.
16///
17/// Zero and negative weights have no drawing to do, and clamping once here is cheaper than every
18/// callsite checking. The shader feathers a sub-pixel line rather than dropping it, so one logical
19/// pixel is a hairline on any display rather than a line that disappears on some of them.
20pub const HAIRLINE: f32 = 1.0;
21
22/// A straight segment of fixed weight between two points.
23///
24/// Placed by its ends rather than by a box: a rule, a connector and a chart's series are all two
25/// positions and a thickness, and a rectangle is the wrong shape to say that in. The two ends are
26/// [`Point`]s in the ordinary placement grammar, so an end can sit on a grid track, half way across
27/// its trunk, or at an anchor's edge -- and the line follows when any of those move.
28///
29/// ```no_run
30/// # use foliage::{Line, Palette, Place, Point, Source};
31/// Line::new()
32///     .weight(2.0)
33///     .color(Palette::Muted)
34///     .between(Point::new(0.px(), 0.px()), Point::new(100.pct(), 0.px()));
35/// ```
36///
37/// A stroke whose ends differ by breakpoint states a [`Trace`], which is the same pair per
38/// breakpoint in the chain a [`Location`](crate::Location) is written in:
39///
40/// ```no_run
41/// # use foliage::{Line, Point, Source, Trace};
42/// Line::new().trace(
43///     Trace::new()
44///         .xs(Point::new(0.px(), 50.pct()), Point::new(100.pct(), 50.pct()))
45///         .md(Point::new(50.pct(), 0.px()), Point::new(50.pct(), 100.pct())),
46/// );
47/// ```
48///
49/// Its box is the rectangle around the two ends, grown by half its weight on every side -- so a
50/// horizontal rule, whose ends share a `y`, still has a box to be clipped, ranked and hit-tested by.
51#[derive(Clone, Debug, Default)]
52pub struct Line {
53    pub(crate) placement: Placement,
54    pub(crate) fill: Fill,
55    pub(crate) cap: Cap,
56}
57
58impl Line {
59    /// A muted hairline, from its trunk's top-left corner to itself.
60    ///
61    /// [`between`](Line::between) is what gives it somewhere to go: a line that says nothing is a
62    /// line of no length rather than one filling its trunk, because two ends have no reading that
63    /// corresponds to "the whole of the parent's box".
64    pub fn new() -> Self {
65        Self {
66            placement: Placement::default(),
67            fill: Fill::Role(Palette::Muted),
68            cap: Cap::default(),
69        }
70    }
71
72    /// Where the two ends are, at every breakpoint.
73    ///
74    /// The ordinary case, and [`trace`](Line::trace) with one link: a stroke that runs the same way
75    /// at every width says so once.
76    pub fn between(self, from: Point, to: Point) -> Self {
77        self.trace(Trace::new().xs(from, to))
78    }
79
80    /// Where the two ends are, per breakpoint.
81    ///
82    /// What [`at`](crate::Boxed::at) is to a box. Separate from it rather than a second form of it,
83    /// because an element has a box or it has ends, and a stroke handed a [`Location`](crate::Location)
84    /// would have nothing to read it with.
85    pub fn trace(mut self, trace: Trace) -> Self {
86        self.placement.traced = Some(trace);
87        self
88    }
89
90    /// How thick the stroke is, in logical pixels. Clamped to [`HAIRLINE`].
91    ///
92    /// Placement rather than decoration, because it is what the line's box is grown by: two ends on
93    /// one line describe a rectangle of no height until the weight says otherwise.
94    pub fn weight(mut self, weight: f32) -> Self {
95        self.placement.stroke = Some(Stroke::new(weight));
96        self
97    }
98
99    /// What the stroke is filled with: a [`Palette`] role, or a [`Color`] stated outright.
100    /// Undeclared, it is [`Palette::Muted`] -- the role a rule and a division are drawn in.
101    pub fn color(mut self, fill: impl Into<Fill>) -> Self {
102        self.fill = fill.into();
103        self
104    }
105
106    /// How the stroke's two ends are finished. Undeclared, they are [`Cap::Butt`].
107    pub fn cap(mut self, cap: Cap) -> Self {
108        self.cap = cap;
109        self
110    }
111}
112
113/// How a stroke's ends are finished.
114///
115/// A stroke is a rectangle, so two of them meeting at an angle leave a wedge open on the outside of
116/// the turn. [`Round`](Cap::Round) covers it: each end is a half-disc of half the weight centred on
117/// the point, so two strokes sharing an end put one disc of exactly that radius over the wedge
118/// between them.
119#[derive(Copy, Clone, Debug, Default, PartialEq, Eq)]
120pub enum Cap {
121    /// Square, ending exactly on the point. What a rule and a division want: an end that reached
122    /// half a weight past where it was told to stop would not line up with anything.
123    #[default]
124    Butt,
125    /// Half-round, reaching half a weight past the point. What a path wants, and what a single
126    /// stroke wants when it is a mark rather than a boundary.
127    Round,
128}
129
130impl Places for Line {
131    fn placement(&mut self) -> &mut Placement {
132        &mut self.placement
133    }
134}

136impl Buds for Line {
137    fn bud(mut self, at: Caller) -> Bud {
138        // A line always has a weight, because its box is derived from one, and always has a trace,
139        // because that is what says it is placed by its ends: one that stated neither is a stroke
140        // of no length at its trunk's corner, and not a box filling the trunk. Everything else
141        // about a placement stays absent when it was not stated.
142        self.placement.stroke.get_or_insert_default();
143        self.placement.traced.get_or_insert_default();
144        Bud {
145            chlorophyll: Chlorophyll::Line,
146            pigment: Some(Pigment::Line(LinePigment {
147                fill: self.fill,
148                cap: self.cap,
149            })),
150            placement: self.placement,
151            at,
152            ..Bud::bare()
153        }
154    }
155}

157/// Where the layout put a stroke's two ends, in the space [`Placed`](crate::rowan::Placed) is in.
158///
159/// Settled by R2b beside the box, and on the same terms: what R2b answered, before any scrolling
160/// ancestor moves it. A stroke in motion is settled at the blend, so this is also what a retarget
161/// snapshots -- a box in motion offers `Placed` for the same purpose.
162///
163/// The box cannot stand in for it. A box is the rectangle around the two ends grown by half the
164/// weight, and a rectangle has two diagonals: which of them the stroke runs along is a fact about
165/// the ends that the rectangle does not carry, and it depends on where the ends resolved rather
166/// than on how they were written.
167#[derive(Component, Copy, Clone, Debug, Default, PartialEq)]
168pub(crate) struct Spanned(pub(crate) Stretched);
169
170/// Where a stroke's two ends actually landed, in the space [`Drawn`](crate::rowan::Drawn) is in.
171///
172/// [`Spanned`] less every scrolling ancestor's accumulated offset, written by R4 beside `Drawn` for
173/// every stroke on every frame -- a region scrolling under a stroke nothing wrote to moves its ends
174/// exactly as it moves its box. What drawing and [`Vein::Ends`](crate::Vein::Ends) read.
175#[derive(Component, Copy, Clone, Debug, Default, PartialEq)]
176pub(crate) struct Stretched {
177    pub(crate) from: Position,
178    pub(crate) to: Position,
179}
180
181impl Stretched {
182    /// Writes one axis of both ends, which is what one pass of the resolver answers.
183    pub(crate) fn set(&mut self, axis: Axis, from: f32, to: f32) {
184        self.from = self.from.set(axis, from);
185        self.to = self.to.set(axis, to);
186    }
187
188    /// Reads one axis of both ends, which is what one pass of the resolver blends toward.
189    pub(crate) fn along(&self, axis: Axis) -> (f32, f32) {
190        (self.from.along(axis), self.to.along(axis))
191    }
192
193    /// Both ends moved by a scrolling ancestor's accumulated offset.
194    pub(crate) fn less(self, offset: Position) -> Self {
195        Self {
196            from: Position::new(self.from.x - offset.x, self.from.y - offset.y),
197            to: Position::new(self.to.x - offset.x, self.to.y - offset.y),
198        }
199    }
200}

202/// How thick a stroked element is drawn.
203#[derive(Component, Copy, Clone, Debug, PartialEq)]
204pub(crate) struct Stroke {
205    pub(crate) weight: f32,
206}
207
208impl Stroke {
209    pub(crate) fn new(weight: f32) -> Self {
210        Self {
211            weight: weight.max(HAIRLINE),
212        }
213    }
214
215    /// How far past its two ends the stroke reaches, which is what its box is grown by.
216    pub(crate) fn half(self) -> f32 {
217        self.weight / 2.0
218    }
219}
220
221impl Default for Stroke {
222    fn default() -> Self {
223        Self::new(HAIRLINE)
224    }
225}

236/// One line, as extraction states it: two ends on the surface, a weight, a cap and a colour.
237///
238/// Deliberately *not* the form the vertex buffer takes. An axis-aligned stroke is put on whole
239/// device pixels so that a one-pixel rule is one lit row rather than two half-lit ones, and only the
240/// backend knows how large a device pixel is. So this is what [`Elm`](crate::elm) compares, and the
241/// snapped form is derived where the density is known, exactly as a glyph's ink is.
242#[derive(Copy, Clone, Debug, PartialEq)]
243pub(crate) struct LineInstance {
244    pub(crate) from: Position,
245    pub(crate) to: Position,
246    pub(crate) color: Color,
247    pub(crate) weight: f32,
248    pub(crate) cap: Cap,
249}
