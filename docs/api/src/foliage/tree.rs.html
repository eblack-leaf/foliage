29/// An element is held in the slot.
30const LIVE: u32 = 1;

36///
37/// *Moving with the content* and *counting toward extent* are the same question, so they get one
38/// answer: a pinned element receives no offset from its nearest scrolling ancestor in R4 and
39/// contributes nothing to that region's extent in R3. Two separate flags could drift out of

60/// The tree itself, seen from the inside.
61///
62/// Holds every element, one slot each at the index of its name, and every field of an element as a
63/// column of those slots: what a pass asks of many elements is read from one array rather than
64/// gathered from wherever each element is. The only place an element's declarations are held.
65///
66/// What only some elements state -- a grid, a scroll offset, a font, a run, an anchor -- is held in
67/// a side table, and only for those that state it. What an element draws is held packed by
68/// renderer, so each renderer's array is as long as what draws with it.

197/// What one renderer was told, for every element carrying it, packed.
198///
199/// An element that withers has the last of them moved into its place, and the element that one
200/// belongs to is told where it went -- so the array is as long as what draws with this renderer and
201/// no longer, and nothing else moves.
202struct Kind<P> {
203    items: Vec<P>,
204    /// The slot each item belongs to.
205    owners: Vec<u32>,
206}

212            owners: Vec::new(),

217impl<P> Kind<P> {
218    /// Holds `item` for the element at `slot`, and hands back where.
219    fn push(&mut self, slot: usize, item: P) -> u32 {
220        self.items.push(item);
221        self.owners.push(slot as u32);
222        (self.items.len() - 1) as u32
223    }
224
225    /// Lets go of the item at `at`, and hands back the slot of the element whose item was moved
226    /// into its place, if one was.
227    fn remove(&mut self, at: u32) -> Option<u32> {
228        let at = at as usize;
229        self.items.swap_remove(at);
230        self.owners.swap_remove(at);
231        self.owners.get(at).copied()
232    }
233}

235/// What only some elements state, held only for those that do.
236///
237/// Each table owns a bit of a slot's flags, set exactly while it holds something for that slot: an
238/// element that states nothing here is answered by the bit rather than by a lookup, and one that
239/// withers lets go of what it held and touches no table it held nothing in.
240struct Side<V> {
241    bit: u32,
242    held: Named<u32, V>,
243}
244
245impl<V> Side<V> {
246    fn new(bit: u32) -> Self {
247        Self {
248            bit,
249            held: Named::default(),
250        }
251    }
252
253    fn get(&self, flags: &[u32], slot: usize) -> Option<&V> {
254        if flags[slot] & self.bit == 0 {
255            return None;
256        }
257        self.held.get(&(slot as u32))
258    }
259
260    fn get_mut(&mut self, flags: &[u32], slot: usize) -> Option<&mut V> {
261        if flags[slot] & self.bit == 0 {
262            return None;
263        }
264        self.held.get_mut(&(slot as u32))
265    }
266
267    fn insert(&mut self, flags: &mut [u32], slot: usize, value: V) {
268        flags[slot] |= self.bit;
269        self.held.insert(slot as u32, value);
270    }
271
272    /// Writes `value` in place where one is held, holding it where none is, and reports whether
273    /// what was held changed.
274    fn overwrite(&mut self, flags: &mut [u32], slot: usize, value: V) -> bool
275    where
276        V: PartialEq,
277    {
278        match self.get_mut(flags, slot) {
279            Some(held) if *held == value => false,
280            Some(held) => {
281                *held = value;
282                true
283            }
284            None => {
285                self.insert(flags, slot, value);
286                true
287            }
288        }
289    }
290
291    fn vacate(&mut self, flags: &mut [u32], slot: usize) {
292        if flags[slot] & self.bit != 0 {
293            flags[slot] &= !self.bit;
294            self.held.remove(&(slot as u32));
295        }
296    }
297}

540    fn sow(
541        &mut self,
542        leaf: Leaf,
543        growth: Growth,
544        under: Option<Leaf>,
545        mut bud: Bud,
546        grown: u32,
547    ) -> bool {
548        let slot = leaf.slot();
549        if slot >= self.generation.len() {
550            self.reach(slot + 1);
551        }
552        if self.generation[slot] != leaf.generation || self.flags[slot] & LIVE != 0 {
553            return false;
554        }
555        // Everything an element declares about how it behaves. What resolution makes of it is not
556        // here: that is held in resolution's own columns ([`Elements`](crate::rowan::Elements)),
557        // which is where every pass reads it.
558        let manner = bud.placement.manner;
559        let mut flags = LIVE | grown;
560        if manner.visible.0 {
561            flags |= VISIBLE;
562        }
563        // Absent unless declared: an element that travels with its region's content and is clipped
564        // by it is the ordinary case.
565        if manner.pinned {
566            flags |= PINNED;
567        }
568        self.flags[slot] = flags;
569        self.growth[slot] = growth;
570        self.spawned_at[slot] = Some(bud.at);
571        self.chlorophyll[slot] = bud.chlorophyll;
572        self.gestures[slot] = manner.gestures;
573        self.opacity[slot] = manner.opacity;
574        self.elevation[slot] = bud.placement.elevation.unwrap_or_default();
575        self.shift[slot] = Shift::default();
576        self.scrolls[slot] = manner.scrolls.map(|scrolls| scrolls.0);
577        // Grown under something that went before the drain reached this is grown at top level,
578        // since there is nothing left to hang it off.
579        match under.and_then(|under| self.held(under)) {
580            Some(trunk) => self.attach(slot, trunk),
581            None => self.trunk[slot] = NONE,
582        }
583        // A stroke is placed by its ends and nothing else is, and is not given a box as well: it
584        // would be held, and carried, and read by nothing, since a trace is what resolution asks
585        // for first and a `Location` written onto a stroke is refused.
586        self.placements[slot] = Some(match bud.placement.traced.take() {
587            Some(trace) => Declared::Traced(trace, bud.placement.stroke.take().unwrap_or_default()),
588            None => Declared::Boxed(bud.placement.location.take().unwrap_or_default()),
589        });
590        self.pigment[slot] = match bud.pigment.take() {
591            // What a renderer was told is held where that renderer reads it, so it is held only
592            // where the element carries that renderer.
593            Some(pigment) if chlorophyll(&pigment) != bud.chlorophyll => NONE,
594            Some(Pigment::Panel(pigment)) => self.panels.push(slot, pigment),
595            Some(Pigment::Text(pigment)) => self.texts.push(slot, pigment),
596            Some(Pigment::Polygon(pigment)) => self.polygons.push(slot, pigment),
597            Some(Pigment::Line(pigment)) => self.lines.push(slot, pigment),
598            Some(Pigment::Icon(pigment)) => self.icons.push(slot, pigment),
599            Some(Pigment::Image(pigment)) => self.images.push(slot, pigment),
600            None => NONE,
601        };
602        let flags = &mut self.flags;
603        if let Some(grid) = bud.placement.grid {
604            self.grids.insert(flags, slot, grid);
605        }
606        if manner.focusing != Focusing::default() {
607            self.focusing.insert(flags, slot, manner.focusing);
608        }
609        if let Some(escape) = manner.floats {
610            self.floats.insert(flags, slot, Floats(escape));
611        }
612        if let Some(lettering) = bud.lettering {
613            self.letterings.insert(flags, slot, lettering);
614        }
615        if let Some(tints) = bud.tints {
616            self.tints.insert(flags, slot, tints);
617        }
618        if let Some(typeface) = bud.placement.typeface {
619            self.typefaces.insert(flags, slot, typeface);
620        }
621        if let Some(anchored) = bud.placement.anchor {
622            self.anchors.insert(flags, slot, anchored);
623        }
624        // What it was grown under has one more element to reach over, and finds that out by the
625        // walk R2m makes from this one toward it.
626        self.declared(leaf);
627        self.lettered.insert(leaf);
628        self.restructured = true;
629        self.grown.push(leaf);
630        true
631    }

654    /// Hangs the element at `slot` off the one at `trunk`, after everything already grown there.
655    fn attach(&mut self, slot: usize, trunk: usize) {
656        let last = self.last[trunk];
657        self.trunk[slot] = trunk as u32;
658        self.previous[slot] = last;
659        self.next[slot] = NONE;
660        match last {
661            NONE => self.first[trunk] = slot as u32,
662            last => self.next[last as usize] = slot as u32,
663        }
664        self.last[trunk] = slot as u32;
665    }

667    /// Takes the element at `slot` out from under what it hangs off, leaving the rest in order.
668    fn detach(&mut self, slot: usize) {
669        let trunk = self.trunk[slot];
670        if trunk == NONE {
671            return;
672        }
673        let (previous, next) = (self.previous[slot], self.next[slot]);
674        match previous {
675            NONE => self.first[trunk as usize] = next,
676            previous => self.next[previous as usize] = next,
677        }
678        match next {
679            NONE => self.last[trunk as usize] = previous,
680            next => self.previous[next as usize] = previous,
681        }
682        self.trunk[slot] = NONE;
683        self.previous[slot] = NONE;
684        self.next[slot] = NONE;
685    }

687    /// Takes each of `leaves` down, and everything beneath each, and reports every name that went.
688    ///
689    /// Each is taken out from under what it hangs off where it lies, which leaves what was grown
690    /// beside it in order and costs the same however many were grown there -- so letting go of many
691    /// off one trunk, as a list cut short does, costs what went and not the trunk's length for each.
692    pub(crate) fn wither(&mut self, leaves: &[Leaf]) -> Vec<Leaf> {
693        let mut gone = Vec::new();
694        let mut beneath = Vec::new();
695        for &leaf in leaves {
696            // Already gone, under one of the others.
697            let Some(slot) = self.held(leaf) else {
698                continue;
699            };
700            if let Some(trunk) = self.trunk(leaf) {
701                self.detach(slot);
702                // What it hung off has one fewer element to reach over, and nothing left below it
703                // to say so: the element that went is what the walk would have started from.
704                self.measures(trunk);
705            }
706            // Everything beneath it, each before what was grown under it and in the order it was
707            // grown -- and each let go of once found, so one of `leaves` under another is not
708            // found twice.
709            beneath.push(slot as u32);
710            while let Some(slot) = beneath.pop() {
711                gone.push(self.named(slot));
712                // What it drew goes with it, and extraction has to know which renderer to tell.
713                self.withered
714                    .push((self.named(slot), self.chlorophyll[slot as usize]));
715                let mut child = self.last[slot as usize];
716                while child != NONE {
717                    beneath.push(child);
718                    child = self.previous[child as usize];
719                }
720                self.vacate(slot as usize);
721            }
722        }
723        // Back to be named again, each a generation on from the name that went.
724        self.naming.release(gone.iter().map(|leaf| Leaf {
725            index: leaf.index,
726            generation: self.generation[leaf.slot()],
727        }));
728        self.restructured = true;
729        // Whatever it said, it does not say any more.
730        self.restated = true;
731        gone
732    }
