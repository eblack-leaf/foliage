30/// The tree itself, seen from the inside.
31///
32/// Owns the world. The only place raw `bevy_ecs` is touched.
33pub(crate) struct Tree {
34    world: World,
35    /// Every element written to since resolution last read this.
36    ///
37    /// A declaration is what resolution reads, so an element nothing has written to resolves to what
38    /// it resolved to last frame. A placement, a trace, a grid or a run written back at what the
39    /// element already held is not recorded -- the setter compares first -- because resolution would
40    /// answer the same box for it, and asking is the cost of the element and of everything resolved
41    /// against it.
42    touched: Leaves,
43    /// Every element written to in a way no pass that places anything reads.
44    ///
45    /// How an element is filled, rounded, tinted or shaped is read by extraction alone, and whether
46    /// it is shown, how opaque it is, whether it is disabled and how far forward it sits are read by
47    /// the passes that run over every element anyway. None of it moves a box, so none of it is a
48    /// reason to resolve one again: an element here is read afresh and drawn again, and its subtree
49    /// is left where it was.
50    restyled: Leaves,
51    /// Every element whose run was written, or that was grown: what R1 measures again.
52    ///
53    /// Apart from `touched`, because R1 reads nothing but the element's own run and face. An element
54    /// that moved says what it said, in the cell it said it in, and has nothing to measure again.
55    lettered: Leaves,
56    /// Every element that has to measure again because what is under it went away.
57    ///
58    /// Separate from `touched`, because the two spread differently: what an element was written to
59    /// travels to everything resolved against it, and what an element measures to travels no further
60    /// than the measure -- unless the measure moves, which R2m finds out and says so.
61    reached: Leaves,
62    /// Whether any run stopped being stated -- a run rewritten, or an element carrying one withered
63    /// -- which is the only thing that can leave the shaping cache holding what nothing wants.
64    restated: bool,
65    /// Whether the elements, or the edges between them, changed -- which is what the order is built
66    /// from and the only thing that makes it worth building again.
67    restructured: bool,
68    /// Whether something every element is read against has changed, which is every element written.
69    invalidated: bool,
70    /// Whether something every element is drawn against has changed -- the scheme, or an asset --
71    /// which is every element drawn again and none of them placed again.
72    repainted: bool,
73}

75/// How an element is placed, borrowed from its declaration.
76pub(crate) enum Placement<'a> {
77    /// By a box.
78    Boxed(&'a Location),
79    /// By two ends, and the weight of the stroke between them.
80    Traced(&'a Trace, Stroke),
81}

83/// What the tree has been written to since resolution last read it.
84///
85/// Borrowed rather than handed over, so a frame that writes to a great many elements is not also a
86/// frame that allocates a set the size of what it wrote.
87pub(crate) struct Written<'a> {
88    /// The elements whose own declarations were written.
89    pub(crate) declared: &'a Leaves,
90    /// The elements written to in a way that moves nothing.
91    pub(crate) restyled: &'a Leaves,
92    /// The elements whose run was written, or that were grown.
93    pub(crate) lettered: &'a Leaves,
94    /// The elements that have to measure again because what is under them went away.
95    pub(crate) measures: &'a Leaves,
96    /// Whether the order has to be built again.
97    pub(crate) restructured: bool,
98    /// Whether a run stopped being stated, so the shaping cache has to be swept.
99    pub(crate) restated: bool,
100    /// Whether everything has to be resolved again.
101    pub(crate) all: bool,
102    /// Whether everything has to be drawn again.
103    pub(crate) repainted: bool,
104}

122    /// Records that `leaf`'s declared state was written.
123    ///
124    /// Every write of something resolution reads passes through here. What resolution *writes* does
125    /// not: those go through [`overwrite`](Tree::overwrite), which is the whole of the difference
126    /// between a declaration and a resolved value.
127    ///
128    /// [`aspen`](crate::aspen) states it directly, because a motion is the one thing resolution
129    /// reads that is not held on the element at all.

132    }

140    }

147    /// Records that everything has to be resolved again.
148    ///
149    /// The viewport, the breakpoint and the short-side reading are each read by every placement in
150    /// the tree, and none of them belongs to an element that could be marked. A breakpoint can move
151    /// what size a run is set at, which leaves the run it was set at unstated.
152    ///
153    /// The order is not built again: none of this changes which elements there are or what any of
154    /// them hangs off.
155    pub(crate) fn invalidate(&mut self) {
156        self.invalidated = true;
157        self.restated = true;
158    }

163    /// alone -- so a repaint or an arrival is every element's fill written, and no box's.
164    pub(crate) fn repaint(&mut self) {
165        self.repainted = true;
166    }

170        Written {

172            restyled: &self.restyled,
173            lettered: &self.lettered,
174            measures: &self.reached,

182    /// Forgets it, once resolution has taken it.
183    ///
184    /// Called the moment the read is made rather than at the end of the frame, so a write made by a
185    /// pass after that point belongs to the next frame -- which is the frame that would resolve it.
186    /// The sets keep the room they had, because a frame that writes a great deal is usually followed
187    /// by another one.
188    pub(crate) fn taken(&mut self) {
189        self.touched.clear();
190        self.restyled.clear();
191        self.lettered.clear();
192        self.reached.clear();
193        self.restated = false;
194        self.restructured = false;
195        self.invalidated = false;
196        self.repainted = false;
197    }

223    /// Grows `leaf`, reporting whether the name was still free to grow into.
224    pub(crate) fn grow(
225        &mut self,
226        leaf: Leaf,
227        growth: Growth,
228        under: Option<Leaf>,
229        bud: Bud,
230    ) -> bool {
231        let Ok(mut entity) = self.world.spawn_at(leaf.0, Grown) else {
232            return false;
233        };
234        entity.insert((
235            SpawnedAt(bud.at),
236            growth,
237            bud.chlorophyll,
238            bud.placement.location.unwrap_or_default(),
239            bud.placement.grid.unwrap_or_default(),
240            bud.placement.elevation.unwrap_or_default(),
241        ));
242        // Everything an element declares about how it behaves. Each is present on every element, so
243        // nothing that reads one has to ask whether the element is the kind of thing that has it.
244        // What resolution makes of them is not here: that is held in resolution's own columns
245        // ([`Elements`](crate::rowan::Elements)), which is where every pass reads it.
246        let manner = bud.placement.manner;
247        entity.insert((
248            manner.gestures,
249            manner.focusing,
250            manner.visible,
251            manner.opacity,
252            Disabled::default(),
253            Offset::default(),
254            Extent::default(),
255        ));
256        if let Some(scrolls) = manner.scrolls {
257            entity.insert(scrolls);
258        }
259        // Absent unless declared: an element that travels with its region's content and is clipped
260        // by it is the ordinary case, and neither is a value to hold.
261        if manner.pinned {
262            entity.insert(Pinned);
263        }
264        if let Some(escape) = manner.floats {
265            entity.insert(Floats(escape));
266        }
267        match bud.pigment {
268            Some(Pigment::Panel(pigment)) => {
269                entity.insert(pigment);
270            }
271            Some(Pigment::Text(pigment)) => {
272                entity.insert(pigment);
273            }
274            Some(Pigment::Polygon(pigment)) => {
275                entity.insert(pigment);
276            }
277            Some(Pigment::Line(pigment)) => {
278                entity.insert(pigment);
279            }
280            Some(Pigment::Icon(pigment)) => {
281                entity.insert(pigment);
282            }
283            Some(Pigment::Image(pigment)) => {
284                entity.insert(pigment);
285            }
286            None => {}
287        }
288        if let Some(lettering) = bud.lettering {
289            entity.insert(lettering);
290        }
291        if let Some(tints) = bud.tints {
292            entity.insert(tints);
293        }
294        // The point-mode placement. Absent on everything placed by a box, which is what makes "has a
295        // trace" the one question the resolver asks to tell the two apart.
296        if let Some(trace) = bud.placement.traced {
297            entity.insert(trace);
298        }
299        if let Some(stroke) = bud.placement.stroke {
300            entity.insert(stroke);
301        }
302        if let Some(typeface) = bud.placement.typeface {
303            entity.insert(typeface);
304        }
305        if let Some(anchored) = bud.placement.anchor {
306            entity.insert(anchored);
307        }
308        if let Some(under) = under {
309            entity.insert(ChildOf(under.0));
310        }
311        // What it was grown under has one more element to reach over, and finds that out by the
312        // walk R2m makes from this one toward it.
313        self.declared(leaf);
314        self.lettered.insert(leaf);
315        self.restructured = true;
316        true
317    }

319    /// Takes `leaf` and everything beneath it down, and reports every name that went.
320    pub(crate) fn wither(&mut self, leaf: Leaf) -> Vec<Leaf> {
321        let mut gone = Vec::new();
322        self.gather(leaf, &mut gone);
323        let trunk = self.trunk(leaf);
324        if let Ok(entity) = self.world.get_entity_mut(leaf.0) {
325            entity.despawn();
326        }
327        // What it hung off has one fewer element to reach over, and nothing left below it to say so:
328        // the element that went is what the walk would have started from.
329        if let Some(trunk) = trunk {
330            self.measures(trunk);
331        }
332        self.restructured = true;
333        // Whatever it said, it does not say any more.
334        self.restated = true;
335        gone
336    }

338    fn gather(&self, leaf: Leaf, into: &mut Vec<Leaf>) {
339        into.push(leaf);
340        let Ok(entity) = self.world.get_entity(leaf.0) else {
341            return;
342        };
343        let Some(children) = entity.get::<Children>() else {
344            return;
345        };
346        for child in children.iter().copied() {
347            self.gather(Leaf(child), into);
348        }
349    }

356    /// The same, borrowed rather than collected.
357    ///
358    /// Resolution asks this of every element on more than one of its passes, so the vector
359    /// [`branches`](Tree::branches) hands back would be an allocation per element per pass.
360    pub(crate) fn branched(&self, leaf: Leaf) -> impl Iterator<Item = Leaf> + '_ {
361        self.world
362            .get_entity(leaf.0)
363            .ok()
364            .and_then(|entity| entity.get::<Children>())
365            .into_iter()
366            .flat_map(|children| children.iter().copied())
367            .filter(|child| {
368                self.world
369                    .get_entity(*child)
370                    .is_ok_and(|child| child.contains::<Grown>())
371            })
372            .map(Leaf)
373    }

396    /// How `leaf` is placed: by its two ends and the weight between them, or by a box. `None` for
397    /// an element that is not live.
398    ///
399    /// One read of the element for the question every pass that resolves geometry asks first, and
400    /// the declaration it goes on to read -- which a pass resolving every element on two axes would
401    /// otherwise ask the world for separately.

403        let entity = self.world.get_entity(leaf.0).ok()?;

405            Some(trace) => Some(Placement::Traced(
406                trace,
407                entity.get::<Stroke>().copied().unwrap_or_default(),

409            None => entity.get::<Location>().map(Placement::Boxed),
410        }
411    }

434    /// Rewrites what `leaf` says, reporting whether it is something with a run to write.
435    pub(crate) fn set_lettering(&mut self, leaf: Leaf, value: String) -> bool {
436        let Ok(mut entity) = self.world.get_entity_mut(leaf.0) else {

550    /// Whether `leaf` is grown somewhere under `trunk`, however deep.
551    ///
552    /// What [`ScrollTo::show`](crate::ScrollTo::show) is asked, because bringing an element into
553    /// view means nothing unless the region is what it is inside.
554    pub(crate) fn grown_under(&self, leaf: Leaf, trunk: Leaf) -> bool {
555        let mut step = self.trunk(leaf);
556        while let Some(above) = step {
557            if above == trunk {
558                return true;
559            }
560            step = self.trunk(above);
561        }
562        false
563    }

565    /// Whether `from` reaches `target` by following anchors.
566    ///
567    /// Bounded by construction: an anchor is refused if it would close a cycle, so the chain this
568    /// walks is always finite.
569    pub(crate) fn reaches(&self, from: Leaf, target: Leaf) -> bool {
570        let mut step = Some(from);
571        while let Some(leaf) = step {
572            if leaf == target {
573                return true;
574            }
575            step = self.anchor(leaf);
576        }
577        false
578    }

580    /// Moves `leaf`, reporting whether it is something with a box to place.
581    ///
582    /// An element placed by its ends has none, and refuses the write rather than taking a
583    /// declaration nothing would read: the resolver asks for a trace first, so a `Location` written
584    /// onto a stroke would sit there being ignored.
585    pub(crate) fn set_location(&mut self, leaf: Leaf, location: Location) -> bool {
586        let Ok(entity) = self.world.get_entity(leaf.0) else {
587            return false;
588        };
589        if entity.contains::<Trace>() {
590            return false;
591        }
592        // Written back at what it already holds is no write at all. Resolution would answer the
593        // same box, and asking it to is the whole cost of the element and everything under it.
594        if self.overwrite(leaf, location) {
595            self.declared(leaf);
596        }
597        true
598    }

962    /// Writes a component that is already there in place, inserting it only the first time, and
963    /// reports whether what it held changed.
964    ///
965    /// `insert` goes through the bundle machinery whatever it is handed, which is what a component
966    /// arriving for the first time needs and what one being overwritten does not. And a write of
967    /// what is already held is reported as none, which is what lets a setter mark an element only
968    /// when something about it actually changed.
969    fn overwrite<C: Component<Mutability = Mutable> + PartialEq>(
970        &mut self,
971        leaf: Leaf,
972        value: C,
973    ) -> bool {
974        let Ok(mut entity) = self.world.get_entity_mut(leaf.0) else {
975            return false;
976        };
977        match entity.get_mut::<C>() {
978            Some(mut held) => {
979                if *held == value {
980                    return false;
981                }
982                *held = value;
983            }
984            None => {
985                entity.insert(value);
986            }
987        }
988        true
989    }
990
991    fn read<C: Component + Copy>(&self, leaf: Leaf) -> Option<C> {
992        self.world.get_entity(leaf.0).ok()?.get::<C>().copied()
993    }
