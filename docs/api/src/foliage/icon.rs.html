14/// A registered mark: one multi-channel distance field, at whatever resolution it was baked.
15///
16/// Handed out by [`Foliage::icon`](crate::Foliage::icon) and named on an element with
17/// [`Icon::new`]. Opaque: there is nothing to be done with one but draw it.
18///
19/// A field rather than a bitmap, and that is the whole reason an icon is not a glyph. A glyph is
20/// cut into coverage at a size, because text is composed at a handful of sizes and a cut is exact.
21/// A mark is stretched to whatever box a layout hands it -- a 16px affordance and a 96px empty
22/// state are the same artwork -- so it is stored once as a distance to its own edge and
23/// reconstructed at any size, sharp at every one of them.
24#[derive(Copy, Clone, PartialEq, Eq, Hash, PartialOrd, Ord, Debug)]
25pub struct Field(pub(crate) u32);
26
27/// A vector mark, filled like text and scaled like a shape.
28///
29/// Sized by its box like any other element; the artwork is fitted into the largest square that box
30/// holds, so a non-square box leaves room around the mark rather than distorting it.
31///
32/// ```no_run
33/// # use foliage::{Boxed, Field, Icon, Location, Palette, Source, left, top};
34/// # fn f(check: Field) {
35/// Icon::new(check)
36///     .color(Palette::Accent)
37///     .at(Location::new().xs(
38///         left(0.px()).width(24.px()),
39///         top(0.px()).height(24.px()),
40///     ));
41/// # }
42/// ```
43///
44/// What it is filled with is entirely the element's: the field carries shape and no colour, exactly
45/// as a glyph's coverage does. So an icon is recoloured, repainted and animated by the same writes a
46/// panel and a run take.
47#[derive(Clone, Debug)]
48pub struct Icon {
49    pub(crate) placement: Placement,
50    pub(crate) field: Field,
51    pub(crate) fill: Fill,
52}
53
54impl Icon {
55    /// A mark drawing the registered `field`, filled with [`Palette::Ink`] -- what is read against a
56    /// surface rather than drawn as one, which is what a mark beside a label is.
57    pub fn new(field: Field) -> Self {
58        Self {
59            placement: Placement::default(),
60            field,
61            fill: Fill::Role(Palette::Ink),
62        }
63    }
64
65    /// What the mark is filled with: a [`Palette`] role, or a [`Color`](crate::Color) stated
66    /// outright.
67    pub fn color(mut self, fill: impl Into<Fill>) -> Self {
68        self.fill = fill.into();
69        self
70    }
71}
72

105/// One icon, as extraction states it.
106///
107/// Not the form the vertex buffer takes: where the field sits on the sheet, and how many screen
108/// pixels its distance range spans at this size, are both the backend's -- the first because only
109/// the backend packs the sheet, the second because it depends on the display's density. What
110/// extraction supplies is the box, the fill, and which field.
111#[derive(Copy, Clone, Debug, PartialEq)]
112pub(crate) struct IconInstance {
113    pub(crate) section: Section,
114    pub(crate) color: Color,
115    pub(crate) field: Field,
116}

118/// Every registered mark's field bytes.
119///
120/// Held on the engine rather than in the backend because a field is registered before there is a
121/// device to hold it: an app names its marks at boot, and the surface arrives when the platform
122/// resumes. The backend packs what is here onto its sheet the first time it draws one.
123#[derive(Default)]
124pub(crate) struct Fields {
125    /// A slot per name. Empty where a name was handed out for a field that has not arrived yet: an
126    /// element drawing one occupies its box and draws nothing until it does, exactly as a picture
127    /// whose pixels are still coming does.
128    marks: Vec<Option<Mark>>,
129}
130
131/// One registered mark: the field itself, and what the shader needs to read it.
132pub(crate) struct Mark {
133    /// The field, `side` by `side` texels of RGBA, row-major.
134    pub(crate) field: Vec<u8>,
135    pub(crate) side: u32,
136    /// How many texels the baked distance range spans. What turns a sampled distance into a
137    /// screen-space edge one pixel wide at whatever size the mark is drawn.
138    pub(crate) range: f32,
139}
140
141impl Fields {
142    /// Fills a name with a field the program stated.
143    ///
144    /// # Panics
145    ///
146    /// If the field is smaller than `side` by `side` texels of RGBA.
147    pub(crate) fn register(&mut self, name: Field, field: &[u8], side: u32, range: f32) {
148        assert!(
149            side > 0 && field.len() as u32 >= side * side * 4,
150            "an icon field is {side}x{side} texels of RGBA, which is {} bytes, and {} were given",
151            side * side * 4,
152            field.len(),
153        );
154        *self.slot(name) = Some(Mark {
155            field: field.to_vec(),
156            side,
157            range: range.max(1.0),
158        });
159        info!(field = name.0, side, range, "icon registered");
160    }
161
162    /// Fills a name with a field that was read.
163    ///
164    /// Refuses rather than asserts, for the reason a fetched font is refused rather than panicked
165    /// on: what arrived from a path or a URL is not something the program stated.
166    pub(crate) fn fill(
167        &mut self,
168        field: Field,
169        bytes: &[u8],
170        side: u32,
171        range: f32,
172    ) -> Result<(), String> {
173        if side == 0 || (bytes.len() as u32) < side * side * 4 {
174            return Err(format!(
175                "a {side}x{side} field is {} bytes of RGBA, and {} arrived",
176                side * side * 4,
177                bytes.len()
178            ));
179        }
180        *self.slot(field) = Some(Mark {
181            field: bytes.to_vec(),
182            side,
183            range: range.max(1.0),
184        });
185        info!(field = field.0, side, range, "icon registered");
186        Ok(())
187    }
188
189    /// Where `field`'s mark goes, growing to reach it.
190    ///
191    /// Names come from [`Naming`](crate::naming::Naming) rather than from this registry, so that one
192    /// can be taken where the registry is out of reach. A name that has not been filled draws
193    /// nothing, which is what any unfilled name does.
194    fn slot(&mut self, field: Field) -> &mut Option<Mark> {
195        let index = field.0 as usize;
196        if index >= self.marks.len() {
197            self.marks.resize_with(index + 1, || None);
198        }
199        &mut self.marks[index]
200    }
201
202    /// The mark `field` names, or `None` where it named nothing or has yet to arrive.
203    pub(crate) fn mark(&self, field: Field) -> Option<&Mark> {
204        self.marks.get(field.0 as usize)?.as_ref()
205    }
206}
