1//! How long a motion takes, and what shape it moves in.
2//!
3//! The two are independent. A duration says when the motion is over; an ease says where it is on
4//! the way, and says nothing about how long that takes. Keeping them apart is what lets one
5//! app-wide shape be stated beside any duration.
6
7use core::time::Duration;
8
9use crate::aspen::Sequence;

11/// The shape of a motion: how its elapsed fraction maps onto its progress.
12///
13/// Every shape here is a cubic bezier through `(0, 0)` and `(1, 1)`, so all of them start where the
14/// motion started and end exactly on its target. The named ones are the three a surface needs:
15/// something arriving, something leaving, and something moving in place.
16///
17/// [`Linear`](Ease::Linear) is unshaped, and is what a [`Timing`] takes when nothing is said. It is
18/// the honest default rather than the flattering one -- a shape an app did not ask for is a shape
19/// it cannot see in the code.
20#[derive(Copy, Clone, Debug, Default, PartialEq)]
21#[non_exhaustive]
22pub enum Ease {
23    /// Constant rate.
24    #[default]
25    Linear,
26    /// Quick to start, gentle to settle. For something arriving: the eye follows it in and is let
27    /// down at the end.
28    Decelerate,
29    /// Gentle to start, quick to leave. For something going away, which needs no help being
30    /// followed because it is already known.
31    Accelerate,
32    /// Eased at both ends with a quick middle. For a change in place that should be noticed
33    /// without being waited on.
34    Emphasis,
35    /// A cubic bezier through two interior control points, each stated as a fraction: `x` along the
36    /// motion's duration, `y` along its progress.
37    ///
38    /// The endpoints are fixed at `(0, 0)` and `(1, 1)` and are not stated. `x` is clamped to
39    /// `0.0..=1.0`, because a control point outside the duration describes no ordering of the
40    /// motion; `y` is not, so a curve may overshoot and come back.
41    Curve {
42        /// The first control point's position along the duration.
43        x1: f32,
44        /// The first control point's position along the progress.
45        y1: f32,
46        /// The second control point's position along the duration.
47        x2: f32,
48        /// The second control point's position along the progress.
49        y2: f32,
50    },
51}
52
53impl Ease {
54    /// The two interior control points, as `(x1, y1, x2, y2)`.
55    fn control(self) -> (f32, f32, f32, f32) {
56        match self {
57            // The cubic that is its own straight line: evenly spaced control points.
58            Ease::Linear => (1.0 / 3.0, 1.0 / 3.0, 2.0 / 3.0, 2.0 / 3.0),
59            Ease::Decelerate => (0.0, 0.0, 0.2, 1.0),
60            Ease::Accelerate => (0.4, 0.0, 1.0, 1.0),
61            Ease::Emphasis => (0.4, 0.0, 0.2, 1.0),
62            Ease::Curve { x1, y1, x2, y2 } => (x1.clamp(0.0, 1.0), y1, x2.clamp(0.0, 1.0), y2),
63        }
64    }
65
66    /// The progress this shape is at, `fraction` of the way through the motion's duration.
67    ///
68    /// Exact at both ends: `0.0` and `1.0` map to themselves whatever the shape, so a motion always
69    /// leaves where it was and lands on its target rather than near it.
70    pub(crate) fn at(self, fraction: f32) -> f32 {
71        let fraction = fraction.clamp(0.0, 1.0);
72        if fraction == 0.0 || fraction == 1.0 || self == Ease::Linear {
73            return fraction;
74        }
75        let (x1, y1, x2, y2) = self.control();
76        cubic(y1, y2, parameter(x1, x2, fraction))
77    }
78}

80/// A cubic bezier through `(0, 0)` and `(1, 1)` with the two interior points given, in its
81/// polynomial form: `((a * u + b) * u + c) * u`.
82fn coefficients(first: f32, second: f32) -> (f32, f32, f32) {
83    let c = 3.0 * first;
84    let b = 3.0 * (second - first) - c;
85    (1.0 - c - b, b, c)
86}
87
88fn cubic(first: f32, second: f32, u: f32) -> f32 {
89    let (a, b, c) = coefficients(first, second);
90    ((a * u + b) * u + c) * u
91}
92
93fn slope(first: f32, second: f32, u: f32) -> f32 {
94    let (a, b, c) = coefficients(first, second);
95    (3.0 * a * u + 2.0 * b) * u + c
96}

98/// The curve parameter whose `x` is `fraction`.
99///
100/// A bezier is parametric, so the curve is not a function of the fraction directly -- the parameter
101/// has to be recovered from it before the progress can be read off. Newton converges in a handful
102/// of steps for every shape that is a usable ease, and bisection finishes the ones where the slope
103/// is too flat for it. Both are bounded, so this cannot fail to return and cannot be slow.
104fn parameter(x1: f32, x2: f32, fraction: f32) -> f32 {
105    let mut u = fraction;
106    for _ in 0..8 {
107        let error = cubic(x1, x2, u) - fraction;
108        if error.abs() < 1e-6 {
109            return u;
110        }
111        let slope = slope(x1, x2, u);
112        if slope.abs() < 1e-6 {
113            break;
114        }
115        u -= error / slope;
116    }
117    let (mut low, mut high) = (0.0, 1.0);
118    u = fraction;
119    for _ in 0..32 {
120        let at = cubic(x1, x2, u);
121        if (at - fraction).abs() < 1e-6 {
122            break;
123        }
124        if at < fraction {
125            low = u;
126        } else {
127            high = u;
128        }
129        u = (low + high) / 2.0;
130    }
131    u
132}

134/// How long a motion takes, when it starts, what shape it moves in, and what it arrives as part of.
135///
136/// ```no_run
137/// # use foliage::{Ease, Timing};
138/// Timing::ms(240).ease(Ease::Decelerate).after(60);
139/// ```
140///
141/// Milliseconds throughout, because that is the scale every motion on a surface is stated at, and
142/// one unit at the callsite is what keeps two motions comparable by reading them.
143#[derive(Copy, Clone, Debug, PartialEq)]
144pub struct Timing {
145    duration: Duration,
146    delay: Duration,
147    ease: Ease,
148    within: Option<Sequence>,
149}
150
151impl Timing {
152    /// A motion lasting `millis`, beginning at once, unshaped, arriving on its own.
153    ///
154    /// A duration of zero is a motion that is over on the frame it starts, which is what makes a
155    /// [`timer`](crate::Grow::timer) of zero the next frame's report rather than a special case.
156    pub fn ms(millis: u64) -> Self {
157        Self {
158            duration: Duration::from_millis(millis),
159            delay: Duration::ZERO,
160            ease: Ease::Linear,
161            within: None,
162        }
163    }
164
165    /// Counts this into a [`Sequence`], so its end is reported as part of the group's rather than
166    /// only as its own.
167    ///
168    /// Here rather than as an argument to [`animate`](crate::Grow::animate) because a sequence is
169    /// about *when a group is over*, which is the one question this type exists to answer. What is
170    /// moving stays out of it: a motion, a channel and a timer join a sequence in the same words,
171    /// and none of them has to be near any of the others to do it.
172    pub fn within(mut self, sequence: Sequence) -> Self {
173        self.within = Some(sequence);
174        self
175    }
176
177    /// The sequence this was counted into, if any.
178    pub(crate) fn sequence(&self) -> Option<Sequence> {
179        self.within
180    }
181
182    /// The shape it moves in.
183    pub fn ease(mut self, ease: Ease) -> Self {
184        self.ease = ease;
185        self
186    }
187
188    /// How long to wait before it starts, in milliseconds.
189    ///
190    /// The motion holds where it was for that long rather than being queued: it is running from the
191    /// frame it was asked for, and a write that cancels it cancels it whether or not it has begun
192    /// to move.
193    pub fn after(mut self, millis: u64) -> Self {
194        self.delay = Duration::from_millis(millis);
195        self
196    }
197
198    /// The progress a motion `elapsed` into this timing has made.
199    pub(crate) fn at(&self, elapsed: Duration) -> f32 {
200        let running = elapsed.saturating_sub(self.delay);
201        // Nothing to be a fraction of. A zero-duration motion is at its end from its first instant.
202        if self.duration.is_zero() {
203            return 1.0;
204        }
205        let fraction = running.as_nanos() as f64 / self.duration.as_nanos() as f64;
206        self.ease.at(fraction.clamp(0.0, 1.0) as f32)
207    }
208
209    /// Whether a motion `elapsed` into this timing has reached its end.
210    pub(crate) fn done(&self, elapsed: Duration) -> bool {
211        elapsed >= self.delay + self.duration
212    }
213}
