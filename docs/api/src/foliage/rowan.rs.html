1//! Rowan -- resolution.
2//!
3//! # Rowan recomputes. Elm decides what changed.
4//!
5//! Resolution runs over everything, every frame. There is no dirty tracking, no invalidation, and
6//! no mechanism by which a value can be stale because something forgot to mark it. The cost is
7//! arithmetic over a handful of numbers per element; the saving is a whole class of bug. Nothing is
8//! re-uploaded on account of it, because extraction compares against a cached copy and sends only
9//! genuine differences.
10//!
11//! The passes each write exactly one thing, and nothing writes what another owns:
12//!
13//! | | Pass | Direction | Produces |
14//! |---|---|---|---|
15//! | R1 | measure | — | [`Cell`], and the max-content width half of [`Intrinsic`] |
16//! | R2a | horizontal | dependency order | the horizontal axis |
17//! | R2m | wrap | **bottom-up** | the measured-height half of [`Intrinsic`] |
18//! | R2b | vertical | dependency order | the vertical axis, and with it [`Placed`] |
19//! | R3 | extent | **bottom-up** | [`Extent`](crate::view::Extent): how far a region's content reaches |
20//! | R4 | scroll | top-down | [`Drawn`]: [`Placed`] less every scrolling ancestor's offset |
21//! | R5 | clip | top-down | [`Clipped`]: what a scrolling ancestor leaves visible |
22//! | R6 | rank | dependency order | `ResolvedElevation`: elevation accumulated, then tie-broken |
23//! | R7 | inherit | top-down | [`Inherited`]: the visible, opacity and disabled products |
24//! | R8 | regions | rank order | the box stack the next frame's dispatch reads |
25//!
26//! Both halves of R2 call the same pure resolver, once per axis. That is only safe because it is
27//! pure: there is no accumulated state for a second call to corrupt.
28//!
29//! Two passes run the other way, and each is a cycle that turns out not to be vicious.
30//!
31//! R2m is one: wrapping makes a height depend on a width, and the width comes from the layout, so a
32//! box sized to its contents looks circular. It is not, because **width flows down and height flows
33//! up** -- a monospaced run's widest unwrapped line is free from its character count, so the
34//! down-pass needs nothing measured. R2m sits between the two halves of R2 for that reason: it is
35//! the one moment when every width is known and no height is, which is exactly what measuring needs.
36//!
37//! R3 is the other: a child resolves against its parent's box, a scrolling parent's extent comes
38//! from where its children landed, and that extent is what its offset is clamped to. Extent affects
39//! only the clamp and never the parent's own box -- so a top-down layout sweep, a bottom-up extent
40//! sweep and a top-down scroll application resolve it in one pass each.
41//!
42//! Neither iterates to convergence. Every pass here runs exactly once.
43

79/// The character cell an element's own font and size make, as R1 measured it.
80///
81/// On every element, because every element may be sized in characters: it is what
82/// [`letters`](crate::Source::letters) and a letter-pitched track are measured in. An element that
83/// named no font and no size has none, and reads zero.
84#[derive(Component, Copy, Clone, Debug, Default)]

87/// What an element measured to, which is what [`content()`](crate::content) reads.
88///
89/// The two halves are written by different passes and mean different questions, which is the whole
90/// of width-down and height-up:
91///
92/// - **width** is max-content, written by R1 -- the widest the element would like to be, unwrapped.
93///   In a monospaced font that is a character count times a cell, so it is free and is available
94///   before any layout has happened.
95/// - **height** is measured, written by R2m -- what the element turned out to be at the width R2a
96///   gave it.
97#[derive(Component, Copy, Clone, Debug, Default)]
98pub(crate) struct Intrinsic(pub(crate) Area);

100/// Steps 6 and 7. Declared state becomes resolved geometry and resolved products, for everything.
101pub(crate) fn run(grove: &mut Grove) {
102    let order = order(&grove.tree);
103    {
104        let _step = trace_span!("resolve", elements = order.len()).entered();
105        measure(grove, &order);
106        let (boxes, ends) = axes(grove, &order);
107        // Both of the passes that shape have run, so what is not held now is a run nothing states.
108        grove.shaping.sweep();
109        extent(grove, &order, &boxes);
110        scroll(grove, &order, &boxes, &ends);
111        clip(grove, &order);
112        rank(grove, &order);
113    }
114    let _step = trace_span!("settle").entered();
115    inherit(grove, &order);
116    regions(grove, &order);
117}

119/// R1. What an element's own font makes of it: its character cell, and the widest its content would
120/// like to be.
121///
122/// Ahead of every other pass, and reading no geometry at all, because neither answer has anything to
123/// do with where the element ended up. A monospaced run's max-content width is its longest line's
124/// character count times its cell, so the whole of the down-pass's input is available before the
125/// down-pass runs -- which is what leaves only the up-pass with anything to measure.
126fn measure(grove: &mut Grove, order: &[Leaf]) {
127    let _pass = trace_span!("measure").entered();
128    let Grove {
129        tree,
130        fonts,
131        shaping,
132        layout,
133        short,
134        ..
135    } = grove;
136    for &leaf in order {
137        // No font and no size is no cell, and nothing measured in one.
138        let Some(typeface) = tree.typeface(leaf) else {
139            continue;
140        };
141        let size = typeface.size.at(*layout, *short);
142        let cell = fonts.cell(typeface.font, size);
143        let width = match tree.lettering(leaf) {
144            Some(value) => shaping
145                .shape(fonts, typeface.font, size, value)
146                .max_content(),
147            None => 0.0,
148        };
149        tree.set_cell(leaf, cell);
150        // The height half is R2m's, and is written before anything reads it.
151        tree.set_intrinsic(leaf, Area::new(width, 0.0));
152    }
153}

155/// R2a, R2m and R2b: the horizontal axis, then the measure it makes possible, then the vertical one.
156///
157/// An element with a placement in motion is resolved twice on each axis -- once for each endpoint,
158/// in the *same* context -- and the two answers are blended. That is what the resolver's purity
159/// buys: the endpoints are consistent with each other because they were asked at the same position
160/// in the same dependency order, against the same settled ancestors, and neither is remembered
161/// afterwards.
162fn axes(grove: &mut Grove, order: &[Leaf]) -> (HashMap<Leaf, Section>, HashMap<Leaf, Stretched>) {
163    let mut boxes: HashMap<Leaf, Section> = HashMap::with_capacity(order.len());
164    let mut ends: HashMap<Leaf, Stretched> = HashMap::new();
165    axis(grove, order, Axis::Horizontal, &mut boxes, &mut ends);
166    wrap(grove, order, &boxes);
167    axis(grove, order, Axis::Vertical, &mut boxes, &mut ends);
168    (boxes, ends)
169}

171/// One axis of the whole tree, in dependency order, through the one pure resolver.
172fn axis(
173    grove: &Grove,
174    order: &[Leaf],
175    axis: Axis,
176    boxes: &mut HashMap<Leaf, Section>,
177    ends: &mut HashMap<Leaf, Stretched>,
178) {
179    let _pass = trace_span!("axis", vertical = (axis == Axis::Vertical)).entered();
180    let viewport = Section::new(Position::default(), grove.viewport);
181    let fallback = Location::default();
182    for &leaf in order {
183        let context = context(grove, boxes, viewport, leaf, axis);
184        let (span, stretched) = geometry(grove, leaf, &fallback, &context, axis);
185        if let Some((from, to)) = stretched {
186            ends.entry(leaf).or_default().set(axis, from, to);
187        }
188        let section = boxes.entry(leaf).or_default();
189        match axis {
190            Axis::Horizontal => {
191                section.position.x = span.near;
192                section.area.width = span.extent();
193            }
194            Axis::Vertical => {
195                section.position.y = span.near;
196                section.area.height = span.extent();
197            }
198        }
199    }
200}

202/// One axis of `leaf`, whichever of the two ways its placement is stated.
203///
204/// A box resolves to a span directly. A **trace** resolves to two positions, and the span is the
205/// distance between them grown by half the stroke's weight on each side -- which is what gives a
206/// rule, whose ends share a coordinate on one axis, a box on that axis at all. Both ends come back
207/// too, because the span cannot say which of its diagonals they are.
208fn geometry(
209    grove: &Grove,
210    leaf: Leaf,
211    fallback: &Location,
212    context: &Context,
213    axis: Axis,
214) -> (Span, Option<(f32, f32)>) {
215    let Some(traced) = grove.tree.traced(leaf) else {
216        return (span(grove, leaf, fallback, context, axis), None);
217    };
218    let half = grove.tree.stroke(leaf).unwrap_or_default().half();
219    let (from, to) = (locate(&traced.from, context), locate(&traced.to, context));
220    (
221        Span {
222            near: from.min(to) - half,
223            far: from.max(to) + half,
224        },
225        Some((from, to)),
226    )
227}

229/// One axis of where `leaf` currently is: what it declares, blended with the endpoint a motion left.
230///
231/// The one place a placement becomes a span, so a measure and a layout are answering the same
232/// question -- an element in motion is measured where it *is* rather than where it is going.
233fn span(grove: &Grove, leaf: Leaf, fallback: &Location, context: &Context, axis: Axis) -> Span {
234    let location = grove.tree.location(leaf).unwrap_or(fallback);
235    let target = resolve(pinned(location, grove, axis), context);
236    match grove.aspen.location(leaf) {
237        Some((departed, at)) => departure(departed, grove, context, axis).blend(target, at),
238        None => target,
239    }
240}

242/// How one axis of a placement is pinned down, at the breakpoint in force.
243///
244/// A lookup rather than arithmetic, which is why it is here and not in the resolver: picking a
245/// configuration out of a placement is not part of resolving one.
246fn pinned<'a>(location: &'a Location, grove: &Grove, axis: Axis) -> &'a Config {
247    let axes = location.axes(grove.layout, grove.short);
248    match axis {
249        Axis::Horizontal => &axes.horizontal,
250        Axis::Vertical => &axes.vertical,
251    }
252}

254/// One axis of the endpoint a motion left.
255///
256/// A placement it left is resolved in the context the target was resolved in, so anything that
257/// moves one end moves both. A snapshot is already an answer: it is the box the element was blended
258/// to when it was retargeted, and it never corresponded to a placement that could be re-resolved.
259fn departure(
260    departed: &Departed<Location, Section>,
261    grove: &Grove,
262    context: &Context,
263    axis: Axis,
264) -> Span {
265    match departed {
266        Departed::Declared(location) => resolve(pinned(location, grove, axis), context),
267        Departed::Snapshot(section) => Span::of(*section, axis),
268    }
269}

271/// R2m. How tall each element's contents turned out at the width R2a gave it.
272///
273/// Bottom-up, so everything inside an element is measured before the element asks. Two things are
274/// measured, and an element takes the greater of them, because both answer the one question
275/// [`content()`](crate::content) asks -- *how large is what is inside me*:
276///
277/// - a run of glyphs wraps at its own resolved width, and its lines times its cell is its height
278/// - anything with elements grown under it takes the furthest any of them reaches down
279///
280/// The second is resolved in the element's own space, with its vertical extent taken as **not yet
281/// known**, which is exactly what it is at this point in the frame. So a child that reads that
282/// extent -- `100.pct()`, a row of a grid, an anchor's edge -- contributes nothing to the measure
283/// and is given its real height by R2b like anything else. That is the correct reading rather than a
284/// limitation: a child sized to its trunk cannot also be what sizes it, and nothing is asked to
285/// converge.
286fn wrap(grove: &mut Grove, order: &[Leaf], boxes: &HashMap<Leaf, Section>) {
287    let _pass = trace_span!("wrap").entered();
288    let fallback = Location::default();
289    for &leaf in order.iter().rev() {
290        let width = boxes.get(&leaf).copied().unwrap_or_default().width();
291        let height = wrapped(grove, leaf, width).max(reach(grove, boxes, &fallback, leaf));
292        let mut intrinsic = grove.tree.intrinsic(leaf);
293        intrinsic.height = height;
294        grove.tree.set_intrinsic(leaf, intrinsic);
295    }
296}

298/// How tall `leaf`'s own run of glyphs is at `width`, or zero if it says nothing.
299fn wrapped(grove: &mut Grove, leaf: Leaf, width: f32) -> f32 {
300    let Grove {
301        tree,
302        fonts,
303        shaping,
304        layout,
305        short,
306        ..
307    } = grove;
308    let Some(typeface) = tree.typeface(leaf) else {
309        return 0.0;
310    };
311    let Some(value) = tree.lettering(leaf) else {
312        return 0.0;
313    };
314    let size = typeface.size.at(*layout, *short);
315    shaping
316        .shape(fonts, typeface.font, size, value)
317        .measure(width)
318}

320/// How far the elements grown under `leaf` reach below its top edge.
321///
322/// Only the children that describe their own extent are counted. One that reads a vertical box --
323/// a percentage of this element, a row of its grid, an anchor's edge -- is asking how tall
324/// something else is, so it cannot be what decides how tall this is. See
325/// [`Config::measurable`](crate::placement::role::Config::measurable).
326fn reach(grove: &Grove, boxes: &HashMap<Leaf, Section>, fallback: &Location, leaf: Leaf) -> f32 {
327    let mut reach: f32 = 0.0;
328    for child in grove.tree.branches(leaf) {
329        if !measurable(grove, child, fallback) {
330            continue;
331        }
332        let context = raised(grove, boxes, leaf, child);
333        reach = reach.max(
334            geometry(grove, child, fallback, &context, Axis::Vertical)
335                .0
336                .far,
337        );
338    }
339    reach
340}

342/// Whether `leaf`'s vertical placement describes its own extent, and so counts toward the measure
343/// of what it is grown under.
344///
345/// One question with two spellings, because a placement has two. A box asks it of its vertical
346/// configuration; a trace asks it of both of its ends, since either one reading a vertical box is
347/// enough to make the answer circular.
348fn measurable(grove: &Grove, leaf: Leaf, fallback: &Location) -> bool {
349    match grove.tree.traced(leaf) {
350        Some(traced) => traced.from.measurable() && traced.to.measurable(),
351        None => pinned(
352            grove.tree.location(leaf).unwrap_or(fallback),
353            grove,
354            Axis::Vertical,
355        )
356        .measurable(),
357    }
358}

360/// What one child resolves its vertical axis against while its trunk is being measured.
361///
362/// The same [`Context`] R2b will build, with every vertical reading taken as zero: no box on this
363/// axis has resolved yet, which is the point of measuring. Every horizontal reading is real, so a
364/// height stated in columns or read off a width still answers.
365fn raised(grove: &Grove, boxes: &HashMap<Leaf, Section>, trunk: Leaf, child: Leaf) -> Context {
366    let flattened = |leaf: Leaf| {
367        let section = boxes.get(&leaf).copied().unwrap_or_default();
368        Basis {
369            section: Section::new(
370                Position::new(section.left(), 0.0),
371                Area::new(section.width(), 0.0),
372            ),
373            intrinsic: grove.tree.intrinsic(leaf),
374            tracks: grove
375                .tree
376                .grid(leaf)
377                .unwrap_or_default()
378                .tracks(grove.layout, grove.short),
379            cell: grove.tree.cell(leaf),
380        }
381    };
382    Context {
383        axis: Axis::Vertical,
384        own: Basis {
385            section: Section::default(),
386            intrinsic: grove.tree.intrinsic(child),
387            tracks: Tracks::default(),
388            cell: grove.tree.cell(child),
389        },
390        trunk: flattened(trunk),
391        anchor: match grove.tree.anchor(child) {
392            Some(anchor) => flattened(anchor),
393            None => Basis::default(),
394        },
395    }
396}

398/// R3. How far each region's content reaches, from where its children landed.
399///
400/// Bottom-up, so a subtree is measured before whatever contains it. Two things are deliberately not
401/// consulted: what is currently drawn, because content scrolled out of sight is exactly what an
402/// extent describes and has to remain reachable; and the offset, because measuring against it would
403/// make the extent depend on the clamp that depends on the extent.
404///
405/// A child that scrolls in its own right contributes its box and not its content -- what overflows
406/// inside it is its own to reach, and is already reachable there.
407///
408/// A **visible child that is simply far away is content**, and is counted. That is the answer
409/// `views.md` gives to the parking footgun, and it is deliberately not a guess about intent: if it
410/// is visible and out there it is reachable, which is coherent where the old behaviour was a
411/// surprise. What removes a child from the extent is the app saying so -- [`visible(false)`] takes
412/// it and its subtree out, and [`pinned`] takes out something that is inside the region without
413/// travelling with it.
414///
415/// [`visible(false)`]: crate::Place::visible
416/// [`pinned`]: crate::Place::pinned
417fn extent(grove: &mut Grove, order: &[Leaf], boxes: &HashMap<Leaf, Section>) {
418    let _pass = trace_span!("extent").entered();
419    // The far corner of everything under an element, in absolute coordinates.
420    let mut reach: HashMap<Leaf, Position> = HashMap::with_capacity(order.len());
421    for &leaf in order.iter().rev() {
422        // A hidden element is not content, and neither is anything under it: that is what makes
423        // hiding the whole answer for parking something out of the way, rather than half of one.
424        if !grove.tree.visible(leaf).0 {
425            continue;
426        }
427        let section = boxes.get(&leaf).copied().unwrap_or_default();
428        let mut far = Position::new(section.right(), section.bottom());
429        for child in grove.tree.branches(leaf) {
430            // The two ways an element grown inside a region is not part of its content, and each
431            // is left out here by the same one declaration that says the rest of what it means --
432            // which is what keeps the halves from disagreeing.
433            //
434            // A pinned child does not move with the content. A floating one moves with it but sits
435            // over the region rather than in it, so it is not content either: an overlay that
436            // invented room to scroll to is the scrollbar nobody ordered.
437            if grove.tree.pinned(child) || grove.tree.floats(child).is_some() {
438                continue;
439            }
440            let Some(child) = reach.get(&child).copied() else {
441                continue;
442            };
443            far = Position::new(far.x.max(child.x), far.y.max(child.y));
444        }
445        if let Some(scroll) = grove.tree.scrolls(leaf) {
446            grove.tree.set_extent(leaf, reached(scroll, section, far));
447            far = Position::new(section.right(), section.bottom());
448        }
449        reach.insert(leaf, far);
450    }
451}

453/// How far a region's content reaches, on each axis.
454///
455/// Three rules, and each of them is what stops a scrollbar nobody ordered:
456///
457/// - **only along a declared axis.** An axis the region does not scroll reads its own box and
458///   nothing else, so a child extending sideways out of a column that scrolls down is simply out of
459///   frame. Most accidental extent came from the axis nobody was scrolling.
460/// - **measured outward from the content origin, clamped at the near side.** A child at a negative
461///   offset creates no range to scroll *back* into: content above the origin is a layout mistake,
462///   and the previous behaviour turned it into a feature.
463/// - **never smaller than the region's own box**, so an empty region has a range of zero rather
464///   than a negative one.
465///
466fn reached(scroll: Scroll, section: Section, far: Position) -> Area {
467    let along = |axis: Axis, reach: f32, near: f32| {
468        let own = section.area.along(axis);
469        if !scroll.covers(axis) {
470            return own;
471        }
472        (reach - near).max(own)
473    };
474    Area::new(
475        along(Axis::Horizontal, far.x, section.left()),
476        along(Axis::Vertical, far.y, section.top()),
477    )
478}

480/// R4. Where each element is drawn: where the layout put it, less what its scrolling ancestors have
481/// moved.
482///
483/// A region's own box does not move under its own offset -- what moves is everything grown inside
484/// it. The offset is clamped here, against the extent R3 just measured, so a region whose content
485/// shrank under it comes back into range on the next frame rather than staying somewhere it can no
486/// longer reach.
487///
488/// The three ways a region is *asked* to move -- a coast still running from a release, a
489/// [`scroll`](crate::Grow::scroll) written this frame, and a
490/// [`Motion::Scroll`](crate::Motion::Scroll) part way through -- are answered first, here rather
491/// than where they were written, because all three need the extent and the extent is one pass old.
492fn scroll(
493    grove: &mut Grove,
494    order: &[Leaf],
495    boxes: &HashMap<Leaf, Section>,
496    ends: &HashMap<Leaf, Stretched>,
497) {
498    let _pass = trace_span!("scroll", coasting = grove.coasting.len()).entered();
499    view::asked(grove, boxes);
500    let mut accumulated: HashMap<Leaf, Position> = HashMap::with_capacity(order.len());
501    // What a *pinned* child of each element receives, which is the accumulation with its nearest
502    // scrolling ancestor left out of it.
503    let mut unpinned: HashMap<Leaf, Position> = HashMap::with_capacity(order.len());
504    for &leaf in order {
505        let placed = boxes.get(&leaf).copied().unwrap_or_default();
506        let trunk = grove.tree.trunk(leaf);
507        let inherited = trunk
508            .and_then(|trunk| accumulated.get(&trunk).copied())
509            .unwrap_or_default();
510        let outside = trunk
511            .and_then(|trunk| unpinned.get(&trunk).copied())
512            .unwrap_or_default();
513        // A pinned element does not receive its nearest scrolling ancestor's offset, and receives
514        // every offset outside that one: pinning is relative to the region the element sits in and
515        // says nothing about what contains that region.
516        let applied = match grove.tree.pinned(leaf) {
517            true => outside,
518            false => inherited,
519        };
520        let drawn = Section::new(
521            Position::new(placed.left() - applied.x, placed.top() - applied.y),
522            placed.area,
523        );
524        grove.tree.settle(leaf, placed, drawn);
525        // A stroke's ends travel with its box, by the same offset, because they are the same
526        // geometry said two ways.
527        if let Some(stretched) = ends.get(&leaf) {
528            grove.tree.set_stretched(leaf, stretched.less(applied));
529        }
530        let (carried, escaped) = match grove.tree.scrolls(leaf) {
531            Some(scroll) => {
532                let clamped = clamp(grove, leaf, scroll, placed);
533                (
534                    Position::new(applied.x + clamped.x, applied.y + clamped.y),
535                    // A pinned child of this element is pinned to *this* region, so its offset is
536                    // the accumulation as it stood before this region's own.
537                    applied,
538                )
539            }
540            // Nothing here to be pinned against, so a pinned child of this element is pinned to
541            // whatever region contains it, exactly as a pinned sibling of this element would be.
542            None => (applied, outside),
543        };
544        accumulated.insert(leaf, carried);
545        unpinned.insert(leaf, escaped);
546    }
547}

549/// A region's offset, held to what it can actually reach.
550///
551/// An axis that was not declared does not scroll, so nothing can have moved along it -- and a
552/// region whose extent shrank under it is brought back into range here rather than left somewhere
553/// it can no longer get to.
554fn clamp(grove: &mut Grove, leaf: Leaf, scroll: Scroll, placed: Section) -> Position {
555    let offset = grove.tree.offset(leaf);
556    let extent = grove.tree.extent(leaf);
557    let mut clamped = Position::default();
558    for axis in Axis::BOTH {
559        if scroll.covers(axis) {
560            clamped = clamped.set(
561                axis,
562                offset
563                    .along(axis)
564                    .clamp(0.0, range(extent, placed.area, axis)),
565            );
566        }
567    }
568    if clamped != offset {
569        grove.tree.set_offset(leaf, clamped);
570    }
571    clamped
572}

574/// R5. What a scrolling ancestor leaves visible of each element.
575///
576/// A rect, and nothing else. A region does not clip itself, only what is grown inside it, and an
577/// element with no scrolling ancestor is clipped by nothing at all. Whether an element is *culled*
578/// is extraction's decision from this rect, and is never recorded on the element -- so there is no
579/// state saying "currently clipped away" for anything else, extent first among them, to read.
580fn clip(grove: &mut Grove, order: &[Leaf]) {
581    let _pass = trace_span!("clip").entered();
582    let mut passed: HashMap<Leaf, Section> = HashMap::with_capacity(order.len());
583    // What a *floating* child of each element is clipped to, which is the intersection with its
584    // nearest scrolling ancestor's own rect left out of it. The same second accumulation R4 carries
585    // for a pinned child, and for the same reason: both say "not part of this region", and both
586    // have to say it about the region the element is actually in rather than about all of them.
587    let mut escaped: HashMap<Leaf, Section> = HashMap::with_capacity(order.len());
588    for &leaf in order {
589        let trunk = grove.tree.trunk(leaf);
590        let inherited = trunk
591            .and_then(|trunk| passed.get(&trunk).copied())
592            .unwrap_or(Clipped::unbounded().0);
593        let outside = trunk
594            .and_then(|trunk| escaped.get(&trunk).copied())
595            .unwrap_or(Clipped::unbounded().0);
596        // A floating element is positioned outside the region on purpose, so cutting it off at the
597        // region's edge would undo the placement that put it there. How far out it reaches is the
598        // element's own statement, because no one answer is right everywhere.
599        let applied = match grove.tree.floats(leaf) {
600            // Held by whatever holds the region it left.
601            Some(Escape::Region) => outside,
602            // Held by nothing, which extraction reads as the surface: a clip is never wider than
603            // what is being drawn on.
604            Some(Escape::Surface) => Clipped::unbounded().0,
605            // Held by the element named, which is what that element passes down to what it holds.
606            Some(Escape::Within(named)) => match within(grove, &passed, leaf, named) {
607                Some(clip) => clip,
608                // Named something that is not above it, so it holds nothing. Falling back to the
609                // near answer rather than the far one: an element escapes no further than it was
610                // told to, and a mis-named ancestor is not permission to leave everything.
611                None => outside,
612            },
613            None => inherited,
614        };
615        grove.tree.set_clip(leaf, applied);
616        let scrolls = grove.tree.scrolls(leaf).is_some();
617        passed.insert(
618            leaf,
619            match scrolls {
620                true => applied.intersect(grove.tree.drawn(leaf)),
621                false => applied,
622            },
623        );
624        escaped.insert(
625            leaf,
626            match scrolls {
627                true => applied,
628                false => outside,
629            },
630        );
631    }
632}
633
634/// What `named` leaves visible of what it holds, if it is above `leaf` at all.
635///
636/// The walk is up the trunks from `leaf`, so an element that names something beside it rather than
637/// above it gets `None` and is answered as though it had named the region it is in. Naming the
638/// element rather than counting regions is what makes this survive a wrapper being added between
639/// the two, which a count would not.
640fn within(
641    grove: &Grove,
642    passed: &HashMap<Leaf, Section>,
643    leaf: Leaf,
644    named: Leaf,
645) -> Option<Section> {
646    let mut step = grove.tree.trunk(leaf);
647    while let Some(above) = step {
648        if above == named {
649            return passed.get(&named).copied();
650        }
651        step = grove.tree.trunk(above);
652    }
653    None
654}

656/// R7. The three off-states, resolved over each element's whole ancestry.
657///
658/// One walk, in the same order the axes used, which puts every trunk before what hangs off it.
659/// Nothing has a cascade to write: an element grown under a disabled trunk is disabled on its first
660/// frame because the pass does not care when it arrived, and enabling that trunk leaves anything
661/// disabled in its own right disabled because the product is over the whole ancestry rather than a
662/// single bit that was overwritten on the way down.
663fn inherit(grove: &mut Grove, order: &[Leaf]) {
664    let _pass = trace_span!("inherit").entered();
665    let mut products: HashMap<Leaf, Inherited> = HashMap::with_capacity(order.len());
666    for &leaf in order {
667        let trunk = grove
668            .tree
669            .trunk(leaf)
670            .and_then(|trunk| products.get(&trunk).copied())
671            .unwrap_or_default();
672        let product = Inherited::under(
673            trunk,
674            grove.tree.visible(leaf),
675            grove.tree.opacity(leaf),
676            grove.tree.disabled(leaf),
677        );
678        products.insert(leaf, product);
679        grove.tree.set_inherited(leaf, product);
680    }
681}

683/// R8. The box stack the next frame's dispatch reads.
684///
685/// Membership is universal: an element is here because it is there. What is left out is only what
686/// is not there at all -- hidden, fully transparent, or clipped away by a region it sits inside.
687/// `intangible` is not a way out of the stack; it is carried on the region and decides what may be
688/// the top of it.
689fn regions(grove: &mut Grove, order: &[Leaf]) {
690    let _pass = trace_span!("regions").entered();
691    let mut ranked = Vec::with_capacity(order.len());
692    for &leaf in order {
693        let inherited = grove.tree.inherited(leaf);
694        if !inherited.present() {
695            continue;
696        }
697        let section = grove.tree.drawn(leaf);
698        let clip = grove.tree.clip(leaf);
699        if section.intersect(clip).is_empty() {
700            continue;
701        }
702        let gestures = grove.tree.gestures(leaf);
703        ranked.push((
704            grove.tree.rank(leaf),
705            Region {
706                leaf,
707                section,
708                clip,
709                shape: gestures.shape,
710                tangible: !gestures.intangible,
711                receives: gestures.receives,
712                disabled: inherited.disabled,
713            },
714        ));
715    }
716    grove.stack.settle(ranked);
717}

719/// R6. Declared elevation accumulates down the tree, and allocation order settles what it leaves
720/// equal.
721///
722/// One walk in the same dependency order the axes used, which puts every trunk before what hangs
723/// off it. Nothing here reads a box: where an element sits in the stack has nothing to do with
724/// where it sits on the surface.
725fn rank(grove: &mut Grove, order: &[Leaf]) {
726    let _pass = trace_span!("rank").entered();
727    let mut stacks: HashMap<Leaf, i32> = HashMap::with_capacity(order.len());
728    for &leaf in order {
729        let trunk = grove
730            .tree
731            .trunk(leaf)
732            .and_then(|trunk| stacks.get(&trunk).copied())
733            .unwrap_or_default();
734        let stack = grove.tree.elevation(leaf).accumulate(trunk);
735        stacks.insert(leaf, stack);
736        grove.tree.set_rank(
737            leaf,
738            ResolvedElevation {
739                stack,
740                growth: grove.tree.growth(leaf).0,
741            },
742        );
743    }
744}

746/// Everything one axis of `leaf` resolves against.
747fn context(
748    grove: &Grove,
749    boxes: &HashMap<Leaf, Section>,
750    viewport: Section,
751    leaf: Leaf,
752    axis: Axis,
753) -> Context {
754    Context {
755        axis,
756        // Its box is the answer being computed, and its own grid divides it for its children
757        // rather than for itself, so neither is readable here.
758        own: Basis {
759            section: Section::default(),
760            intrinsic: grove.tree.intrinsic(leaf),
761            tracks: Tracks::default(),
762            cell: grove.tree.cell(leaf),
763        },
764        // A top-level element has no trunk, and fills the viewport instead.
765        trunk: basis(grove, boxes, grove.tree.trunk(leaf), viewport),
766        // A placement that reads an anchor it has not been given resolves against a zero box.
767        anchor: basis(grove, boxes, grove.tree.anchor(leaf), Section::default()),
768    }
769}

771/// What `leaf` offers a placement reading it, or `fallback` in place of a box when there is no such
772/// element or it has not resolved yet.
773fn basis(
774    grove: &Grove,
775    boxes: &HashMap<Leaf, Section>,
776    leaf: Option<Leaf>,
777    fallback: Section,
778) -> Basis {
779    let Some(leaf) = leaf else {
780        return Basis {
781            section: fallback,
782            ..Basis::default()
783        };
784    };
785    Basis {
786        section: boxes.get(&leaf).copied().unwrap_or(fallback),
787        intrinsic: grove.tree.intrinsic(leaf),
788        tracks: grove
789            .tree
790            .grid(leaf)
791            .unwrap_or_default()
792            .tracks(grove.layout, grove.short),
793        cell: grove.tree.cell(leaf),
794    }
795}

797/// Every live element, ordered so that nothing resolves before what it depends on.
798///
799/// An anchor may point anywhere -- a later sibling, a cousin, an element in another subtree -- so
800/// this is ordered by dependency rather than by tree depth: an element resolves after its parent
801/// *and* after whatever it anchors to.
802///
803/// A chain of anchors cannot close on itself, because one that would is refused where it is written.
804/// A trunk and an anchor still can, between them: an element anchored to something grown under it
805/// waits on a box that is waiting on its own. That is not the same contradiction -- both boxes
806/// resolve, just not both against a settled other -- so the remainder is ordered by allocation order
807/// and resolves against what its dependency last was. **Every live element is in the order**, which
808/// is the property the passes downstream rely on: an element left out would have no box, no rank and
809/// no place in the stack, and nothing would say so.
810fn order(tree: &Tree) -> Vec<Leaf> {
811    let leaves = tree.leaves();
812    let mut waiting: HashMap<Leaf, usize> = HashMap::with_capacity(leaves.len());
813    let mut dependents: HashMap<Leaf, Vec<Leaf>> = HashMap::with_capacity(leaves.len());
814    for &leaf in &leaves {
815        let depends_on = [tree.trunk(leaf), tree.anchor(leaf)];
816        let mut count = 0;
817        for on in depends_on.into_iter().flatten() {
818            if tree.is_live(on) {
819                count += 1;
820                dependents.entry(on).or_default().push(leaf);
821            }
822        }
823        waiting.insert(leaf, count);
824    }
825    let mut ready: Vec<Leaf> = leaves
826        .iter()
827        .copied()
828        .filter(|leaf| waiting[leaf] == 0)
829        .collect();
830    let mut order = Vec::with_capacity(leaves.len());
831    let mut next = 0;
832    while next < ready.len() {
833        let leaf = ready[next];
834        next += 1;
835        order.push(leaf);
836        for dependent in dependents.get(&leaf).into_iter().flatten() {
837            let waiting = waiting.get_mut(dependent).expect("a live dependent");
838            *waiting -= 1;
839            if *waiting == 0 {
840                ready.push(*dependent);
841            }
842        }
843    }
844    if order.len() != leaves.len() {
845        order.extend(leaves.iter().copied().filter(|leaf| waiting[leaf] != 0));
846    }
847    order
848}
