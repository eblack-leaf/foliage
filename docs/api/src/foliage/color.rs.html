7/// An sRGB color with an alpha channel, one channel per field.
8///
9/// Channels are separate floats in `0.0..=1.0` rather than packed bytes, because each is
10/// interpolated independently when a color is animated and because that is the form a shader takes.
11///
12/// This is what a [`Palette`](crate::Palette) tone resolves to, and the form a palette is stated
13/// in. An element declares the tone rather than the color, so that what a tone resolves to can be
14/// changed in one place.
15///
16/// `#[repr(C)]` over four floats, which is the layout a shader reads a `vec4` in.
17#[repr(C)]
18#[derive(Copy, Clone, Debug, PartialEq, Pod, Zeroable)]
19pub struct Color {
20    /// The red channel, in `0.0..=1.0`.
21    pub red: f32,
22    /// The green channel, in `0.0..=1.0`.
23    pub green: f32,
24    /// The blue channel, in `0.0..=1.0`.
25    pub blue: f32,
26    /// Coverage, in `0.0..=1.0`: `0.0` paints nothing and `1.0` paints over what is behind it.
27    pub alpha: f32,
28}
29
30impl Color {
31    /// An opaque color.
32    pub fn rgb(red: f32, green: f32, blue: f32) -> Self {
33        Self::rgba(red, green, blue, 1.0)
34    }
35
36    /// A color with an alpha channel. Every channel is clamped to `0.0..=1.0`.
37    pub fn rgba(red: f32, green: f32, blue: f32, alpha: f32) -> Self {
38        Self {
39            red: red.clamp(0.0, 1.0),
40            green: green.clamp(0.0, 1.0),
41            blue: blue.clamp(0.0, 1.0),
42            alpha: alpha.clamp(0.0, 1.0),
43        }
44    }
45
46    /// This color a fraction `at` of the way to `other`, channel by channel.
47    ///
48    /// What a [`Motion::Color`](crate::Motion::Color) resolves to each frame. Channel-wise because
49    /// that is what separate floats are for, and it is why a color is held as four of them rather
50    /// than as packed bytes.
51    pub(crate) fn blend(self, other: Self, at: f32) -> Self {
52        Self::rgba(
53            blend(self.red, other.red, at),
54            blend(self.green, other.green, at),
55            blend(self.blue, other.blue, at),
56            blend(self.alpha, other.alpha, at),
57        )
58    }
59
60    /// This color at a fraction of its own alpha.
61    ///
62    /// What an element's resolved opacity does to what it is painted in. Taken at extraction rather
63    /// than written onto the element, so nothing holds a second copy of a color that a repaint
64    /// would have to find.
65    pub(crate) fn faded(self, opacity: f32) -> Self {
66        Self {
67            alpha: self.alpha * opacity.clamp(0.0, 1.0),
68            ..self
69        }
70    }
71
72    /// This color as OKLab lightness and its two opponent axes.
73    ///
74    /// The space a [`Scheme`](crate::Scheme) walks a ramp in. Lightness there is perceptual, so a
75    /// fixed offset reads as the same size of change at either end of a ramp, and the two axes carry
76    /// hue and chroma together -- which is what lets a step move lightness while leaving the rest of
77    /// the color where the seed put it. Interpolating sRGB channels toward black or white does
78    /// neither: it desaturates and drifts in hue.
79    ///
80    /// Alpha is not part of the conversion and is carried separately.
81    pub(crate) fn oklab(self) -> (f32, f32, f32) {
82        let red = linear(self.red);
83        let green = linear(self.green);
84        let blue = linear(self.blue);
85        let long = (0.412_221_46 * red + 0.536_332_55 * green + 0.051_445_995 * blue).cbrt();
86        let medium = (0.211_903_5 * red + 0.680_699_5 * green + 0.107_396_96 * blue).cbrt();
87        let short = (0.088_302_46 * red + 0.281_718_85 * green + 0.629_978_7 * blue).cbrt();
88        (
89            0.210_454_26 * long + 0.793_617_8 * medium - 0.004_072_047 * short,
90            1.977_998_5 * long - 2.428_592_2 * medium + 0.450_593_7 * short,
91            0.025_904_037 * long + 0.782_771_77 * medium - 0.808_675_77 * short,
92        )
93    }
94
95    /// The color at these OKLab coordinates, in sRGB.
96    ///
97    /// Shifting lightness at a fixed hue and chroma can leave the sRGB gamut, most often at a ramp's
98    /// bright end for a saturated seed. Rather than clip a channel -- which shifts hue, the one
99    /// thing a ramp holds -- chroma is backed off toward the neutral of the same lightness until the
100    /// color fits, so the step lands as close to its seed's hue as sRGB can carry. The `bool` is
101    /// whether it had to be, which is what a ramp traces.
102    pub(crate) fn from_oklab(lightness: f32, a: f32, b: f32, alpha: f32) -> (Self, bool) {
103        let direct = oklab_linear(lightness, a, b);
104        if in_gamut(direct) {
105            return (encode(direct, alpha), false);
106        }
107        // Chroma zero is the neutral of this lightness and always fits, so the low end of the search
108        // is known to be good from the start and the loop only ever raises it.
109        let (mut fits, mut over) = (0.0f32, 1.0f32);
110        for _ in 0..16 {
111            let between = 0.5 * (fits + over);
112            if in_gamut(oklab_linear(lightness, a * between, b * between)) {
113                fits = between;
114            } else {
115                over = between;
116            }
117        }
118        (
119            encode(oklab_linear(lightness, a * fits, b * fits), alpha),
120            true,
121        )
122    }
123}

125/// One sRGB channel with its transfer function removed.
126fn linear(channel: f32) -> f32 {
127    if channel <= 0.040_45 {
128        channel / 12.92
129    } else {
130        ((channel + 0.055) / 1.055).powf(2.4)
131    }
132}
133
134/// One linear channel with the sRGB transfer function applied.
135fn transfer(channel: f32) -> f32 {
136    if channel <= 0.003_130_8 {
137        channel * 12.92
138    } else {
139        1.055 * channel.powf(1.0 / 2.4) - 0.055
140    }
141}
142
143/// These OKLab coordinates as linear RGB, which is where the gamut is a range rather than a shape.
144fn oklab_linear(lightness: f32, a: f32, b: f32) -> (f32, f32, f32) {
145    let long = (lightness + 0.396_337_78 * a + 0.215_803_76 * b).powi(3);
146    let medium = (lightness - 0.105_561_346 * a - 0.063_854_17 * b).powi(3);
147    let short = (lightness - 0.089_484_18 * a - 1.291_485_5 * b).powi(3);
148    (
149        4.076_741_7 * long - 3.307_711_6 * medium + 0.230_969_94 * short,
150        -1.268_438 * long + 2.609_757_4 * medium - 0.341_319_38 * short,
151        -0.004_196_086_3 * long - 0.703_418_6 * medium + 1.707_614_7 * short,
152    )
153}

155/// Whether linear RGB names a color sRGB can carry, with the slack a round trip costs.
156fn in_gamut((red, green, blue): (f32, f32, f32)) -> bool {
157    const SLACK: f32 = 1e-4;
158    [red, green, blue]
159        .into_iter()
160        .all(|channel| channel >= -SLACK && channel <= 1.0 + SLACK)
161}
162
163/// Linear RGB as a [`Color`].
164fn encode((red, green, blue): (f32, f32, f32), alpha: f32) -> Color {
165    Color::rgba(transfer(red), transfer(green), transfer(blue), alpha)
