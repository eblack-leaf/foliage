1//! The soft keyboard -- raised by focus, and by nothing else.
2//!
3//! A phone has no keyboard until something asks for one, and the thing that asks is a field holding
4//! focus. That is the whole rule, and it needs no declaration of its own: focus already rests only
5//! on what said [`interactive`](crate::Place::interactive), and a [`TextInput`](crate::TextInput)
6//! is the only thing in the engine that is typed *into*. So the keyboard is raised for the field
7//! that holds focus and lowered for everything else, decided in the same breath as the caret's
8//! visibility and from the same state.
9//!
10//! What a field may say is which **[`Keypad`]** it wants, because a telephone number behind a full
11//! alphabet is the one thing the platform cannot be left to guess.
12//!
13//! # The web owns the keyboard, so the web owns the keys
14//!
15//! There is no way to raise a browser's keyboard except by giving DOM focus to a real input, and
16//! doing that takes keyboard focus off the canvas winit is listening on -- so every key typed after
17//! it would be swallowed by an input the person cannot see. The hidden input therefore hands them
18//! back: what it composes arrives as text and what it does not is read from the key itself, both
19//! entering [`Incoming`](crate::interaction::input::Incoming) exactly where a translated winit
20//! event enters it.
21//!
22//! Off the web nothing is raised. Android would be the other platform with a keyboard to raise and
23//! there is no Android build to raise it from, so what is here is the seam rather than a stub.
24

29/// Which keys a soft keyboard offers.
30///
31/// A hint and not a constraint: it changes what is easy to type and never what a field accepts, so
32/// a [`Keypad::Number`] field can still be pasted a word into and is still the app's to validate.
33/// Ignored where the platform has no keyboard of its own to raise.
34#[derive(Component, Copy, Clone, Debug, Default, PartialEq, Eq)]
35pub enum Keypad {
36    /// Letters. What a field asks for unless it says otherwise.
37    #[default]
38    Text,
39    /// Digits, for a quantity or a code.
40    Number,
41    /// A dialling pad.
42    Telephone,
43}
44
45impl Keypad {
46    /// What the browser is told to offer, as `inputmode` names it.
47    #[cfg_attr(not(target_family = "wasm"), allow(dead_code))]
48    fn mode(self) -> &'static str {
49        match self {
50            Self::Text => "text",
51            Self::Number => "numeric",
52            Self::Telephone => "tel",
53        }
54    }
55}

57/// The soft keyboard, and what it is currently showing.
58///
59/// What is raised is engine state on every target, and raising it is the platform's -- so the rule
60/// that focus decides the keyboard is proven headlessly while the DOM half stays the edge the suite
61/// cannot reach, exactly as the winit translation is.
62#[derive(Default)]
63pub(crate) struct Keyboard {
64    /// What is up, if anything is. Kept so the platform is asked only when the answer changes.
65    raised: Option<Keypad>,
66    /// The hidden input the browser raises its keyboard for. Absent until
67    /// [`attach`](Keyboard::attach), and absent for good under the headless suite.
68    #[cfg(target_family = "wasm")]
69    trigger: Option<web::Trigger>,
70}
71
72impl Keyboard {
73    /// What is raised, if anything is.
74    ///
75    /// Read by the headless suite, which is the only side of the seam that can observe it: asking
76    /// for a keyboard is the engine's and raising one is the host's.
77    #[cfg(test)]
78    pub(crate) fn raised(&self) -> Option<Keypad> {
79        self.raised
80    }
81
82    /// Builds what the platform raises its keyboard for, once, at boot.
83    ///
84    /// `wake` is what a key captured outside a frame rouses the loop with -- while the hidden input
85    /// holds DOM focus the canvas hears nothing, so nothing else would ask for the frame that
86    /// delivers it.
87    #[cfg_attr(not(target_family = "wasm"), allow(unused_variables))]
88    pub(crate) fn attach(&mut self, wake: &crate::queue::Wake) {
89        #[cfg(target_family = "wasm")]
90        {
91            self.trigger = web::Trigger::build(wake.clone());
92        }
93    }
94
95    /// Raises the keyboard `wanted` asks for, or lowers whatever is up.
96    ///
97    /// Called every frame with the whole answer rather than told about changes, because the answer
98    /// is a product of focus and is recomputed like every other product.
99    ///
100    /// What is *up* is compared, so the mode is written and the trace is made only where it moved.
101    /// DOM focus is not: the page can take it off the hidden input without the engine hearing about
102    /// it -- a press on the canvas does exactly that, and is how every tap from one field to
103    /// another arrives -- so it is asserted on every frame instead. Focusing what already holds
104    /// focus does nothing, which is what makes asserting it free.
105    pub(crate) fn raise(&mut self, wanted: Option<Keypad>) {
106        let moved = wanted != self.raised;
107        if moved {
108            self.raised = wanted;
109            debug!(keypad = ?wanted, "keyboard");
110        }
111        #[cfg(target_family = "wasm")]
112        if let Some(trigger) = &self.trigger {
113            match wanted {
114                Some(keypad) => {
115                    if moved {
116                        trigger.mode(keypad);
117                    }
118                    trigger.focus();
119                }
120                None if moved => trigger.blur(),
121                None => {}
122            }
123        }
124    }
125
126    /// What the hidden input captured since the last frame, in the order it arrived.
127    ///
128    /// Empty everywhere but the web, and empty there until something has been typed into a keyboard
129    /// this raised.
130    pub(crate) fn captured(&mut self) -> Vec<crate::interaction::input::Input> {
131        #[cfg(target_family = "wasm")]
132        if let Some(trigger) = &self.trigger {
133            return trigger.captured();
134        }
135        Vec::new()
136    }
137}

139/// The browser's half: one hidden input, and the keys it takes from the canvas given back.
140#[cfg(target_family = "wasm")]
141mod web {
142    use std::cell::RefCell;
143    use std::rc::Rc;
144
145    use wasm_bindgen::JsCast;
146    use wasm_bindgen::closure::Closure;
147
148    use super::Keypad;
149    use crate::interaction::input::{Input, Key, Modifiers};

152    /// The id the hidden input is found by. One element rather than one per [`Keypad`]: the pad is
153    /// `inputmode`, which is written before focus moves, so there is nothing three elements would
154    /// say that one cannot.
155    const TRIGGER: &str = "foliage-keypad";
156
157    /// Off screen rather than `display:none`, because an element that is not laid out cannot take
158    /// focus and an element that cannot take focus raises no keyboard. Everything that would let it
159    /// second-guess what is typed is off: the value is read and cleared on every event, so an
160    /// autocorrection would be applied to a value the person never sees.
161    const STYLE: &str = "position:absolute;left:-1px;top:-1px;width:0;height:0;\
162        min-width:0;min-height:0;padding:0;border:0;opacity:0";
163
164    /// What the hidden input has captured and no frame has taken yet.
165    ///
166    /// The browser is single-threaded, so an `Rc` is the whole of what sharing this with the DOM's
167    /// own callbacks needs.
168    #[derive(Clone, Default)]
169    struct Captured(Rc<RefCell<Vec<Input>>>);
170
171    /// The hidden input, and what it has taken.
172    pub(super) struct Trigger {
173        element: web_sys::HtmlInputElement,
174        captured: Captured,
175    }
176
177    impl Trigger {
178        /// Puts the input in the page and wires it to the input stream.
179        ///
180        /// `None` where there is no document to put it in, which is every host that is not a
181        /// browser -- the same absence the headless suite has, reached a different way.
182        pub(super) fn build(wake: Wake) -> Option<Self> {
183            let document = web_sys::window()?.document()?;
184            let element = document
185                .create_element("input")
186                .ok()?
187                .dyn_into::<web_sys::HtmlInputElement>()
188                .ok()?;
189            element.set_id(TRIGGER);
190            element.set_type("text");
191            for (name, value) in [
192                ("style", STYLE),
193                ("autocapitalize", "off"),
194                ("autocomplete", "off"),
195                ("autocorrect", "off"),
196                ("spellcheck", "false"),
197                ("aria-hidden", "true"),
198            ] {
199                element.set_attribute(name, value).ok()?;
200            }
201            document.body()?.append_child(&element).ok()?;
202            let captured = Captured::default();
203            listen(&element, &captured, &wake);
204            Some(Self { element, captured })
205        }
206
207        /// Says which keys to offer, before anything is offered.
208        ///
209        /// A mode written to an input that already holds focus changes nothing the person can see
210        /// until the keyboard has been dismissed and raised again, so this is only ever written
211        /// while the keyboard is down.
212        pub(super) fn mode(&self, keypad: Keypad) {
213            let _ = self.element.set_attribute("inputmode", keypad.mode());
214        }
215
216        /// Gives the input DOM focus, which is the only thing a browser raises a keyboard for.
217        pub(super) fn focus(&self) {
218            let _ = self.element.focus();
219        }
220
221        pub(super) fn blur(&self) {
222            let _ = self.element.blur();
223        }
224
225        pub(super) fn captured(&self) -> Vec<Input> {
226            core::mem::take(&mut *self.captured.0.borrow_mut())
227        }
228    }
229
230    /// Wires the two events a hidden input has to give back.
231    ///
232    /// `input` is what was composed -- a plain letter, a dead-key sequence, a whole word an IME
233    /// committed -- and is the only one that answers for text, because the browser is what resolved
234    /// the layout. `keydown` is for the keys that produce no text and would otherwise be spent on
235    /// the input's own editing, and for a character held with a modifier, which produces no `input`
236    /// event because it is a command rather than something to insert.
237    fn listen(element: &web_sys::HtmlInputElement, captured: &Captured, wake: &Wake) {
238        let composed = {
239            let (captured, wake) = (captured.clone(), wake.clone());
240            Closure::wrap(Box::new(move |event: web_sys::Event| {
241                let Some(input) = event
242                    .target()
243                    .and_then(|target| target.dyn_into::<web_sys::HtmlInputElement>().ok())
244                else {
245                    return;
246                };
247                let value = input.value();
248                if value.is_empty() {
249                    return;
250                }
251                // Cleared at once. What was typed is carried as keystrokes from here on, and a
252                // value left in the input would be composed against by whatever is typed next.
253                input.set_value("");
254                captured.push(
255                    value
256                        .chars()
257                        // The rule `photosynthesize` reads a winit key by: a control character is
258                        // what a key that means something produces where the platform hands text
259                        // as well, and inserting one would put it in the value.
260                        .filter(|character| !character.is_control())
261                        .map(|character| Input::Keyed(Key::Typed(character)))
262                        .collect(),
263                );
264                wake.rouse();
265            }) as Box<dyn FnMut(_)>)
266        };
267        let pressed = {
268            let (captured, wake) = (captured.clone(), wake.clone());
269            Closure::wrap(Box::new(move |event: web_sys::KeyboardEvent| {
270                // Read from this event rather than from what the engine last heard: the canvas
271                // stopped receiving `ModifiersChanged` the moment this input took focus, so the
272                // engine's own tracker is exactly as starved as the key events are.
273                let modifiers = Modifiers {
274                    shift: event.shift_key(),
275                    control: event.ctrl_key() || event.meta_key(),
276                };
277                let Some(key) = named(&event.key(), modifiers) else {
278                    return;
279                };
280                // The input's own editing would otherwise spend the key -- a backspace deleting
281                // from a value that is always empty, an arrow moving a caret nobody can see.
282                event.prevent_default();
283                captured.push(vec![
284                    Input::Modifiers(modifiers),
285                    Input::Keyed(key),
286                    Input::Modifiers(Modifiers::default()),
287                ]);
288                wake.rouse();
289            }) as Box<dyn FnMut(_)>)
290        };
291        for (name, callback) in [("input", composed.as_ref()), ("keydown", pressed.as_ref())] {
292            let _ = element.add_event_listener_with_callback(name, callback.unchecked_ref());
293        }
294        // The listeners outlive this call by exactly as long as the input does, which is the life
295        // of the program: there is one hidden input and nothing ever takes it out of the page.
296        composed.forget();
297        pressed.forget();
298    }

300    /// Which engine key a browser's `KeyboardEvent.key` is, where it is one this path has to carry.
301    ///
302    /// Mirrors `photosynthesize`'s reading of a winit key, and for the same reason: a key that
303    /// produced text is left to the `input` event that carries it, so the two paths do not both
304    /// answer for the same press.
305    fn named(key: &str, modifiers: Modifiers) -> Option<Key> {
306        let named = match key {
307            "Backspace" => Key::Backspace,
308            "Delete" => Key::Delete,
309            "ArrowLeft" => Key::Left,
310            "ArrowRight" => Key::Right,
311            "ArrowUp" => Key::Up,
312            "ArrowDown" => Key::Down,
313            "Home" => Key::Home,
314            "End" => Key::End,
315            "Enter" => Key::Enter,
316            "Tab" => Key::Tab,
317            "Escape" => Key::Escape,
318            // A character held with control is a command and produces no `input` event, so this is
319            // the only path it has. A character held with nothing is text and already has one.
320            _ => {
321                let mut characters = key.chars();
322                let character = characters.next()?;
323                if characters.next().is_some() || !modifiers.control {
324                    return None;
325                }
326                Key::Typed(character)
327            }
328        };
329        Some(named)
330    }

335        }
