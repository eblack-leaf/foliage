1//! The soft keyboard -- raised by focus, and by nothing else.
2//!
3//! A phone has no keyboard until something asks for one, and the thing that asks is a field holding
4//! focus. That is the whole rule, and it needs no declaration of its own: focus already rests only
5//! on what said [`interactive`](crate::Place::interactive), and a [`TextInput`](crate::TextInput)
6//! is the only thing in the engine that is typed *into*. So the keyboard is raised for the field
7//! that holds focus and lowered for everything else, decided in the same breath as the caret's
8//! visibility and from the same state.
9//!
10//! What a field may say is which **[`Keypad`]** it wants, because a telephone number behind a full
11//! alphabet is the one thing the platform cannot be left to guess.
12//!
13//! # The web owns the keyboard, so the web owns the keys
14//!
15//! There is no way to raise a browser's keyboard except by giving DOM focus to a real input, and
16//! doing that takes keyboard focus off the canvas winit is listening on -- so every key typed after
17//! it would be swallowed by an input the person cannot see. The hidden input therefore hands them
18//! back: what it composes arrives as text and what it does not is read from the key itself, both
19//! entering [`Incoming`](crate::interaction::input::Incoming) exactly where a translated winit
20//! event enters it.
21//!
22//! # Android raises one and keeps the keys
23//!
24//! The other platform with a keyboard of its own, and the easier half: the activity raises and
25//! lowers it directly, and what is typed arrives as ordinary key events on the same road a physical
26//! keyboard's would. So there is nothing to hand back and no hidden element to hold -- only the
27//! activity handle, which is the one thing the engine cannot reach for itself.
28//!
29//! Nothing is raised anywhere else, because nothing else has one.
30

35/// Which keys a soft keyboard offers.
36///
37/// A hint and not a constraint: it changes what is easy to type and never what a field accepts, so
38/// a [`Keypad::Number`] field can still be pasted a word into and is still the app's to validate.
39/// Ignored where the platform has no keyboard of its own to raise.
40#[derive(Component, Copy, Clone, Debug, Default, PartialEq, Eq)]
41pub enum Keypad {
42    /// Letters. What a field asks for unless it says otherwise.
43    #[default]
44    Text,
45    /// Digits, for a quantity or a code.
46    Number,
47    /// A dialling pad.
48    Telephone,
49}
50
51impl Keypad {
52    /// What the browser is told to offer, as `inputmode` names it.
53    #[cfg_attr(not(target_family = "wasm"), allow(dead_code))]
54    fn mode(self) -> &'static str {
55        match self {
56            Self::Text => "text",
57            Self::Number => "numeric",
58            Self::Telephone => "tel",
59        }
60    }
61
62    /// What Android is told to offer, as `EditorInfo.inputType` names it.
63    ///
64    /// The class only. A variation would say what *kind* of text -- an address, a password, a URI --
65    /// which is a distinction this engine does not make and one the field itself would have to
66    /// declare.
67    #[cfg(target_os = "android")]
68    fn class(self) -> winit::platform::android::activity::input::InputType {
69        use winit::platform::android::activity::input::InputType;
70
71        match self {
72            Self::Text => InputType::TYPE_CLASS_TEXT,
73            Self::Number => InputType::TYPE_CLASS_NUMBER,
74            Self::Telephone => InputType::TYPE_CLASS_PHONE,
75        }
76    }
77}

79/// The soft keyboard, and what it is currently showing.
80///
81/// What is raised is engine state on every target, and raising it is the platform's -- so the rule
82/// that focus decides the keyboard is proven headlessly while the DOM half stays the edge the suite
83/// cannot reach, exactly as the winit translation is.
84#[derive(Default)]
85pub(crate) struct Keyboard {
86    /// What is up, if anything is. Kept so the platform is asked only when the answer changes.
87    raised: Option<Keypad>,
88    /// The hidden input the browser raises its keyboard for. Absent until
89    /// [`attach`](Keyboard::attach), and absent for good under the headless suite.
90    #[cfg(target_family = "wasm")]
91    trigger: Option<web::Trigger>,
92    /// The activity Android raises its keyboard through. Absent until
93    /// [`activity`](Keyboard::activity), and absent for good under the headless suite.
94    #[cfg(target_os = "android")]
95    activity: Option<winit::platform::android::activity::AndroidApp>,
96}
97
98impl Keyboard {
99    /// What is raised, if anything is.
100    ///
101    /// Read by the headless suite, which is the only side of the seam that can observe it: asking
102    /// for a keyboard is the engine's and raising one is the host's.
103    #[cfg(test)]
104    pub(crate) fn raised(&self) -> Option<Keypad> {
105        self.raised
106    }
107
108    /// Builds what the platform raises its keyboard for, once, at boot.
109    ///
110    /// `wake` is what a key captured outside a frame rouses the loop with -- while the hidden input
111    /// holds DOM focus the canvas hears nothing, so nothing else would ask for the frame that
112    /// delivers it.
113    #[cfg_attr(not(target_family = "wasm"), allow(unused_variables))]
114    pub(crate) fn attach(&mut self, wake: &crate::queue::Wake) {
115        #[cfg(target_family = "wasm")]
116        {
117            self.trigger = web::Trigger::build(wake.clone());
118        }
119    }
120
121    /// Takes the activity Android raises its keyboard through, once, at boot.
122    ///
123    /// Separate from [`attach`](Keyboard::attach) because it is not the engine's to find: the
124    /// handle enters the program at `android_main` and reaches here only because
125    /// [`photosynthesize`](crate::Foliage::photosynthesize) is handed it. Nothing takes it under the
126    /// headless suite, so nothing is raised there.
127    #[cfg(target_os = "android")]
128    pub(crate) fn activity(&mut self, activity: winit::platform::android::activity::AndroidApp) {
129        self.activity = Some(activity);
130    }
131
132    /// Raises the keyboard `wanted` asks for, or lowers whatever is up.
133    ///
134    /// Called every frame with the whole answer rather than told about changes, because the answer
135    /// is a product of focus and is recomputed like every other product.
136    ///
137    /// What is *up* is compared, so the mode is written and the trace is made only where it moved.
138    /// DOM focus is not: the page can take it off the hidden input without the engine hearing about
139    /// it -- a press on the canvas does exactly that, and is how every tap from one field to
140    /// another arrives -- so it is asserted on every frame instead. Focusing what already holds
141    /// focus does nothing, which is what makes asserting it free.
142    pub(crate) fn raise(&mut self, wanted: Option<Keypad>) {
143        let moved = wanted != self.raised;
144        if moved {
145            self.raised = wanted;
146            debug!(keypad = ?wanted, "keyboard");
147        }
148        #[cfg(target_family = "wasm")]
149        if let Some(trigger) = &self.trigger {
150            match wanted {
151                Some(keypad) => {
152                    if moved {
153                        trigger.mode(keypad);
154                    }
155                    trigger.focus();
156                }
157                None if moved => trigger.blur(),
158                None => {}
159            }
160        }
161        // Only where it moved, unlike the web: there is no element here whose focus the platform can
162        // take away behind the engine's back, so what was raised stays raised until this lowers it.
163        #[cfg(target_os = "android")]
164        if moved && let Some(activity) = &self.activity {
165            use winit::platform::android::activity::input::{ImeOptions, TextInputAction};
166
167            match wanted {
168                Some(keypad) => {
169                    // Written before the keyboard is raised. An input type given to one already up
170                    // changes nothing a person can see until it has been dismissed and raised again.
171                    activity.set_ime_editor_info(
172                        keypad.class(),
173                        // What the enter key does is the app's business and this engine has no way
174                        // to say -- `submitted` is reported for the key whatever it is labelled.
175                        TextInputAction::Unspecified,
176                        ImeOptions::empty(),
177                    );
178                    // Implicit: focus asked for this keyboard, not a person reaching for one. The
179                    // distinction is the platform's to act on -- an implicit keyboard is the one a
180                    // back gesture is allowed to dismiss.
181                    activity.show_soft_input(true);
182                }
183                // Everything, not only what was raised implicitly. Focus has left the field, so
184                // there is nothing left for a keyboard to type into either way.
185                None => activity.hide_soft_input(false),
186            }
187        }
188    }
189
190    /// What the hidden input captured since the last frame, in the order it arrived.
191    ///
192    /// Empty everywhere but the web, and empty there until something has been typed into a keyboard
193    /// this raised.
194    pub(crate) fn captured(&mut self) -> Vec<crate::interaction::input::Input> {
195        #[cfg(target_family = "wasm")]
196        if let Some(trigger) = &self.trigger {
197            return trigger.captured();
198        }
199        Vec::new()
200    }
201}

203/// The browser's half: one hidden input, and the keys it takes from the canvas given back.

205mod web {
206    use std::cell::RefCell;

216    /// The id the hidden input is found by. One element rather than one per [`Keypad`]: the pad is
217    /// `inputmode`, which is written before focus moves, so there is nothing three elements would
218    /// say that one cannot.
219    const TRIGGER: &str = "foliage-keypad";
220
221    /// Off screen rather than `display:none`, because an element that is not laid out cannot take
222    /// focus and an element that cannot take focus raises no keyboard. Everything that would let it
223    /// second-guess what is typed is off: the value is read and cleared on every event, so an
224    /// autocorrection would be applied to a value the person never sees.
225    const STYLE: &str = "position:absolute;left:-1px;top:-1px;width:0;height:0;\
226        min-width:0;min-height:0;padding:0;border:0;opacity:0";
227
228    /// What the hidden input has captured and no frame has taken yet.
229    ///
230    /// The browser is single-threaded, so an `Rc` is the whole of what sharing this with the DOM's
231    /// own callbacks needs.
232    #[derive(Clone, Default)]
233    struct Captured(Rc<RefCell<Vec<Input>>>);
234
235    /// The hidden input, and what it has taken.
236    pub(super) struct Trigger {
237        element: web_sys::HtmlInputElement,
238        captured: Captured,
239    }
240
241    impl Trigger {
242        /// Puts the input in the page and wires it to the input stream.
243        ///
244        /// `None` where there is no document to put it in, which is every host that is not a
245        /// browser -- the same absence the headless suite has, reached a different way.
246        pub(super) fn build(wake: Wake) -> Option<Self> {
247            let document = web_sys::window()?.document()?;
248            let element = document
249                .create_element("input")
250                .ok()?
251                .dyn_into::<web_sys::HtmlInputElement>()
252                .ok()?;
253            element.set_id(TRIGGER);
254            element.set_type("text");
255            for (name, value) in [
256                ("style", STYLE),
257                ("autocapitalize", "off"),
258                ("autocomplete", "off"),
259                ("autocorrect", "off"),
260                ("spellcheck", "false"),
261                ("aria-hidden", "true"),
262            ] {
263                element.set_attribute(name, value).ok()?;
264            }
265            document.body()?.append_child(&element).ok()?;
266            let captured = Captured::default();
267            listen(&element, &captured, &wake);
268            Some(Self { element, captured })
269        }
270
271        /// Says which keys to offer, before anything is offered.
272        ///
273        /// A mode written to an input that already holds focus changes nothing the person can see
274        /// until the keyboard has been dismissed and raised again, so this is only ever written
275        /// while the keyboard is down.
276        pub(super) fn mode(&self, keypad: Keypad) {
277            let _ = self.element.set_attribute("inputmode", keypad.mode());
278        }
279
280        /// Gives the input DOM focus, which is the only thing a browser raises a keyboard for.
281        pub(super) fn focus(&self) {
282            let _ = self.element.focus();
283        }
284
285        pub(super) fn blur(&self) {
286            let _ = self.element.blur();
287        }
288
289        pub(super) fn captured(&self) -> Vec<Input> {
290            core::mem::take(&mut *self.captured.0.borrow_mut())
291        }
292    }

294    /// Wires the three events a hidden input has to give back.
295    ///
296    /// `input` is what was composed -- a plain letter, a dead-key sequence, a whole word an IME
297    /// committed -- and is the only one that answers for text, because the browser is what resolved
298    /// the layout. `keydown` is for the keys that produce no text and would otherwise be spent on
299    /// the input's own editing, and for a character held with a modifier, which produces no `input`
300    /// event because it is a command rather than something to insert. `paste` carries what another
301    /// program put on the clipboard, which is the only way this target ever sees it.
302    fn listen(element: &web_sys::HtmlInputElement, captured: &Captured, wake: &Wake) {
303        let composed = {
304            let (captured, wake) = (captured.clone(), wake.clone());
305            Closure::wrap(Box::new(move |event: web_sys::Event| {
306                let Some(input) = event
307                    .target()
308                    .and_then(|target| target.dyn_into::<web_sys::HtmlInputElement>().ok())
309                else {
310                    return;
311                };
312                let value = input.value();
313                if value.is_empty() {
314                    return;
315                }
316                // Cleared at once. What was typed is carried as keystrokes from here on, and a
317                // value left in the input would be composed against by whatever is typed next.
318                input.set_value("");
319                captured.push(
320                    value
321                        .chars()
322                        // The rule `photosynthesize` reads a winit key by: a control character is
323                        // what a key that means something produces where the platform hands text
324                        // as well, and inserting one would put it in the value.
325                        .filter(|character| !character.is_control())
326                        .map(|character| Input::Keyed(Key::Typed(character)))
327                        .collect(),
328                );
329                wake.rouse();
330            }) as Box<dyn FnMut(_)>)
331        };
332        let pressed = {
333            let (captured, wake) = (captured.clone(), wake.clone());
334            Closure::wrap(Box::new(move |event: web_sys::KeyboardEvent| {
335                // Read from this event rather than from what the engine last heard: the canvas
336                // stopped receiving `ModifiersChanged` the moment this input took focus, so the
337                // engine's own tracker is exactly as starved as the key events are.
338                let modifiers = Modifiers {
339                    shift: event.shift_key(),
340                    control: event.ctrl_key() || event.meta_key(),
341                };
342                let Some(key) = named(&event.key(), modifiers) else {
343                    return;
344                };
345                // The input's own editing would otherwise spend the key -- a backspace deleting
346                // from a value that is always empty, an arrow moving a caret nobody can see.
347                event.prevent_default();
348                captured.push(vec![
349                    Input::Modifiers(modifiers),
350                    Input::Keyed(key),
351                    Input::Modifiers(Modifiers::default()),
352                ]);
353                wake.rouse();
354            }) as Box<dyn FnMut(_)>)
355        };
356        let pasted = {
357            let (captured, wake) = (captured.clone(), wake.clone());
358            Closure::wrap(Box::new(move |event: web_sys::ClipboardEvent| {
359                // Empty for a paste that is not text -- a picture, a file -- which the browser
360                // fires this for all the same. Nothing was pasted as far as this engine is
361                // concerned, and letting it through would be an edit that wrote nothing.
362                let Some(text) = event
363                    .clipboard_data()
364                    .and_then(|data| data.get_data("text/plain").ok())
365                    .filter(|text| !text.is_empty())
366                else {
367                    return;
368                };
369                // The value is where the browser would put this, and the `input` event that
370                // followed would then insert it a character at a time as if it had been typed.
371                event.prevent_default();
372                captured.push(vec![Input::Pasted(text)]);
373                wake.rouse();
374            }) as Box<dyn FnMut(_)>)
375        };
376        for (name, callback) in [
377            ("input", composed.as_ref()),
378            ("keydown", pressed.as_ref()),
379            ("paste", pasted.as_ref()),
380        ] {
381            let _ = element.add_event_listener_with_callback(name, callback.unchecked_ref());
382        }
383        // The listeners outlive this call by exactly as long as the input does, which is the life
384        // of the program: there is one hidden input and nothing ever takes it out of the page.
385        composed.forget();
386        pressed.forget();
387        pasted.forget();
388    }

390    /// Which engine key a browser's `KeyboardEvent.key` is, where it is one this path has to carry.
391    ///
392    /// Mirrors `photosynthesize`'s reading of a winit key, and for the same reason: a key that
393    /// produced text is left to the `input` event that carries it, so the two paths do not both
394    /// answer for the same press. `Ctrl+V` is the same rule reaching the third listener --
395    /// answering it here would take the key, and taking the key is what stops the browser ever
396    /// firing the `paste` that carries the text.
397    ///
398    /// Nothing is lost at the root by declining it: this listener hears a key only while the hidden
399    /// input holds DOM focus, which is only ever while a field holds the engine's, and a chord
400    /// pressed with focus nowhere reaches the canvas through winit like every other key.
401    fn named(key: &str, modifiers: Modifiers) -> Option<Key> {
402        // A browser reports the letter of a chord in the case the shift key put it in.
403        if modifiers.control && key.eq_ignore_ascii_case("v") {
404            return None;
405        }
406        let named = match key {
407            "Backspace" => Key::Backspace,
408            "Delete" => Key::Delete,
409            "ArrowLeft" => Key::Left,
410            "ArrowRight" => Key::Right,
411            "ArrowUp" => Key::Up,
412            "ArrowDown" => Key::Down,
413            "Home" => Key::Home,
414            "End" => Key::End,
415            "Enter" => Key::Enter,
416            "Tab" => Key::Tab,
417            "Escape" => Key::Escape,
418            // A character held with control is a command and produces no `input` event, so this is
419            // the only path it has. A character held with nothing is text and already has one.
420            _ => {
421                let mut characters = key.chars();
422                let character = characters.next()?;
423                if characters.next().is_some() || !modifiers.control {
424                    return None;
425                }
426                Key::Typed(character)
427            }
428        };
429        Some(named)
430    }
