12/// What the tree put out this frame.
13///
14/// A set you interrogate, not a list you walk: ask it about the elements you own. There is no
15/// order to read, apart from the sequences that are ordered in their own right.
16#[derive(Clone, Default)]
17pub struct Pollen(Arc<Drift>);

40    /// Whether a gesture went down on `leaf`.
41    ///
42    /// The hook for a pressed visual. It says a gesture is being held, not what it will turn out to
43    /// be -- [`clicked`](Pollen::clicked), [`held`](Pollen::held) and
44    /// [`drag_started`](Pollen::drag_started) are what it can become, and
45    /// [`disengaged`](Pollen::disengaged) is where a pressed visual is put back however it ended.
46    pub fn engaged(&self, leaf: Leaf) -> bool {
47        self.0.engaged.contains(&leaf)
48    }

50    /// Whether `leaf` stopped holding a gesture, however it stopped.
51    ///
52    /// It ended, or it became a drag this element does not take and passed to a region containing
53    /// it. Either way this element is no longer holding anything, so there is always somewhere to
54    /// put a pressed visual back.
55    pub fn disengaged(&self, leaf: Leaf) -> bool {
56        self.0.disengaged.contains(&leaf)
57    }

59    /// Whether `leaf` was tapped: a gesture that began on it and ended without ever becoming a
60    /// drag.
61    ///
62    /// Not a click that was issued and then taken back. Nothing is emitted while a gesture is
63    /// resolving, so there is nothing to retract when it turns out to be a drag.
64    pub fn clicked(&self, leaf: Leaf) -> bool {
65        self.0.clicked.contains_key(&leaf)
66    }
67
68    /// Whether `leaf` was pressed: tapped, or sent `Enter` or the space bar while it held focus.
69    ///
70    /// The one reading of a press that a keyboard can make as well as a pointer, so what focus steps
71    /// to can be pressed where it lands. A text field is sent its own `Enter` and space as edits and
72    /// as a [`submitted`](Pollen::submitted), which is what a field's press is; this is for the
73    /// things that are pressed rather than typed into.
74    pub fn activated(&self, leaf: Leaf) -> bool {
75        self.clicked(leaf)
76            || self
77                .keys(leaf)
78                .iter()
79                .any(|stroke| matches!(stroke.key, Key::Enter | Key::Typed(' ')))
80    }

82    /// Where `leaf` was tapped, if it was.
83    ///
84    /// The point the gesture *began* at, which is the point it was hit-tested against and therefore
85    /// the only one that is certainly on the element. A gesture that wandered and came back is
86    /// still a tap, and it is a tap where it started.
87    pub fn clicked_at(&self, leaf: Leaf) -> Option<Position> {
88        self.0.clicked.get(&leaf).copied()
89    }

91    /// Whether the press `leaf` is holding has been held past [`Hold::after`](crate::Hold) without
92    /// becoming a drag.
93    ///
94    /// A gesture fact of its own, and the second way a press stops being one that might still be a
95    /// tap -- so a hold is never also a [`clicked`](Pollen::clicked), and a press released after one
96    /// moves no focus. Reported once, in the frame the duration passed in.
97    ///
98    /// What follows is the holder's. The drag out of a hold belongs to whoever took it, whatever
99    /// that element declared with [`drags`](crate::Place::drags) and whichever way it goes, so a
100    /// press-and-hold that turns into a drag reports [`drag_started`](Pollen::drag_started) and then
101    /// [`dragged`](Pollen::dragged) like any other.
102    pub fn held(&self, leaf: Leaf) -> bool {
103        self.0.held.contains_key(&leaf)
104    }
105
106    /// Where `leaf` was held, if it was.
107    ///
108    /// The point the press landed at, which is the point it was hit-tested against. A hold is where
109    /// a menu opens and where a selection begins, and neither has anywhere else to take its place
110    /// from -- the pointer has not moved since, by definition of the gesture still resolving.
111    pub fn held_at(&self, leaf: Leaf) -> Option<Position> {
112        self.0.held.get(&leaf).copied()
113    }
114
115    /// Whether the gesture `leaf` is holding has become a drag.
116    ///
117    /// Reported once per gesture, and only to an element that declared
118    /// [`drags`](crate::Place::drags) on the axis the gesture went, or that took the hold this drag
119    /// came out of -- a hold has already settled who is holding the gesture, so there is nothing
120    /// left for an axis to decide. An element that takes no drags hears
121    /// [`disengaged`](Pollen::disengaged) instead, because it has let the gesture go.
122    pub fn drag_started(&self, leaf: Leaf) -> bool {
123        self.0.drag_started.contains(&leaf)
124    }
125
126    /// How far the drag `leaf` is holding has moved, if it moved this frame.
127    pub fn dragged(&self, leaf: Leaf) -> Option<Drag> {
128        self.0.dragged.get(&leaf).copied()
129    }

131    /// Whether a motion on `leaf` reached its end.
132    ///
133    /// The element settling, and the cheap hook for whatever happens next -- pruning what has faded
134    /// out, hiding what has slid away. There is nothing to settle: what the element declares was
135    /// already the target from the moment the motion started, so this reports an arrival rather than
136    /// asking for one.
137    ///
138    /// One report for the element rather than one per property, so it says an element arrived and
139    /// not which of its properties did. Two motions ending together are one arrival to an app, and
140    /// telling one of an element's motions from another is what the name
141    /// [`animate`](crate::Grow::animate) hands back is for.
142    pub fn landed(&self, leaf: Leaf) -> bool {
143        self.0.landed.contains(&leaf)
144    }
145
146    /// How far a tween has come this frame, for as long as it is running.
147    ///
148    /// A channel's value between the two ends it was given, and a motion's eased progress from `0.0`
149    /// to `1.0` -- the eased progress and not the fraction of the duration elapsed, because what a
150    /// motion is part way through is what it looks like, and the two differ under every
151    /// [`Ease`](crate::Ease) but [`Linear`](crate::Ease::Linear).
152    ///
153    /// The frame it ends reports its end value and [`finished`](Pollen::finished) together, so
154    /// there is never an end value to infer from an absence. After that there is nothing to read.
155    pub fn tween(&self, tween: Tween) -> Option<f32> {
156        self.0.tweens.get(&tween).copied()
157    }
158
159    /// Whether the motion, channel or [`timer`](crate::Grow::timer) `tween` names ended this frame
160    /// **on the end it was going to**.
161    ///
162    /// An arrival, and the hook a chain hangs off: it reports a tween that ran its course, and one
163    /// [`finish`](crate::Grow::finish) ended early, which is the app declaring the same arrival. It
164    /// does not report a name that stopped being the one running -- [`stop`](crate::Grow::stop)ped,
165    /// cancelled by a direct write, replaced by a second motion on the property, or taken down with
166    /// its element. A chain waiting on any of those never runs, which is what makes stopping a
167    /// motion the way to break one.
168    pub fn finished(&self, tween: Tween) -> bool {
169        self.0.finished.contains(&tween)
170    }
171
172    /// Whether the last tween running under a [`Sequence`](crate::Sequence) ended this frame.
173    ///
174    /// The group's own arrival, as against each member's. Keyed on the sequence rather than on a
175    /// [`Leaf`], because a group is not about an element -- its whole purpose is to time things
176    /// together that have nothing else in common.
177    ///
178    /// However the members ended: landed, stopped, cancelled by a direct write, or taken down with
179    /// their element. A group being over is one fact and gets one report -- whether a *particular*
180    /// motion arrived is [`finished`](Pollen::finished), which is a different question and is
181    /// answered separately, so neither report has to carry the other's meaning.
182    pub fn sequence_finished(&self, sequence: Sequence) -> bool {
183        self.0.sequences.contains(&sequence)
184    }

186    /// Whether the person at the keyboard changed what `leaf` says.
187    ///
188    /// A [`TextInput`](crate::TextInput) or a [`TextArea`](crate::TextArea) only, and only what
189    /// was typed into it: a value the app
190    /// wrote with [`text`](crate::Grow::text) is not reported back, because an app that wrote one
191    /// already knows what it wrote. What it now says is [`Vein::Text`](crate::Vein::Text).
192    pub fn edited(&self, leaf: Leaf) -> bool {
193        self.0.edited.contains(&leaf)
194    }

196    /// Whether `Enter` was pressed in `leaf` -- `Ctrl+Enter` on a [`TextArea`](crate::TextArea),
197    /// whose bare `Enter` is a newline.
198    ///
199    /// What that means is the app's: a field says the key was pressed and holds no opinion about
200    /// whether anything is to be submitted.

202        self.0.submitted.contains(&leaf)
203    }

208    }

210    /// Whether `leaf` lost focus.
211    ///
212    /// It was moved elsewhere, or `leaf` stopped being something focus can rest on -- it withered,
213    /// was hidden, or was disabled.
214    pub fn unfocused(&self, leaf: Leaf) -> bool {

218    /// Whether what `of` names has arrived and been registered.
219    ///
220    /// One question over a [`Font`](crate::Font), a [`Field`](crate::Field) and a
221    /// [`Plate`](crate::Plate) alike, because one road fills all three. Reported once, in the frame
222    /// the bytes landed in.
223    ///
224    /// Mostly nothing has to wait on it: an element drawing a picture or a mark that has not arrived
225    /// occupies its box and draws nothing, and appears the frame it does. A **font** is the one worth
226    /// waiting for, because a run composed in one that has yet to land is measured in the bundled
227    /// face and reflows when the real one arrives.
228    pub fn loaded(&self, of: impl Into<Arrival>) -> bool {
229        self.0.loaded.contains(&of.into())
230    }
231
232    /// Whether what `of` names could not be read, or could not be used once it was.
233    ///
234    /// A path that is not there, a fetch that answered with a status, a picture in a format foliage
235    /// does not decode, a font that turned out to be proportional. One report, because there is one
236    /// thing an app can do about any of them.
237    ///
238    /// The name stays valid and unfilled: what drew nothing goes on drawing nothing, and a second
239    /// attempt is a second call.
240    pub fn missing(&self, of: impl Into<Arrival>) -> bool {
241        self.0.missing.contains(&of.into())
242    }
243
244    /// What the clipboard held, if a [`paste`](crate::Grow::paste) this app asked for was answered
245    /// this frame.
246    ///
247    /// Never in the frame that asked for it. Empty text is the answer for an empty clipboard and
248    /// for one the host would not let the engine read, because there is one thing an app does about
249    /// either -- the reason it could not be read is traced rather than reported.
250    ///
251    /// A [`TextInput`](crate::TextInput) answers `Ctrl+V` for itself, and what it wrote is
252    /// [`edited`](Pollen::edited) like anything else typed into it. Nothing arrives here for that.
253    pub fn pasted(&self) -> Option<&str> {
254        self.0.pasted.as_deref()
255    }
256
257    /// What was typed at `leaf` this frame, in the order it arrived.
258    ///
259    /// A key goes to whatever holds focus, and focus rests only on what declared
260    /// [`interactive`](crate::Place::interactive) -- so an element that can be focused hears keys
261    /// with nothing further declared, and one that cannot never does. What an element makes of a
262    /// key is the app's: an activation on [`Key::Enter`](crate::Key::Enter), a step through a list
263    /// on [`Key::Up`](crate::Key::Up)/[`Key::Down`](crate::Key::Down), anything else it wants to
264    /// answer.
265    ///
266    /// ```
267    /// # use foliage::{Grove, Grow, Key, Leaf, Pollen};
268    /// # fn f(grove: &mut Grove, pollen: &Pollen, button: Leaf, menu: Leaf) {
269    /// for stroke in pollen.keys(button) {
270    ///     match stroke.key {
271    ///         Key::Enter => grove.prune(menu),
272    ///         Key::Down => grove.focus_next(),
273    ///         _ => {}
274    ///     }
275    /// }
276    /// # }
277    /// ```
278    ///
279    /// **Ordered, which nothing else here is.** [`Pollen`] is a set to interrogate rather than a
280    /// list to walk, and keystrokes are the one exception, because two keys in a frame mean
281    /// different things in each order.
282    ///
283    /// A [`TextInput`](crate::TextInput) is told what it was sent as well, so a field's own keys are
284    /// readable beside the [`edited`](Pollen::edited) they produced.
285    pub fn keys(&self, leaf: Leaf) -> &[Keystroke] {
286        self.0.keys.get(&leaf).map_or(&[], Vec::as_slice)
287    }
288
289    /// What was typed this frame while focus rested nowhere, in the order it arrived.
290    ///
291    /// Keys reach an element by holding focus, and a page that has never been pressed holds none --
292    /// so a chord that is about the whole app rather than about anything in it would have nowhere
293    /// to land. It lands here instead.
294    ///
295    /// A key is never reported both here and to an element: focus rests somewhere or it does not.
296    pub fn root_keys(&self) -> &[Keystroke] {
297        &self.0.root_keys
298    }

302    }

311/// What the frame is collecting, before it is sealed into a [`Pollen`] and handed over.
312#[derive(Default, Debug, PartialEq)]
313pub(crate) struct Drift {
314    pub(crate) withered: HashSet<Leaf>,
315    pub(crate) resized: Option<Area>,
316    pub(crate) repainted: bool,
317    pub(crate) engaged: HashSet<Leaf>,
318    pub(crate) disengaged: HashSet<Leaf>,
319    /// Where each tap landed, which is where its gesture began. Carried rather than counted,
320    /// because an element that has somewhere to put a point needs the point.
321    pub(crate) clicked: HashMap<Leaf, Position>,
322    /// Where each hold landed, which is where its press did. Carried for the reason a tap's point
323    /// is: a menu and a selection both begin at a point, and neither has another to use.
324    pub(crate) held: HashMap<Leaf, Position>,
325    pub(crate) drag_started: HashSet<Leaf>,
326    pub(crate) dragged: HashMap<Leaf, Drag>,
327    /// What each focused element was sent, in arrival order -- the one ordered thing here, because
328    /// two keys in a frame mean different things in each order.
329    pub(crate) keys: HashMap<Leaf, Vec<Keystroke>>,
330    /// The same for keys that arrived while focus rested nowhere, which are the app's own.
331    pub(crate) root_keys: Vec<Keystroke>,
332    /// What a `paste` the app asked for came back with. A field's own paste is an `edited` instead,
333    /// because what the person at the keyboard did to a value is one report however they did it.
334    pub(crate) pasted: Option<String>,
335    /// What arrived from a path or a URL this frame, and what could not.
336    pub(crate) loaded: HashSet<Arrival>,
337    pub(crate) missing: HashSet<Arrival>,
338    pub(crate) edited: HashSet<Leaf>,
339    pub(crate) submitted: HashSet<Leaf>,
340    pub(crate) focused: HashSet<Leaf>,
341    pub(crate) unfocused: HashSet<Leaf>,
342    pub(crate) landed: Leaves,
343    pub(crate) tweens: Named<Tween, f32>,
344    pub(crate) finished: HashSet<Tween, EntityHash>,
345    pub(crate) sequences: HashSet<Sequence>,
346}
347
348impl Drift {
349    /// Whether there is anything here to tell the app.
350    ///
351    /// A frame is owed while this is true (F9): steps 4 through 7 emit into the drift, and step 3
352    /// of the *next* frame is where an app is handed it (F7). Without this the loop could idle
353    /// holding a report nothing would ever be run to deliver.
354    ///
355    /// Compared against an empty drift rather than field by field, so a field added later is
356    /// counted without anything having to remember to count it.
357    pub(crate) fn pending(&self) -> bool {
358        *self != Drift::default()
359    }
360
361    /// Adds one frame's worth of movement to what `leaf` is being told about its drag.
362    ///
363    /// Several moves can arrive in one frame, and a reader is handed one answer: where the drag is
364    /// now, and how far it came this frame. Which platform events that was made of is engine
365    /// bookkeeping.
366    pub(crate) fn dragged(&mut self, leaf: Leaf, drag: Drag) {
367        let carried = self.dragged.entry(leaf).or_insert(Drag {
368            delta: Position::default(),
369            ..drag
370        });
371        carried.current = drag.current;
372        carried.delta = Position::new(
373            carried.delta.x + drag.delta.x,
374            carried.delta.y + drag.delta.y,
375        );
376    }
