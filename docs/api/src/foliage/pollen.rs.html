11/// What the tree put out this frame.
12///
13/// A set you interrogate, not a list you walk: ask it about the elements you own. There is no
14/// order to read, apart from the sequences that are ordered in their own right.
15#[derive(Clone, Default)]
16pub struct Pollen(Arc<Drift>);
17
18impl Pollen {
19    /// Whether `leaf` was taken down. Reported once, in the frame after it went.
20    pub fn withered(&self, leaf: Leaf) -> bool {
21        self.0.withered.contains(&leaf)
22    }
23
24    /// The surface's new size, if it changed this frame.
25    pub fn resized(&self) -> Option<Area> {
26        self.0.resized
27    }
28
29    /// Whether a gesture went down on `leaf`.
30    ///
31    /// The hook for a pressed visual. It says a gesture is being held, not what it will turn out to
32    /// be -- [`clicked`](Pollen::clicked), [`held`](Pollen::held) and
33    /// [`drag_started`](Pollen::drag_started) are what it can become, and
34    /// [`disengaged`](Pollen::disengaged) is where a pressed visual is put back however it ended.
35    pub fn engaged(&self, leaf: Leaf) -> bool {
36        self.0.engaged.contains(&leaf)
37    }
38
39    /// Whether `leaf` stopped holding a gesture, however it stopped.
40    ///
41    /// It ended, or it became a drag this element does not take and passed to a region containing
42    /// it. Either way this element is no longer holding anything, so there is always somewhere to
43    /// put a pressed visual back.
44    pub fn disengaged(&self, leaf: Leaf) -> bool {
45        self.0.disengaged.contains(&leaf)
46    }
47
48    /// Whether `leaf` was tapped: a gesture that began on it and ended without ever becoming a
49    /// drag.
50    ///
51    /// Not a click that was issued and then taken back. Nothing is emitted while a gesture is
52    /// resolving, so there is nothing to retract when it turns out to be a drag.
53    pub fn clicked(&self, leaf: Leaf) -> bool {
54        self.0.clicked.contains_key(&leaf)
55    }
56
57    /// Where `leaf` was tapped, if it was.
58    ///
59    /// The point the gesture *began* at, which is the point it was hit-tested against and therefore
60    /// the only one that is certainly on the element. A gesture that wandered and came back is
61    /// still a tap, and it is a tap where it started.
62    pub fn clicked_at(&self, leaf: Leaf) -> Option<Position> {
63        self.0.clicked.get(&leaf).copied()
64    }
65
66    /// Whether the press `leaf` is holding has been held past [`Hold::after`](crate::Hold) without
67    /// becoming a drag.
68    ///
69    /// A gesture fact of its own, and the second way a press stops being one that might still be a
70    /// tap -- so a hold is never also a [`clicked`](Pollen::clicked), and a press released after one
71    /// moves no focus. Reported once, in the frame the duration passed in.
72    ///
73    /// What follows is the holder's. The drag out of a hold belongs to whoever took it, whatever
74    /// that element declared with [`drags`](crate::Place::drags) and whichever way it goes, so a
75    /// press-and-hold that turns into a drag reports [`drag_started`](Pollen::drag_started) and then
76    /// [`dragged`](Pollen::dragged) like any other.
77    pub fn held(&self, leaf: Leaf) -> bool {
78        self.0.held.contains_key(&leaf)
79    }
80
81    /// Where `leaf` was held, if it was.
82    ///
83    /// The point the press landed at, which is the point it was hit-tested against. A hold is where
84    /// a menu opens and where a selection begins, and neither has anywhere else to take its place
85    /// from -- the pointer has not moved since, by definition of the gesture still resolving.
86    pub fn held_at(&self, leaf: Leaf) -> Option<Position> {
87        self.0.held.get(&leaf).copied()
88    }
89
90    /// Whether the gesture `leaf` is holding has become a drag.
91    ///
92    /// Reported once per gesture, and only to an element that declared
93    /// [`drags`](crate::Place::drags) on the axis the gesture went, or that took the hold this drag
94    /// came out of -- a hold has already settled who is holding the gesture, so there is nothing
95    /// left for an axis to decide. An element that takes no drags hears
96    /// [`disengaged`](Pollen::disengaged) instead, because it has let the gesture go.
97    pub fn drag_started(&self, leaf: Leaf) -> bool {
98        self.0.drag_started.contains(&leaf)
99    }
100
101    /// How far the drag `leaf` is holding has moved, if it moved this frame.
102    pub fn dragged(&self, leaf: Leaf) -> Option<Drag> {
103        self.0.dragged.get(&leaf).copied()
104    }
105
106    /// Whether a motion on `leaf` reached its end.
107    ///
108    /// The element settling, and the cheap hook for whatever happens next -- pruning what has faded
109    /// out, hiding what has slid away. There is nothing to settle: what the element declares was
110    /// already the target from the moment the motion started, so this reports an arrival rather than
111    /// asking for one.
112    ///
113    /// One report for the element rather than one per property, so it says an element arrived and
114    /// not which of its properties did. Two motions ending together are one arrival to an app, and
115    /// telling one of an element's motions from another is what the name
116    /// [`animate`](crate::Grow::animate) hands back is for.
117    pub fn landed(&self, leaf: Leaf) -> bool {
118        self.0.landed.contains(&leaf)
119    }
120
121    /// How far a tween has come this frame, for as long as it is running.
122    ///
123    /// A channel's value between the two ends it was given, and a motion's eased progress from `0.0`
124    /// to `1.0` -- the eased progress and not the fraction of the duration elapsed, because what a
125    /// motion is part way through is what it looks like, and the two differ under every
126    /// [`Ease`](crate::Ease) but [`Linear`](crate::Ease::Linear).
127    ///
128    /// The frame it ends reports its end value and [`finished`](Pollen::finished) together, so
129    /// there is never an end value to infer from an absence. After that there is nothing to read.
130    pub fn tween(&self, tween: Tween) -> Option<f32> {
131        self.0.tweens.get(&tween).copied()
132    }
133
134    /// Whether the motion, channel or [`timer`](crate::Grow::timer) `tween` names ended this frame
135    /// **on the end it was going to**.
136    ///
137    /// An arrival, and the hook a chain hangs off: it reports a tween that ran its course, and one
138    /// [`finish`](crate::Grow::finish) ended early, which is the app declaring the same arrival. It
139    /// does not report a name that stopped being the one running -- [`stop`](crate::Grow::stop)ped,
140    /// cancelled by a direct write, replaced by a second motion on the property, or taken down with
141    /// its element. A chain waiting on any of those never runs, which is what makes stopping a
142    /// motion the way to break one.
143    pub fn finished(&self, tween: Tween) -> bool {
144        self.0.finished.contains(&tween)
145    }
146
147    /// Whether the last tween running under a [`Sequence`](crate::Sequence) ended this frame.
148    ///
149    /// The group's own arrival, as against each member's. Keyed on the sequence rather than on a
150    /// [`Leaf`], because a group is not about an element -- its whole purpose is to time things
151    /// together that have nothing else in common.
152    ///
153    /// However the members ended: landed, stopped, cancelled by a direct write, or taken down with
154    /// their element. A group being over is one fact and gets one report -- whether a *particular*
155    /// motion arrived is [`finished`](Pollen::finished), which is a different question and is
156    /// answered separately, so neither report has to carry the other's meaning.
157    pub fn sequence_finished(&self, sequence: Sequence) -> bool {
158        self.0.sequences.contains(&sequence)
159    }
160
161    /// Whether the person at the keyboard changed what `leaf` says.
162    ///
163    /// A [`TextInput`](crate::TextInput) only, and only what was typed into it: a value the app
164    /// wrote with [`text`](crate::Grow::text) is not reported back, because an app that wrote one
165    /// already knows what it wrote. What it now says is [`Vein::Text`](crate::Vein::Text).
166    pub fn edited(&self, leaf: Leaf) -> bool {
167        self.0.edited.contains(&leaf)
168    }
169
170    /// Whether `Enter` was pressed in `leaf`.
171    ///
172    /// What that means is the app's: a field says the key was pressed and holds no opinion about
173    /// whether anything is to be submitted.
174    pub fn submitted(&self, leaf: Leaf) -> bool {
175        self.0.submitted.contains(&leaf)
176    }
177
178    /// Whether `leaf` took focus.
179    pub fn focused(&self, leaf: Leaf) -> bool {
180        self.0.focused.contains(&leaf)
181    }
182
183    /// Whether `leaf` lost focus.
184    ///
185    /// It was moved elsewhere, or `leaf` stopped being something focus can rest on -- it withered,
186    /// was hidden, or was disabled.
187    pub fn unfocused(&self, leaf: Leaf) -> bool {
188        self.0.unfocused.contains(&leaf)
189    }
190
191    /// Whether what `of` names has arrived and been registered.
192    ///
193    /// One question over a [`Font`](crate::Font), a [`Field`](crate::Field) and a
194    /// [`Plate`](crate::Plate) alike, because one road fills all three. Reported once, in the frame
195    /// the bytes landed in.
196    ///
197    /// Mostly nothing has to wait on it: an element drawing a picture or a mark that has not arrived
198    /// occupies its box and draws nothing, and appears the frame it does. A **font** is the one worth
199    /// waiting for, because a run composed in one that has yet to land is measured in the bundled
200    /// face and reflows when the real one arrives.
201    pub fn loaded(&self, of: impl Into<Arrival>) -> bool {
202        self.0.loaded.contains(&of.into())
203    }
204
205    /// Whether what `of` names could not be read, or could not be used once it was.
206    ///
207    /// A path that is not there, a fetch that answered with a status, a picture in a format foliage
208    /// does not decode, a font that turned out to be proportional. One report, because there is one
209    /// thing an app can do about any of them.
210    ///
211    /// The name stays valid and unfilled: what drew nothing goes on drawing nothing, and a second
212    /// attempt is a second call.
213    pub fn missing(&self, of: impl Into<Arrival>) -> bool {
214        self.0.missing.contains(&of.into())
215    }
216
217    /// What the clipboard held, if a [`paste`](crate::Grow::paste) this app asked for was answered
218    /// this frame.
219    ///
220    /// Never in the frame that asked for it. Empty text is the answer for an empty clipboard and
221    /// for one the host would not let the engine read, because there is one thing an app does about
222    /// either -- the reason it could not be read is traced rather than reported.
223    ///
224    /// A [`TextInput`](crate::TextInput) answers `Ctrl+V` for itself, and what it wrote is
225    /// [`edited`](Pollen::edited) like anything else typed into it. Nothing arrives here for that.
226    pub fn pasted(&self) -> Option<&str> {
227        self.0.pasted.as_deref()
228    }
229
230    /// What was typed at `leaf` this frame, in the order it arrived.
231    ///
232    /// A key goes to whatever holds focus, and focus rests only on what declared
233    /// [`interactive`](crate::Place::interactive) -- so an element that can be focused hears keys
234    /// with nothing further declared, and one that cannot never does. What an element makes of a
235    /// key is the app's: an activation on [`Key::Enter`](crate::Key::Enter), a step through a list
236    /// on [`Key::Up`](crate::Key::Up)/[`Key::Down`](crate::Key::Down), anything else it wants to
237    /// answer.
238    ///
239    /// ```
240    /// # use foliage::{Grove, Grow, Key, Leaf, Pollen};
241    /// # fn f(grove: &mut Grove, pollen: &Pollen, button: Leaf, menu: Leaf) {
242    /// for stroke in pollen.keys(button) {
243    ///     match stroke.key {
244    ///         Key::Enter => grove.prune(menu),
245    ///         Key::Down => grove.focus_next(),
246    ///         _ => {}
247    ///     }
248    /// }
249    /// # }
250    /// ```
251    ///
252    /// **Ordered, which nothing else here is.** [`Pollen`] is a set to interrogate rather than a
253    /// list to walk, and keystrokes are the one exception, because two keys in a frame mean
254    /// different things in each order.
255    ///
256    /// A [`TextInput`](crate::TextInput) is told what it was sent as well, so a field's own keys are
257    /// readable beside the [`edited`](Pollen::edited) they produced.
258    pub fn keys(&self, leaf: Leaf) -> &[Keystroke] {
259        self.0.keys.get(&leaf).map_or(&[], Vec::as_slice)
260    }
261
262    /// What was typed this frame while focus rested nowhere, in the order it arrived.
263    ///
264    /// Keys reach an element by holding focus, and a page that has never been pressed holds none --
265    /// so a chord that is about the whole app rather than about anything in it would have nowhere
266    /// to land. It lands here instead.
267    ///
268    /// A key is never reported both here and to an element: focus rests somewhere or it does not.
269    pub fn root_keys(&self) -> &[Keystroke] {
270        &self.0.root_keys
271    }
272
273    pub(crate) fn seal(drift: Drift) -> Self {
274        Self(Arc::new(drift))
275    }
276}

284/// What the frame is collecting, before it is sealed into a [`Pollen`] and handed over.
285#[derive(Default, Debug, PartialEq)]
286pub(crate) struct Drift {
287    pub(crate) withered: HashSet<Leaf>,
288    pub(crate) resized: Option<Area>,
289    pub(crate) engaged: HashSet<Leaf>,
290    pub(crate) disengaged: HashSet<Leaf>,
291    /// Where each tap landed, which is where its gesture began. Carried rather than counted,
292    /// because an element that has somewhere to put a point needs the point.
293    pub(crate) clicked: HashMap<Leaf, Position>,
294    /// Where each hold landed, which is where its press did. Carried for the reason a tap's point
295    /// is: a menu and a selection both begin at a point, and neither has another to use.
296    pub(crate) held: HashMap<Leaf, Position>,
297    pub(crate) drag_started: HashSet<Leaf>,
298    pub(crate) dragged: HashMap<Leaf, Drag>,
299    /// What each focused element was sent, in arrival order -- the one ordered thing here, because
300    /// two keys in a frame mean different things in each order.
301    pub(crate) keys: HashMap<Leaf, Vec<Keystroke>>,
302    /// The same for keys that arrived while focus rested nowhere, which are the app's own.
303    pub(crate) root_keys: Vec<Keystroke>,
304    /// What a `paste` the app asked for came back with. A field's own paste is an `edited` instead,
305    /// because what the person at the keyboard did to a value is one report however they did it.
306    pub(crate) pasted: Option<String>,
307    /// What arrived from a path or a URL this frame, and what could not.
308    pub(crate) loaded: HashSet<Arrival>,
309    pub(crate) missing: HashSet<Arrival>,
310    pub(crate) edited: HashSet<Leaf>,
311    pub(crate) submitted: HashSet<Leaf>,
312    pub(crate) focused: HashSet<Leaf>,
313    pub(crate) unfocused: HashSet<Leaf>,
314    pub(crate) landed: HashSet<Leaf>,
315    pub(crate) tweens: HashMap<Tween, f32>,
316    pub(crate) finished: HashSet<Tween>,
317    pub(crate) sequences: HashSet<Sequence>,
318}
319
320impl Drift {
321    /// Whether there is anything here to tell the app.
322    ///
323    /// A frame is owed while this is true (F9): steps 4 through 7 emit into the drift, and step 3
324    /// of the *next* frame is where an app is handed it (F7). Without this the loop could idle
325    /// holding a report nothing would ever be run to deliver.
326    ///
327    /// Compared against an empty drift rather than field by field, so a field added later is
328    /// counted without anything having to remember to count it.
329    pub(crate) fn pending(&self) -> bool {
330        *self != Drift::default()
331    }
332
333    /// Adds one frame's worth of movement to what `leaf` is being told about its drag.
334    ///
335    /// Several moves can arrive in one frame, and a reader is handed one answer: where the drag is
336    /// now, and how far it came this frame. Which platform events that was made of is engine
337    /// bookkeeping.
338    pub(crate) fn dragged(&mut self, leaf: Leaf, drag: Drag) {
339        let carried = self.dragged.entry(leaf).or_insert(Drag {
340            delta: Position::default(),
341            ..drag
342        });
343        carried.current = drag.current;
344        carried.delta = Position::new(
345            carried.delta.x + drag.delta.x,
346            carried.delta.y + drag.delta.y,
347        );
348    }
