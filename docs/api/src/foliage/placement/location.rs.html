10/// An element's placement: one [`Horizontal`] and one [`Vertical`], per breakpoint.
11///
12/// The two axes are distinct types and are given in that order, so they cannot be written the wrong
13/// way round.
14///
15/// ```ignore
16/// Location::new()
17///     .xs(
18///         left(20.px()).right(100.pct() - 16.px()),
19///         top(anchor().bottom() + 8.px()).height(content()),
20///     )
21///     .md(left(2.col()).right(3.col()), top(0.px()).height(120.px()))
22/// ```
23///
24/// Each link states one breakpoint, both axes together, so what an element looks like at a given
25/// width is read in one place. A breakpoint with nothing of its own takes the nearest smaller one
26/// that has, and a chain that runs out falls back to the whole of the parent's box -- so an element
27/// that only differs above a certain width states that width and nothing else.
28///
29/// # A value is a source and a role
30///
31/// The two are independent, which is what lets any source fill any coordinate. A **source** is
32/// where a number comes from -- [`20.px()`](crate::Source::px), [`50.pct()`](crate::Source::pct),
33/// [`2.col()`](crate::Source::col), [`8.letters()`](crate::Source::letters),
34/// [`anchor()`](crate::anchor)`.right()`, [`content()`](crate::content), and arithmetic on any of
35/// them. A **role** is what it means for this element: left, right, top, bottom, width, height,
36/// centre.
37///
38/// The role is written first, because it says what the value describes before the value has to be
39/// parsed: "my left is my anchor's right" is read in the order it is understood. The pairing is a
40/// type rather than a rule -- each opener returns a value carrying only the completions legal with
41/// it, so an axis has exactly four spellings and the illegal ones cannot be written at all.
42///
43/// ```ignore
44/// left(anchor().right()).width(140.px())
45/// left(20.px()).right(100.pct() - 16.px())
46/// center_x(50.pct()).width(140.px())
47/// top(0.px()).height(content())
48/// ```
49///
50/// Every edge role is a coordinate in the parent's space, including `right` and `bottom`, so
51/// sixteen in from the right is `right(100.pct() - 16.px())`. Insets would read more nicely for
52/// that one case, but an anchor's edges are already positions, so an inset would put two coordinate
53/// spaces in one expression.
54///
55/// # Width flows down. Height flows up.
56///
57/// Text wrapping makes height depend on width, and width comes from layout, so a container sized to
58/// its contents is a cycle in the general case. It is bounded here at two passes, with no iteration
59/// to convergence: the horizontal axis resolves for the whole tree, each text run wraps at its
60/// now-known width, and then the vertical axis resolves and reads the heights that came out of it.
61///
62/// A monospaced run's widest unwrapped width is its character count times the cell width -- exact,
63/// free, and available before any layout has happened. That is what makes the down-pass complete,
64/// and why only the up-pass needs a real measurement.
65///
66/// The order is visible in the types: a [`VerticalLength`](crate::VerticalLength) is refused by a
67/// horizontal role, because the horizontal pass runs first and cannot read a height.
68#[derive(Component, Clone, Debug, PartialEq)]
69pub struct Location(pub(crate) Breakpoints<Axes>);
70
71impl Location {
72    /// An empty placement: the whole of the parent's box, at every breakpoint.
73    ///
74    /// Each of [`xs`](Location::xs) upward states one breakpoint's placement, and a breakpoint
75    /// with none of its own takes the nearest smaller one that has.
76    pub fn new() -> Self {
77        Self(Breakpoints::new())
78    }
79
80    /// States the placement from the smallest breakpoint up, which is to say everywhere that a
81    /// larger one does not override it.
82    pub fn xs(self, horizontal: Horizontal, vertical: Vertical) -> Self {
83        self.set(Override::Xs, horizontal, vertical)
84    }
85
86    /// Overrides the placement from the `sm` breakpoint up.
87    pub fn sm(self, horizontal: Horizontal, vertical: Vertical) -> Self {
88        self.set(Override::Sm, horizontal, vertical)
89    }
90
91    /// Overrides the placement from the `md` breakpoint up.
92    pub fn md(self, horizontal: Horizontal, vertical: Vertical) -> Self {
93        self.set(Override::Md, horizontal, vertical)
94    }
95
96    /// Overrides the placement from the `lg` breakpoint up.
97    pub fn lg(self, horizontal: Horizontal, vertical: Vertical) -> Self {
98        self.set(Override::Lg, horizontal, vertical)
99    }
100
101    /// Overrides the placement at the `xl` breakpoint.
102    pub fn xl(self, horizontal: Horizontal, vertical: Vertical) -> Self {
103        self.set(Override::Xl, horizontal, vertical)
104    }
105
106    /// Overrides the placement whenever the viewport is vertically cramped, whatever its width.
107    pub fn short(self, horizontal: Horizontal, vertical: Vertical) -> Self {
108        self.set(Override::Short, horizontal, vertical)
109    }
110
111    fn set(mut self, at: Override, horizontal: Horizontal, vertical: Vertical) -> Self {
112        self.0.set(
113            at,
114            Axes {
115                horizontal: horizontal.0,
116                vertical: vertical.0,
117            },
118        );
119        self
120    }
121
122    pub(crate) fn axes(&self, layout: Layout, short: Short) -> &Axes {
123        self.0.at(layout, short)
124    }
125}
126
127impl Default for Location {
128    /// The whole of the parent's box.
129    fn default() -> Self {
130        Self::new()
131    }
132}
133
134/// One breakpoint's placement: both axes, each pinned down.
135#[derive(Clone, Debug, PartialEq)]
136pub(crate) struct Axes {
137    pub(crate) horizontal: Config,
138    pub(crate) vertical: Config,
139}
140
141impl Default for Axes {
142    /// The whole of the parent's box, which is what an element that says nothing fills.
143    fn default() -> Self {
144        Self {
145            horizontal: left(0.px()).right(100.pct()).0,
146            vertical: top(0.px()).bottom(100.pct()).0,
147        }
148    }
149}
