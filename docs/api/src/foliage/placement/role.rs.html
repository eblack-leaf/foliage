1//! Roles: what a source means for this element.
2//!
3//! A role is written first, because it says what the value is describing before the value has to be
4//! parsed. `left(anchor().right())` is read in the order it is understood.
5//!
6//! Each opener returns a type carrying only the completions that are legal with it, so an axis has
7//! exactly four spellings and no illegal pairing can be written at all. `left(..).center_x(..)` and
8//! `width(..).width(..)` are not rejected -- the method is not there.
9
10use crate::coordinate::Axis;
11use crate::placement::source::{
12    Against, Coord, Expr, HorizontalCoordinate, Kind, Length, Origin, VerticalCoordinate,
13    VerticalLength,
14};

16/// How one axis is pinned down: two of its coordinates, and any clamp on the extent between them.
17#[derive(Clone, Debug, PartialEq)]
18pub(crate) struct Config {
19    pub(crate) form: Form,
20    pub(crate) least: Option<Expr>,
21    pub(crate) most: Option<Expr>,
22}
23
24impl Config {
25    /// Whether anything in this axis of a placement reads the measured extent of what `against`
26    /// names, which is the one thing about it a measure that moved changes when its box did not.
27    pub(crate) fn reads_content(&self, against: Against) -> bool {
28        let form = match &self.form {
29            Form::NearExtent { near, extent } => [&near.expr, extent],
30            Form::NearFar { near, far } => [&near.expr, &far.expr],
31            Form::FarExtent { far, extent } => [&far.expr, extent],
32            Form::CenterExtent { center, extent } => [&center.expr, extent],
33        };
34        form.into_iter()
35            .chain(&self.least)
36            .chain(&self.most)
37            .any(|expr| expr.reads_content(against))
38    }
39
40    fn new(form: Form) -> Self {
41        Self {
42            form,
43            least: None,
44            most: None,
45        }
46    }
47
48    /// Whether this axis can be resolved before any box on it is known, and so whether it counts
49    /// toward the measure of the element it is grown under on `axis` -- R1m across, R2m down.
50    ///
51    /// An element sized by its contents is being asked how large what is inside it turned out. A
52    /// child that answers *by reading a box on that axis* -- a percentage of its trunk, a track of
53    /// its trunk's grid -- is not describing how large it is, it is asking someone else. It cannot be
54    /// what decides the answer without deciding it circularly, so it is left out of the measure and
55    /// given its real extent by the axis's own pass like anything else.
56    ///
57    /// Everything that describes an extent in its own terms counts: pixels, letters, its own
58    /// content -- and on the vertical axis any horizontal reading, because the horizontal axis has
59    /// already resolved by then. See [`known`] for the whole of it.
60    ///
61    /// Reading its **anchor** on the axis is the one case in between. The anchor's box is not laid
62    /// out yet either, but where the anchor is grown under the same element and itself counts, the
63    /// measure has already found where it reaches -- so a stack of things each placed under the
64    /// last is measured whole. That is [`Measure::Leans`], and the measure decides it, because only
65    /// it knows where the anchor is.
66    pub(crate) fn measurable(&self, axis: Axis) -> Measure {
67        let terms = |expr: &Expr| {
68            expr.terms().iter().fold(Measure::Own, |measure, term| {
69                measure.and(known(term.kind, axis))
70            })
71        };
72        let coordinate = |coordinate: &Coord| {
73            // An anchor's near edge is a position on the axis, known only where the anchor is.
74            let origin = match coordinate.origin {
75                Origin::Anchor => Measure::Leans,
76                Origin::Trunk | Origin::Surface => Measure::Own,
77            };
78            origin.and(terms(&coordinate.expr))
79        };
80        let form = match &self.form {
81            Form::NearExtent { near, extent } => coordinate(near).and(terms(extent)),
82            Form::NearFar { near, far } => coordinate(near).and(coordinate(far)),
83            Form::FarExtent { far, extent } => coordinate(far).and(terms(extent)),
84            Form::CenterExtent { center, extent } => coordinate(center).and(terms(extent)),
85        };
86        let clamp = |clamp: &Option<Expr>| clamp.as_ref().map_or(Measure::Own, terms);
87        form.and(clamp(&self.least)).and(clamp(&self.most))
88    }
89}

91/// How a placement on one axis counts toward the measure of what it is grown under.
92#[derive(Copy, Clone, Debug, Default, PartialEq, Eq)]
93pub(crate) enum Measure {
94    /// It reads a box on the axis that nothing has laid out yet, so it is left out.
95    #[default]
96    Never,
97    /// It describes its own extent, and counts.
98    Own,
99    /// It describes its extent against its anchor's, and counts where the anchor is grown under
100    /// the same element and counts too.
101    Leans,
102}
103
104impl Measure {
105    /// Both at once: left out if either is, leaning if either leans.
106    pub(crate) fn and(self, other: Self) -> Self {
107        match (self, other) {
108            (Self::Never, _) | (_, Self::Never) => Self::Never,
109            (Self::Leans, _) | (_, Self::Leans) => Self::Leans,
110            (Self::Own, Self::Own) => Self::Own,
111        }
112    }
113}
114
115/// Whether one term of a placement on `axis` reads something already settled when that axis is
116/// measured -- across before any width is laid out, and down once every width is.
117pub(crate) fn known(kind: Kind, axis: Axis) -> Measure {
118    let own = |known: bool| match known {
119        true => Measure::Own,
120        false => Measure::Never,
121    };
122    // What reads the anchor on the axis being measured leans on it; anything else that reads a
123    // box on that axis is not known.
124    let leans = |against: Against| match against {
125        Against::Anchor => Measure::Leans,
126        Against::Trunk | Against::Own => Measure::Never,
127    };
128    match kind {
129        // Reads no geometry at all, or reads a cell -- which R1 answered before either axis.
130        Kind::Px(_) => Measure::Own,
131        Kind::Letters { .. } => Measure::Own,
132        // A place in a run is read off the width the run resolved to, which the horizontal axis
133        // settles: known down, once it has, and not across, where it is the width being asked for.
134        Kind::Character { .. } | Kind::Span { .. } => own(axis == Axis::Vertical),
135        // The element's own measure, which the bottom-up sweep answers below a trunk before the
136        // trunk asks; and an anchor's, which it answers too where the anchor is under that trunk.
137        Kind::Content { against } => match against {
138            Against::Own => Measure::Own,
139            other => leans(other),
140        },
141        // An extent or a track on the other axis, once that axis has resolved -- which is only ever
142        // the horizontal one, read while measuring down. An anchor's extent on this axis is what
143        // the measure found it reaching to.
144        Kind::Extent {
145            axis: read,
146            against,
147        } => match read == axis {
148            true => leans(against),
149            false => own(axis == Axis::Vertical && read == Axis::Horizontal),
150        },
151        Kind::Cell { axis: read, .. } => own(axis == Axis::Vertical && read == Axis::Horizontal),
152        // A fraction of the axis being measured, which nothing has laid out yet -- unless it is a
153        // fraction of an anchor the measure has already found.
154        Kind::Pct { against, .. } => leans(against),
155        // Already a position on a surface no box on this axis has been laid out on yet, but for an
156        // anchor's, which the measure has placed.
157        Kind::Edge { against, .. } => leans(against),
158    }
159}

161/// The four legal ways to pin an axis down.
162///
163/// Near is left or top, far is right or bottom, and extent is width or height.
164#[derive(Clone, Debug, PartialEq)]
165pub(crate) enum Form {
166    NearExtent { near: Coord, extent: Expr },
167    NearFar { near: Coord, far: Coord },
168    FarExtent { far: Coord, extent: Expr },
169    CenterExtent { center: Coord, extent: Expr },
170}

172/// One axis of a placement, complete: the horizontal half.
173///
174/// Complete, so there is nothing left to say about the axis. A second extent has no method to be
175/// written with:
176///
177/// ```compile_fail,E0599
178/// use foliage::{Source, left};
179/// left(0.px()).width(10.px()).width(20.px());
180/// ```
181#[derive(Clone, Debug, PartialEq)]
182pub struct Horizontal(pub(crate) Config);
183
184/// One axis of a placement, complete: the vertical half.
185///
186/// A distinct type from [`Horizontal`], so the two cannot be given in the wrong order:
187///
188/// ```compile_fail,E0308
189/// use foliage::{Location, Source, left, top};
190/// Location::new().xs(top(0.px()).height(10.px()), left(0.px()).width(10.px()));
191/// ```
192#[derive(Clone, Debug, PartialEq)]
193pub struct Vertical(pub(crate) Config);
194
195impl Horizontal {
196    /// Refuses to resolve narrower than this.
197    ///
198    /// Named for the bound rather than `min`, because `width(content()).min(300)` puts two
199    /// unrelated senses of the word in one expression: one meaning the content's natural size, the
200    /// other a floor.
201    pub fn at_least(mut self, extent: impl Into<Length>) -> Self {
202        self.0.least = Some(extent.into().0);
203        self
204    }
205
206    /// Refuses to resolve wider than this. Over [`content()`](crate::content) this is fit-content:
207    /// whichever of the two is smaller.
208    pub fn at_most(mut self, extent: impl Into<Length>) -> Self {
209        self.0.most = Some(extent.into().0);
210        self
211    }
212}

214impl Vertical {

218        self
219    }
220
221    /// Refuses to resolve taller than this.
222    pub fn at_most(mut self, extent: impl Into<VerticalLength>) -> Self {

224        self
225    }

264/// A [`left`] awaiting the value that completes the axis.
265///
266/// It carries only the two completions that are legal with it, so an illegal pairing is not
