1//! Roles: what a source means for this element.
2//!
3//! A role is written first, because it says what the value is describing before the value has to be
4//! parsed. `left(anchor().right())` is read in the order it is understood.
5//!
6//! Each opener returns a type carrying only the completions that are legal with it, so an axis has
7//! exactly four spellings and no illegal pairing can be written at all. `left(..).center_x(..)` and
8//! `width(..).width(..)` are not rejected -- the method is not there.
9
10use crate::coordinate::Axis;
11use crate::placement::source::{
12    Against, Coord, Expr, HorizontalCoordinate, Kind, Length, Origin, VerticalCoordinate,
13    VerticalLength,
14};

16/// How one axis is pinned down: two of its coordinates, and any clamp on the extent between them.
17#[derive(Clone, Debug, PartialEq)]
18pub(crate) struct Config {
19    pub(crate) form: Form,
20    pub(crate) least: Option<Expr>,
21    pub(crate) most: Option<Expr>,
22}
23
24impl Config {
25    /// Whether anything in this axis of a placement reads the measured extent of what `against`
26    /// names, which is the one thing about it a measure that moved changes when its box did not.
27    pub(crate) fn reads_content(&self, against: Against) -> bool {
28        let form = match &self.form {
29            Form::NearExtent { near, extent } => [&near.expr, extent],
30            Form::NearFar { near, far } => [&near.expr, &far.expr],
31            Form::FarExtent { far, extent } => [&far.expr, extent],
32            Form::CenterExtent { center, extent } => [&center.expr, extent],
33        };
34        form.into_iter()
35            .chain(&self.least)
36            .chain(&self.most)
37            .any(|expr| expr.reads_content(against))
38    }
39
40    fn new(form: Form) -> Self {
41        Self {
42            form,
43            least: None,
44            most: None,
45        }
46    }
47
48    /// Whether this axis can be resolved before any box on it is known, and so whether it counts
49    /// toward the measure of the element it is grown under on `axis` -- R1m across, R2m down.
50    ///
51    /// An element sized by its contents is being asked how large what is inside it turned out. A
52    /// child that answers *by reading a box on that axis* -- a percentage of its trunk, a track of
53    /// its trunk's grid, an edge of an anchor -- is not describing how large it is, it is asking
54    /// someone else. It cannot be what decides the answer without deciding it circularly, so it is
55    /// left out of the measure and given its real extent by the axis's own pass like anything else.
56    ///
57    /// Everything that describes an extent in its own terms counts: pixels, letters, its own
58    /// content -- and on the vertical axis any horizontal reading, because the horizontal axis has
59    /// already resolved by then. See [`known`] for the whole of it.
60    pub(crate) fn measurable(&self, axis: Axis) -> bool {
61        let terms = |expr: &Expr| expr.terms().iter().all(|term| known(term.kind, axis));
62        let coordinate = |coordinate: &Coord| {
63            // An anchor's near edge is a position on the axis, which is exactly what is not known.
64            coordinate.origin != Origin::Anchor && terms(&coordinate.expr)
65        };
66        let form = match &self.form {
67            Form::NearExtent { near, extent } => coordinate(near) && terms(extent),
68            Form::NearFar { near, far } => coordinate(near) && coordinate(far),
69            Form::FarExtent { far, extent } => coordinate(far) && terms(extent),
70            Form::CenterExtent { center, extent } => coordinate(center) && terms(extent),
71        };
72        form && self.least.as_ref().is_none_or(terms) && self.most.as_ref().is_none_or(terms)
73    }
74}

76/// Whether one term of a placement on `axis` reads something already settled when that axis is
77/// measured -- across before any width is laid out, and down once every width is.
78pub(crate) fn known(kind: Kind, axis: Axis) -> bool {
79    match kind {
80        // Reads no geometry at all, or reads a cell -- which R1 answered before either axis.
81        Kind::Px(_) => true,
82        Kind::Letters { .. } => true,
83        // A place in a run is read off the width the run resolved to, which the horizontal axis
84        // settles: known down, once it has, and not across, where it is the width being asked for.
85        Kind::Character { .. } | Kind::Span { .. } => axis == Axis::Vertical,
86        // The element's own measure, which the bottom-up sweep answers below a trunk before the trunk
87        // asks. Anyone else's is still being measured.
88        Kind::Content { against } => against == Against::Own,
89        // An extent or a track on the other axis, once that axis has resolved -- which is only ever
90        // the horizontal one, read while measuring down.
91        Kind::Extent { axis: read, .. } | Kind::Cell { axis: read, .. } => {
92            axis == Axis::Vertical && read == Axis::Horizontal
93        }
94        // A fraction of the axis being measured, which nothing has laid out yet.
95        Kind::Pct { .. } => false,
96        // Already a position on a surface no box on this axis has been laid out on yet.
97        Kind::Edge { .. } => false,
98    }
99}

101/// The four legal ways to pin an axis down.
102///
103/// Near is left or top, far is right or bottom, and extent is width or height.
104#[derive(Clone, Debug, PartialEq)]
105pub(crate) enum Form {
106    NearExtent { near: Coord, extent: Expr },
107    NearFar { near: Coord, far: Coord },
108    FarExtent { far: Coord, extent: Expr },
109    CenterExtent { center: Coord, extent: Expr },
110}
111
112/// One axis of a placement, complete: the horizontal half.
113///
114/// Complete, so there is nothing left to say about the axis. A second extent has no method to be
115/// written with:
116///
117/// ```compile_fail,E0599
118/// use foliage::{Source, left};
119/// left(0.px()).width(10.px()).width(20.px());
120/// ```
121#[derive(Clone, Debug, PartialEq)]
122pub struct Horizontal(pub(crate) Config);
123
124/// One axis of a placement, complete: the vertical half.
125///
126/// A distinct type from [`Horizontal`], so the two cannot be given in the wrong order:
127///
128/// ```compile_fail,E0308
129/// use foliage::{Location, Source, left, top};
130/// Location::new().xs(top(0.px()).height(10.px()), left(0.px()).width(10.px()));
131/// ```
132#[derive(Clone, Debug, PartialEq)]
133pub struct Vertical(pub(crate) Config);
134
135impl Horizontal {
136    /// Refuses to resolve narrower than this.
137    ///
138    /// Named for the bound rather than `min`, because `width(content()).min(300)` puts two
139    /// unrelated senses of the word in one expression: one meaning the content's natural size, the
140    /// other a floor.
141    pub fn at_least(mut self, extent: impl Into<Length>) -> Self {
142        self.0.least = Some(extent.into().0);
143        self
144    }
145
146    /// Refuses to resolve wider than this. Over [`content()`](crate::content) this is fit-content:
147    /// whichever of the two is smaller.
148    pub fn at_most(mut self, extent: impl Into<Length>) -> Self {
149        self.0.most = Some(extent.into().0);
150        self
151    }
152}
153
154impl Vertical {
155    /// Refuses to resolve shorter than this.
156    pub fn at_least(mut self, extent: impl Into<VerticalLength>) -> Self {
157        self.0.least = Some(extent.into().0);
158        self
159    }
160
161    /// Refuses to resolve taller than this.
162    pub fn at_most(mut self, extent: impl Into<VerticalLength>) -> Self {
163        self.0.most = Some(extent.into().0);
164        self
165    }
166}

168/// The element's left edge, as a coordinate in its parent's space.
169///
170/// Complete it with a [`width`](Left::width) or with the [`right`](Left::right) edge.
171pub fn left(coordinate: impl Into<HorizontalCoordinate>) -> Left {
172    Left(coordinate.into().into())
173}

175/// The element's right edge, as a coordinate in its parent's space.
176///
177/// Every edge role is a coordinate, including this one, so sixteen in from the right is

179/// sources are already positions, so an inset would put two coordinate spaces in one expression.
180pub fn right(coordinate: impl Into<HorizontalCoordinate>) -> Right {

182}

204/// A [`left`] awaiting the value that completes the axis.
205///
206/// It carries only the two completions that are legal with it, so an illegal pairing is not
207/// rejected -- it cannot be written:
208///
209/// ```compile_fail,E0599
210/// use foliage::{Source, left};
211/// left(0.px()).center_x(50.pct());
212/// ```
213#[derive(Clone, Debug, PartialEq)]
214pub struct Left(Coord);
215
216/// A [`right`] awaiting the value that completes the axis.
217///
218/// A far edge takes an extent. The near-and-far pair is spelled from the near edge, so each of the
219/// four forms has exactly one spelling:
220///
221/// ```compile_fail,E0599
222/// use foliage::{Source, right};
223/// right(100.pct()).left(0.px());
224/// ```
225#[derive(Clone, Debug, PartialEq)]
226pub struct Right(Coord);
227
228/// A [`center_x`] awaiting the value that completes the axis.
