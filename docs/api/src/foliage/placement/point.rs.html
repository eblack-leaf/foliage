1//! Points: the placement grammar, read as a position rather than as a box.
2//!
3//! A [`Line`](crate::Line) has no box to state. What it has is two ends, and an end is a position
4//! on each axis -- which is exactly what a [`Coord`] already is. So a point is one coordinate per
5//! axis in the grammar every other placement is written in, and it resolves through the same pure
6//! resolver, in the same context, in the same two passes.
7//!
8//! That is the whole of the point mode. Nothing here is a second way to describe geometry: `px`,
9//! `pct`, `col`, `letters`, `content()`, `anchor()` and every arithmetic on them read identically,
10//! so a line's end can sit on a grid track, half way across its trunk, or at an anchor's edge.
11//!
12//! What is responsive is the pair and not the point: a [`Trace`](crate::Trace) states two of these
13//! per breakpoint, in the chain a [`Location`](crate::Location) is written in.
14
15use crate::coordinate::Axis;
16use crate::placement::role::known;
17use crate::placement::source::{Against, Coord, HorizontalCoordinate, Origin, VerticalCoordinate};
18
19/// A position stated in the placement grammar: one coordinate on each axis.
20///
21/// Read by whatever states its geometry as vertices rather than as a rectangle. The two halves are
22/// the same [`HorizontalCoordinate`] and [`VerticalCoordinate`] a box's edges take, which is what
23/// keeps one grammar rather than two.
24///
25/// A value rather than a role, so it is constructed rather than opened. The bare grammar openers --
26/// [`left`](crate::left), [`top`](crate::top) and the rest -- are free functions because each names
27/// what a value *describes* before the value is read, and returns something the grammar has yet to
28/// complete. A point describes nothing and is already complete.
29#[derive(Clone, Debug, PartialEq)]
30pub struct Point {
31    pub(crate) x: Coord,
32    pub(crate) y: Coord,
33}
34
35impl Point {
36    /// Whether this point on `axis` reads the measured extent of what `against` names.
37    pub(crate) fn reads_content(&self, against: Against, axis: Axis) -> bool {
38        match axis {
39            Axis::Horizontal => self.x.expr.reads_content(against),
40            Axis::Vertical => self.y.expr.reads_content(against),
41        }
42    }
43
44    /// A position, from one coordinate on each axis.
45    ///
46    /// The horizontal half is written first, as it is everywhere else in the grammar, and the two
47    /// are distinct types -- so the pair cannot be given the wrong way round:
48    ///
49    /// ```compile_fail,E0277
50    /// use foliage::{Point, Source, left, top};
51    /// Point::new(top(0.px()), left(0.px()));
52    /// ```
53    ///
54    /// Bare lengths read the trunk, exactly as they do in a box's placement:
55    ///
56    /// ```no_run
57    /// # use foliage::{Point, Source, anchor};
58    /// Point::new(0.px(), 0.px());
59    /// Point::new(50.pct(), 100.pct() - 8.px());
60    /// Point::new(2.col(), anchor().bottom());
61    /// ```
62    pub fn new(x: impl Into<HorizontalCoordinate>, y: impl Into<VerticalCoordinate>) -> Self {
63        Self {
64            x: x.into().into(),
65            y: y.into().into(),
66        }
67    }
68
69    /// This point's horizontal half, as the grammar states a coordinate.
70    ///
71    /// What puts a box *at* a point: a marker on a series, a round joint filling the wedge two
72    /// strokes leave open where they meet. Without it the same expression is written twice -- once
73    /// for the point and once for the box on it -- and the two are free to drift apart.
74    ///
75    /// Safe in the direction it is written and in no other: the halves were separated by
76    /// [`new`](Point::new) from values that were already typed, so putting one back where it came
77    /// from cannot cross the axes.
78    pub fn x(&self) -> HorizontalCoordinate {
79        HorizontalCoordinate {
80            expr: self.x.expr.clone(),
81            origin: self.x.origin,
82        }
83    }
84
85    /// This point's vertical half, as the grammar states a coordinate.
86    pub fn y(&self) -> VerticalCoordinate {
87        VerticalCoordinate {
88            expr: self.y.expr.clone(),
89            origin: self.y.origin,
90        }
91    }
92
93    /// Whether this point's half on `axis` can be answered before any box on that axis has
94    /// resolved, and so whether an element ending here counts toward the measure of the element it
95    /// is grown under on that axis.
96    ///
97    /// The same question [`Config::measurable`](crate::placement::role::Config::measurable) asks of
98    /// a box, asked of a vertex: a point that reads an extent on the axis is asking how large
99    /// something else is, and cannot be what decides how large this is.
100    pub(crate) fn measurable(&self, axis: Axis) -> bool {
101        let coordinate = match axis {
102            Axis::Horizontal => &self.x,
103            Axis::Vertical => &self.y,
104        };
105        coordinate.origin != Origin::Anchor
106            && coordinate
107                .expr
108                .terms()
109                .iter()
110                .all(|term| known(term.kind, axis))
111    }
112}
