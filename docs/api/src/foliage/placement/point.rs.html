1//! Points: the placement grammar, read as a position rather than as a box.
2//!
3//! A [`Line`](crate::Line) has no box to state. What it has is two ends, and an end is a position
4//! on each axis -- which is exactly what a [`Coord`] already is. So a point is one coordinate per
5//! axis in the grammar every other placement is written in, and it resolves through the same pure
6//! resolver, in the same context, in the same two passes.
7//!
8//! That is the whole of the point mode. Nothing here is a second way to describe geometry: `px`,
9//! `pct`, `col`, `letters`, `content()`, `anchor()` and every arithmetic on them read identically,
10//! so a line's end can sit on a grid track, half way across its trunk, or at an anchor's edge.
11
12use crate::placement::role::known;
13use crate::placement::source::{Coord, HorizontalCoordinate, Origin, VerticalCoordinate};
14
15/// A position stated in the placement grammar: one coordinate on each axis.
16///
17/// Read by whatever states its geometry as vertices rather than as a rectangle. The two halves are
18/// the same [`HorizontalCoordinate`] and [`VerticalCoordinate`] a box's edges take, which is what
19/// keeps one grammar rather than two.
20///
21/// A value rather than a role, so it is constructed rather than opened. The bare grammar openers --
22/// [`left`](crate::left), [`top`](crate::top) and the rest -- are free functions because each names
23/// what a value *describes* before the value is read, and returns something the grammar has yet to
24/// complete. A point describes nothing and is already complete.
25#[derive(Clone, Debug, PartialEq)]
26pub struct Point {
27    pub(crate) x: Coord,
28    pub(crate) y: Coord,
29}
30
31impl Point {
32    /// A position, from one coordinate on each axis.
33    ///
34    /// The horizontal half is written first, as it is everywhere else in the grammar, and the two
35    /// are distinct types -- so the pair cannot be given the wrong way round:
36    ///
37    /// ```compile_fail,E0277
38    /// use foliage::{Point, Source, left, top};
39    /// Point::new(top(0.px()), left(0.px()));
40    /// ```
41    ///
42    /// Bare lengths read the trunk, exactly as they do in a box's placement:
43    ///
44    /// ```no_run
45    /// # use foliage::{Point, Source, anchor};
46    /// Point::new(0.px(), 0.px());
47    /// Point::new(50.pct(), 100.pct() - 8.px());
48    /// Point::new(2.col(), anchor().bottom());
49    /// ```
50    pub fn new(x: impl Into<HorizontalCoordinate>, y: impl Into<VerticalCoordinate>) -> Self {
51        Self {
52            x: x.into().into(),
53            y: y.into().into(),
54        }
55    }
56
57    /// This point's horizontal half, as the grammar states a coordinate.
58    ///
59    /// What puts a box *at* a point: a marker on a series, a round joint filling the wedge two
60    /// strokes leave open where they meet. Without it the same expression is written twice -- once
61    /// for the point and once for the box on it -- and the two are free to drift apart.
62    ///
63    /// Safe in the direction it is written and in no other: the halves were separated by
64    /// [`new`](Point::new) from values that were already typed, so putting one back where it came
65    /// from cannot cross the axes.
66    pub fn x(&self) -> HorizontalCoordinate {
67        HorizontalCoordinate {
68            expr: self.x.expr.clone(),
69            origin: self.x.origin,
70        }
71    }
72
73    /// This point's vertical half, as the grammar states a coordinate.
74    pub fn y(&self) -> VerticalCoordinate {
75        VerticalCoordinate {
76            expr: self.y.expr.clone(),
77            origin: self.y.origin,
78        }
79    }
80
81    /// Whether this point's vertical half can be answered before any vertical box has resolved, and
82    /// so whether an element ending here counts toward the measure of the element it is grown under.
83    ///
84    /// The same question [`Config::measurable`](crate::placement::role::Config::measurable) asks of
85    /// a box, asked of a vertex: a point that reads a vertical extent is asking how tall something
86    /// else is, and cannot be what decides how tall this is.
87    pub(crate) fn measurable(&self) -> bool {
88        self.y.origin != Origin::Anchor && self.y.expr.terms.iter().all(|term| known(term.kind))
89    }
90}
