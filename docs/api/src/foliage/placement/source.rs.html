1//! Sources: where a number in a placement comes from.
2//!
3//! A source says nothing about what it is *for*. `20.px()` is neither a position nor a length until
4//! a role names it, and the role is also what decides which axis it resolves on. [`content()`] is
5//! the clearest case: in a width role it is the widest the element wants to be, and in a height
6//! role it is how tall it turned out at the width the horizontal pass gave it.
7//!
8//! Four types carry that, and the split is the resolution model stated as types. The horizontal
9//! axis resolves before the vertical one, so:
10//!
11//! - a [`Length`] is available to any role
12//! - a [`VerticalLength`] is available only to a vertical role, because only the vertical pass
13//!   knows the answer
14//! - a coordinate never crosses axes at all, because a position on one axis has no reading on the
15//!   other
16//!
17//! So `height(2.col())` is a two-column span used as a height, and `width(2.row())` does not
18//! compile.
19//!
20//! # Whose geometry
21//!
22//! Every term that reads geometry names whose it reads, as an [`Against`]. The bare spellings read
23//! the trunk, which is the ordinary case; [`trunk()`](crate::trunk) and
24//! [`anchor()`](crate::anchor) say it outright, and the second is what lets an element keep
25//! addressing a grid after it has been grown somewhere else.

31/// Whose geometry a term reads.
32///
33/// One question, asked once. A grid, a character cell, a measured size and a box all belong to some
34/// element, and a term that reads one has to say which -- otherwise half the grammar can only ever
35/// describe the trunk, and an element grown somewhere else loses the vocabulary it was written in.
36#[derive(Copy, Clone, Debug, PartialEq, Eq)]
37pub(crate) enum Against {
38    /// The element itself. Its declared character cell and its measured size, which are the only
39    /// two things about itself an element can read: its box is what is being solved for.
40    Own,
41    /// The element this one was grown under.
42    Trunk,
43    /// The one other element the placement may read.
44    Anchor,
45}
46
47/// What a term reads.
48#[derive(Copy, Clone, Debug, PartialEq)]
49pub(crate) enum Kind {
50    /// Logical pixels, as written. The only source that reads no geometry at all.
51    Px(f32),
52    /// A fraction of an extent on the resolving axis, where `1.0` is the whole of it.
53    Pct { fraction: f32, against: Against },
54    /// An extent on the named axis, whichever axis is resolving. Axis-explicit, so
55    /// `height(anchor().width())` is an element as tall as its anchor is wide.
56    Extent { axis: Axis, against: Against },
57    /// A one-based track index into a grid on the named axis. Which edge of that track it means is
58    /// the role's decision.
59    Cell {
60        index: i32,
61        axis: Axis,
62        against: Against,
63    },
64    /// A count of character cells on the resolving axis, in the named element's font.
65    Letters { letters: f32, against: Against },
66    /// A measured intrinsic extent on the resolving axis.
67    Content { against: Against },
68    /// One edge of a box. Already a position on the surface, so it is measured from nothing.
69    Edge { edge: Edge, against: Against },
70}
71
72/// Which edge of a box a term reads.
73#[derive(Copy, Clone, Debug, PartialEq)]
74pub(crate) enum Edge {
75    Left,
76    Right,
77    CenterX,
78    Top,
79    Bottom,
80    CenterY,
81}

83/// A sum of scaled terms.
84///
85/// Every operator a source supports -- addition, subtraction, and scaling by a plain number --
86/// keeps an expression linear, so this shape is total rather than a simplification. It is built
87/// once, where the placement is written, and never allocates during resolution.
88///
89/// Terms in one expression may read different elements: `anchor().bottom() + 50.pct()` is half the
90/// trunk's extent below the anchor's bottom edge, and each half names its own.
91#[derive(Clone, Debug, Default, PartialEq)]
92pub(crate) struct Expr {
93    pub(crate) terms: Vec<Term>,
94}
95
96/// One addend of an expression: a source, and the factor it was scaled by.
97#[derive(Copy, Clone, Debug, PartialEq)]
98pub(crate) struct Term {
99    pub(crate) scale: f32,
100    pub(crate) kind: Kind,
101}
102
103impl Term {
104    fn new(kind: Kind) -> Self {
105        Self { scale: 1.0, kind }
106    }
107}
108
109impl Expr {
110    fn of(kind: Kind) -> Self {
111        Self {
112            terms: vec![Term::new(kind)],
113        }
114    }
115
116    fn plus(mut self, other: Expr) -> Self {
117        self.terms.extend(other.terms);
118        self
119    }
120
121    fn minus(self, other: Expr) -> Self {
122        self.plus(other.negated())
123    }
124
125    fn negated(self) -> Self {
126        self.scaled(-1.0)
127    }
128
129    fn scaled(mut self, by: f32) -> Self {
130        for term in &mut self.terms {
131            term.scale *= by;
132        }
133        self
134    }
135}

137/// The near edge a coordinate's terms are measured from.
138///
139/// A coordinate is one origin and a sum of deltas. Which element supplies the origin is the basis
140/// the coordinate was opened against: a bare length takes the trunk, `anchor().col(2)` takes the
141/// anchor, and an edge is already a position on the surface and takes nothing.
142#[derive(Copy, Clone, Debug, PartialEq, Eq)]
143pub(crate) enum Origin {
144    /// The near edge of the trunk's box on the resolving axis.
145    Trunk,
146    /// The near edge of the anchor's box on the resolving axis.
147    Anchor,
148    /// The surface, which is where an edge already is.
149    Surface,
150}

177/// A length: an extent, with no position of its own.
178///
179/// Produced by [`Source`] and by the extent readings of a basis, such as
180/// [`anchor().width()`](crate::Anchor::width). Legal in every role -- as a size directly, and as a
181/// position measured from the trunk's near edge.
182#[derive(Clone, Debug, PartialEq)]
183pub struct Length(pub(crate) Expr);
184
185/// A length only the vertical axis can answer.
186///
187/// Produced by [`row`](Source::row) and by [`anchor().height()`](crate::Anchor::height). The
188/// horizontal axis resolves first and cannot see a vertical result, so these are legal in vertical
189/// roles only. A [`Length`] converts into one, never the reverse.
190///
191/// ```compile_fail,E0277
192/// use foliage::{Source, left};
193/// left(0.px()).width(2.row());
194/// ```
195///
196/// ```compile_fail,E0277
197/// use foliage::{Source, anchor, left};
198/// left(0.px()).width(anchor().height());
199/// ```
200///
201/// The other direction is fine, and useful -- `height(2.col())` is a two-column span used as a
202/// height.
203#[derive(Clone, Debug, PartialEq)]
204pub struct VerticalLength(pub(crate) Expr);
205
206/// A position on the horizontal axis.
207///
208/// Either a position read from a basis -- an edge, or a track of its grid -- or a [`Length`]
209/// measured from the trunk's left.
210///
211/// A position is not an extent, so an edge cannot be a size:
212///
213/// ```compile_fail,E0277
214/// use foliage::{Source, anchor, left};
215/// left(0.px()).width(anchor().left());
216/// ```
217///
218/// Subtracting two of them is what gives the [`Length`] between them, which is how two edges are
219/// used as a size. Adding two is not an operation:
220///
221/// ```compile_fail,E0308
222/// use foliage::{Source, anchor, left};
223/// left(anchor().left() + anchor().right()).width(10.px());
224/// ```
225///
226/// A position on one axis has no reading on the other, so coordinates never cross:
227///
228/// ```compile_fail,E0277
229/// use foliage::{Source, anchor, left};
230/// left(anchor().bottom()).width(10.px());
231/// ```
232#[derive(Clone, Debug, PartialEq)]
233pub struct HorizontalCoordinate {
234    pub(crate) expr: Expr,
235    pub(crate) origin: Origin,
236}

301/// Plain numbers as placement sources.
302///
303/// Every unit names itself at the call site, which is what keeps an expression readable when it
304/// mixes them: `right(100.pct() - 16.px())`.
305///
306/// These read the trunk. The same units against another element are on
307/// [`trunk()`](crate::trunk) and [`anchor()`](crate::anchor).
308pub trait Source: Sized {
309    /// Logical pixels.
310    fn px(self) -> Length;
311
312    /// A percentage of the trunk's extent on the axis the role names. `100.pct()` is the whole of
313    /// it, so `right(100.pct())` is the trunk's right edge.
314    fn pct(self) -> Length;
315
316    /// A one-based column of the trunk's grid.
317    ///
318    /// The role decides which part of that column is meant: a near role gives its left edge, a far
319    /// role its right edge, a centre role its middle, and a size role the width of a span of that
320    /// many columns, gaps included. So `left(1.col()).right(1.col())` is exactly the first column.
321    fn col(self) -> Length;
322
323    /// A one-based row of the trunk's grid, read the same way [`col`](Source::col) is.
324    fn row(self) -> VerticalLength;
325
326    /// A count of character cells, at the element's own font size.
327    ///
328    /// The right tool whenever the count is genuinely known ahead of time: it costs nothing to
329    /// resolve and it says what it means. Where the count is not known ahead of time,
330    /// [`content()`] measures instead.
331    ///
332    /// The element's own font, because that is the only one it is composed in. A count in the
333    /// font of another element is [`anchor().letters(n)`](crate::Anchor::letters).
334    fn letters(self) -> Length;
335}
336
337/// The element's own intrinsic extent, which is a different question per axis.
338///
339/// In a width role it is **max-content**: the widest the element wants to be, unwrapped. In a
340/// monospaced font that is the character count times the cell width, so it is exact and free, and
341/// it is available before any layout has happened.
342///
343/// In a height role it is what the element measured to *after* wrapping, at the width the
344/// horizontal pass gave it. One word, and the axis supplies the question.
345///
346/// Under [`at_most`](crate::Horizontal::at_most) this is fit-content: the smaller of what the
347/// content wants and what the ceiling allows.
348///
349/// An element with nothing in it has an intrinsic extent of zero. Another element's measured extent
350/// is [`anchor().content()`](crate::Anchor::content), which is a different number from its box
351/// whenever it was given more room than it asked for.
352pub fn content() -> Length {
353    Length::of(Kind::Content {
354        against: Against::Own,
355    })
356}

358macro_rules! source {
359    ($($number:ty),*) => {$(
360        impl Source for $number {
361            fn px(self) -> Length {
362                Length::of(Kind::Px(self as f32))
363            }
364
365            fn pct(self) -> Length {
366                Length::of(Kind::Pct {
367                    fraction: self as f32 / 100.0,
368                    against: Against::Trunk,
369                })
370            }
371
372            fn col(self) -> Length {
373                Length::of(Kind::Cell {
374                    index: self as i32,
375                    axis: Axis::Horizontal,
376                    against: Against::Trunk,
377                })
378            }
379
380            fn row(self) -> VerticalLength {
381                VerticalLength::of(Kind::Cell {
382                    index: self as i32,
383                    axis: Axis::Vertical,
384                    against: Against::Trunk,
385                })
386            }
387
388            fn letters(self) -> Length {
389                Length::of(Kind::Letters {
390                    letters: self as f32,
391                    against: Against::Own,
392                })
393            }
394        }
395    )*};
396}

433macro_rules! length_arithmetic {
434    ($name:ident) => {
435        impl Add for $name {
436            type Output = $name;
437            fn add(self, rhs: $name) -> $name {
438                $name(self.0.plus(rhs.0))
439            }
440        }
441
442        impl Sub for $name {
443            type Output = $name;
444            fn sub(self, rhs: $name) -> $name {
445                $name(self.0.minus(rhs.0))
446            }
447        }
448
449        impl Mul<f32> for $name {
450            type Output = $name;
451            fn mul(self, rhs: f32) -> $name {
452                $name(self.0.scaled(rhs))
453            }
454        }
455
456        impl Neg for $name {
457            type Output = $name;
458            fn neg(self) -> $name {
459                $name(self.0.negated())
460            }
461        }
462    };
463}

496macro_rules! coordinate_arithmetic {
497    ($name:ident, $($length:ident),*) => {$(
498        impl Add<$length> for $name {
499            type Output = $name;
500            fn add(self, rhs: $length) -> $name {
501                $name { expr: self.expr.plus(rhs.0), origin: self.origin }
502            }
503        }
504
505        impl Sub<$length> for $name {
506            type Output = $name;
507            fn sub(self, rhs: $length) -> $name {
508                $name { expr: self.expr.minus(rhs.0), origin: self.origin }
509            }
510        }
511
512        impl Add<$name> for $length {
513            type Output = $name;
514            fn add(self, rhs: $name) -> $name {
515                $name { expr: self.0.plus(rhs.expr), origin: rhs.origin }
516            }
517        }
518    )*};
519}
