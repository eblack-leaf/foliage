1//! Bases: the elements a placement may read, and the one vocabulary each is read in.
2//!
3//! An element resolves against three things -- itself, the trunk it was grown under, and the one
4//! element it anchors to. [`trunk()`] and [`anchor()`] open the last two, and both carry the same
5//! readings, so nothing is sayable about one that is not sayable about the other.
6//!
7//! The element's own two readings have no opener: its box is what is being solved for, so
8//! [`content()`](crate::content) and [`letters`](crate::Source::letters) are the whole of what it
9//! can ask about itself.
10
11use crate::coordinate::Axis;
12use crate::placement::source::{
13    Against, Edge, HorizontalCoordinate, Kind, Length, Origin, VerticalCoordinate, VerticalLength,
14};
15
16/// The element this one was grown under.
17///
18/// The bare sources on [`Source`](crate::Source) already read the trunk, so this is for the two
19/// readings they cannot spell -- [`content`](Trunk::content) and [`letters`](Trunk::letters), which
20/// belong to the element that owns them and are not the same as the reader's own.
21pub fn trunk() -> Trunk {
22    Trunk
23}
24
25/// The one other element this one's placement may read.
26///
27/// Set with [`anchored`](crate::Place::anchored) when the element is described, or with
28/// [`anchor`](crate::Grow::anchor) at any time after. An element has at most one, and a placement
29/// that reads an anchor it has not been given resolves against a zero box.
30///
31/// Anchoring reads one already-resolved box, which is cheap and exact, and is the right answer for
32/// "sit below that thing, wherever it ended up" -- it holds up under wrapping and stacking where a
33/// fixed offset does not. Sizing to content answers a different question, and neither substitutes
34/// for the other.
35///
36/// It carries the whole vocabulary a trunk does, which is what makes growing an element elsewhere a
37/// real move rather than a downgrade: an element that leaves its trunk to escape a stack or a clip
38/// anchors back to it and goes on addressing the same grid.
39pub fn anchor() -> Anchor {
40    Anchor
41}
42
43/// What [`trunk()`] reads.
44#[derive(Copy, Clone, Debug)]
45pub struct Trunk;
46
47/// What [`anchor()`] reads.
48#[derive(Copy, Clone, Debug)]
49pub struct Anchor;
50
51macro_rules! basis {
52    ($name:ident, $against:expr, $origin:expr) => {
53        impl $name {
54            /// Its left edge.
55            pub fn left(self) -> HorizontalCoordinate {
56                HorizontalCoordinate::edge(Edge::Left, $against)
57            }
58
59            /// Its right edge.
60            pub fn right(self) -> HorizontalCoordinate {
61                HorizontalCoordinate::edge(Edge::Right, $against)
62            }
63
64            /// Its horizontal midpoint.
65            pub fn center_x(self) -> HorizontalCoordinate {
66                HorizontalCoordinate::edge(Edge::CenterX, $against)
67            }
68
69            /// Its top edge.
70            pub fn top(self) -> VerticalCoordinate {
71                VerticalCoordinate::edge(Edge::Top, $against)
72            }
73
74            /// Its bottom edge.
75            pub fn bottom(self) -> VerticalCoordinate {
76                VerticalCoordinate::edge(Edge::Bottom, $against)
77            }
78
79            /// Its vertical midpoint.
80            pub fn center_y(self) -> VerticalCoordinate {
81                VerticalCoordinate::edge(Edge::CenterY, $against)
82            }
83
84            /// How wide it is.
85            ///
86            /// A length rather than a coordinate, so it is legal on either axis:
87            /// `height(anchor().width())` is an element as tall as its anchor is wide.
88            pub fn width(self) -> Length {
89                Length::of(Kind::Extent {
90                    axis: Axis::Horizontal,
91                    against: $against,
92                })
93            }
94
95            /// How tall it is.
96            ///
97            /// Vertical only, because the horizontal pass runs first and cannot read a height.
98            pub fn height(self) -> VerticalLength {
99                VerticalLength::of(Kind::Extent {
100                    axis: Axis::Vertical,
101                    against: $against,
102                })
103            }
104
105            /// A one-based column of its grid, read the way [`col`](crate::Source::col) is.
106            ///
107            /// A position rather than a length, because a track of someone else's grid is somewhere
108            /// on the surface rather than an offset into this element's own trunk. So it is legal
109            /// where an edge is legal, and refused where an edge is:
110            ///
111            /// ```compile_fail,E0277
112            /// use foliage::{Source, anchor, left};
113            /// left(0.px()).width(anchor().col(2));
114            /// ```
115            ///
116            /// A span of another element's tracks is the distance between two of them, exactly as
117            /// it is for edges.
118            pub fn col(self, index: i32) -> HorizontalCoordinate {
119                HorizontalCoordinate::cell(index, $against, $origin)
120            }
121
122            /// A one-based row of its grid, read the way [`row`](crate::Source::row) is.
123            ///
124            /// A position on the vertical axis, so it does not cross:
125            ///
126            /// ```compile_fail,E0277
127            /// use foliage::{Source, anchor, left};
128            /// left(anchor().row(2)).width(10.px());
129            /// ```
130            pub fn row(self, index: i32) -> VerticalCoordinate {
131                VerticalCoordinate::cell(index, $against, $origin)
132            }
133
134            /// A count of character cells in *its* font, which is not the reader's.
135            ///
136            /// An element composes in its own font, so [`letters`](crate::Source::letters) is
137            /// always the reader's. This is the other one, and it is what a letter-pitched grid is
138            /// measured in.
139            pub fn letters(self, count: f32) -> Length {
140                Length::of(Kind::Letters {
141                    letters: count,
142                    against: $against,
143                })
144            }
145
146            /// Its measured intrinsic extent on the resolving axis.
147            ///
148            /// A different number from its box whenever it was given more room than it asked for,
149            /// which is what makes "as wide as that label's text" sayable at all.
150            pub fn content(self) -> Length {
151                Length::of(Kind::Content { against: $against })
152            }
153        }
154    };
155}
156
157basis!(Trunk, Against::Trunk, Origin::Trunk);
158basis!(Anchor, Against::Anchor, Origin::Anchor);
