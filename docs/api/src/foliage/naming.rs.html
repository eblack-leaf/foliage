1//! Where every name the engine hands out comes from.
2//!
3//! One source, drawn from by both sides of the boundary: a name taken inside a frame and one taken
4//! from a [`Sprig`](crate::Sprig) on another thread can never collide, and the order they were asked
5//! for is the order they were asked for. Every counter here is atomic for that reason -- naming takes
6//! `&self` wherever it happens, and an op issued off the frame is ordered against the frame's own by
7//! nothing but when it arrived.
8//!
9//! What is named here is exactly what can be named before the thing exists, which is everything a
10//! name is handed out for: an element, a channel, a group, a picture, a face, a mark. Each registry
11//! holds what has been filled and grows to meet a name it has not seen, so nothing has to reach one
12//! to be given a name for it.
13//!
14//! Every counter is monotonic and **nothing is ever handed out twice**, which is what makes a stale
15//! handle inert rather than dangerous: there is no later thing for it to come to address. A
16//! generation would be what made reuse safe, and so there is none to carry -- except on an element,
17//! where the index behind a [`Leaf`] does return here when it withers, and the generation it comes
18//! back with is what tells the two apart.
19
20use std::sync::atomic::{AtomicU32, AtomicU64, AtomicUsize, Ordering};
21use std::sync::{Arc, Mutex, PoisonError};
22
23use crate::aspen::{Sequence, Tween};
24use crate::icon::Field;
25use crate::image::Plate;
26use crate::leaf::{Growth, Leaf};
27use crate::text::Font;
28
29/// The one source of names, shared by the frame and by every [`Sprig`](crate::Sprig).
30#[derive(Clone)]
31pub(crate) struct Naming(Arc<Names>);
32
33struct Names {
34    /// One past every index an element has been named by: where a name comes from when none has
35    /// gone back.
36    fresh: AtomicU32,
37    /// The names of elements that withered, each at the generation it is handed out at next. The
38    /// last to go back is the first handed out, so what is held per index stays as few indices
39    /// long as there have ever been elements at once.
40    free: Mutex<Vec<Leaf>>,
41    /// How many are in `free`, read without taking it: a frame growing a great deal with nothing
42    /// gone back -- the first frame, a page loading -- takes no lock for any of it.
43    freed: AtomicUsize,
44    growth: AtomicU64,
45    tweens: AtomicU64,
46    sequences: AtomicU64,
47    plates: AtomicU32,
48    /// Starts past the bundled face, which is [`Font::DEFAULT`] and is registered before anything
49    /// can ask for a name.
50    faces: AtomicU32,
51    marks: AtomicU32,
52}
53
54impl Naming {
55    pub(crate) fn new() -> Self {
56        Self(Arc::new(Names {
57            fresh: AtomicU32::new(0),
58            free: Mutex::new(Vec::new()),
59            freed: AtomicUsize::new(0),
60            growth: AtomicU64::new(0),
61            tweens: AtomicU64::new(0),
62            sequences: AtomicU64::new(0),
63            plates: AtomicU32::new(0),
64            faces: AtomicU32::new(Font::DEFAULT.0 + 1),
65            marks: AtomicU32::new(0),
66        }))
67    }
68
69    /// A name for one element, and its place in allocation order.
70    ///
71    /// Both are taken here rather than at the drain, so the order is the order `plant` and `branch`
72    /// were called in and not the order the drain reached them. That order settles the elevation
73    /// tie-break, and it is total across the boundary because there is one counter rather than one
74    /// per side.
75    pub(crate) fn leaf(&self) -> (Leaf, Growth) {
76        let names = &self.0;
77        let returned = match names.freed.load(Ordering::Acquire) {
78            0 => None,
79            _ => {
80                let mut free = names.free.lock().unwrap_or_else(PoisonError::into_inner);
81                let leaf = free.pop();
82                names.freed.store(free.len(), Ordering::Release);
83                leaf
84            }
85        };
86        // Nothing went back, or another thread took the last of it between the count and the lock.
87        let leaf = returned.unwrap_or_else(|| Leaf {
88            index: names.fresh.fetch_add(1, Ordering::Relaxed),
89            generation: 0,
90        });
91        (leaf, Growth(names.growth.fetch_add(1, Ordering::Relaxed)))
92    }
93
94    /// Hands the indices of elements that withered back, each at the generation its next name
95    /// carries. The frame's alone to call, since only the frame withers anything.
96    pub(crate) fn release(&self, leaves: impl IntoIterator<Item = Leaf>) {
97        let mut free = self.0.free.lock().unwrap_or_else(PoisonError::into_inner);
98        free.extend(leaves);
99        self.0.freed.store(free.len(), Ordering::Release);
100    }
101
102    /// A name for one channel. Never reused, so a stale one is inert.
103    pub(crate) fn tween(&self) -> Tween {
104        Tween(self.0.tweens.fetch_add(1, Ordering::Relaxed))
105    }
106
107    /// A name for one group.
108    pub(crate) fn sequence(&self) -> Sequence {
109        Sequence(self.0.sequences.fetch_add(1, Ordering::Relaxed))
110    }
111
112    /// A name for one picture, whose pixels may not exist yet.
113    ///
114    /// Taken from a counter rather than from the registry that holds pictures, because a decode that
115    /// finishes on another thread names what it decoded where the registry is out of reach. The
116    /// registry grows to meet the name when it is filled.
117    pub(crate) fn plate(&self) -> Plate {
118        Plate(self.0.plates.fetch_add(1, Ordering::Relaxed))
119    }
120
121    /// A name for one typeface, whose bytes may not be parsed yet.
122    ///
123    /// Never [`Font::DEFAULT`]: the bundled face holds that name for the life of the run, which is
124    /// what lets a face that has not arrived be measured as one.
125    pub(crate) fn face(&self) -> Font {
126        Font(self.0.faces.fetch_add(1, Ordering::Relaxed))
127    }
128
129    /// A name for one mark, whose field may not have arrived yet.
130    pub(crate) fn mark(&self) -> Field {
131        Field(self.0.marks.fetch_add(1, Ordering::Relaxed))
132    }
133}
