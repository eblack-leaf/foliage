1//! The three ways an element can be off, and the products they resolve to.
2//!
3//! They are genuinely different, and stating them together is the point:
4//!
5//! | | Draws | In the box stack | Receives input |
6//! |---|---|---|---|
7//! | `visible(false)` | no | no | no |
8//! | `disable` | yes | **yes** | no -- swallows presses, passes scrolling on |
9//! | `opacity(0.0)` | nothing to see | no | no |
10//!
11//! A disabled element still draws and still blocks, which is what makes it different from
12//! decoration and what makes disabling a page enough on its own when a drawer opens over it. A
13//! fully transparent one is not there at all, which closes the case where an element faded out went
14//! on taking presses.
15//!
16//! Each is declared on one element and **inherited as a product** over the whole ancestry, computed
17//! in [`inherit`](crate::rowan) every frame. Nothing has a cascade to write and nothing has one to
18//! get wrong: an element grown under a disabled trunk is disabled on its first frame, and enabling
19//! the trunk leaves anything disabled in its own right disabled.
20
21/// Whether the app has hidden the element.
22///
23/// App intent, and only that. Being scrolled out of view is not a kind of hidden -- culling is a
24/// decision extraction makes from the clip rect and is never recorded here, so content scrolled
25/// past still counts toward its region's extent and can be scrolled back to.
26#[derive(Copy, Clone, Debug, PartialEq)]
27pub(crate) struct Visible(pub(crate) bool);
28
29impl Default for Visible {
30    fn default() -> Self {
31        Self(true)
32    }
33}
34
35/// How opaque the element is, in `0.0..=1.0`.
36#[derive(Copy, Clone, Debug, PartialEq)]
37pub(crate) struct Opacity(pub(crate) f32);
38
39impl Opacity {
40    pub(crate) fn new(opacity: f32) -> Self {
41        Self(opacity.clamp(0.0, 1.0))
42    }
43}
44
45impl Default for Opacity {
46    fn default() -> Self {
47        Self(1.0)
48    }
49}

51/// Whether the element was disabled in its own right, as against by an ancestor.
52///
53/// Held separately from the product for exactly that reason: re-enabling an ancestor recomputes the
54/// product over the whole ancestry, and an element that turned itself off stays off.
55#[derive(Copy, Clone, Debug, Default, PartialEq)]
56pub(crate) struct Disabled(pub(crate) bool);
57
58/// What the three resolved to over an element's whole ancestry.
59///
60/// R7's one output, and what the box stack and extraction read. Nothing reads the declarations
61/// directly, so there is no path by which an element could act on its own value while ignoring the
62/// subtree it sits in. Held in resolution's own columns rather than on the element.
63#[derive(Copy, Clone, Debug, PartialEq)]
64pub(crate) struct Inherited {
65    pub(crate) visible: bool,
66    /// The whole product, over every ancestor: how faint it looks, and what decides whether it is
67    /// there at all.
68    pub(crate) opacity: f32,
69    pub(crate) disabled: bool,
70    /// The picture it is drawn into, where something it is grown under -- or it -- is drawn
71    /// [`together`](crate::Place::together) and fading: one more than the slot of the element
72    /// that opened it, or zero.
73    pub(crate) layer: u32,
74    /// How opaque it is drawn into that picture: the product from the element that opened it down,
75    /// that element's own and everything above it left out, since the picture is faded by those
76    /// once, whole. The same as [`opacity`](Self::opacity) where there is no picture.
77    pub(crate) drawn: f32,
78}
79
80impl Inherited {
81    /// This element's own declarations resolved against what its trunk resolved to. `together`
82    /// is the picture it would open, where it is declared [`together`](crate::Place::together).
83    pub(crate) fn under(
84        trunk: Inherited,
85        visible: bool,
86        opacity: f32,
87        disabled: bool,
88        together: Option<u32>,
89    ) -> Self {
90        let product = trunk.opacity * opacity;
91        let (layer, drawn) = match (trunk.layer, together) {
92            // Already in a picture: drawn into it, faded by everything between.
93            (layer, _) if layer != 0 => (layer, trunk.drawn * opacity),
94            // Opens one, but only while there is a fade to apply to it whole. At full opacity
95            // there is nothing to show through, and nothing is drawn apart.
96            (_, Some(layer)) if product < 1.0 => (layer, 1.0),
97            _ => (0, product),
98        };
99        Self {
100            visible: trunk.visible && visible,
101            opacity: product,
102            disabled: trunk.disabled || disabled,
103            layer,
104            drawn,
105        }
106    }
107
108    /// Whether the element opens the picture it is drawn into, as against being drawn into one
109    /// something above it opened. `slot` is its own.
110    pub(crate) fn opens(&self, slot: usize) -> bool {
111        self.layer != 0 && self.layer == slot as u32 + 1
112    }
113
114    /// Whether the element is in the box stack at all.
115    ///
116    /// Hidden or fully transparent is not there, wherever the fade was declared -- on the element,
117    /// on the root of the part it belongs to, or on anything above -- so fading something out is
118    /// enough to be rid of it, and no part's inner structure decides whether a fade leaves an
119    /// invisible control behind. A press target that shows nothing is said by what it draws
120    /// rather than by fading it: a stem, or a clear color, at full opacity. Disabled is there: it
121    /// blocks, which is the whole difference between a disabled control and decoration.
122    pub(crate) fn present(&self) -> bool {
123        self.visible && self.opacity > 0.0
124    }
125}
126
127impl Default for Inherited {
128    fn default() -> Self {
129        Self {
130            visible: true,
131            opacity: 1.0,
132            disabled: false,
133            layer: 0,
134            drawn: 1.0,
135        }
136    }
137}
