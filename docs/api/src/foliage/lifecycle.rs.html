1//! The three ways an element can be off, and the products they resolve to.
2//!
3//! They are genuinely different, and stating them together is the point:
4//!
5//! | | Draws | In the box stack | Receives input |
6//! |---|---|---|---|
7//! | `visible(false)` | no | no | no |
8//! | `disable` | yes | **yes** | no -- and swallows |
9//! | `opacity(0.0)` | nothing to see | no | no |
10//!
11//! A disabled element still draws and still blocks, which is what makes it different from
12//! decoration and what makes disabling a page enough on its own when a drawer opens over it. A
13//! fully transparent one is not there at all, which closes the case where an element faded out went
14//! on taking presses.
15//!
16//! Each is declared on one element and **inherited as a product** over the whole ancestry, computed
17//! in [`inherit`](crate::rowan) every frame. Nothing has a cascade to write and nothing has one to
18//! get wrong: an element grown under a disabled trunk is disabled on its first frame, and enabling
19//! the trunk leaves anything disabled in its own right disabled.
20
21use bevy_ecs::component::Component;
22
23/// Whether the app has hidden the element.
24///
25/// App intent, and only that. Being scrolled out of view is not a kind of hidden -- culling is a
26/// decision extraction makes from the clip rect and is never recorded here, so content scrolled
27/// past still counts toward its region's extent and can be scrolled back to.
28#[derive(Component, Copy, Clone, Debug, PartialEq)]
29pub(crate) struct Visible(pub(crate) bool);
30
31impl Default for Visible {
32    fn default() -> Self {
33        Self(true)
34    }
35}
36
37/// How opaque the element is, in `0.0..=1.0`.
38#[derive(Component, Copy, Clone, Debug, PartialEq)]
39pub(crate) struct Opacity(pub(crate) f32);
40
41impl Opacity {
42    pub(crate) fn new(opacity: f32) -> Self {
43        Self(opacity.clamp(0.0, 1.0))
44    }
45}
46
47impl Default for Opacity {
48    fn default() -> Self {
49        Self(1.0)
50    }
51}
52
53/// Whether the element was disabled in its own right, as against by an ancestor.
54///
55/// Held separately from the product for exactly that reason: re-enabling an ancestor recomputes the
56/// product over the whole ancestry, and an element that turned itself off stays off.
57#[derive(Component, Copy, Clone, Debug, Default, PartialEq)]
58pub(crate) struct Disabled(pub(crate) bool);
59
60/// What the three resolved to over an element's whole ancestry.
61///
62/// R7's one output, and what the box stack and extraction read. Nothing reads the declarations
63/// directly, so there is no path by which an element could act on its own value while ignoring the
64/// subtree it sits in.
65#[derive(Component, Copy, Clone, Debug, PartialEq)]
66pub(crate) struct Inherited {
67    pub(crate) visible: bool,
68    pub(crate) opacity: f32,
69    pub(crate) disabled: bool,
70}
71
72impl Inherited {
73    /// This element's own declarations resolved against what its trunk resolved to.
74    pub(crate) fn under(
75        trunk: Inherited,
76        visible: Visible,
77        opacity: Opacity,
78        disabled: Disabled,
79    ) -> Self {
80        Self {
81            visible: trunk.visible && visible.0,
82            opacity: trunk.opacity * opacity.0,
83            disabled: trunk.disabled || disabled.0,
84        }
85    }
86
87    /// Whether the element is in the box stack at all.
88    ///
89    /// Hidden or fully transparent is not there. Disabled is: it blocks, which is the whole
90    /// difference between a disabled control and decoration.
91    pub(crate) fn present(&self) -> bool {
92        self.visible && self.opacity > 0.0
93    }
94}
95
96impl Default for Inherited {
97    fn default() -> Self {
98        Self {
99            visible: true,
100            opacity: 1.0,
101            disabled: false,
102        }
103    }
104}
