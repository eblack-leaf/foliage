1//! The box stack, and the one read taken of it.
2//!
3//! # What is at this point, and who receives it, are two questions
4//!
5//! At a pointer position the **box stack** is every element whose box contains that point and which
6//! is not clipped away, hidden or fully transparent, ordered top-first by resolved elevation.
7//!
8//! **Membership is universal.** Nothing opts in. An element is in the stack because it is there,
9//! which is geometry, and it is what makes occlusion per-point and automatic: there is no such
10//! thing as an obscured element, only an element that is below another *at this point*. A panel
11//! covering half a button is above it for those pixels and absent for the rest, with no special
12//! handling anywhere.
13//!
14//! # The read does not search
15//!
16//! A gesture goes to the top of the stack -- the element nearest the viewer at that point, whatever
17//! it draws and whether or not it receives. [`intangible`](crate::Place::intangible) is what a
18//! gesture passes through rather than landing on, so the element beneath it is the top. An element
19//! that is at the top without receiving eats the gesture, and a backdrop, a sheet backing and a
20//! menu's padding are all that without declaring anything to be it.
21//!
22//! Reading the top is the whole of it. The engine never continues downward looking for an element
23//! willing to take the gesture, and never passes over one because it is undeclared or draws
24//! nothing. A search would have to judge, at each element it passed, whether that element is *part
25//! of* what it covers or a *layer over* it -- and at a point those are the same picture, so it
26//! would answer by inference. Inference is wrong silently, and it is wrong twice: where a gesture
27//! lands is where a following drag looks for its scrolling region, so a press attributed to the
28//! wrong element scrolls the wrong region.
29
30use crate::coordinate::{Position, Section};
31use crate::elevation::ResolvedElevation;
32use crate::interaction::Shape;
33use crate::leaf::Leaf;

35/// One element as the hit test sees it: where it is, what shape it is, and what it declared.
36#[derive(Copy, Clone, Debug)]
37pub(crate) struct Region {
38    pub(crate) leaf: Leaf,
39    /// Where it was drawn.
40    pub(crate) section: Section,
41    /// What a scrolling ancestor left of it. A point outside this is outside the element, however
42    /// far its own box extends.
43    pub(crate) clip: Section,
44    pub(crate) shape: Shape,
45    /// Whether a gesture lands here. False for an element declared
46    /// [`intangible`](crate::Place::intangible), which is in the stack but never the top of it.
47    pub(crate) tangible: bool,
48    /// Declared [`interactive`](crate::Place::interactive).
49    pub(crate) receives: bool,
50    /// Disabled, in its own right or by an ancestor. Blocks, and gives nothing back.
51    pub(crate) disabled: bool,
52}
53
54impl Region {
55    /// Whether the element covers `point`.
56    fn holds(&self, point: Position) -> bool {
57        if !self.clip.contains(point) {
58            return false;
59        }
60        match self.shape {
61            Shape::Box => self.section.contains(point),
62            // The ellipse inscribed in the box, so a round control does not take presses in the
63            // square corners it does not draw.
64            Shape::Round => {
65                if self.section.is_empty() {
66                    return false;
67                }
68                let center = self.section.center();
69                let x = (point.x - center.x) / (self.section.width() / 2.0);
70                let y = (point.y - center.y) / (self.section.height() / 2.0);
71                x * x + y * y <= 1.0
72            }
73        }
74    }
75}

77/// Every element that was drawn, front-most first.
78///
79/// Rebuilt by R8 each frame and read by the next frame's dispatch, which is what makes hit-testing
80/// run against what was drawn: a pointer event was produced by a person looking at the screen, and
81/// the screen is the previous frame's render.
82#[derive(Default)]
83pub(crate) struct Stack {
84    regions: Vec<Region>,
85}
86
87impl Stack {
88    /// Takes this frame's regions, ordering them front-most first.
89    ///
90    /// The order is total -- a resolved elevation carries allocation order as its tie-break -- so
91    /// two identical runs read the same element at the same point.
92    pub(crate) fn settle(&mut self, mut ranked: Vec<(ResolvedElevation, Region)>) {
93        ranked.sort_by(|left, right| right.0.cmp(&left.0));
94        self.regions.clear();
95        self.regions
96            .extend(ranked.into_iter().map(|(_, region)| region));
97    }
98
99    /// The top of the stack at `point`: the front-most tangible element there.
100    ///
101    /// One read, and the whole of the hit test.
102    pub(crate) fn top(&self, point: Position) -> Option<Region> {
103        self.regions
104            .iter()
105            .find(|region| region.tangible && region.holds(point))
106            .copied()
107    }
108
109    /// How many elements were in the stack. Reported to the trace, and nothing else reads it.
110    pub(crate) fn len(&self) -> usize {
111        self.regions.len()
112    }
113}
