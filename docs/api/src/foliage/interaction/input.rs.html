1//! What arrives from the platform, before anything has been decided about it.
2//!
3//! One shape for every device. A mouse, a finger and a stylus produce the same four pointer
4//! statements, and the wheel is separate because it is a discrete pulse rather than a gesture with a
5//! lifecycle: it moves what is under it and is over.
6//!
7//! Nothing here reads the tree. Translation from the platform's own events happens at the edge, and
8//! the headless suite writes these directly -- which is what makes the two paths the same path, and
9//! what keeps the winit translation layer the only part of input the suite cannot reach.
10
11use core::time::Duration;
12
13use web_time::Instant;
14
15use crate::coordinate::Position;
16
17/// One input event, as the platform reported it.
18#[derive(Clone, Debug, PartialEq)]
19pub(crate) enum Input {
20    /// The pointer went down. Opens a gesture.
21    Pressed(Position),
22    /// The pointer moved while down.
23    Moved(Position),
24    /// The pointer came up.
25    Released(Position),
26    /// The gesture was taken away rather than completed -- the window lost focus, a touch was
27    /// cancelled by the system. Never produces a tap.
28    Cancelled,
29    /// A wheel notch at a point, as the movement a drag of the same distance would have been.
30    Wheeled { at: Position, delta: Position },
31    /// A key went down. Ordered against everything else here, because keystrokes are the one thing
32    /// in the engine whose order is part of what they mean.
33    ///
34    /// It names the key alone. What was held with it is [`Modifiers`], which arrives as its own
35    /// event in this same stream -- so what a key was pressed with is read from the order the two
36    /// arrived in rather than carried alongside.
37    Keyed(Key),
38    /// What is now held down. Ordered here with everything else for the reason above: a modifier is
39    /// only ever a statement about the keys pressed after it.
40    Modifiers(Modifiers),
41    /// Text the host pasted, carrying what it holds rather than a request to go and read it.
42    ///
43    /// The one event here with a payload, and the reason is that some hosts answer a paste and are
44    /// never willing to be asked one: a browser hands the text to the gesture that pasted it and
45    /// refuses the same read a frame later. So where a host says what was pasted, it is taken here
46    /// and ordered against the keystrokes around it; where it does not,
47    /// [`Key::Typed('v')`](Key::Typed) with control asks and
48    /// [`Op::Pasted`](crate::op::Op::Pasted) answers a frame or more later.
49    ///
50    /// The web is the only host that announces one today, so a native build outside the suite
51    /// carries the variant and constructs none.
52    #[cfg_attr(not(target_family = "wasm"), allow(dead_code))]
53    Pasted(String),
54}

56/// One keystroke: which key, and what was held with it.
57///
58/// What [`Pollen::keys`](crate::Pollen::keys) hands an element, and what
59/// [`Pollen::root_keys`](crate::Pollen::root_keys) hands an app with nothing focused.
60///
61/// Assembled at dispatch from the key and whatever [`Modifiers`] the stream had reached by then,
62/// which is what makes it a single value the drain can carry.
63#[derive(Copy, Clone, Debug, PartialEq, Eq)]
64pub struct Keystroke {
65    /// Which key it was.
66    pub key: Key,
67    /// What was held down with it.
68    pub modifiers: Modifiers,
69}
70
71/// A key, as the platform's own layout resolved it.
72///
73/// [`Typed`](Key::Typed) is a character to insert and nothing more: which key produced it, whether
74/// it took a dead key or two, and what the layout is are all the platform's, answered before this.
75/// Everything else here is a key that means something rather than says something.
76///
77/// Non-exhaustive, because a key set grows: an app matches the keys it acts on and lets the rest
78/// fall through.
79#[derive(Copy, Clone, Debug, PartialEq, Eq)]
80#[non_exhaustive]
81pub enum Key {
82    /// A character the layout produced, to be taken as itself.
83    Typed(char),
84    /// The left arrow.
85    Left,
86    /// The right arrow.
87    Right,
88    /// The up arrow.
89    Up,
90    /// The down arrow.
91    Down,
92    /// Home.
93    Home,
94    /// End.
95    End,
96    /// Backspace, which deletes behind the caret.
97    Backspace,
98    /// Delete, which deletes ahead of it.
99    Delete,
100    /// Enter, which submits a [`TextInput`](crate::TextInput) and puts a newline in a
101    /// [`TextArea`](crate::TextArea).
102    Enter,
103    /// Tab.
104    Tab,
105    /// Escape, which drops focus.
106    Escape,
107}
108
109/// What was held down with a key.
110///
111/// One flag per modifier that changes what a key means rather than which character it produced --
112/// a layout has already answered the second before anything arrives here. `shift` extends a
113/// selection where a bare arrow moves a caret; `control` is what makes a key a command rather than
114/// a character.
115#[derive(Copy, Clone, Debug, Default, PartialEq, Eq)]
116pub struct Modifiers {
117    /// Whether shift was down.
118    pub shift: bool,
119    /// Whether control was down.
120    pub control: bool,
121}
122
123/// One pointer event as the platform reported it, before any gesture was made of it.
124///
125/// What [`Grove::pointer`](crate::Grove::pointer) hands an app: every one that arrived since the
126/// last frame, in order and unmerged. A [`Drag`](crate::Drag) is what a gesture *did*, one per
127/// frame and with the travel spent claiming it left out; this is what the hand did, which is what
128/// something tracing the hand itself -- a pen, a signature -- needs instead.
129///
130/// It is a second view of the same events, not a second road for them. Every one of these also
131/// went to dispatch as an ordinary press, move or release, and gestures resolve exactly as if
132/// nothing were reading here.
133#[derive(Copy, Clone, Debug, PartialEq)]
134pub struct Pointer {
135    /// What the pointer did.
136    pub phase: Phase,
137    /// Where, in logical pixels.
138    pub at: Position,
139    /// Which pointer, as the platform numbers them. A mouse is always 0; each touch keeps its own
140    /// id from down to up, so a palm resting beside a pen is told apart from it here even though
141    /// dispatch takes both as the one pointer.
142    pub id: u64,
143    /// What kind of device it came from, as far as the platform says.
144    pub device: Device,
145    /// When the window received it, from the first pointer event the engine saw.
146    ///
147    /// Arrival rather than the moment the device sampled it, which the platform layer does not
148    /// pass on. Events arriving in a burst carry times close together.
149    pub time: Duration,
150}
151
152/// What a [`Pointer`] did.
153#[derive(Copy, Clone, Debug, PartialEq, Eq)]
154pub enum Phase {
155    /// It went down.
156    Down,
157    /// It moved while down.
158    Moved,
159    /// It came up.
160    Up,
161    /// It was taken away rather than lifted -- focus lost, a touch cancelled by the system.
162    Cancelled,
163}
164
165/// What a [`Pointer`] came from.
166#[derive(Copy, Clone, Debug, PartialEq, Eq)]
167pub enum Device {
168    /// A mouse, or anything the platform presents as one.
169    Mouse,
170    /// A touch screen: a finger, or a pen the platform presents as a touch.
171    Touch,
172}

174/// Everything input has arrived with and everything it has built up: what is waiting for the next
175/// frame, the gesture in progress, and what the keyboard is holding.
176///
177/// One of these for the whole engine. There is one pointer -- multi-touch resolves to a primary one
178/// at the platform edge, and a second finger is not a second gesture here -- and one keyboard, so
179/// what a key was pressed with and where a hand is are the same object's business.
180#[derive(Default)]
181pub(crate) struct Incoming {
182    /// Events waiting for the next frame's dispatch, in arrival order.
183    pub(crate) pending: Vec<Input>,
184    /// The gesture that is open, from the press that opened it until it ends.
185    pub(crate) gesture: Option<super::Gesture>,
186    /// The last tap, for as long as the next one could still follow it.
187    pub(crate) tapped: Option<super::Tapped>,
188    /// What is held down as of the last [`Input::Modifiers`] dispatch reached.
189    ///
190    /// Engine state rather than the platform's, so the headless suite reaches it by writing the
191    /// same event a window does and there is no second path for a test to miss.
192    pub(crate) modifiers: Modifiers,
193    /// Pointer events as they arrived, waiting for the next frame to hand them to the app.
194    pub(crate) raw: Vec<Pointer>,
195    /// The first pointer event's arrival, which every [`Pointer::time`] is measured from.
196    origin: Option<Instant>,
197}
198
199impl Incoming {
200    /// Takes an event from the platform. Nothing is resolved here: dispatch is a frame phase, and
201    /// input that arrived between frames is handled in the frame it arrived in.
202    pub(crate) fn take(&mut self, input: Input) {
203        self.pending.push(input);
204    }
205
206    /// Keeps a pointer event as it arrived, stamped now.
207    ///
208    /// Called at the platform edge beside [`take`](Incoming::take), never instead of it.
209    pub(crate) fn contact(&mut self, phase: Phase, at: Position, id: u64, device: Device) {
210        let now = Instant::now();
211        let origin = *self.origin.get_or_insert(now);
212        self.raw.push(Pointer {
213            phase,
214            at,
215            id,
216            device,
217            time: now.saturating_duration_since(origin),
218        });
219    }
220
221    /// Whether the open gesture could still become a hold, which is a frame the loop owes (F9).
222    ///
223    /// A press that is not moving is the one gesture that changes with nothing arriving to change
224    /// it, so it is the one that has to be waited on rather than woken for.
225    pub(crate) fn awaiting_hold(&self) -> bool {
226        self.gesture
227            .as_ref()
228            .is_some_and(super::Gesture::awaiting_hold)
229    }
230}
