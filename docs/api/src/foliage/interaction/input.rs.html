1//! What arrives from the platform, before anything has been decided about it.
2//!
3//! One shape for every device. A mouse, a finger and a stylus produce the same four pointer
4//! statements, and the wheel is separate because it is a discrete pulse rather than a gesture with a
5//! lifecycle: it moves what is under it and is over.
6//!
7//! Nothing here reads the tree. Translation from the platform's own events happens at the edge, and
8//! the headless suite writes these directly -- which is what makes the two paths the same path, and
9//! what keeps the winit translation layer the only part of input the suite cannot reach.
10
11use crate::coordinate::Position;
12
13/// One input event, as the platform reported it.
14#[derive(Copy, Clone, Debug, PartialEq)]
15pub(crate) enum Input {
16    /// The pointer went down. Opens a gesture.
17    Pressed(Position),
18    /// The pointer moved while down.
19    Moved(Position),
20    /// The pointer came up.
21    Released(Position),
22    /// The gesture was taken away rather than completed -- the window lost focus, a touch was
23    /// cancelled by the system. Never produces a tap.
24    Cancelled,
25    /// A wheel notch at a point, as the movement a drag of the same distance would have been.
26    Wheeled { at: Position, delta: Position },
27    /// A key went down. Ordered against everything else here, because keystrokes are the one thing
28    /// in the engine whose order is part of what they mean.
29    ///
30    /// It names the key alone. What was held with it is [`Modifiers`], which arrives as its own
31    /// event in this same stream -- so what a key was pressed with is read from the order the two
32    /// arrived in rather than carried alongside.
33    Keyed(Key),
34    /// What is now held down. Ordered here with everything else for the reason above: a modifier is
35    /// only ever a statement about the keys pressed after it.
36    Modifiers(Modifiers),
37}
38
39/// One keystroke: which key, and what was held with it.
40///
41/// What [`Pollen::keys`](crate::Pollen::keys) hands an element, and what
42/// [`Pollen::root_keys`](crate::Pollen::root_keys) hands an app with nothing focused.
43///
44/// Assembled at dispatch from the key and whatever [`Modifiers`] the stream had reached by then,
45/// which is what makes it a single value the drain can carry.
46#[derive(Copy, Clone, Debug, PartialEq, Eq)]
47pub struct Keystroke {
48    /// Which key it was.
49    pub key: Key,
50    /// What was held down with it.
51    pub modifiers: Modifiers,
52}
53
54/// A key, as the platform's own layout resolved it.
55///
56/// [`Typed`](Key::Typed) is a character to insert and nothing more: which key produced it, whether
57/// it took a dead key or two, and what the layout is are all the platform's, answered before this.
58/// Everything else here is a key that means something rather than says something.
59///
60/// Non-exhaustive, because a key set grows: an app matches the keys it acts on and lets the rest
61/// fall through.
62#[derive(Copy, Clone, Debug, PartialEq, Eq)]
63#[non_exhaustive]
64pub enum Key {
65    /// A character the layout produced, to be taken as itself.
66    Typed(char),
67    /// The left arrow.
68    Left,
69    /// The right arrow.
70    Right,
71    /// The up arrow.
72    Up,
73    /// The down arrow.
74    Down,
75    /// Home.
76    Home,
77    /// End.
78    End,
79    /// Backspace, which deletes behind the caret.
80    Backspace,
81    /// Delete, which deletes ahead of it.
82    Delete,
83    /// Enter, which submits a [`TextInput`](crate::TextInput).
84    Enter,
85    /// Tab.
86    Tab,
87    /// Escape, which drops focus.
88    Escape,
89}
90
91/// What was held down with a key.
92///
93/// One flag per modifier that changes what a key means rather than which character it produced --
94/// a layout has already answered the second before anything arrives here. `shift` extends a
95/// selection where a bare arrow moves a caret; `control` is what makes a key a command rather than
96/// a character.
97#[derive(Copy, Clone, Debug, Default, PartialEq, Eq)]
98pub struct Modifiers {
99    /// Whether shift was down.
100    pub shift: bool,
101    /// Whether control was down.
102    pub control: bool,
103}
104
105/// Everything input has arrived with and everything it has built up: what is waiting for the next
106/// frame, the gesture in progress, and what the keyboard is holding.
107///
108/// One of these for the whole engine. There is one pointer -- multi-touch resolves to a primary one
109/// at the platform edge, and a second finger is not a second gesture here -- and one keyboard, so
110/// what a key was pressed with and where a hand is are the same object's business.
111#[derive(Default)]
112pub(crate) struct Incoming {
113    /// Events waiting for the next frame's dispatch, in arrival order.
114    pub(crate) pending: Vec<Input>,
115    /// The gesture that is open, from the press that opened it until it ends.
116    pub(crate) gesture: Option<super::Gesture>,
117    /// What is held down as of the last [`Input::Modifiers`] dispatch reached.
118    ///
119    /// Engine state rather than the platform's, so the headless suite reaches it by writing the
120    /// same event a window does and there is no second path for a test to miss.
121    pub(crate) modifiers: Modifiers,
122}
123
124impl Incoming {
125    /// Takes an event from the platform. Nothing is resolved here: dispatch is a frame phase, and
126    /// input that arrived between frames is handled in the frame it arrived in.
127    pub(crate) fn take(&mut self, input: Input) {
128        self.pending.push(input);
129    }
130
131    /// Whether the open gesture could still become a hold, which is a frame the loop owes (F9).
132    ///
133    /// A press that is not moving is the one gesture that changes with nothing arriving to change
134    /// it, so it is the one that has to be waited on rather than woken for.
135    pub(crate) fn awaiting_hold(&self) -> bool {
136        self.gesture
137            .as_ref()
138            .is_some_and(super::Gesture::awaiting_hold)
139    }
140}
