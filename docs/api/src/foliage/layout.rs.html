1//! Breakpoints: the responsive state a placement is read against.
2//!
3//! Both are properties of the surface rather than of any element, so there is one answer for the
4//! whole tree at any moment. They are recomputed at intake, from the viewport, and every element
5//! resolves against the same pair for the rest of the frame.
6
7use crate::coordinate::Area;
8
9/// The width breakpoint in force.
10///
11/// A [`Location`](crate::Location) and a [`Grid`](crate::Grid) may state a configuration per
12/// breakpoint, and each falls back to the nearest smaller one that was given, so only `xs` is ever
13/// required.
14#[derive(Copy, Clone, PartialEq, Eq, PartialOrd, Ord, Hash, Debug)]
15pub enum Layout {
16    /// Narrower than [`SM`](Layout::SM). The only breakpoint every placement must state.
17    Xs,
18    /// At least [`SM`](Layout::SM) wide.
19    Sm,
20    /// At least [`MD`](Layout::MD) wide.
21    Md,
22    /// At least [`LG`](Layout::LG) wide.
23    Lg,
24    /// At least [`XL`](Layout::XL) wide.
25    Xl,
26}
27
28impl Layout {
29    /// Lower bound of [`Sm`](Layout::Sm), in logical pixels. Below this is [`Xs`](Layout::Xs).
30    pub const SM: f32 = 420.0;
31    /// Lower bound of [`Md`](Layout::Md).
32    pub const MD: f32 = 600.0;
33    /// Lower bound of [`Lg`](Layout::Lg).
34    pub const LG: f32 = 840.0;
35    /// Lower bound of [`Xl`](Layout::Xl).
36    pub const XL: f32 = 1200.0;
37
38    /// The breakpoint a viewport of `viewport` falls in.
39    pub fn of(viewport: Area) -> Self {
40        if viewport.width >= Self::XL {
41            Self::Xl
42        } else if viewport.width >= Self::LG {
43            Self::Lg
44        } else if viewport.width >= Self::MD {
45            Self::Md
46        } else if viewport.width >= Self::SM {
47            Self::Sm
48        } else {
49            Self::Xs
50        }
51    }
52}
53
54/// Whether the viewport is vertically cramped.
55///
56/// Deliberately orthogonal to [`Layout`] rather than a sixth breakpoint: width and height are
57/// independent, and `Layout`'s fallback chain is a total order that only means anything along one
58/// axis. A phone held landscape is genuinely `Md`-wide *and* cramped-tall, and one enum has to
59/// discard half of that.
60///
61/// A configuration keyed to [`Short::Yes`] wins over the width-derived one. Nothing without such a
62/// configuration is affected, so this changes no layout until a placement opts in.
63#[derive(Copy, Clone, PartialEq, Eq, Debug)]
64pub enum Short {
65    /// The viewport is cramped, and a configuration keyed to it wins over the width-derived one.
66    Yes,
67    /// It is not, and only [`Layout`] decides.
68    No,
69}
70
71impl Short {
72    /// Below this height, in logical pixels, the viewport becomes [`Short::Yes`].
73    pub const ENTER: f32 = 400.0;
74    /// At or above this height it returns to [`Short::No`].
75    ///
76    /// The gap between the two is a deadband, and it is asymmetric on purpose. A single threshold
77    /// thrashes: on mobile web the address bar hides and shows as the page scrolls, so a viewport
78    /// resting near the boundary would cross it on every scroll and re-resolve the whole tree. It
79    /// leans toward `Yes` because a false `Yes` is a layout more conservative than it needed to be,
80    /// while a false `No` is content running off the bottom of the screen.
81    pub const EXIT: f32 = 440.0;
82
83    /// The next state, given the current one. Between [`ENTER`](Short::ENTER) and
84    /// [`EXIT`](Short::EXIT) the answer is whatever it already was.
85    pub fn next(self, viewport: Area) -> Self {
86        match self {
87            Self::No if viewport.height < Self::ENTER => Self::Yes,
88            Self::Yes if viewport.height >= Self::EXIT => Self::No,
89            current => current,
90        }
91    }
92}
