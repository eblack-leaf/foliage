8/// How many roles a [`Scheme`] answers.
9const ROLES: usize = 6;
10
11/// How many steps a role's ramp holds.
12const STEPS: usize = 5;
13
14/// How far one step moves a role's seed, in OKLab lightness.
15///
16/// Large enough that a step reads as a different color against the one beside it, small enough that
17/// two steps do not read as a different role.
18const NOTCH: f32 = 0.06;
19
20/// What a color is for, rather than what it is.
21///
22/// An element declares a tone and the [`Scheme`] decides what it resolves to, so a treatment is
23/// stated once and every element carrying that tone follows when it changes. A literal has no way to
24/// be changed together with the others, which is why one cannot be written here.
25///
26/// A tone is a role and a step on that role's ramp. The role is what the color is for and is what a
27/// scheme is written in terms of; the step is how far the tone sits from the ground the scheme is
28/// read against, which is what a state -- a hover, a press, something disabled -- is expressed as
29/// without leaving the scheme. Each of the six roles names its own base step, so `Palette::Accent`
30/// is a tone in its own right and every element that declares one says nothing about steps.
31///
32/// Resolution happens in extraction, against the tone the element declared. Nothing on the element
33/// holds a resolved color, so there is no copy to fall out of date and repainting is one write.
34#[derive(Component, Copy, Clone, Debug, Default, PartialEq, Eq, Hash)]
35pub struct Palette {
36    role: Role,
37    step: Step,
38}
39
40// The six roles, each named at its base step. Cased as the roles they name rather than as the
41// constants they are, because a tone is what a caller writes and `Palette::Accent` is its name.
42#[allow(non_upper_case_globals)]
43impl Palette {
44    /// The ordinary fill, and what an element that says nothing takes.
45    pub const Surface: Self = Self::base(Role::Surface);
46    /// A surface in front of another: a card against the page it sits on.
47    pub const Raised: Self = Self::base(Role::Raised);
48    /// A quieter fill, for a division or a rule.
49    pub const Muted: Self = Self::base(Role::Muted);
50    /// The one emphatic color.
51    pub const Accent: Self = Self::base(Role::Accent);
52    /// What is read against a surface rather than drawn as one.
53    pub const Ink: Self = Self::base(Role::Ink);
54    /// What is read against [`Accent`](Palette::Accent) rather than against a surface.
55    ///
56    /// The accent is the one role whose color a scheme is expected to move furthest, and a label on
57    /// top of it cannot follow with [`Ink`](Palette::Ink) -- ink is legible against a surface by
58    /// construction and against an accent only by accident. Stating both is what lets an accent be
59    /// rehued without silently making everything written on it unreadable.
60    pub const Contrast: Self = Self::base(Role::Contrast);
61}
62
63impl Palette {
64    /// This role at its base step.
65    const fn base(role: Role) -> Self {
66        Self {
67            role,
68            step: Step::Base,
69        }
70    }
71
72    /// The same role at `step`.
73    pub const fn at(self, step: Step) -> Self {
74        Self {
75            role: self.role,
76            step,
77        }
78    }
79
80    /// The same role one step farther into the ground, and itself at [`Step::Farthest`].
81    ///
82    /// What something quieted -- disabled, placeholding, resting under something else -- is drawn
83    /// in.
84    pub const fn recede(self) -> Self {
85        self.at(match self.step {
86            Step::Farthest | Step::Far => Step::Farthest,
87            Step::Base => Step::Far,
88            Step::Near => Step::Base,
89            Step::Nearest => Step::Near,
90        })
91    }
92
93    /// The same role one step nearer out of the ground, and itself at [`Step::Nearest`].
94    ///
95    /// What something brought forward -- hovered, held, carrying focus -- is drawn in.
96    pub const fn advance(self) -> Self {
97        self.at(match self.step {
98            Step::Farthest => Step::Far,
99            Step::Far => Step::Base,
100            Step::Base => Step::Near,
101            Step::Near | Step::Nearest => Step::Nearest,
102        })
103    }
104
105    /// Where a [`Scheme`] holds this tone's color.
106    fn index(self) -> usize {
107        self.role.index() * STEPS + self.step.index()
108    }
109}

111/// What a color is for.
112///
113/// Not public: a role is reached as the tone at its base step -- [`Palette::Accent`] -- and moved
114/// along its ramp from there, so there is one name for a role rather than two.
115#[derive(Copy, Clone, Debug, Default, PartialEq, Eq, Hash)]
116enum Role {
117    #[default]
118    Surface,
119    Raised,
120    Muted,
121    Accent,
122    Ink,
123    Contrast,
124}
125
126impl Role {
127    /// Which ramp in a [`Scheme`] is this role's.
128    fn index(self) -> usize {
129        match self {
130            Role::Surface => 0,
131            Role::Raised => 1,
132            Role::Muted => 2,
133            Role::Accent => 3,
134            Role::Ink => 4,
135            Role::Contrast => 5,
136        }
137    }
138}

140/// How far a tone stands from the ground its scheme is read against.
141///
142/// Named for where a step stands rather than for which way it moves in lightness, because the two
143/// readings a scheme has move opposite ways: standing nearer means lighter against a dark ground and
144/// darker against a light one. A state written once as [`advance`](Palette::advance) is therefore
145/// correct in both, which is what makes a light and a dark scheme the same app rather than two.
146#[derive(Copy, Clone, Debug, Default, PartialEq, Eq, Hash)]
147pub enum Step {
148    /// Two steps back into the ground.
149    Farthest,
150    /// One step back into the ground.
151    Far,
152    /// What the role resolves to undeclared, and the color a [`Scheme`] states the role in.
153    #[default]
154    Base,
155    /// One step out of the ground.
156    Near,
157    /// Two steps out of the ground.
158    Nearest,
159}
160
161impl Step {
162    /// Every step, deepest into the ground first, which is the order a ramp is held in.
163    const ALL: [Step; STEPS] = [
164        Step::Farthest,
165        Step::Far,
166        Step::Base,
167        Step::Near,
168        Step::Nearest,
169    ];
170
171    /// Where in a role's ramp this step sits.
172    fn index(self) -> usize {
173        match self {
174            Step::Farthest => 0,
175            Step::Far => 1,
176            Step::Base => 2,
177            Step::Near => 3,
178            Step::Nearest => 4,
179        }
180    }
181
182    /// How many notches out from the ground this step stands, relative to the base.
183    fn notches(self) -> f32 {
184        self.index() as f32 - Step::Base.index() as f32
185    }
186}

188/// What an element is filled with: a tone, or a color stated outright.
189///
190/// A tone is the ordinary answer and the reason [`Palette`] exists -- a treatment stated once, and
191/// moved for every element carrying it by one [`repaint`](crate::Grow::repaint).
192///
193/// A literal is the opt-out, and it is deliberately visible as one. An element filled with a color
194/// is not part of any scheme, so a repaint does not move it; that is the whole of the difference,
195/// and holding both in one type is what keeps it a difference a reader can see rather than two
196/// parallel paths through the renderer. It is also what lets a fill be animated to either -- a
197/// motion writes the target to the element, and a target has to be something the element can hold.
198#[derive(Copy, Clone, Debug, PartialEq)]
199pub enum Fill {
200    /// A tone, resolved against whatever [`Scheme`] is in force when it is drawn.
201    Role(Palette),
202    /// A color stated outright, which no repaint moves.
203    Literal(Color),
204}
205
206impl Fill {
207    /// The color this resolves to under `scheme`. A literal resolves to itself.
208    pub(crate) fn color(self, scheme: &Scheme) -> Color {
209        match self {
210            Fill::Role(tone) => scheme.color(tone),
211            Fill::Literal(color) => color,
212        }
213    }
214}
215
216impl Default for Fill {
217    /// The ordinary fill's tone, which is what an element that says nothing takes.
218    fn default() -> Self {
219        Self::Role(Palette::default())
220    }
221}
222
223impl From<Palette> for Fill {
224    fn from(tone: Palette) -> Self {
225        Self::Role(tone)
226    }
227}
228
229impl From<Color> for Fill {
230    fn from(color: Color) -> Self {
231        Self::Literal(color)
232    }
233}

235/// Which way a scheme's ramps run.
236///
237/// Not public: a reading is chosen by which constructor built the scheme, and the tones it produced
238/// are the whole of what anything downstream reads.
239#[derive(Copy, Clone, Debug, PartialEq, Eq)]
240enum Reading {
241    Dark,
242    Light,
243}
244
245impl Reading {
246    /// Which way in OKLab lightness a step out from the ground goes.
247    fn advancing(self) -> f32 {
248        match self {
249            Reading::Dark => 1.0,
250            Reading::Light => -1.0,
251        }
252    }
253}

255/// What each [`Palette`] tone resolves to.
256///
257/// Six roles of five steps each, written at boot or at any frame after it with
258/// [`repaint`](crate::Grow::repaint). Changing it changes every element carrying an affected tone
259/// and nothing else: extraction resolves the tone each frame and compares the result, so the
260/// elements that moved are exactly the ones that were painted in a color that changed.
261///
262/// A scheme is stated in six colors, one per role, and derives the other four steps of each ramp
263/// from that seed -- so a theme is six decisions rather than thirty. Derivation holds the seed's hue
264/// and chroma and moves only its lightness, in OKLab, away from the ground the reading names; a step
265/// that leaves sRGB has its chroma backed off until it fits. A single step can be replaced outright
266/// where a derived one will not do.
267#[derive(Copy, Clone, Debug, PartialEq)]
268pub struct Scheme {
269    tones: [Color; ROLES * STEPS],
270    reading: Reading,
271}
272
273impl Scheme {
274    /// The scheme every tone resolves to until one is given: a dark neutral ground with a green
275    /// accent.
276    pub fn new() -> Self {
277        Self::default()
278    }
279
280    /// The same six roles read against a light ground.
281    ///
282    /// A separate scheme rather than a flag on this one, because which colors a role is seeded with
283    /// is a decision and not a transform: a green that carries an accent against near-black is not
284    /// the green that carries it against near-white. What the reading decides is only which way the
285    /// ramps run from those seeds, so a tone an app declared is correct in both without moving.
286    pub fn light() -> Self {
287        Self::seeded(
288            [
289                Color::rgb(0.96, 0.96, 0.97),
290                Color::rgb(1.0, 1.0, 1.0),
291                Color::rgb(0.66, 0.68, 0.72),
292                Color::rgb(0.20, 0.52, 0.34),
293                Color::rgb(0.10, 0.11, 0.13),
294                Color::rgb(0.97, 0.98, 0.99),
295            ],
296            Reading::Light,
297        )
298    }
299
300    /// What one tone resolves to.
301    ///
302    /// A role's base step is its seed: writing it states the role and re-derives the rest of its
303    /// ramp. Writing any other step replaces that step alone and leaves the ramp it sits in as it
304    /// was, which is the way out when a derived step is not the one wanted.
305    pub fn set(mut self, tone: Palette, color: Color) -> Self {
306        if tone.step == Step::Base {
307            let at = tone.role.index() * STEPS;
308            self.tones[at..at + STEPS].copy_from_slice(&ramp(color, self.reading));
309        } else {
310            self.tones[tone.index()] = color;
311        }
312        self
313    }
314
315    /// The color `tone` resolves to.
316    pub fn color(&self, tone: Palette) -> Color {
317        self.tones[tone.index()]
318    }
319
320    /// Six seeds and a reading, as the ramps they derive.
321    fn seeded(seeds: [Color; ROLES], reading: Reading) -> Self {
322        let mut tones = [Color::rgb(0.0, 0.0, 0.0); ROLES * STEPS];
323        for (role, seed) in seeds.into_iter().enumerate() {
324            let at = role * STEPS;
325            tones[at..at + STEPS].copy_from_slice(&ramp(seed, reading));
326        }
327        Self { tones, reading }
328    }
329
330    /// How many tones differ between this scheme and `other`.
331    ///
332    /// What a repaint reports. Every element painted in one of them is re-extracted and no other is,
333    /// so this is the size of the repaint rather than a description of it.
334    pub(crate) fn moved(&self, other: &Self) -> usize {
335        self.tones
336            .iter()
337            .zip(other.tones.iter())
338            .filter(|(held, given)| held != given)
339            .count()
340    }
341}
342
343impl Default for Scheme {
344    fn default() -> Self {
345        Self::seeded(
346            [
347                Color::rgb(0.09, 0.10, 0.12),
348                Color::rgb(0.15, 0.16, 0.19),
349                Color::rgb(0.28, 0.30, 0.34),
350                Color::rgb(0.38, 0.71, 0.51),
351                Color::rgb(0.93, 0.94, 0.96),
352                Color::rgb(0.07, 0.08, 0.09),
353            ],
354            Reading::Dark,
355        )
356    }
357}

359/// One role's five steps, derived from its seed.
360///
361/// The seed is written at [`Step::Base`] rather than derived at zero offset, so a color a scheme was
362/// given is exactly the color that role resolves to and a round trip through OKLab cannot move it.
363///
364/// The two halves of a ramp are sized independently. A role seeded near black or near white has two
365/// notches of room on one side and less than that on the other, and the short half is compressed to
366/// what is left rather than run off the end, so a ramp is ordered from its farthest step to its
367/// nearest whatever it was seeded with. What that costs is a smaller step on the short side,
368/// which is the most a seed at an extreme can be given: an ink already at white has no brighter
369/// reading to offer. A seed sitting exactly on black or white has no room at all on that side, and
370/// every step of that half answers the seed.
371fn ramp(seed: Color, reading: Reading) -> [Color; STEPS] {
372    let (lightness, a, b) = seed.oklab();
373    let advancing = reading.advancing();
374    let ground = if advancing > 0.0 {
375        lightness
376    } else {
377        1.0 - lightness
378    };
379    let (out, back) = (notch(1.0 - ground), notch(ground));
380    let mut steps = [seed; STEPS];
381    for step in Step::ALL {
382        let notches = step.notches();
383        let size = if notches > 0.0 { out } else { back };
384        let moved = (lightness + advancing * notches * size).clamp(0.0, 1.0);
385        // The base, and any step of a half with no room in it. Left as the seed rather than
386        // converted back from a lightness it never left, which a round trip would move by a bit.
387        if moved == lightness {
388            continue;
389        }
390        let (color, refitted) = Color::from_oklab(moved, a, b, seed.alpha);
391        if refitted {
392            trace!(?step, "ramp step backed off to fit sRGB");
393        }
394        steps[step.index()] = color;
395    }
396    steps
397}

399/// How far one step moves, given the lightness left between the seed and that end of the range.
400///
401/// A full notch where two of them fit, and half of what is there where they do not.
402fn notch(headroom: f32) -> f32 {
403    NOTCH.min(headroom / 2.0)
