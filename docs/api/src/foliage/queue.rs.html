5/// The one op queue.
6///
7/// Every op lands here whichever side of the boundary issued it, and its position is fixed by
8/// when it arrived and by nothing else.
9#[derive(Clone, Default)]
10pub(crate) struct Queue(Arc<Mutex<Vec<Op>>>);
11
12impl Queue {
13    pub(crate) fn push(&self, op: Op) {
14        self.lock().push(op);
15    }
16
17    /// Everything queued since the last drain, in arrival order.
18    pub(crate) fn take(&self) -> Vec<Op> {
19        core::mem::take(&mut *self.lock())
20    }
21
22    pub(crate) fn is_empty(&self) -> bool {
23        self.lock().is_empty()
24    }
25
26    fn lock(&self) -> MutexGuard<'_, Vec<Op>> {
27        self.0
28            .lock()
29            .unwrap_or_else(|poisoned| poisoned.into_inner())
30    }
31}

33/// What is called to rouse the platform's loop when something arrives from outside a frame.
34///
35/// The queue is shared and an op may be pushed into it from anywhere, but a loop that sleeps until
36/// the platform speaks will never run the frame that drains one: a retrieval finishing on a thread
37/// arrives with nothing to wake it. So an arrival pushes its op and then calls this.
38///
39/// **A rouse is a frame owed.** The loop wakes and asks what is owed (F9), and an op it can see is
40/// -- but not everything that arrives is an op. A key the hidden input captured sits in the
41/// keyboard until intake moves it, and the loop has no clause for it. So a rouse is remembered
42/// until a frame runs, and that frame is owed on the rouse's account alone; what it announced is
43/// then found where it was put, at intake or at the drain.
44///
45/// Installed by `photosynthesize` and absent everywhere else -- the headless suite runs frames by
46/// hand, and a frame it did not ask for is not a frame it could observe. The record of a rouse is
47/// kept either way, so the suite can see that one was asked for.
48///
49/// **Shared rather than copied.** A handle taken before the loop is running is one that would
50/// otherwise hold an empty wake for the rest of the run, and a [`Sprig`](crate::Sprig) handed out at
51/// boot is exactly that: the whole reason it can push an op is that the frame it needs will be run.
52#[derive(Clone, Default)]
53pub(crate) struct Wake(Arc<Roused>);
54
55#[derive(Default)]
56struct Roused {
57    rouse: OnceLock<Arc<Rouse>>,
58    /// Whether a rouse has been made that no frame has yet answered.
59    pending: AtomicBool,
60}
61
62/// A wake is called from wherever the arrival happened, which on every target with threads means
63/// from another one. On the web there are none, and requiring `Send` there would only be a bound
64/// no caller could satisfy.
65#[cfg(not(target_family = "wasm"))]
66type Rouse = dyn Fn() + Send + Sync + 'static;
67#[cfg(target_family = "wasm")]
68type Rouse = dyn Fn() + 'static;
69
70impl Wake {
71    /// Installs the one wake, which is the loop's. A second is dropped: there is one loop.
72    #[cfg(not(target_family = "wasm"))]
73    pub(crate) fn install(&self, rouse: impl Fn() + Send + Sync + 'static) {
74        let _ = self.0.rouse.set(Arc::new(rouse));
75    }
76
77    #[cfg(target_family = "wasm")]
78    pub(crate) fn install(&self, rouse: impl Fn() + 'static) {
79        let _ = self.0.rouse.set(Arc::new(rouse));
80    }
81
82    /// Asks for the frame that will take what was just put down. Wakes nothing where no loop is
83    /// running, but is remembered regardless.
84    ///
85    /// Recorded before the loop is woken, so a loop that asks what is owed on waking finds the
86    /// answer already there. Whatever the caller put down, it put down before calling this, so the
87    /// frame that answers finds that too.
88    pub(crate) fn rouse(&self) {
89        self.0.pending.store(true, Ordering::Release);
90        if let Some(rouse) = self.0.rouse.get() {
91            rouse();
92        }
93    }
94
95    /// Whether a rouse has been made that no frame has yet answered.
96    pub(crate) fn pending(&self) -> bool {
97        self.0.pending.load(Ordering::Acquire)
98    }
99
100    /// Answers every rouse made so far. Called at intake, before anything a rouse could have
101    /// announced is looked for -- so a rouse arriving after this is answered by the next frame,
102    /// and one arriving before it announced something this frame will find.
103    pub(crate) fn answer(&self) {
104        self.0.pending.store(false, Ordering::Release);
105    }
106}
