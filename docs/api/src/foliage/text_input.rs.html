1//! TextInput -- an editable run, and the first [`Frond`](crate::frond): a leaf that is divided.
2//!
3//! # What a field is made of
4//!
5//! Everything else foliage grows is one element. A field is four, because the parts move
6//! independently of each other and each is already something the engine draws:
7//!
8//! | | Part | Is |
9//! |---|---|---|
10//! | | `run` | the value, as an ordinary [`Text`](crate::Text) |
11//! | | `hint` | the placeholder, shown only while the value is empty |
12//! | | `caret` | a [`Panel`](crate::Panel) [`CARET`] wide, shown only while the field holds focus |
13//! | | `selection` | a [`Panel`](crate::Panel) behind the run, from one end of the selection to the other |
14//!
15//! The app names the field and nothing else. Every verb and every read is addressed to it, and the
16//! parts are grown, placed and hidden from here -- so a field is one [`Leaf`] to hold, and what it
17//! is made of is not a surface to keep in step with.
18//!
19//! # The caret is placed in the ordinary grammar
20//!
21//! A caret at character `n` is `anchor().left() + anchor().letters(n)` against the run, and a
22//! selection is the span between two of those. That is the whole of the geometry: no pass measures
23//! a caret, because [`letters`](crate::Source::letters) already resolves a character count against
24//! the font the run composes in, and it moves when the run moves because an anchor does.
25//!
26//! The run is drawn in front of both marks. A caret sits on the boundary between two character
27//! cells and is as wide as it needs to be seen, so a caret drawn over the run would take a bite out
28//! of the glyph it stands before -- at a small size, a quarter of it.
29//!
30//! # The field is a scrolling region
31//!
32//! One line, as wide as its own value, inside a box that clips it. That is a region, so it is
33//! declared as one -- which is also what makes the caret stay in view: every edit asks the field to
34//! [`show`](crate::ScrollTo::show) the caret, and R4 answers it against the extent the same frame
35//! measured.

65/// How wide the caret is drawn, in logical pixels.
66///
67/// Stated rather than derived from the character cell: a caret is a mark between two characters and
68/// not a character, so it is the same width in a heading as in a caption.
69const CARET: f32 = 2.0;

71/// An editable run of glyphs.
72///
73/// One line. What it holds is a value, a placeholder shown while that value is empty, and a caret
74/// with a selection behind it -- and the app holds one [`Leaf`] for the whole of it:
75///
76/// ```no_run
77/// # use foliage::{Boxed, FontSize, Location, Palette, Place, Source, TextInput, left, top};
78/// TextInput::new()
79///     .value("")
80///     .placeholder("Search")
81///     .color(Palette::Ink)
82///     .font_size(FontSize::new().xs(14))
83///     .at(Location::new().xs(
84///         left(0.px()).right(100.pct()),
85///         top(0.px()).height(32.px()),
86///     ));
87/// ```
88///
89/// It draws no border and no ground of its own. A field is a run and a caret inside whatever box an
90/// app puts it in, and a chrome the engine drew would be one more thing to talk an app out of.
91///
92/// A tap puts the caret where it landed and takes focus -- the second because focus goes to
93/// whatever a tap lands on, which a field is by virtue of receiving at all. An app that wants focus
94/// somewhere else writes [`focus`](crate::Grow::focus) from [`clicked`](crate::Pollen::clicked) and
95/// wins, because the tap settled focus a frame before the app is handed it.
96///
97/// A drag across a field **scrolls its value**, because a field declares no drags and so is left
98/// to the region it is. Selecting is a press that was [`held`](crate::Pollen::held) and then
99/// dragged: the two motions are identical until the hold separates them, which is the only thing
100/// that can.
101///
102/// What it types is reported as [`edited`](crate::Pollen::edited), and an `Enter` as
103/// [`submitted`](crate::Pollen::submitted).
104///
105/// It answers the clipboard itself: `Ctrl+C` and `Ctrl+X` put the selected span on it and `Ctrl+V`
106/// asks for what is there, which lands **in a later frame** because what a clipboard holds is the
107/// host's to say. A paste is reported as [`edited`](crate::Pollen::edited) like anything else the
108/// person at the keyboard did, so an app hears it where it hears the typing.
109///
110/// On a platform with a soft keyboard it raises the [`keypad`](TextInput::keypad) it named
111/// whenever it holds focus, and nothing declares that beyond being a field.
112#[derive(Clone, Debug)]

114    pub(crate) placement: Placement,
115    pub(crate) value: String,
116    pub(crate) placeholder: String,

118    pub(crate) hint: Fill,
119    pub(crate) caret: Fill,
120    pub(crate) selection: Fill,
121    pub(crate) keypad: Keypad,
122}

204impl Buds for TextInput {
205    fn bud(mut self, at: Caller) -> Bud {
206        // A field is measured in characters throughout -- its caret, its selection and its own
207        // value -- so it carries a typeface whether or not one was named, exactly as a run does.
208        self.placement.typeface.get_or_insert_default();
209        // Declared here rather than left to the app, because both are what make a field a field
210        // rather than a preference about one. It receives, so a gesture can reach it and focus can
211        // rest on it -- which is what makes a tap focus it, since that is where focus goes for
212        // anything that receives; and it is a region across, which is what clips a value longer
213        // than its box and what the caret is kept in view by.
214        //
215        // It declares no drags. A drag across a field is then the region's, and the region is the
216        // field, so the value moves under its box; selection comes out of a hold, which claims a
217        // drag whatever an element declared.
218        self.placement.manner.gestures.receives = true;
219        self.placement.manner.scrolls = Some(Scrolls(
220            Scroll::new(Axes::Horizontal).contain(Axes::Horizontal),
221        ));
222        Bud {
223            chlorophyll: Chlorophyll::None,
224            placement: self.placement,
225            sprout: Some(Box::new(Sprout {
226                value: self.value,
227                placeholder: self.placeholder,
228                fill: self.fill,
229                hint: self.hint,
230                caret: self.caret,
231                selection: self.selection,
232                keypad: self.keypad,
233            })),
234            at,
235            ..Bud::bare()
236        }
237    }
238}

265/// Where the caret is and what is selected, in characters of the value.
266///
267/// Two indices rather than a range, because which end moves is what a shifted arrow key needs to
268/// know: `anchor` is where the selection was begun and `caret` is where it has reached, so the two
269/// are equal exactly when nothing is selected.
270#[derive(Component, Copy, Clone, Debug, Default, PartialEq, Eq)]
271pub(crate) struct Editing {
272    pub(crate) caret: usize,
273    pub(crate) anchor: usize,
274}
275
276impl Editing {
277    /// A caret at `index`, with nothing selected.
278    pub(crate) fn at(index: usize) -> Self {
279        Self {
280            caret: index,
281            anchor: index,
282        }
283    }
284
285    /// The selected span, low end first.
286    pub(crate) fn span(self) -> Range<usize> {
287        match self.caret <= self.anchor {
288            true => self.caret..self.anchor,
289            false => self.anchor..self.caret,
290        }
291    }
292
293    /// Whether anything is selected.
294    fn collapsed(self) -> bool {
295        self.caret == self.anchor
296    }
297
298    /// The same, held inside a value `length` characters long.
299    fn clamped(self, length: usize) -> Self {
300        Self {
301            caret: self.caret.min(length),
302            anchor: self.anchor.min(length),
303        }
304    }
305}

307/// What one keystroke did.
308pub(crate) enum Applied {
309    /// The value changed, and this is what it and the caret became.
310    Wrote(String, Editing),
311    /// The caret moved and the value did not. A selection is a caret with its anchor left behind,
312    /// so selecting is this as well.
313    Moved(Editing),
314    /// The selected span, to go on the clipboard. The value is unchanged.
315    Copied(String),
316    /// The same span, with the value it came out of and where that leaves the caret.
317    Cut {
318        text: String,
319        written: String,
320        editing: Editing,
321    },
322    /// The clipboard was asked for, to be written in at the caret when it answers.
323    ///
324    /// The one keystroke a field cannot finish on its own: what is on the clipboard is the host's
325    /// to say and it does not say it now, so the write happens in the frame the answer lands.
326    Pasting,
327    /// `Enter`.
328    Submitted,
329    /// Nothing a field does.
330    Nothing,
331}

333/// One keystroke against a value and a caret. The whole of what editing means.
334///
335/// Pure, and the only place the rules live: what a key does to a value is arithmetic over character
336/// indices, and separating it from the tree is what lets every case be stated as one.
337///
338/// Indices are **characters**, not bytes -- the space [`Shaped`](crate::text::shape) lays a run out
339/// in and the space a [`tint`](crate::Grow::tint) is written in, so a caret, a highlight and a
340/// range all mean the same thing by the same number.
341pub(crate) fn applied(value: &str, editing: Editing, stroke: Keystroke) -> Applied {
342    let characters: Vec<char> = value.chars().collect();
343    let editing = editing.clamped(characters.len());
344    let extend = stroke.modifiers.shift;
345    let span = editing.span();
346    // A command rather than a character. Held, a key says what to do with the value instead of
347    // what to put in it, so it is answered before the key's own meaning is.
348    if stroke.modifiers.control {
349        // A selection is what the clipboard verbs work over, and an empty one is nothing to work
350        // over: a field with its caret between two characters has no span to copy and none to cut.
351        // Taking the whole value instead would be a rule nobody asked for.
352        let selected = || characters[span.start..span.end].iter().collect::<String>();
353        return match stroke.key {
354            Key::Typed('a') | Key::Typed('A') => Applied::Moved(Editing {
355                anchor: 0,
356                caret: characters.len(),
357            }),
358            Key::Typed('c') | Key::Typed('C') if !editing.collapsed() => {
359                Applied::Copied(selected())
360            }
361            Key::Typed('x') | Key::Typed('X') if !editing.collapsed() => Applied::Cut {
362                text: selected(),
363                written: spliced(&characters, span.clone(), ""),
364                editing: Editing::at(span.start),
365            },
366            Key::Typed('v') | Key::Typed('V') => Applied::Pasting,
367            _ => Applied::Nothing,
368        };
369    }
370    match stroke.key {
371        Key::Typed(character) => Applied::Wrote(
372            spliced(&characters, span.clone(), &character.to_string()),
373            Editing::at(span.start + 1),
374        ),
375        Key::Backspace => {
376            // A selection is what is removed where there is one; otherwise the character before the
377            // caret. The two are one rule -- delete the span -- with an empty span reaching back by
378            // one first.
379            let removed = match editing.collapsed() {
380                true => span.start.saturating_sub(1)..span.end,
381                false => span,
382            };
383            if removed.is_empty() {
384                return Applied::Nothing;
385            }
386            Applied::Wrote(
387                spliced(&characters, removed.clone(), ""),
388                Editing::at(removed.start),
389            )
390        }
391        Key::Delete => {
392            let removed = match editing.collapsed() {
393                true => span.start..(span.end + 1).min(characters.len()),
394                false => span,
395            };
396            if removed.is_empty() {
397                return Applied::Nothing;
398            }
399            Applied::Wrote(
400                spliced(&characters, removed.clone(), ""),
401                Editing::at(removed.start),
402            )
403        }
404        Key::Left => Applied::Moved(moved(editing, characters.len(), Toward::Left, extend)),
405        Key::Right => Applied::Moved(moved(editing, characters.len(), Toward::Right, extend)),
406        Key::Home => Applied::Moved(moved(editing, characters.len(), Toward::Start, extend)),
407        Key::End => Applied::Moved(moved(editing, characters.len(), Toward::End, extend)),
408        // Not an edit and not a caret: the app is told, and what that means is the app's.
409        Key::Enter => Applied::Submitted,
410        // Both answered before a field is ever asked, because both move focus wherever focus is:
411        // `Tab` steps it and `Escape` takes it away. A field never sees either.
412        Key::Tab | Key::Escape => Applied::Nothing,
413        // A field is one line, so there is no line above or below to move to. An app reading its
414        // own keys is what steps a list of them.
415        Key::Up | Key::Down => Applied::Nothing,
416    }
417}

419/// Text put in at the caret, replacing whatever was selected.
420///
421/// The rule a paste is applied by, and the general case of the one a typed character is: what
422/// arrives is a string rather than a character, and it is written in at exactly the same place.
423///
424/// Control characters are dropped, which is the rule a keystroke already arrives under -- a field
425/// is one line, so a newline that came in on the clipboard would put a character in the value that
426/// no cell can be measured for and no caret can stand between.
427pub(crate) fn inserted(value: &str, editing: Editing, text: &str) -> Applied {
428    let characters: Vec<char> = value.chars().collect();
429    let span = editing.clamped(characters.len()).span();
430    let written: String = text
431        .chars()
432        .filter(|character| !character.is_control())
433        .collect();
434    if written.is_empty() {
435        return Applied::Nothing;
436    }
437    Applied::Wrote(
438        spliced(&characters, span.clone(), &written),
439        Editing::at(span.start + written.chars().count()),
440    )
441}

443/// The value with a span of it replaced by `text`, which is empty where the span is being removed.
444fn spliced(characters: &[char], span: Range<usize>, text: &str) -> String {
445    characters[..span.start]
446        .iter()
447        .copied()
448        .chain(text.chars())
449        .chain(characters[span.end..].iter().copied())
450        .collect()
451}

453/// Which way a caret was asked to go.
454#[derive(Copy, Clone, PartialEq, Eq)]
455enum Toward {
456    Left,
457    Right,
458    Start,
459    End,
460}
461
462/// Where the caret lands, and what that leaves selected.
463///
464/// An unshifted arrow against a selection **collapses it to the edge it points at** rather than
465/// stepping from the caret. That is what every editor does and what a reader means by it: the
466/// selection was the thing being moved away from.
467fn moved(editing: Editing, length: usize, toward: Toward, extend: bool) -> Editing {
468    let span = editing.span();
469    let caret = match (toward, extend, editing.collapsed()) {
470        (Toward::Left, false, false) => span.start,
471        (Toward::Right, false, false) => span.end,
472        (Toward::Left, _, _) => editing.caret.saturating_sub(1),
473        (Toward::Right, _, _) => (editing.caret + 1).min(length),
474        (Toward::Start, _, _) => 0,
475        (Toward::End, _, _) => length,
476    };
477    match extend {
478        true => Editing {
479            caret,
480            anchor: editing.anchor,
481        },
482        false => Editing::at(caret),
483    }
484}

492/// Grows a field's four parts under it, in the drain that grew the field.
493fn sprout(grove: &mut Grove, field: Leaf, sprout: Sprout) {
494    let at = grove
495        .tree
496        .spawned_at(field)
497        .unwrap_or(core::panic::Location::caller());
498    let typeface = grove.tree.typeface(field);
499    // Every name first, so the parts can be placed against each other as they are grown.
500    let (run, run_growth) = grove.naming.leaf();
501    let (hint, hint_growth) = grove.naming.leaf();
502    let (caret, caret_growth) = grove.naming.leaf();
503    let (selection, selection_growth) = grove.naming.leaf();
504    let parts = Parts {
505        run,
506        hint,
507        caret,
508        selection,
509    };
510    // A run as wide as its own value, one line tall, centred in whatever box the field was given.
511    // Its width is what the field's extent is measured from, so a value longer than the box is what
512    // makes the field scrollable and nothing has to say so. It sits in front of both marks, because
513    // a mark drawn over it would cut into the glyph it stands against.
514    let line = |elevation: i32| Placement {
515        location: Some(Location::new().xs(
516            left(0.px()).width(content()),
517            center_y(50.pct()).height(1.letters()),
518        )),
519        elevation: Some(Elevation::up(elevation)),
520        typeface,
521        manner: Manner {
522            // The hit test reads the top of the stack and stops, so every part of a field has to be
523            // intangible or the field itself would never be the top of it.
524            gestures: Gestures {
525                intangible: true,
526                ..Gestures::default()
527            },
528            ..Manner::default()
529        },
530        ..Placement::default()
531    };
532    // The caret and the selection are spans of the run's own character cells, so both are anchored
533    // to it and both are rewritten by `refresh` whenever the caret moves.
534    let against = |elevation: i32, visible: bool| Placement {
535        location: Some(span(0, 0)),
536        anchor: Some(Anchored { to: run, at }),
537        elevation: Some(Elevation::up(elevation)),
538        manner: Manner {
539            gestures: Gestures {
540                intangible: true,
541                ..Gestures::default()
542            },
543            visible: Visible(visible),
544            ..Manner::default()
545        },
546        ..Placement::default()
547    };
548    grove.tree.grow(
549        selection,
550        selection_growth,
551        Some(field),
552        Bud {
553            chlorophyll: Chlorophyll::Panel,
554            pigment: Some(Pigment::Panel(PanelPigment {
555                fill: sprout.selection,
556                rounding: Default::default(),
557            })),
558            placement: against(0, false),
559            at,
560            ..Bud::bare()
561        },
562    );
563    let empty = sprout.value.is_empty();
564    grove.tree.grow(
565        run,
566        run_growth,
567        Some(field),
568        Bud {
569            chlorophyll: Chlorophyll::Text,
570            pigment: Some(Pigment::Text(TextPigment { fill: sprout.fill })),
571            lettering: Some(Lettering(sprout.value)),
572            placement: line(2),
573            at,
574            ..Bud::bare()
575        },
576    );
577    let mut hint_placement = line(2);
578    hint_placement.manner.visible = Visible(empty);
579    grove.tree.grow(
580        hint,
581        hint_growth,
582        Some(field),
583        Bud {
584            chlorophyll: Chlorophyll::Text,
585            pigment: Some(Pigment::Text(TextPigment { fill: sprout.hint })),
586            lettering: Some(Lettering(sprout.placeholder)),
587            placement: hint_placement,
588            at,
589            ..Bud::bare()
590        },
591    );
592    grove.tree.grow(
593        caret,
594        caret_growth,
595        Some(field),
596        Bud {
597            chlorophyll: Chlorophyll::Panel,
598            pigment: Some(Pigment::Panel(PanelPigment {
599                fill: sprout.caret,
600                rounding: Default::default(),
601            })),
602            // A caret always has something to show, so it declares itself shown and lets focus be
603            // the only thing that hides it. The selection says otherwise for itself, below.
604            placement: against(1, true),
605            at,
606            ..Bud::bare()
607        },
608    );
609    grove.tree.set_parts(field, parts);
610    grove.tree.set_editing(field, Editing::default());
611    grove.tree.set_keypad(field, sprout.keypad);
612    refresh(grove, field);
613    debug!(leaf = field.id(), "field sprouted");
614}

616/// Where the caret sits at character `index` of the run.
617fn caret_at(index: usize) -> Location {
618    Location::new().xs(
619        left(anchor().left() + anchor().letters(index as f32)).width(CARET.px()),
620        center_y(anchor().center_y()).height(anchor().letters(1.0)),
621    )
622}

624/// The box covering characters `from..to` of the run.
625fn span(from: usize, to: usize) -> Location {
626    Location::new().xs(
627        left(anchor().left() + anchor().letters(from as f32))
628            .right(anchor().left() + anchor().letters(to as f32)),
629        center_y(anchor().center_y()).height(anchor().letters(1.0)),
630    )
631}

633/// Puts the parts back in step with the value and the caret.
634///
635/// Runs at the drain, after whatever changed either of them, so the frame that took the keystroke is
636/// the frame the caret has moved in.
637pub(crate) fn refresh(grove: &mut Grove, field: Leaf) {
638    let Some(parts) = grove.tree.parts(field) else {
639        return;
640    };
641    let length = grove
642        .tree
643        .lettering(parts.run)
644        .map(|value| value.chars().count())
645        .unwrap_or_default();
646    let editing = grove.tree.editing(field).clamped(length);
647    grove.tree.set_editing(field, editing);
648    let selected = editing.span();
649    grove
650        .tree
651        .set_location(parts.caret, caret_at(editing.caret));
652    grove
653        .tree
654        .set_location(parts.selection, span(selected.start, selected.end));
655    grove
656        .tree
657        .set_visible(parts.selection, !selected.is_empty());
658    grove.tree.set_visible(parts.hint, length == 0);
659    // The caret is kept in view by the region the field already is, against the extent this frame's
660    // R3 measures rather than the one the last frame left -- which is what makes typing past the
661    // right edge of the box scroll it in the same frame.
662    grove.sought.push((field, ScrollTo::show(parts.caret)));
663}

665/// One keystroke, delivered to the element holding focus.
666///
667/// Anything but a field makes nothing of one, which is ordinary rather than a drop: the key was
668/// delivered and reported, and having no use for it is what most elements do.
669pub(crate) fn typed(grove: &mut Grove, field: Leaf, stroke: Keystroke) {
670    let Some(parts) = grove.tree.parts(field) else {
671        return;
672    };
673    match applied(&value(grove, parts), grove.tree.editing(field), stroke) {
674        Applied::Wrote(written, editing) => wrote(grove, field, parts, written, editing),
675        Applied::Moved(editing) => {
676            grove.tree.set_editing(field, editing);
677            refresh(grove, field);
678        }
679        Applied::Copied(text) => grove.clipboard.write(text),
680        Applied::Cut {
681            text,
682            written,
683            editing,
684        } => {
685            grove.clipboard.write(text);
686            wrote(grove, field, parts, written, editing);
687        }
688        // The one keystroke that finishes in another frame. The read is started here and the write
689        // happens where its answer is drained, which is `pasted`.
690        Applied::Pasting => {
691            let (queue, wake) = (grove.queue.clone(), grove.wake.clone());
692            grove.clipboard.read(&queue, &wake, Some(field));
693        }
694        Applied::Submitted => {
695            grove.drift.submitted.insert(field);
696            debug!(leaf = field.id(), "submitted");
697        }
698        Applied::Nothing => {}
699    }
700}
701
702/// What the clipboard answered, written in at the caret of the field that asked for it.
703///
704/// A frame or more after the `Ctrl+V` that asked, which is the whole reason the answer is an op:
705/// what a clipboard holds is the host's to say and it does not say it now. Everything from here is
706/// the ordinary write the keystroke would have made, so the paste is
707/// [`edited`](crate::Pollen::edited) like anything else the person at the keyboard did.
708pub(crate) fn pasted(grove: &mut Grove, field: Leaf, text: &str) {
709    let Some(parts) = grove.tree.parts(field) else {
710        return;
711    };
712    if let Applied::Wrote(written, editing) =
713        inserted(&value(grove, parts), grove.tree.editing(field), text)
714    {
715        wrote(grove, field, parts, written, editing);
716    }
717}

719/// What the field currently says.
720fn value(grove: &Grove, parts: Parts) -> String {
721    grove
722        .tree
723        .lettering(parts.run)
724        .unwrap_or_default()
725        .to_string()
726}

728/// Puts a new value and caret on a field, however it was arrived at.
729///
730/// Typing, cutting and pasting are one write with three causes, and each is what the person at the
731/// keyboard did to the value -- so each is one [`edited`](crate::Pollen::edited).
732fn wrote(grove: &mut Grove, field: Leaf, parts: Parts, written: String, editing: Editing) {
733    grove.tree.set_lettering(parts.run, written);
734    grove.tree.set_editing(field, editing);
735    grove.drift.edited.insert(field);
736    refresh(grove, field);
737    debug!(leaf = field.id(), "edited");
738}

740/// The kind, for the two questions [`Fronds`] asks of every field at once.
741pub(crate) struct Field;
742
743impl Fronds for Field {
744    fn gestured(&self, grove: &mut Grove) {
745        gestured(grove)
746    }
747
748    fn settled(&self, grove: &mut Grove) {
749        settled(grove)
750    }
751}

753/// What a field makes of the gestures dispatch reported this frame.
754///
755/// **A field reads interaction; interaction knows nothing about fields.** A tap is a statement
756/// about where the caret goes, a hold is one about where a selection begins, and a drag is one
757/// about how far it reaches -- but each is a field's reading of an ordinary gesture, not something
758/// a gesture carries. So this asks what was reported about the leaves it owns and queues the same
759/// [`select`](crate::Grow::select) an app would write.
760fn gestured(grove: &mut Grove) {
761    for (field, parts) in grove.tree.fields() {
762        // A tap says where the caret goes and collapses whatever was selected. A gesture that
763        // became a drag was never a tap, so a drag out of a field to scroll the page behind it
764        // leaves the field exactly as it found it.
765        if let Some(at) = grove.drift.clicked.get(&field).copied() {
766            let index = index_at(grove, parts, at);
767            grove.queue.push(Op::Select {
768                leaf: field,
769                range: index..index,
770            });
771        }
772        // A hold says the same thing a tap does about where the caret goes, and takes focus with
773        // it: a held press is not a tap, so nothing has moved either. It is what begins a
774        // selection, since the drag out of a hold is the field's whatever it declared.
775        if let Some(at) = grove.drift.held.get(&field).copied() {
776            let index = index_at(grove, parts, at);
777            grove.queue.push(Op::Focus(Intent::To(field)));
778            grove.queue.push(Op::Select {
779                leaf: field,
780                range: index..index,
781            });
782        }
783        // A drag selects from where it began to where it has reached. Stated as one span rather
784        // than as two moves, because that is what it is -- and low end last where the drag went
785        // leftwards, which is how the anchor stays the end that was pressed.
786        //
787        // The anchor is read from the field and not from where the drag began. It is a character,
788        // and the hold that started the selection already put it there; a *point* is not, because
789        // the value scrolls under it the moment the drag reaches the edge -- so re-reading one
790        // would move the end the selection was measured from as the field followed the caret.
791        //
792        // Where it has reached is the **open gesture** where there is one, and the move where the
793        // gesture has already closed -- which is the release, and is the last of the movement. A
794        // pointer held still reports nothing, and a drag that stopped moving is still a drag.
795        let reached = grove
796            .drift
797            .dragged
798            .get(&field)
799            .map(|drag| drag.current)
800            .or_else(|| interaction::dragging(grove, field));
801        if let Some(to) = reached {
802            let anchor = grove.tree.editing(field).anchor;
803            let caret = index_at(grove, parts, to);
804            grove.queue.push(Op::Select {
805                leaf: field,
806                range: anchor..caret,
807            });
808            // A pointer past the edge is a reader still asking for more of the value, and the
809            // caret it is being taken to is further out every frame. So the field owes itself the
810            // frames that get there, the way anything driving its own motion does.
811            if beyond(grove, field, to) {
812                grove.again();
813            }
814        }
815    }
816}

818/// Puts every field's parts back in step with focus.
819///
820/// At the end of the drain, where focus is already final: a tap settles it at dispatch and an
821/// [`Op::Focus`] settles it here, so by now there is nothing left that could move it. That is what
822/// keeps a caret an ordinary [`visible`](crate::Grow::visible) write inherited by R7 like anything
823/// else, instead of a product patched up after the pass that composes it.
824///
825/// What is selected is not cleared with focus. A selection is state and focus is not, so a field
826/// stepped away from and back into is as it was left -- while a *tap* back into it collapses the
827/// selection, because a tap says where the caret goes.
828///
829/// The soft keyboard is settled here too, and it is the same statement the caret is: a platform's
830/// own keyboard is raised for the field that holds focus and lowered for everything else, so
831/// nothing about a phone is declared anywhere -- being a field is the whole of it.
832fn settled(grove: &mut Grove) {
833    let focused = grove.focus.held();
834    for (field, parts) in grove.tree.fields() {
835        let showing = Some(field) == focused;
836        grove.tree.set_visible(parts.caret, showing);
837        let selected = grove.tree.editing(field).span();
838        grove
839            .tree
840            .set_visible(parts.selection, showing && !selected.is_empty());
841    }
842    let wanted = focused.and_then(|leaf| grove.tree.keypad(leaf));
843    grove.keyboard.raise(wanted);
844}

846/// Selects a span of the value outright.
847///
848/// `range` is read as anchor-then-caret rather than low-then-high, so a span whose end precedes its
849/// start is a selection reaching backwards -- which is what a drag leftwards is, and what a shifted
850/// arrow continues from.
851pub(crate) fn select(grove: &mut Grove, field: Leaf, range: Range<usize>) {
852    if grove.tree.parts(field).is_none() {
853        debug!(leaf = field.id(), "select dropped: not a field");
854        return;
855    }
856    grove.tree.set_editing(
857        field,
858        Editing {
859            anchor: range.start,
860            caret: range.end,
861        },
862    );
863    refresh(grove, field);
864}

866/// Rewrites the whole value, leaving the caret at the end of it.
867///
868/// Takes the parts the drain already looked up to tell a field from a run, so the one question that
869/// answers it is asked once.
870///
871/// Not reported as [`edited`](crate::Pollen::edited): that is what the person at the keyboard did,
872/// and an app that wrote the value already knows what it wrote.
873pub(crate) fn lettered(grove: &mut Grove, field: Leaf, parts: Parts, value: String) {
874    let length = value.chars().count();
875    grove.tree.set_lettering(parts.run, value);
876    grove.tree.set_editing(field, Editing::at(length));
877    refresh(grove, field);
878}

880/// Whether a point has left the field across, which is the axis it scrolls.
881///
882/// Across and not the whole box: a pointer below a one-line field is at a character like any other,
883/// and the field has nowhere to go for it.
884fn beyond(grove: &Grove, field: Leaf, at: Position) -> bool {
885    let box_of = grove.tree.drawn(field);
886    at.x < box_of.left() || at.x > box_of.right()
887}

889/// Which character of the run a point falls on.
890///
891/// Rounded rather than floored, so pressing past the middle of a character puts the caret after it
892/// -- which is where a hand aiming between two characters means.
893fn index_at(grove: &Grove, parts: Parts, at: Position) -> usize {
894    let cell = grove.tree.cell(parts.run);
895    if cell.width <= 0.0 {
896        return 0;
897    }
898    let length = grove
899        .tree
900        .lettering(parts.run)
901        .map(|value| value.chars().count())
902        .unwrap_or_default();
903    let across = at.x - grove.tree.drawn(parts.run).left();
904    ((across / cell.width).round().max(0.0) as usize).min(length)
905}
