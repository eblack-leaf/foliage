22pub(crate) enum Op {
23    Plant {
24        leaf: Leaf,
25        growth: Growth,
26        bud: Bud,
27    },
28    Branch {
29        leaf: Leaf,
30        growth: Growth,
31        under: Leaf,
32        bud: Bud,
33    },
34    Prune(Leaf),
35    Place {
36        leaf: Leaf,
37        location: Location,
38    },
39    /// Moves the two ends of a stroke. The point-mode counterpart to [`Place`](Op::Place), and
40    /// separate for the same reason the declarations are: an element states a box or two ends, and
41    /// an op that could write either would be able to write the one the element does not have.
42    Trace {
43        leaf: Leaf,
44        from: Point,
45        to: Point,
46    },
47    Divide {
48        leaf: Leaf,
49        grid: Grid,
50    },
51    Anchor {
52        leaf: Leaf,
53        to: Leaf,
54        at: Caller,
55    },
56    Elevate {
57        leaf: Leaf,
58        elevation: Elevation,
59    },
60    Recolor {
61        leaf: Leaf,
62        fill: Fill,
63    },
64    /// Rewrites what a run says. The measure follows in the same frame, because measuring is a pass
65    /// and not a reaction to the write.
66    Letter {
67        leaf: Leaf,
68        value: String,
69    },
70    /// One keystroke, against whatever holds focus. Which element that is is dispatch's answer;
71    /// what the key does is the drain's.
72    ///
73    /// `target` is `None` where focus rests nowhere. The key is still carried, still ordered against
74    /// everything else in the queue, and still reported -- to the app itself rather than to an
75    /// element, which is what lets a chord reach a page nobody has pressed.
76    Keyed {
77        target: Option<Leaf>,
78        stroke: Keystroke,
79    },
80    /// Selects a span of a field's value outright.
81    ///
82    /// Anchor-then-caret rather than low-then-high, so a span whose end precedes its start is a
83    /// selection reaching backwards -- which is what a drag leftwards is. The one op a gesture in a
84    /// field produces, because where a gesture landed is a position and what a field makes of one is
85    /// a span: the field converts, and nothing needs an op that speaks in pixels.
86    Select {
87        leaf: Leaf,
88        range: core::ops::Range<usize>,
89    },
90    /// Refills part of a run, over a range of its own index space.
91    Tint {
92        leaf: Leaf,
93        tints: Tints,
94    },
95    Round {
96        leaf: Leaf,
97        rounding: Corners,
98    },
99    /// Reshapes a regular polygon: how many sides, how round, how far turned.
100    Reshape {
101        leaf: Leaf,
102        shape: crate::polygon::Shape,
103    },
104    /// Fills a registered picture's name with pixels.
105    ///
106    /// An op like any other, rather than a call on the engine, because a picture may arrive at any
107    /// frame -- from a fetch, a decode on another thread, a re-render at a higher resolution -- and
108    /// arriving is a change to the tree's contents in exactly the way a rewritten run is.
109    Load {
110        plate: Plate,
111        pixels: Vec<u8>,
112        size: Area,
113    },
114    /// Moves a region. Recorded here and answered in R4, which is the one place that has both this
115    /// frame's extent and the offset it clamps.
116    Scroll {
117        leaf: Leaf,
118        to: ScrollTo,
119    },
120    Disable {
121        leaf: Leaf,
122        disabled: bool,
123    },
124    Reveal {
125        leaf: Leaf,
126        visible: bool,
127    },
128    Fade {
129        leaf: Leaf,
130        opacity: f32,
131    },
132    /// Moves a property over time. Applied here, which is what makes a write to the same property
133    /// earlier in the same drain cancel it, and one later replace it.
134    Animate {
135        leaf: Leaf,
136        motion: Motion,
137        timing: Timing,
138        /// What its progress is reported under and its ending is reported by. Named at the callsite
139        /// like a channel's, so the name is in hand before the frame that starts the motion.
140        tween: Tween,
141    },
142    /// A scalar channel, reported outward every frame and written nowhere.
143    Channel {
144        tween: Tween,
145        from: f32,
146        to: f32,
147        timing: Timing,
148    },
149    /// Ends a tween before it has run out, on the end it was going to.
150    ///
151    /// `reported` is the whole difference between the two verbs that push this: `finish` ends it as
152    /// an arrival and a chain waiting on it runs, `stop` ends it in silence and the chain does not.
153    Stop {
154        tween: Tween,
155        reported: bool,
156    },
157    /// Where focus is to go. Applied here and answered at settle, against the geometry and the
158    /// inherited state this frame resolves.
159    Focus(Intent),
160    /// The one op that names no element: what every role resolves to, for the whole tree.
161    Repaint(Scheme),
162    /// Bytes that were read from somewhere outside the frame, and the name they were read for.
163    ///
164    /// Pushed by whatever finished the retrieval rather than by an app, which is the whole reason it
165    /// is an op: it arrives at a moment nothing chose, and the queue is what gives it a place in the
166    /// order anyway.
167    Arrived {
168        destination: Destination,
169        bytes: Retrieved,
170    },
171    /// Text to put on the clipboard.
172    Copy(String),
173    /// A request for what the clipboard holds, and who it is for: the field that was asked to paste
174    /// into itself, or the app where it asked for itself.
175    ///
176    /// Separate from the answer because reading a clipboard is not instant on either target -- a
177    /// promise on the web, a round trip to whoever owns the selection off it.
178    Paste {
179        into: Option<Leaf>,
180    },
181    /// What the clipboard turned out to hold, and who asked for it.
182    ///
183    /// Pushed by whatever finished the read rather than by an app, which is what makes it an op for
184    /// the reason [`Arrived`](Op::Arrived) is one: it lands at a moment nothing chose, and the queue
185    /// is what gives it a place in the order anyway.
186    Pasted {
187        into: Option<Leaf>,
188        text: String,
189    },
190    /// A URL to go to.
191    Navigate(String),
192    /// A URL to hand the host to save.
193    Download(String),
194    /// One property of one element, to be read at the end of every frame from here on and published
195    /// for a [`Sprig`](crate::Sprig) to read.
196    ///
197    /// An op rather than a call, because a standing read has a place in the order like anything
198    /// else: watching an element in the same breath as growing it is one sequence, and it works for
199    /// the reason a write to a leaf planted a line earlier works.
200    Watch {
201        leaf: Leaf,
202        vein: Vein,
203    },
204    /// Ends a [`Watch`](Op::Watch).
205    Unwatch {
206        leaf: Leaf,
207        vein: Vein,
208    },
209}

211/// An element formed and not yet open: what the queue carries between the call that described it
212/// and the drain that grows it.
213///
214/// It carries the components the element will hold rather than a second enum describing them: a bud
215/// is the element before it exists, not an account of one. The pigment is present exactly when the
216/// chlorophyll is a renderer that has one, and the lettering exactly when there is a run to read.
217pub(crate) struct Bud {
218    pub(crate) chlorophyll: Chlorophyll,
219    pub(crate) pigment: Option<Pigment>,
220    pub(crate) lettering: Option<Lettering>,
221    /// Present only where part of a run is filled differently from the rest of it.
222    pub(crate) tints: Option<Tints>,
223    /// Present only on a [`Frond`](crate::frond) -- a leaf that is divided. The drain takes this and
224    /// grows the leaflets underneath what it just grew.
225    pub(crate) sprout: Option<Box<dyn Sprouts>>,
226    pub(crate) placement: Placement,
227    pub(crate) at: Caller,
228}
229
230impl Bud {
231    /// An element carrying nothing but a placement: no renderer and nothing to read.
232    ///
233    /// What every seed fills in from, so a seed states what it *is* rather than restating what it
234    /// is not.
235    pub(crate) fn bare() -> Self {
236        Self {
237            chlorophyll: Chlorophyll::None,
238            pigment: None,
239            lettering: None,
240            tints: None,
241            sprout: None,
242            placement: Placement::default(),
243            // Overwritten by every real callsite, which takes the caller's own.
244            at: core::panic::Location::caller(),
245        }
246    }
