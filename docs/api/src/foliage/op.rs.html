22pub(crate) enum Op {
23    Plant {
24        leaf: Leaf,
25        growth: Growth,
26        bud: Bud,
27    },
28    Branch {
29        leaf: Leaf,
30        growth: Growth,
31        under: Leaf,
32        bud: Bud,
33    },
34    Prune(Leaf),
35    Place {
36        leaf: Leaf,
37        location: Location,
38    },
39    /// Moves the two ends of a stroke. The point-mode counterpart to [`Place`](Op::Place), and
40    /// separate for the same reason the declarations are: an element states a box or two ends, and
41    /// an op that could write either would be able to write the one the element does not have.
42    Trace {
43        leaf: Leaf,
44        trace: Trace,
45    },
46    Divide {
47        leaf: Leaf,
48        grid: Grid,
49    },
50    Anchor {
51        leaf: Leaf,
52        to: Leaf,
53        at: Caller,
54    },
55    Elevate {
56        leaf: Leaf,
57        elevation: Elevation,
58    },
59    Recolor {
60        leaf: Leaf,
61        fill: Fill,
62    },
63    /// Rewrites what a run says. The measure follows in the same frame, because measuring is a pass
64    /// and not a reaction to the write.
65    Letter {
66        leaf: Leaf,
67        value: String,
68    },
69    /// One keystroke, against whatever holds focus. Which element that is is dispatch's answer;
70    /// what the key does is the drain's.
71    ///
72    /// `target` is `None` where focus rests nowhere. The key is still carried, still ordered against
73    /// everything else in the queue, and still reported -- to the app itself rather than to an
74    /// element, which is what lets a chord reach a page nobody has pressed.
75    Keyed {
76        target: Option<Leaf>,
77        stroke: Keystroke,
78    },
79    /// Selects a span of a field's value outright.
80    ///
81    /// Anchor-then-caret rather than low-then-high, so a span whose end precedes its start is a
82    /// selection reaching backwards -- which is what a drag leftwards is. The one op a gesture in a
83    /// field produces, because where a gesture landed is a position and what a field makes of one is
84    /// a span: the field converts, and nothing needs an op that speaks in pixels.
85    Select {
86        leaf: Leaf,
87        range: core::ops::Range<usize>,
88    },
89    /// Refills part of a run, over a range of its own index space.
90    Tint {
91        leaf: Leaf,
92        tints: Tints,
93    },
94    Round {
95        leaf: Leaf,
96        rounding: Corners,
97    },
98    /// Reshapes a regular polygon: how many sides, how round, how far turned.
99    Reshape {
100        leaf: Leaf,
101        shape: crate::polygon::Shape,
102    },
103    /// Fills a registered picture's name with pixels.
104    ///
105    /// An op like any other, rather than a call on the engine, because a picture may arrive at any
106    /// frame -- from a fetch, a decode on another thread, a re-render at a higher resolution -- and
107    /// arriving is a change to the tree's contents in exactly the way a rewritten run is.
108    Load {
109        plate: Plate,
110        pixels: Vec<u8>,
111        size: Area,
112    },
113    /// Moves a region. Recorded here and answered in R4, which is the one place that has both this
114    /// frame's extent and the offset it clamps.
115    Scroll {
116        leaf: Leaf,
117        to: ScrollTo,
118    },
119    Disable {
120        leaf: Leaf,
121        disabled: bool,
122    },
123    Reveal {
124        leaf: Leaf,
125        visible: bool,
126    },
127    Fade {
128        leaf: Leaf,
129        opacity: f32,
130    },
131    /// Moves a property over time. Applied here, which is what makes a write to the same property
132    /// earlier in the same drain cancel it, and one later replace it.
133    Animate {
134        leaf: Leaf,
135        motion: Motion,
136        timing: Timing,
137        /// What its progress is reported under and its ending is reported by. Named at the callsite
138        /// like a channel's, so the name is in hand before the frame that starts the motion.
139        tween: Tween,
140    },
141    /// A scalar channel, reported outward every frame and written nowhere.
142    Channel {
143        tween: Tween,
144        from: f32,
145        to: f32,
146        timing: Timing,
147    },
148    /// Ends a tween before it has run out, on the end it was going to.
149    ///
150    /// `reported` is the whole difference between the two verbs that push this: `finish` ends it as
151    /// an arrival and a chain waiting on it runs, `stop` ends it in silence and the chain does not.
152    Stop {
153        tween: Tween,
154        reported: bool,
155    },
156    /// Where focus is to go. Applied here and answered at settle, against the geometry and the
157    /// inherited state this frame resolves.
158    Focus(Intent),
159    /// The one op that names no element: what every role resolves to, for the whole tree.
160    Repaint(Scheme),
161    /// Bytes that were read from somewhere outside the frame, and the name they were read for.
162    ///
163    /// Pushed by whatever finished the retrieval rather than by an app, which is the whole reason it
164    /// is an op: it arrives at a moment nothing chose, and the queue is what gives it a place in the
165    /// order anyway.
166    Arrived {
167        destination: Destination,
168        bytes: Retrieved,
169    },
170    /// Text to put on the clipboard.
171    Copy(String),
172    /// A request for what the clipboard holds, and who it is for: the field that was asked to paste
173    /// into itself, or the app where it asked for itself.
174    ///
175    /// Separate from the answer because reading a clipboard is not instant on either target -- a
176    /// promise on the web, a round trip to whoever owns the selection off it.
177    Paste {
178        into: Option<Leaf>,
179    },
180    /// What the clipboard turned out to hold, and who asked for it.
181    ///
182    /// Pushed by whatever finished the read rather than by an app, which is what makes it an op for
183    /// the reason [`Arrived`](Op::Arrived) is one: it lands at a moment nothing chose, and the queue
184    /// is what gives it a place in the order anyway.
185    Pasted {
186        into: Option<Leaf>,
187        text: String,
188    },
189    /// A URL to go to.
190    Navigate(String),
191    /// A URL to hand the host to save.
192    Download(String),
193    /// One property of one element, to be read at the end of every frame from here on and published
194    /// for a [`Sprig`](crate::Sprig) to read.
195    ///
196    /// An op rather than a call, because a standing read has a place in the order like anything
197    /// else: watching an element in the same breath as growing it is one sequence, and it works for
198    /// the reason a write to a leaf planted a line earlier works.
199    Watch {
200        leaf: Leaf,
201        vein: Vein,
202    },
203    /// Ends a [`Watch`](Op::Watch).
204    Unwatch {
205        leaf: Leaf,
206        vein: Vein,
207    },
208}

210/// An element formed and not yet open: what the queue carries between the call that described it
211/// and the drain that grows it.
212///
213/// It carries the components the element will hold rather than a second enum describing them: a bud
214/// is the element before it exists, not an account of one. The pigment is present exactly when the
215/// chlorophyll is a renderer that has one, and the lettering exactly when there is a run to read.
216pub(crate) struct Bud {
217    pub(crate) chlorophyll: Chlorophyll,
218    pub(crate) pigment: Option<Pigment>,
219    pub(crate) lettering: Option<Lettering>,
220    /// Present only where part of a run is filled differently from the rest of it.
221    pub(crate) tints: Option<Tints>,
222    /// Present only on a [`Frond`](crate::frond) -- a leaf that is divided. The drain takes this and
223    /// grows the leaflets underneath what it just grew.
224    pub(crate) sprout: Option<Box<dyn Sprouts>>,
225    pub(crate) placement: Placement,
226    pub(crate) at: Caller,
227}
228
229impl Bud {
230    /// An element carrying nothing but a placement: no renderer and nothing to read.
231    ///
232    /// What every seed fills in from, so a seed states what it *is* rather than restating what it
233    /// is not.
234    pub(crate) fn bare() -> Self {
235        Self {
236            chlorophyll: Chlorophyll::None,
237            pigment: None,
238            lettering: None,
239            tints: None,
240            sprout: None,
241            placement: Placement::default(),
242            // Overwritten by every real callsite, which takes the caller's own.
243            at: core::panic::Location::caller(),
244        }
245    }
