20/// What an op sink has to be able to do: take an op, and hand out the names an op may need.
21///
22/// Naming an element hands back its place in allocation order along with the name, because that
23/// order is fixed where the name is asked for and not where the element is grown.
24pub(crate) trait Queues {
25    fn queue(&mut self, op: Op);
26    fn allocate(&self) -> (Leaf, Growth);
27    fn name(&self) -> Tween;
28    fn group(&self) -> Sequence;
29    /// A name for a picture whose pixels are on their way. Taken here rather than at the drain so
30    /// that an element can be grown against it in the frame it was asked for.
31    fn picture(&mut self) -> Plate;
32}

34/// Everything an app can ask the engine to do.
35///
36/// A change reads identically wherever it is issued. Sealed: it can be called, never implemented.
37#[allow(private_bounds)]
38pub trait Grow: Queues {
39    /// Grows a top-level element and hands back the [`Leaf`] naming it. Usable immediately,
40    /// including as a trunk in the same frame.
41    #[track_caller]
42    fn plant(&mut self, seed: impl Seed) -> Leaf {
43        let (leaf, growth) = self.allocate();
44        self.queue(Op::Plant {
45            leaf,
46            growth,
47            bud: seed.bud(core::panic::Location::caller()),
48        });
49        leaf
50    }

52    /// Grows an element off `under`.
53    #[track_caller]
54    fn branch(&mut self, under: Leaf, seed: impl Seed) -> Leaf {
55        let (leaf, growth) = self.allocate();
56        self.queue(Op::Branch {
57            leaf,
58            growth,
59            under,
60            bud: seed.bud(core::panic::Location::caller()),
61        });
62        leaf
63    }

65    /// Takes an element and everything beneath it down. Each one is reported as
66    /// [`withered`](crate::Pollen::withered).

68        self.queue(Op::Prune(leaf));

71    /// Moves an element, replacing its whole placement.
72    ///
73    /// A placement is one value rather than a set of edges, so there is no half-written state
74    /// between two of these and no question of which edge a later write meant.
75    ///
76    /// Dropped, like any op naming something it does not apply to, if the element is placed by its
77    /// ends rather than by a box -- which is [`between`](Grow::between).
78    fn at(&mut self, leaf: Leaf, location: Location) {
79        self.queue(Op::Place { leaf, location });
80    }

82    /// Moves both ends of a stroke.
83    ///
84    /// The point-mode counterpart to [`at`](Grow::at), and separate for the same reason the two
85    /// declarations are: an element has a box or it has ends, and a verb that wrote either would be
86    /// able to write the one the element does not have.
87    ///
88    /// Both ends together, because a stroke is one thing: a verb per end would leave a frame in
89    /// which the line is half moved.
90    ///
91    /// Dropped if the element is placed by a box.
92    fn between(&mut self, leaf: Leaf, from: Point, to: Point) {
93        self.queue(Op::Trace { leaf, from, to });
94    }

96    /// Redivides an element's box for the elements grown under it.
97    fn grid(&mut self, leaf: Leaf, grid: Grid) {
98        self.queue(Op::Divide { leaf, grid });
99    }

101    /// Points an element's placement at the one other element it may read.
102    ///
103    /// Replaces any anchor it already had.
104    ///
105    /// # Panics
106    ///
107    /// If the anchor would close a cycle, on the same terms as
108    /// [`anchored`](crate::Place::anchored).
109    #[track_caller]
110    fn anchor(&mut self, leaf: Leaf, to: Leaf) {
111        self.queue(Op::Anchor {
112            leaf,
113            to,
114            at: core::panic::Location::caller(),
115        });
116    }

119    ///
120    /// Elevation accumulates down the tree, so this moves a whole subtree by one write and nothing
121    /// inside it is touched.

124    }

126    /// Refills an element, with a [`Palette`](crate::Palette) role or with a [`Color`](crate::Color) stated
127    /// outright.
128    ///
129    /// A role is the ordinary answer, and a literal is an element saying it is not part of the
130    /// scheme: a [`repaint`](Grow::repaint) moves the first and not the second.
131    ///
132    /// Dropped, like any op naming something it does not apply to, if the element draws nothing.
133    fn color(&mut self, leaf: Leaf, fill: impl Into<Fill>) {
134        self.queue(Op::Recolor {
135            leaf,
136            fill: fill.into(),
137        });
138    }

140    /// Rewrites what a run of glyphs says.
141    ///
142    /// The measure follows in the same frame: R1 shapes what is written here and R2m wraps it, both
143    /// of them before anything reads a box, so an element sized to its own content is the right size
144    /// on the frame the string changed rather than the frame after.
145    ///
146    /// Dropped, like any op naming something it does not apply to, if the element is not a
147    /// [`Text`](crate::Text).
148    fn text(&mut self, leaf: Leaf, value: impl Into<String>) {
149        self.queue(Op::Letter {
150            leaf,
151            value: value.into(),
152        });
153    }

155    /// Fills parts of a run differently from the rest of it, over its own index space.
156    ///
157    /// Replaces every tint on the run rather than adding one, for the reason a placement is one
158    /// value: there is no half-written state between two of these, and no question of which range a
159    /// later write meant. Handing it nothing clears them.
160    ///
161    /// ```no_run
162    /// # use foliage::{Grow, Grove, Leaf, Palette};
163    /// # fn f(grove: &mut Grove, run: Leaf) {
164    /// grove.tint(run, [(0..2, Palette::Accent), (7..12, Palette::Muted)]);
165    /// # }
166    /// ```
167    ///
168    /// [`untint`](Grow::untint) is how they come off, rather than handing this an empty set: the
169    /// fill type of a set with nothing in it cannot be inferred, and a verb that has to be told the
170    /// type of what it is not writing is a worse surface than a second verb.
171    ///
172    /// Dropped, like any op naming something it does not apply to, if the element is not a
173    /// [`Text`](crate::Text).
174    fn tint<F: Into<Fill>>(
175        &mut self,
176        leaf: Leaf,
177        tints: impl IntoIterator<Item = (Range<usize>, F)>,
178    ) {
179        self.queue(Op::Tint {
180            leaf,
181            tints: Tints(
182                tints
183                    .into_iter()
184                    .map(|(range, fill)| (range, fill.into()))
185                    .collect(),
186            ),
187        });
188    }

190    /// Takes every tint off a run, leaving the whole of it in the run's own
191    /// [`color`](Grow::color).
192    ///
193    /// Dropped, like any op naming something it does not apply to, if the element is not a
194    /// [`Text`](crate::Text).
195    fn untint(&mut self, leaf: Leaf) {
196        self.queue(Op::Tint {
197            leaf,
198            tints: Tints::default(),
199        });
200    }

202    /// Selects a span of a [`TextInput`](crate::TextInput)'s value, in characters.
203    ///
204    /// The caret goes to the range's end and the selection is anchored at its start, so an empty
205    /// range places a caret and nothing else -- which is what makes this one verb rather than two

208    /// Both ends are held inside the value, so a range past it selects to the end.
209    ///
210    /// Dropped, like any op naming something it does not apply to, if the element is not a field.
211    fn select(&mut self, leaf: Leaf, range: Range<usize>) {

213    }

219    /// [`Polygon`](crate::Polygon)'s corners are its own, and are moved with

228    /// Reshapes a regular polygon: how many sides, how round its corners, how far it is turned.
229    ///
230    /// One value rather than three verbs, because it is one thought and because
231    /// [`Motion::Polygon`](crate::Motion::Polygon) moves it as one.
232    ///
233    /// Dropped if the element is not a [`Polygon`](crate::Polygon).
234    fn reshape(&mut self, leaf: Leaf, shape: Shape) {

236    }

238    /// Registers a picture and hands back the name elements draw it by.
239    ///
240    /// `pixels` is RGBA, one byte per channel, row-major, `size` texels across -- pixels the app
241    /// made rather than a file it has. An encoded picture, and one read from a path or a URL, is
242    /// [`Grove::image`](crate::Grove::image), which decodes and takes its size from what it decoded.
243    ///
244    /// Usable at any frame. A name taken here is valid immediately -- elements can be grown against
245    /// it in the same frame -- and writing the same name again replaces what it holds, so a picture
246    /// re-rendered at a higher resolution reaches every element drawing it with one write.
247    fn pixels(&mut self, pixels: impl Into<Vec<u8>>, size: Area) -> Plate {
248        let plate = self.plate();
249        self.load(plate, pixels, size);
250        plate
251    }
252
253    /// Names a picture whose pixels have not arrived.
254    ///
255    /// The two halves of [`pixels`](Grow::pixels), for when they happen at different times. A name is
256    /// valid the moment it is handed out, so elements can be grown against it now and
257    /// [`load`](Grow::load)ed when a fetch or a decode finishes -- an element drawing a plate with
258    /// nothing behind it occupies its box, draws nothing, and appears on the frame its pixels do.
259    ///
260    /// That is what keeps "is it loaded yet" out of an app's state. There is no readback for it and
261    /// deliberately so: the answer only ever changes what is on screen, and the engine already
262    /// changes that.
263    fn plate(&mut self) -> Plate {
264        self.picture()
265    }
266
267    /// Fills a picture's name with pixels.
268    ///
269    /// `pixels` is RGBA, one byte per channel, row-major, `size` texels across. Writing a name that
270    /// already holds a picture replaces it, so a re-fetch at a higher resolution reaches every
271    /// element drawing it without any of them being named.
272    ///
273    /// # Panics
274    ///
275    /// If `pixels` is smaller than `size` texels of RGBA.
276    fn load(&mut self, plate: Plate, pixels: impl Into<Vec<u8>>, size: Area) {
277        self.queue(Op::Load {
278            plate,
279            pixels: pixels.into(),
280            size,
281        });
282    }

284    /// Moves a scrolling region to a stated place.
285    ///
286    /// The destination is answered against the extent of the frame it lands in, so scrolling to the
287    /// end of a list that grew in the same frame lands at the end of the list rather than where the
288    /// list used to stop. It is clamped to what the region can reach, and reading the offset back
289    /// afterwards returns the pixels it settled at.
290    ///
291    /// ```no_run
292    /// # use foliage::{Grove, Grow, Leaf, ScrollTo};
293    /// # fn f(grove: &mut Grove, column: Leaf, section: Leaf) {
294    /// grove.scroll(column, ScrollTo::px(240.0));
295    /// grove.scroll(column, ScrollTo::show(section));
296    /// # }
297    /// ```
298    ///
299    /// A direct write, so it cancels a [`Motion::Scroll`](crate::Motion::Scroll) still moving the
300    /// region, and it ends a coast the reader's last drag left running.
301    ///
302    /// Dropped, like any op naming something it does not apply to, if the element does not scroll,
303    /// if the destination leaves no axis to move, or if
304    /// [`ScrollTo::show`](ScrollTo::show) names an element not grown under it.
305    fn scroll(&mut self, leaf: Leaf, to: ScrollTo) {
306        self.queue(Op::Scroll { leaf, to });
307    }

309    /// Makes an element inert without taking it out of the picture.
310    ///
311    /// It still draws -- a greyed control is still a control -- and it still occupies the box
312    /// stack, so it **swallows**: a press on it reaches neither the element itself nor anything
313    /// behind it, and a drag over it scrolls nothing. That is the whole difference between disabled
314    /// and decoration, and it is what makes disabling a page enough on its own when a drawer opens
315    /// over it: the page goes inert without a scrim to arrange.
316    ///
317    /// Cascades to everything grown under it, as a product recomputed every frame rather than a
318    /// write pushed down -- so a child grown under a disabled element is disabled on its first
319    /// frame. Focus leaves the subtree if it was in it.
320    fn disable(&mut self, leaf: Leaf) {
321        self.queue(Op::Disable {
322            leaf,
323            disabled: true,
324        });
325    }

327    /// Undoes [`disable`](Grow::disable) on this element.
328    ///
329    /// Symmetric: an element inside the subtree that was disabled in its own right stays disabled,
330    /// because what is recomputed is the product over the whole ancestry and not one inherited bit
331    /// that was overwritten on the way down.
332    fn enable(&mut self, leaf: Leaf) {
333        self.queue(Op::Disable {
334            leaf,
335            disabled: false,
336        });
337    }

350    /// Moves a property of an element over time.
351    ///
352    /// The target is written to the element at once and the tween carries what it left, so the
353    /// element declares where it is going from the moment it is told to go there. Nothing has to be
354    /// undone when it arrives: the blend at the end is the plain reading of the declaration.
355    ///
356    /// A second `animate` on a property already moving replaces it, starting from where the element
357    /// currently is rather than from where the first one began. A **direct write** to that property
358    /// -- [`at`](Grow::at), [`color`](Grow::color), [`opacity`](Grow::opacity) -- cancels it, and
359    /// the element is at what was written. So a property that is animated somewhere does not have
360    /// to be animated everywhere.
361    ///
362    /// ```no_run
363    /// # use foliage::{Color, Ease, Grow, Grove, Motion, Palette, Timing};
364    /// # fn f(grove: &mut Grove, leaf: foliage::Leaf) {
365    /// grove.animate(leaf, Motion::Palette(Palette::Accent), Timing::ms(180));
366    /// grove.animate(leaf, Motion::Color(Color::rgb(1.0, 0.4, 0.2)), Timing::ms(180));
367    /// grove.animate(leaf, Motion::Opacity(0.0), Timing::ms(240).ease(Ease::Accelerate));
368    /// # }
369    /// ```
370    ///
371    /// # The name it hands back
372    ///
373    /// This motion's own, for the app that has to tell one of an element's motions from another:
374    /// [`Pollen::tween`] is how far it has come and [`Pollen::finished`] is the frame it arrived,
375    /// against [`Pollen::landed`], which is the element settling and says nothing about which of its
376    /// properties did. [`stop`](Grow::stop) ends it, and [`Timing::within`](crate::Timing::within)
377    /// counts it into a group as ever.
378    ///
379    /// ```no_run
380    /// # use foliage::{Grow, Grove, Motion, Pollen, Timing};
381    /// # fn f(grove: &mut Grove, pollen: &Pollen, card: foliage::Leaf) {
382    /// let fade = grove.animate(card, Motion::Opacity(0.0), Timing::ms(240));
383    /// if pollen.finished(fade) {
384    ///     grove.prune(card);
385    /// }
386    /// # }
387    /// ```
388    ///
389    /// A name reports for as long as the motion it named is the one running, and goes quiet
390    /// otherwise: a motion replaced by a second `animate` on the same property, cancelled by a
391    /// direct write, stopped, or taken down with its element reports nothing further. So does one
392    /// that never ran -- dropped, like any op naming something it does not apply to, because the
393    /// element cannot carry the motion, as a fill cannot be moved on something that draws nothing.
394    ///
395    /// [`Pollen::tween`]: crate::Pollen::tween
396    /// [`Pollen::finished`]: crate::Pollen::finished
397    /// [`Pollen::landed`]: crate::Pollen::landed
398    fn animate(&mut self, leaf: Leaf, motion: Motion, timing: Timing) -> Tween {
399        let tween = self.name();
400        self.queue(Op::Animate {
401            leaf,
402            motion,
403            timing,
404            tween,
405        });
406        tween
407    }

409    /// Runs a number from `from` to `to`, reported each frame as [`Pollen::tween`] and written
410    /// nowhere.
411    ///
412    /// The engine's clock and easing, made available to a value it has no concept of. This is the
413    /// answer for everything [`Motion`] deliberately leaves out and for anything an app invents:
414    /// [`Motion`] is closed because the *engine's* obligations should be, not because an app's are.
415    ///
416    /// The frame it ends reports its end value and [`Pollen::finished`] together.
417    ///
418    /// [`Pollen::tween`]: crate::Pollen::tween
419    /// [`Pollen::finished`]: crate::Pollen::finished
420    fn tween(&mut self, from: f32, to: f32, timing: Timing) -> Tween {
421        let tween = self.name();
422        self.queue(Op::Channel {
423            tween,
424            from,
425            to,
426            timing,
427        });
428        tween
429    }

431    /// A [`tween`](Grow::tween) whose value is not read: what is wanted is the report that the time
432    /// is up.
433    ///
434    /// A timer of zero fires no earlier than the next frame -- queued at step 3, applied at 4,
435    /// advanced at 5, and reported at step 3 of the frame after. Honest rather than special-cased.
436    fn timer(&mut self, timing: Timing) -> Tween {
437        self.tween(0.0, 1.0, timing)
438    }

440    /// Names a group of tweens, so their *last* ending is reported as well as each of their own.
441    ///
442    /// A handle and nothing else. Anything that runs on the clock joins one with
443    /// [`Timing::within`](crate::Timing::within), from any callsite and at any frame, which is the
444    /// whole point: a sequence exists to time together things that have no reason to be written
445    /// together, and one that had to be declared in a single call would not be able to.
446    ///
447    /// ```no_run
448    /// # use foliage::{Ease, Grow, Grove, Motion, Palette, Timing};
449    /// # fn f(grove: &mut Grove, first: foliage::Leaf, second: foliage::Leaf) {
450    /// let intro = grove.sequence();
451    /// grove.animate(first, Motion::Opacity(1.0), Timing::ms(200).within(intro));
452    /// grove.animate(
453    ///     second,
454    ///     Motion::Palette(Palette::Accent),
455    ///     Timing::ms(200).after(80).ease(Ease::Decelerate).within(intro),
456    /// );
457    /// # }
458    /// ```
459    ///
460    /// The offsets stay on each tween, where [`after`](crate::Timing::after) already puts them: a
461    /// group says *when it is over*, not when its members start, so there is one place a delay is
462    /// stated rather than two that can disagree.
463    ///
464    /// Reported as [`sequence_finished`](crate::Pollen::sequence_finished) the frame nothing is
465    /// running under it any more. A name that has emptied can be filled again.
466    fn sequence(&mut self) -> Sequence {
467        self.group()
468    }

470    /// Ends a tween before it has run out. **Nothing is reported**, so a chain waiting on it does
471    /// not run.
472    ///
473    /// A motion is left on its target: that is what the element was told to be from the moment the
474    /// motion started, and the duration was only how long it was to take getting there, so giving up
475    /// the duration leaves it where it was meant to end. To end it somewhere *else* is to write that
476    /// property, which cancels the motion and puts the element at what was written -- the engine
477    /// never writes back a value from part way along, because a blend part way along is a
478    /// synthesized intermediate the app never declared.
479    ///
480    /// Nothing is reported because what stopped it already knows: this is the app's own act, and an
481    /// app's own writes are never reported back to it. Stopping the rest of a chain is calling this
482    /// on their names at the same callsite.
483    ///
484    /// A group counting it is told, because a group is over when nothing is running under it however
485    /// each member ended -- so stopping the last member ends the group. It has no reach beyond the
486    /// tween named: a [`Sequence`] is a name for when a set is quiet, not something that owns what
487    /// joined it, and stopping one member is nothing to the others.
488    ///
489    /// Dropped silently if it has already ended.
490    fn stop(&mut self, tween: Tween) {
491        self.queue(Op::Stop {
492            tween,
493            reported: false,
494        });
495    }

497    /// Ends a tween before it has run out, **as an arrival**: a chain waiting on it runs this frame.
498    ///
499    /// [`stop`](Grow::stop) and nothing else, except that the ending is reported. A motion reports
500    /// [`landed`](crate::Pollen::landed) and [`finished`](crate::Pollen::finished) as though it had
501    /// run its course; a channel reports its end value and its finish together, as it would have on
502    /// its own last frame; a [`timer`](Grow::timer) fires.
503    ///
504    /// What an interruption wants whenever the thing waiting on the motion is the point of it.
505    /// Dismissing a card mid-fade should still take the card down, and it should do so through the
506    /// one path that takes it down:
507    ///
508    /// ```no_run
509    /// # use foliage::{Grow, Grove, Motion, Pollen, Timing};
510    /// # fn f(grove: &mut Grove, pollen: &Pollen, card: foliage::Leaf, fade: foliage::Tween) {
511    /// if pollen.root_keys().is_empty() {
512    ///     grove.finish(fade);
513    /// }
514    /// if pollen.finished(fade) {
515    ///     grove.prune(card);
516    /// }
517    /// # }
518    /// ```
519    ///
520    /// Dropped silently if it has already ended.
521    fn finish(&mut self, tween: Tween) {
522        self.queue(Op::Stop {
523            tween,
524            reported: true,
525        });
526    }

569    /// Asks what the system clipboard holds, reported as [`pasted`](crate::Pollen::pasted).
570    ///
571    /// Never in the frame that asked: a clipboard is a promise in a browser and a round trip to
572    /// whichever program owns the selection on a desktop, so what it holds arrives the way bytes
573    /// from a path do and is drained in the frame it arrives in.
574    ///
575    /// A field answers `Ctrl+V` for itself and does not go through this. What comes back here is
576    /// the app's, whatever holds focus.
577    ///
578    /// An empty answer is what an app gets for an empty clipboard and for one the host would not
579    /// let it read -- a browser outside a user gesture, a desktop with no display server. There is
580    /// one thing to do about either, so they are one outcome and the reason is traced.
581    fn paste(&mut self) {
582        self.queue(Op::Paste { into: None });
583    }
584
585    /// Goes to `url`.
586    ///
587    /// On the web this **replaces the page**, and deliberately: a new tab is blocked outside a user
588    /// gesture, and a frame is not one. Off the web the desktop is asked to open the URL, which is
589    /// what going somewhere means where there is no page to replace.
590    fn navigate(&mut self, url: impl Into<String>) {
591        self.queue(Op::Navigate(url.into()));
592    }
593
594    /// Asks the host to save what is at `url`.
595    ///
596    /// The web's, and only the web's -- a browser is what turns a URL into a file in someone's
597    /// downloads. Off the web it is traced and nothing else, so an app naming what it offers names
598    /// it once.
599    fn download(&mut self, url: impl Into<String>) {
600        self.queue(Op::Download(url.into()));
601    }
602
603    /// States what every [`Palette`](crate::Palette) tone resolves to, for the whole tree.
604    ///
605    /// The one write that names no element, because a tone belongs to the scheme and not to any of
606    /// the elements declaring it. Everything painted in a tone whose color changed is re-extracted
607    /// and nothing else is, which is what makes a theme one op rather than a walk.
608    fn repaint(&mut self, scheme: Scheme) {
609        self.queue(Op::Repaint(scheme));
610    }
