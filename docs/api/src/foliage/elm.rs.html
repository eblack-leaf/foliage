1//! Elm -- change extraction.
2//!
3//! # Rowan recomputes. Elm decides what changed.
4//!
5//! Step 8, and the reason recomputing everything costs nothing downstream. Every renderer keeps what
6//! the backend is holding and compares this frame's resolved values against it, so an element that
7//! did not change costs one comparison and no upload.
8//!
9//! The cache is *what the backend holds*, not what the element last was. That is what makes a
10//! skipped frame unable to lose a change: there is no flag to miss, only a value that still differs
11//! at the next comparison. It is also a contract on the backend, which has to apply every batch it
12//! is handed -- a dropped batch leaves the cache claiming something the backend does not have, and
13//! nothing afterwards will correct it.
14//!
15//! # What is compared is not always what is uploaded
16//!
17//! A [`Panel`](crate::Panel) and a [`Polygon`](crate::Polygon) are described entirely in logical
18//! pixels, so one value is both what is compared here and what the vertex buffer holds. The other
19//! four are not, and each for the same reason: turning what the element declares into what the GPU
20//! draws needs the display's density, and the density stops at the backend
21//! ([`Ginkgo`](crate::ginkgo)).
22//!
23//! | | declares | the backend derives |
24//! |---|---|---|
25//! | [`Text`](crate::Text) | cells and characters | the cut ink, snapped to device pixels |
26//! | [`Line`](crate::Line) | two ends and a weight | four corners, axis-aligned ones snapped |
27//! | [`Icon`](crate::Icon) | a box and a field | the sheet rect, and the field's screen-space range |
28//! | [`Image`](crate::Image) | a box and a plate | which texture to bind |
29//!
30//! So extraction is written in logical pixels throughout and compares logical values, and the
31//! derivation happens once per written instance rather than once per frame.
32

56/// Which renderer an element carries, and so which instances it is gathered into.
57///
58/// Decided when the element is described and never afterwards: what an element draws is part of
59/// what it is, and one that is to draw something else is a different element. Nothing writes this
60/// -- there is no op that can, and the only place it is set is where the element is grown.
61///
62/// Extraction walks the tree once and routes on this, so a further renderer costs a variant rather
63/// than a pass over everything. It is the *only* thing that says what an element is: a set of
64/// components that happens to look like a panel is not a panel.
65#[derive(Component, Copy, Clone, Debug, Default, PartialEq, Eq)]
66pub(crate) enum Chlorophyll {
67    /// Nothing. Carrying no renderer is the whole of what makes an element a [`Stem`](crate::Stem).
68    #[default]
69    None,
70    /// A filled rectangle.
71    Panel,
72    /// A run of monospaced glyphs.
73    Text,
74    /// A filled regular polygon.
75    Polygon,
76    /// A straight stroke between two points.
77    Line,
78    /// A vector mark, from a distance field.
79    Icon,
80    /// A picture, fitted into its box.
81    Image,
82}

85///
86/// Carried by the [`Bud`](crate::op::Bud) as one value so that growing an element inserts the
87/// pigment its own renderer reads and no other. Past that point each is an ordinary component and
88/// nothing asks which kind it was.
89pub(crate) enum Pigment {
90    Panel(PanelPigment),
91    Text(TextPigment),
92    Polygon(PolygonPigment),
93    Line(LinePigment),
94    Icon(IconPigment),
95    Image(ImagePigment),
96}

99///

101/// neither. Ordinary declared state, and what [`color`](crate::Grow::color) and
102/// [`round`](crate::Grow::round) write -- the decision beside it stays untouched, because an

110/// What the backend is holding, one set per renderer.
111#[derive(Default)]
112pub(crate) struct Elm {
113    pub(crate) panels: Instances<PanelInstance>,
114    pub(crate) polygons: Instances<PolygonInstance>,
115    pub(crate) lines: Instances<LineInstance>,
116    pub(crate) icons: Instances<IconInstance>,
117    pub(crate) images: Instances<ImageInstance>,
118    pub(crate) texts: Runs,
119    /// Where one run's glyphs are gathered before they are compared against what is held. Kept
120    /// between frames for its capacity, and reused by every run in turn: a frame that changes
121    /// nothing must not allocate.
122    glyphs: Vec<Glyph>,
123}
124
125impl Elm {
126    /// How much this frame's batch moves, across every renderer. Reported to the trace, and the one
127    /// number that says whether an unchanged frame really cost nothing.
128    pub(crate) fn moved(&self) -> (usize, usize) {
129        let written = self.panels.written.len()
130            + self.polygons.written.len()
131            + self.lines.written.len()
132            + self.icons.written.len()
133            + self.images.written.len()
134            + self.texts.written.len();
135        let withdrawn = self.panels.withdrawn.len()
136            + self.polygons.withdrawn.len()
137            + self.lines.withdrawn.len()
138            + self.icons.withdrawn.len()
139            + self.images.withdrawn.len()
140            + self.texts.withdrawn.len();
141        (written, withdrawn)
142    }
143
144    /// Drops everything the backend is held to, so the next extraction writes the tree entire.
145    ///
146    /// The one thing that invalidates a comparison against what the backend holds: the backend's
147    /// copy is in device pixels and this one is not, so a display whose density changed leaves every
148    /// derived instance -- a cut glyph, a snapped stroke, a field's screen-space range -- correct
149    /// against a density that is gone, while the logical values they came from are unchanged and
150    /// compare equal forever.
151    ///
152    /// Total rather than per renderer, because the density is not any renderer's.
153    pub(crate) fn recut(&mut self) {
154        self.panels.forget();
155        self.polygons.forget();
156        self.lines.forget();
157        self.icons.forget();
158        self.images.forget();
159        self.texts.forget();
160    }
161}

163/// One glyph of a run: the cell it occupies, which character occupies it, and what it is filled
164/// with.
165///
166/// The cell is in logical pixels and already offset by the run's own box, because where a character
167/// lands is what wrapping decided and wrapping is the engine's. Where the *ink* sits inside that
168/// cell is not here: that is the rasteriser's, which is the only thing that knows what shape it made
169/// and at what density it made it.
170///
171/// The colour is per glyph because a [`tint`](crate::Grow::tint) is a fill over a range of the run's
172/// own index space. A run with no tints resolves the same colour for every one of them, which costs
173/// the comparison it would have cost anyway.
174#[derive(Copy, Clone, Debug, PartialEq)]
175pub(crate) struct Glyph {
176    pub(crate) cell: Section,
177    pub(crate) character: char,
178    pub(crate) color: Color,
179}
180
181/// The runs, and every glyph of each.
182///
183/// Deliberately not an [`Instances`]: a run is **one** entry in the one stack whose renderer holds
184/// many things under it, and that is the whole difference. What is diffed here is the run entire --
185/// its glyphs, its face, its rank and its clip -- because a run's glyphs move, refill and restack
186/// together, and finding which of them changed would cost more than rewriting the run.
187#[derive(Default)]
188pub(crate) struct Runs {
189    held: HashMap<Key, Run>,
190    /// Runs the backend holds at something other than what is now wanted, or does not hold at all.
191    /// Keys rather than values: what the backend is to apply is what is held for them, which it
192    /// reads back at [`run`](Runs::run).
193    pub(crate) written: Vec<Key>,
194    /// Runs the backend holds and should not, in a stable order.
195    pub(crate) withdrawn: Vec<Key>,
196    /// Which extraction is running. An entry left at an older one is no longer wanted.
197    pass: u64,
198}
199
200/// One run, as the backend is to hold it.
201pub(crate) struct Run {
202    /// Every glyph that leaves ink, in reading order. A space advances the wrap and is not one.
203    pub(crate) glyphs: Vec<Glyph>,
204    /// Which face the glyphs are cut from, and at what size. The backend rasterises against these,
205    /// so they are part of what the run is rather than of how it was measured.
206    pub(crate) font: Font,
207    pub(crate) size: u32,
208    pub(crate) rank: ResolvedElevation,
209    pub(crate) clip: Section,
210    seen: u64,
211}
212
213impl Runs {
214    /// Opens an extraction, dropping what the last one reported.
215    fn open(&mut self) {
216        self.written.clear();
217        self.withdrawn.clear();
218    }
219
220    /// Takes one run as it now stands, and reports it written if the backend is holding it
221    /// otherwise.
222    fn want(
223        &mut self,
224        key: Key,
225        rank: ResolvedElevation,
226        clip: Section,
227        font: Font,
228        size: u32,
229        glyphs: &[Glyph],
230    ) {
231        let pass = self.pass;
232        match self.held.entry(key) {
233            Entry::Occupied(mut held) => {
234                let held = held.get_mut();
235                held.seen = pass;
236                if held.font == font
237                    && held.size == size
238                    && held.rank == rank
239                    && held.clip == clip
240                    && held.glyphs == glyphs
241                {
242                    return;
243                }
244                held.font = font;
245                held.size = size;
246                held.rank = rank;
247                held.clip = clip;
248                // Rewritten in place, so a run that changed costs its own glyphs and no allocation.
249                held.glyphs.clear();
250                held.glyphs.extend_from_slice(glyphs);
251            }
252            Entry::Vacant(slot) => {
253                slot.insert(Run {
254                    glyphs: glyphs.to_vec(),
255                    font,
256                    size,
257                    rank,
258                    clip,
259                    seen: pass,
260                });
261            }
262        }
263        self.written.push(key);
264    }
265
266    /// Closes the extraction: what nothing wanted this frame is withdrawn.
267    fn extract(&mut self) {
268        let pass = self.pass;
269        let withdrawn = &mut self.withdrawn;
270        self.held.retain(|key, held| {
271            if held.seen == pass {
272                return true;
273            }
274            withdrawn.push(*key);
275            false
276        });
277        // Nothing may depend on the order a map iterates, and two identical runs have to extract
278        // identically.
279        withdrawn.sort();
280        self.pass += 1;
281    }
282
283    /// Drops what the backend is held to. See [`Elm::recut`].
284    fn forget(&mut self) {
285        self.held.clear();
286    }
287
288    /// What is held for one run, which is what the backend is to be holding for it.
289    pub(crate) fn run(&self, key: Key) -> Option<&Run> {
290        self.held.get(&key)
291    }
292}

294/// What one instance is held under.
295///
296/// A plain number, and deliberately not a [`Leaf`]. An element is one of these -- its name's own
297/// bits -- and so is a glyph inside a run, so a renderer whose element draws *many* things keeps its
298/// own [`Instances`] under its own numbering and hands the one stack a single entry. Nothing about
299/// holding, diffing or uploading has to know which of the two it is looking at.
300#[derive(Copy, Clone, PartialEq, Eq, Hash, PartialOrd, Ord, Debug)]
301pub(crate) struct Key(pub(crate) u64);
302
303impl From<Leaf> for Key {
304    fn from(leaf: Leaf) -> Self {
305        Self(leaf.id())
306    }
307}

309/// One renderer's instances: what the backend holds, and the difference the last extraction found.
310///
311/// Generic over the instance, because what a renderer sends is the renderer's own business. This
312/// owns the comparison and knows nothing about what is being compared.
313///
314/// The rank is the one thing every renderer has in common, so it is carried here rather than
315/// inside each renderer's instance: where an element sits in the one stack is a fact about the
316/// element, not about what it happens to draw, and the backend needs it in a different form from
317/// the one the resolver produced. Keeping it out is also what leaves an instance free to be
318/// exactly the bytes a vertex buffer takes, for the renderers whose instance is those bytes.
319pub(crate) struct Instances<I> {
320    held: HashMap<Key, Held<I>>,
321    /// What should be drawn this frame, gathered before it is compared. Kept between frames for its
322    /// capacity: a frame that changes nothing must not allocate.
323    wanted: Vec<Stacked<I>>,
324    /// Instances the backend does not hold, or holds at a different value or rank.
325    pub(crate) written: Vec<Stacked<I>>,
326    /// Instances the backend holds and should not, in a stable order.
327    pub(crate) withdrawn: Vec<Key>,
328    /// Which extraction is running. An entry left at an older one is no longer wanted.
329    pass: u64,
330}
331
332/// One instance, where in the one stack it is to be drawn, and what it is clipped to.
333///
334/// The clip is beside the instance rather than inside it for the same reason the rank is: it is not
335/// the renderer's data. It says where the backend is allowed to paint, which is a property of the
336/// region the element sits in, and the backend applies it to the pass rather than to the panel.
337#[derive(Copy, Clone, Debug)]
338pub(crate) struct Stacked<I> {
339    pub(crate) key: Key,
340    pub(crate) rank: ResolvedElevation,
341    pub(crate) clip: Section,
342    pub(crate) instance: I,
343}
344
345/// One instance the backend holds, and the extraction that last asked for it.
346struct Held<I> {
347    instance: I,
348    rank: ResolvedElevation,
349    clip: Section,
350    seen: u64,
351}
352
353impl<I> Default for Instances<I> {
354    fn default() -> Self {
355        Self {
356            held: HashMap::new(),
357            wanted: Vec::new(),
358            written: Vec::new(),
359            withdrawn: Vec::new(),
360            pass: 0,
361        }
362    }
363}
364
365impl<I: Copy + PartialEq> Instances<I> {
366    /// Adds one instance to what should be drawn this frame, at the rank it resolved to and inside
367    /// the clip it resolved under.
368    fn want(&mut self, key: impl Into<Key>, rank: ResolvedElevation, clip: Section, instance: I) {
369        self.wanted.push(Stacked {
370            key: key.into(),
371            rank,
372            clip,
373            instance,
374        });
375    }
376
377    /// Diffs what should be drawn against what the backend holds, and takes the result as the new
378    /// holding.
379    ///
380    /// The holding is updated in place. Rebuilding it would allocate a map per renderer per frame,
381    /// which is a cost on every frame including the ones with nothing in them -- and an unchanged
382    /// frame costing nothing is the whole claim of this phase.
383    fn extract(&mut self) {
384        self.written.clear();
385        self.withdrawn.clear();
386        self.pass += 1;
387        let pass = self.pass;
388        for index in 0..self.wanted.len() {
389            let wanted = self.wanted[index];
390            match self.held.entry(wanted.key) {
391                Entry::Occupied(mut held) => {
392                    let held = held.get_mut();
393                    if held.instance != wanted.instance
394                        || held.rank != wanted.rank
395                        || held.clip != wanted.clip
396                    {
397                        held.instance = wanted.instance;
398                        held.rank = wanted.rank;
399                        held.clip = wanted.clip;
400                        self.written.push(wanted);
401                    }
402                    held.seen = pass;
403                }
404                Entry::Vacant(slot) => {
405                    slot.insert(Held {
406                        instance: wanted.instance,
407                        rank: wanted.rank,
408                        clip: wanted.clip,
409                        seen: pass,
410                    });
411                    self.written.push(wanted);
412                }
413            }
414        }
415        self.wanted.clear();
416        let withdrawn = &mut self.withdrawn;
417        self.held.retain(|key, held| {
418            if held.seen == pass {
419                return true;
420            }
421            withdrawn.push(*key);
422            false
423        });
424        // Nothing may depend on the order a map iterates, and two identical runs have to extract
425        // identically.
426        withdrawn.sort();
427    }
428
429    /// Drops what the backend is held to. See [`Elm::recut`].
430    fn forget(&mut self) {
431        self.held.clear();
432    }
433
434    /// How many instances the backend is holding.
435    pub(crate) fn len(&self) -> usize {
436        self.held.len()
437    }
438
439    /// What the backend is holding for `key`.
440    pub(crate) fn holding(&self, key: impl Into<Key>) -> Option<I> {
441        self.held.get(&key.into()).map(|held| held.instance)
442    }
443}

445/// Step 8. Resolved state becomes instances, and only where it differs from what is already drawn.
446pub(crate) fn run(grove: &mut Grove) {
447    let step = trace_span!(
448        "extract",
449        written = Empty,
450        withdrawn = Empty,
451        glyphs = Empty
452    );
453    let _entered = step.enter();
454    grove.elm.texts.open();
455    // Detached for the walk so that gathering a run's glyphs -- which reads the shaping cache -- and
456    // handing them over -- which writes what is held -- are not the same borrow. It goes back below,
457    // with whatever capacity the widest run this frame gave it.
458    let mut glyphs = core::mem::take(&mut grove.elm.glyphs);
459    let mut total = 0;
460    for leaf in grove.tree.leaves() {
461        let chlorophyll = grove.tree.chlorophyll(leaf);
462        if chlorophyll == Chlorophyll::None {
463            continue;
464        }
465        let Some(painted) = painted(grove, leaf) else {
466            continue;
467        };
468        let rank = grove.tree.rank(leaf);
469        match chlorophyll {
470            // Answered above: carrying no renderer is the whole of what makes an element a stem.
471            Chlorophyll::None => {}
472            Chlorophyll::Panel => {
473                // Grown together and by nothing else, so a panel always has one.
474                let Some(pigment) = grove.tree.panel_pigment(leaf) else {
475                    continue;
476                };
477                let instance = PanelInstance::new(
478                    painted.section,
479                    tint(grove, leaf, pigment.fill).faded(painted.opacity),
480                    pigment.rounding,
481                );
482                grove.elm.panels.want(leaf, rank, painted.clip, instance);
483            }
484            Chlorophyll::Polygon => {
485                let Some(pigment) = grove.tree.polygon_pigment(leaf) else {
486                    continue;
487                };
488                // The shape is read plainly, with no blend applied here: a shape blends to a shape,
489                // so a motion moving one writes it back over the declaration every frame and what
490                // the element holds is already where the motion has reached.
491                let instance = PolygonInstance::new(
492                    painted.section,
493                    tint(grove, leaf, pigment.fill).faded(painted.opacity),
494                    pigment.shape,
495                );
496                grove.elm.polygons.want(leaf, rank, painted.clip, instance);
497            }
498            // A stroke's ends are resolved geometry in their own right, settled beside the box the
499            // way [`Drawn`](crate::rowan::Drawn) is: the box is the rectangle around them grown by
500            // half the weight, and which of its two diagonals the stroke runs along is not
501            // something a rectangle can say.
502            Chlorophyll::Line => {
503                let (Some(pigment), Some(stretched), Some(stroke)) = (
504                    grove.tree.line_pigment(leaf),
505                    grove.tree.stretched(leaf),
506                    grove.tree.stroke(leaf),
507                ) else {
508                    continue;
509                };
510                let instance = LineInstance {
511                    from: stretched.from,
512                    to: stretched.to,
513                    color: tint(grove, leaf, pigment.fill).faded(painted.opacity),
514                    weight: stroke.weight,
515                    cap: pigment.cap,
516                };
517                grove.elm.lines.want(leaf, rank, painted.clip, instance);
518            }
519            Chlorophyll::Icon => {
520                let Some(pigment) = grove.tree.icon_pigment(leaf) else {
521                    continue;
522                };
523                // A field that has not arrived draws nothing and occupies its box, exactly as a
524                // picture with no pixels does. Absent from the batch rather than held as blank: the
525                // sheet cuts a mark when the batch names one, so an instance written before the
526                // field landed would be a mark nothing ever asked the sheet for again.
527                if grove.fields.mark(pigment.field).is_none() {
528                    continue;
529                }
530                let instance = IconInstance {
531                    // Square, because a distance field is: the mark sits in the largest square its
532                    // box holds rather than stretching to the box's own ratio.
533                    section: squared(painted.section),
534                    color: tint(grove, leaf, pigment.fill).faded(painted.opacity),
535                    field: pigment.field,
536                };
537                grove.elm.icons.want(leaf, rank, painted.clip, instance);
538            }
539            Chlorophyll::Image => {
540                let Some(pigment) = grove.tree.image_pigment(leaf) else {
541                    continue;
542                };
543                // A plate whose pixels have not arrived draws nothing and occupies its box. It is
544                // absent from the batch rather than held as blank, so the frame the pixels land is
545                // the frame it appears, with nothing to undo.
546                let Some(picture) = grove.plates.size(pigment.plate) else {
547                    continue;
548                };
549                let (section, crop) = fitted(painted.section, picture, pigment.fit);
550                let instance = ImageInstance {
551                    section,
552                    crop,
553                    radii: pigment.rounding.radii(section),
554                    opacity: painted.opacity,
555                    plate: pigment.plate,
556                };
557                grove.elm.images.want(leaf, rank, painted.clip, instance);
558            }
559            // A run's box, fill, rank and clip resolve exactly as a panel's do. What is different is
560            // that it draws *many* things at one rank: it is one entry in the one stack, and its
561            // renderer holds its glyphs under its own numbering -- which is what a [`Key`] is a
562            // number rather than a [`Leaf`] for.
563            Chlorophyll::Text => {
564                // Grown together and by nothing else, so a run always has both.
565                let (Some(pigment), Some(typeface)) =
566                    (grove.tree.text_pigment(leaf), grove.tree.typeface(leaf))
567                else {
568                    continue;
569                };
570                let color = tint(grove, leaf, pigment.fill).faded(painted.opacity);
571                let size = typeface.size.at(grove.layout, grove.short);
572                let Some(value) = grove.tree.lettering(leaf) else {
573                    continue;
574                };
575                // R1 shapes every run that is measured at all, so one that is not held is one
576                // nothing is laying out. Extraction reads that cache and never adds to it.
577                let Some(shaped) = grove.shaping.shaped(typeface.font, size, value) else {
578                    continue;
579                };
580                let tints = grove.tree.tints(leaf);
581                let origin = painted.section.position;
582                let cell = shaped.cell();
583                glyphs.clear();
584                // Wrapped at the width the run resolved to, which is the width it was measured at.
585                shaped.place(painted.section.width(), |character, index, at| {
586                    glyphs.push(Glyph {
587                        cell: Section::new(origin.moved(at), cell),
588                        character,
589                        // The run's own fill, unless a tint claims this character. Resolved here
590                        // because this is where a fill becomes a colour, which is the same reason
591                        // the run's own is.
592                        color: match tints.and_then(|tints| tints.over(index)) {
593                            Some(fill) => fill.color(&grove.scheme).faded(painted.opacity),
594                            None => color,
595                        },
596                    });
597                });
598                total += glyphs.len();
599                grove.elm.texts.want(
600                    leaf.into(),
601                    rank,
602                    painted.clip,
603                    typeface.font,
604                    size,
605                    &glyphs,
606                );
607            }
608        }
609    }
610    grove.elm.glyphs = glyphs;
611    grove.elm.panels.extract();
612    grove.elm.polygons.extract();
613    grove.elm.lines.extract();
614    grove.elm.icons.extract();
615    grove.elm.images.extract();
616    grove.elm.texts.extract();
617    let (written, withdrawn) = grove.elm.moved();
618    step.record("written", written);
619    step.record("withdrawn", withdrawn);
620    step.record("glyphs", total);
621}

623/// Where an element is painted, and inside what.
624struct Painted {
625    section: Section,
626    clip: Section,
627    opacity: f32,
628}
629
630/// What is painted of `leaf`, or `None` if nothing is.
631///
632/// Hidden is the app's intent and culled is this pass's decision, taken here from the clip rect and
633/// recorded nowhere: an element scrolled out of its region is absent from the batch and unchanged in
634/// every other respect, so scrolling back to it needs nothing to be undone.
635fn painted(grove: &Grove, leaf: Leaf) -> Option<Painted> {
636    let inherited = grove.tree.inherited(leaf);
637    let section = grove.tree.drawn(leaf);
638    // What a scrolling ancestor leaves visible, never wider than the surface: a clip is what the
639    // backend scissors the pass to, and nothing outside the surface is painted whatever the rect
640    // says.
641    let clip = grove
642        .tree
643        .clip(leaf)
644        .intersect(Section::new(Position::default(), grove.viewport));
645    if !inherited.visible || section.intersect(clip).is_empty() {
646        return None;
647    }
648    Some(Painted {
649        section,
650        clip,
651        opacity: inherited.opacity,
652    })
653}

655/// The largest square `section` holds, centred in it.
656fn squared(section: Section) -> Section {
657    let side = section.width().min(section.height());
658    Section::new(
659        Position::new(
660            section.left() + (section.width() - side) / 2.0,
661            section.top() + (section.height() - side) / 2.0,
662        ),
663        Area::new(side, side),
664    )
665}

667/// The box a picture is drawn into, and what part of it is shown.
668///
669/// One of the two moves, never both: fitting inside the box changes the box and shows the whole
670/// picture, and filling the box keeps the box and shows part of the picture. Stretching does
671/// neither and is the only one that distorts.
672fn fitted(section: Section, picture: Area, fit: Fit) -> (Section, [f32; 4]) {
673    const WHOLE: [f32; 4] = [0.0, 0.0, 1.0, 1.0];
674    if picture.width <= 0.0 || picture.height <= 0.0 || section.is_empty() {
675        return (section, WHOLE);
676    }
677    let picture_ratio = picture.width / picture.height;
678    let box_ratio = section.width() / section.height();
679    match fit {
680        Fit::Stretch => (section, WHOLE),
681        Fit::Aspect => {
682            let area = match box_ratio > picture_ratio {
683                true => Area::new(section.height() * picture_ratio, section.height()),
684                false => Area::new(section.width(), section.width() / picture_ratio),
685            };
686            let position = Position::new(
687                section.left() + (section.width() - area.width) / 2.0,
688                section.top() + (section.height() - area.height) / 2.0,
689            );
690            (Section::new(position, area), WHOLE)
691        }
692        Fit::Crop => {
693            let (width, height) = match box_ratio > picture_ratio {
694                true => (1.0, picture_ratio / box_ratio),
695                false => (box_ratio / picture_ratio, 1.0),
696            };
697            (
698                section,
699                [(1.0 - width) / 2.0, (1.0 - height) / 2.0, width, height],
700            )
701        }
702    }
703}

705/// What a fill currently paints as.
706///
707/// A blend of two fills is a color rather than a fill, so a motion on one is applied here -- where a
708/// fill becomes a color -- and not written onto the element. Both ends are read against the same
709/// scheme at the same instant, so a repaint mid-motion moves whichever of them is a role.
710fn tint(grove: &Grove, leaf: Leaf, fill: Fill) -> Color {
711    let target = fill.color(&grove.scheme);
712    match grove.aspen.fill(leaf) {
713        Some((Departed::Declared(fill), at)) => fill.color(&grove.scheme).blend(target, at),
714        Some((Departed::Snapshot(color), at)) => color.blend(target, at),
715        None => target,
716    }
717}
