1//! Text -- a run of monospaced glyphs, and the character cell everything else is measured in.
2//!
3//! # Width flows down. Height flows up.
4//!
5//! Wrapping makes a run's height depend on its width, and its width comes from the layout. That is
6//! a cycle in the general case, and it is what stopped anything being sized to text that wraps.
7//!
8//! Monospace is what unties it. A run's **max-content width** -- the widest it would like to be,
9//! unwrapped -- is its longest line's character count times the cell width: exact, free, and
10//! available before any layout has happened at all. So the pass going down has everything it needs,
11//! and only the pass coming up has to measure anything, by which time it has a width to measure
12//! against.
13//!
14//! | | Pass | Produces |
15//! |---|---|---|
16//! | R1 | measure | the character cell, and max-content width |
17//! | R2m | wrap | the lines the run takes at its resolved width, and so its height |
18//!
19//! Both are [`Rowan`](crate::rowan)'s. What this module owns is the vocabulary they are stated in:
20//! which font, at what size, and what the run says.
21

37/// A run of monospaced glyphs.
38///
39/// Where it sits, how it is divided and what it anchors to read exactly as they do on a
40/// [`Stem`](crate::Stem). What it holds beyond that is what it says, what it is filled with, and
41/// which font it composes in.
42///
43/// It is the one element whose box can be **measured rather than declared**. In a width role
44/// [`content()`](crate::content) is the widest the run would like to be; in a height role it is how
45/// tall the run turned out at the width the layout gave it:
46///
47/// ```no_run
48/// # use foliage::{Boxed, FontSize, Location, Palette, Place, Source, Text, content, left, top};
49/// Text::new("as tall as it wraps to")
50///     .color(Palette::Ink)
51///     .font_size(FontSize::new().xs(14).lg(18))
52///     .at(Location::new().xs(
53///         left(0.px()).right(100.pct()),
54///         top(0.px()).height(content()),
55///     ));
56/// ```
57///
58/// [`letters`](crate::Source::letters) is the other half, and it stays: where the count is genuinely
59/// known ahead of time it costs nothing to resolve and says what it means. Measuring is for where it
60/// is not.
61#[derive(Clone, Debug, Default)]
62pub struct Text {
63    pub(crate) placement: Placement,
64    pub(crate) value: String,
65    pub(crate) fill: Fill,
66    pub(crate) tints: Tints,
67}
68
69impl Text {
70    /// A run saying `value`, filled with [`Palette::Ink`] -- what is read against a surface rather
71    /// than drawn as one.
72    pub fn new(value: impl Into<String>) -> Self {
73        Self {
74            placement: Placement::default(),
75            value: value.into(),
76            fill: Fill::Role(Palette::Ink),
77            tints: Tints::default(),
78        }
79    }
80
81    /// What the glyphs are filled with: a [`Palette`] role, or a [`Color`](crate::Color) stated
82    /// outright.
83    ///
84    /// The same [`Fill`] a panel takes, so a run is repainted, animated and cancelled by the same
85    /// writes -- and so a per-character tint, when it lands, is stated in the vocabulary the run
86    /// already is rather than in a second one beside it.
87    pub fn color(mut self, fill: impl Into<Fill>) -> Self {
88        self.fill = fill.into();
89        self
90    }
91
92    /// Fills part of the run differently from the rest of it.
93    ///
94    /// The range is in **characters of the value**, spaces included -- the same index space a caret
95    /// and a selection are addressed in, and the space the string itself is written in. Anything
96    /// untinted stays the run's own [`color`](Text::color), so a run with no tints is exactly the
97    /// run it would otherwise be.
98    ///
99    /// ```no_run
100    /// # use foliage::{Color, Palette, Text};
101    /// Text::new("ok  warn  error")
102    ///     .color(Palette::Ink)
103    ///     .tint(4..8, Palette::Accent)
104    ///     .tint(10..15, Color::rgb(0.9, 0.35, 0.3));
105    /// ```
106    ///
107    /// A tint is a [`Fill`] like any other, so a role follows a
108    /// [`repaint`](crate::Grow::repaint) and a literal does not. Where two overlap the later one
109    /// wins, which is the rule a reader can predict without knowing what else was written.
110    pub fn tint(mut self, range: Range<usize>, fill: impl Into<Fill>) -> Self {
111        self.tints.0.push((range, fill.into()));
112        self
113    }
114}
115
116impl Places for Text {
117    fn placement(&mut self) -> &mut Placement {
118        &mut self.placement
119    }
120}

122impl Buds for Text {
123    fn bud(mut self, at: Caller) -> Bud {
124        // Every element may carry a typeface; a run of glyphs is the one that must, because there is
125        // no reading of it at all without a font to read it in.
126        self.placement.typeface.get_or_insert_default();
127        Bud {
128            chlorophyll: Chlorophyll::Text,
129            pigment: Some(Pigment::Text(TextPigment { fill: self.fill })),
130            lettering: Some(Lettering(self.value)),
131            // Absent unless part of the run is filled differently, so a plain run carries nothing
132            // for the tint lookup to walk.
133            tints: (!self.tints.0.is_empty()).then_some(self.tints),
134            placement: self.placement,
135            at,
136            ..Bud::bare()
137        }
138    }
139}

143/// What a run says.
144///
145/// Content rather than renderer state: it is what R1 shapes and what R2m wraps, so it is read by
146/// passes that have nothing to do with drawing. [`text`](crate::Grow::text) writes it.
147#[derive(Component, Clone, Debug, Default, PartialEq, Eq)]
148pub(crate) struct Lettering(pub(crate) String);
149
150/// What the text renderer was told.
151///
152/// Grown alongside [`Chlorophyll::Text`] and by nothing else, so an element carries both or neither.
153#[derive(Component, Copy, Clone, Debug, Default, PartialEq)]
154pub(crate) struct TextPigment {
155    pub(crate) fill: Fill,
156}
157
158/// Fills over parts of a run's own index space.
159///
160/// Under the run's own fill rather than beside it: a run declares one fill, and a tint says that
161/// some of its characters are not drawn in it. Everything untinted is the run's, so a run with no
162/// tints is exactly the run it was before tints existed.
163///
164/// Content rather than renderer state, like [`Lettering`] and for the same reason: a range is stated
165/// against the value, so the two have to be written and read in the same terms.
166#[derive(Component, Clone, Debug, Default, PartialEq)]
167pub(crate) struct Tints(pub(crate) Vec<(Range<usize>, Fill)>);
168
169impl Tints {
170    /// What covers character `index`, if anything does.
171    ///
172    /// The last written wins where two overlap. That is the rule a reader can predict without
173    /// knowing what else was written -- the alternative is a precedence order between ranges that
174    /// have no natural one.
175    pub(crate) fn over(&self, index: usize) -> Option<Fill> {
176        self.0
177            .iter()
178            .rev()
179            .find(|(range, _)| range.contains(&index))
180            .map(|(_, fill)| *fill)
181    }
182}
