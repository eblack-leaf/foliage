1//! Sprig -- a cutting of the tree, carried off the frame.
2//!
3//! Everything the engine does is done in a frame, on the thread the frame runs on. A `Sprig` is how
4//! work that is on neither -- a thread decoding something large, a promise that has resolved, a
5//! timer of the host's own -- reaches the tree anyway: it carries [`Grow`](crate::Grow) entire and
6//! pushes onto the one queue, so what it writes is drained where every other change is drained and
7//! is ordered against them by when it arrived.
8//!
9//! It registers as well as writes. A name comes from the one allocator and every registry grows to
10//! meet one, so a worker that built a face, a field or a picture hands the bytes over as the op a
11//! retrieval would have pushed -- which is what makes a thread that *made* an asset and one that
12//! *read* one the same case.
13//!
14//! What it cannot do is **sample**. A read needs the world, and the world belongs to the frame -- so
15//! the reads a worker needs are pushed to it instead: [`Conditions`] is what is true of the tree as a
16//! whole, published every frame, and [`watch`](Sprig::watch) is how one property of one element joins
17//! them.

43/// The engine, reached from off the frame.
44///
45/// Cloneable and, where the platform has threads, `Send`: hand one to a worker and it can grow,
46/// move, fill and take down elements without the frame having to relay anything. It carries
47/// [`Grow`](crate::Grow) exactly as [`Grove`] does, so **a change reads identically wherever it is
48/// issued** and the same code does the same thing on either side of the boundary.
49///
50/// ```no_run
51/// # use foliage::{Grove, Grow, Stem, Vein};
52/// # fn f(grove: &mut Grove) {
53/// let mut sprig = grove.sprig();
54/// std::thread::spawn(move || {
55///     let leaf = sprig.plant(Stem::new());
56///     sprig.watch(leaf, Vein::Drawn);
57/// });
58/// # }
59/// ```
60///
61/// # What is different, and what is not
62///
63/// An op from here lands in the drain of the frame that was running when it arrived, if it arrived
64/// before that frame's drain, and in the next frame's otherwise. That is the whole of the
65/// difference, and it is concurrency's rather than the engine's: **nothing about how an op is
66/// applied depends on which side it came from.**
67///
68/// A [`Leaf`] named here may have withered by the time this thread writes to it, which is safe on
69/// the same terms it is safe anywhere -- every op naming a withered leaf is dropped.
70///
71/// # One handle
72///
73/// Every clone, and every [`Grove::sprig`] the engine hands out, is the same handle: one queue, one
74/// stream of reports, one set of watches. Two workers holding one read a single stream between them
75/// rather than each being sent a copy of it, which is what keeps the reports finite when nothing is
76/// listening and the watches finite when everything is.
77#[derive(Clone)]
78pub struct Sprig(Arc<Cutting>);
79
80/// What every clone of a handle shares.
81struct Cutting {
82    queue: Queue,
83    /// How the loop is roused when an op arrives from off the frame. Shared with the engine rather
84    /// than copied out of it, because a handle is usually taken before there is a loop to rouse.
85    wake: Wake,
86    naming: Naming,
87    /// Last frame's ambient state. Written whether or not anyone is reading, because it is a fixed
88    /// handful of `Copy` values -- there is nothing here to gate.
89    conditions: Mutex<Option<Conditions>>,
90    /// What every standing watch last read. Current rather than a history, so a reader that has not
91    /// asked in a while is behind by no more than a frame instead of by everything that happened.
92    readings: Mutex<HashMap<(Leaf, Vein), Sap>>,
93    /// What the frame has reported and this side has not taken. A history rather than a state,
94    /// because a report is about a moment and nothing else will say it happened.
95    inbox: Mutex<Vec<Pollen>>,
96    /// Set by the first [`pollen`](Sprig::pollen) call. Until then the frame has nowhere to deliver
97    /// and does not, which is what keeps a handle that only ever writes from filling an inbox
98    /// nothing will read.
99    listening: AtomicBool,
100}

102impl Sprig {
103    pub(crate) fn new(queue: Queue, wake: Wake, naming: Naming) -> Self {
104        Self(Arc::new(Cutting {
105            queue,
106            wake,
107            naming,
108            conditions: Mutex::new(None),
109            readings: Mutex::new(HashMap::new()),
110            inbox: Mutex::new(Vec::new()),
111            listening: AtomicBool::new(false),
112        }))
113    }
114
115    /// Registers a font and hands back the name elements compose in.
116    ///
117    /// [`Grove::font`] for a thread that cannot reach the registry: bytes a worker holds, or an
118    /// [`Origin`](crate::Origin) to read them from, and the name comes back at once either way. A
119    /// face that has yet to land is composed in the bundled one until it does.
120    ///
121    /// # What differs from the frame's
122    ///
123    /// **Nothing here panics.** [`Grove::font`] refuses a proportional face outright when it was
124    /// given the bytes, because bytes written at a callsite are a statement the program made and the
125    /// mistake is worth stopping for. A worker holds bytes it built or was sent rather than ones it
126    /// stated, and a panic on a thread ends the thread rather than naming a callsite -- so a face
127    /// that turns out to be proportional is reported as [`missing`](crate::Pollen::missing) here, as
128    /// a fetched one is. What lands is [`loaded`](crate::Pollen::loaded), on the same terms.
129    pub fn font(&mut self, bytes: impl Into<Bytes>) -> Font {
130        let font = self.0.naming.face();
131        self.supply(Destination::Face(font), bytes.into());
132        font
133    }
134
135    /// Registers a mark and hands back the name elements draw it by.
136    ///
137    /// [`Grove::icon`] for a thread that cannot reach the registry, and refusing rather than
138    /// asserting for the reason [`font`](Sprig::font) does: a field smaller than it was said to be
139    /// is reported as [`missing`](crate::Pollen::missing).
140    pub fn icon(&mut self, field: impl Into<Bytes>, side: u32, range: f32) -> Field {
141        let mark = self.0.naming.mark();
142        self.supply(Destination::Mark(mark, side, range), field.into());
143        mark
144    }
145
146    /// Registers a picture and hands back the name elements draw it by.
147    ///
148    /// [`Grove::image`] for a thread that cannot reach the registry: PNG or JPEG, decoded when it is
149    /// drained. Pixels the worker made itself are [`pixels`](crate::Grow::pixels), which needs no
150    /// decode and states a size because there is nothing to read one from.
151    pub fn image(&mut self, bytes: impl Into<Bytes>) -> Plate {
152        let plate = self.0.naming.plate();
153        self.supply(Destination::Picture(plate), bytes.into());
154        plate
155    }
156
157    /// One road for both ways bytes are given.
158    ///
159    /// Bytes a worker holds arrive as the op a read would have pushed, and an
160    /// [`Origin`](crate::Origin) is read exactly as the frame reads one -- a retrieval needs the
161    /// queue and the wake, and this holds both. So a registration is drained in the frame it
162    /// reached, whichever way it was supplied, and a worker that fetches writes the same line as one
163    /// that already has the bytes.
164    fn supply(&mut self, destination: Destination, bytes: Bytes) {
165        match bytes.0 {
166            Supply::Held(bytes) => self.queue(Op::Arrived {
167                destination,
168                bytes: Ok(bytes),
169            }),
170            Supply::At(origin) => retrieve(&self.0.queue, &self.0.wake, destination, origin),
171        }
172    }
173
174    /// What is true of the tree as a whole, as of the last frame, or `None` before the first one has
175    /// run.
176    ///
177    /// The reads that need no watching: not per-element and not large, so the frame publishes them
178    /// outright. Taken together from one frame, so the viewport and the breakpoint in hand always
179    /// agree with each other.
180    pub fn conditions(&self) -> Option<Conditions> {
181        *self.lock(&self.0.conditions)
182    }
183
184    /// Asks to be told what one property of an element is, and told again whenever it changes.
185    ///
186    /// The read path for a thread that cannot sample: the value is pushed rather than asked for, so
187    /// nothing here borrows the world or waits on a frame. Read it with [`tap`](Sprig::tap).
188    ///
189    /// The first reading is taken at the end of the frame this is drained in, with the value as it
190    /// already stands -- so a watch does not have to be seeded by waiting for something to move.
191    /// Watching the same property twice is one watch, and it ends when the element withers or
192    /// [`unwatch`](Sprig::unwatch) says so.
193    pub fn watch(&mut self, leaf: Leaf, vein: Vein) {
194        self.queue(Op::Watch { leaf, vein });
195    }
196
197    /// Ends a [`watch`](Sprig::watch). Unwatching what was never watched does nothing.
198    ///
199    /// The reading goes with it: [`tap`](Sprig::tap) answers `None` from the frame this is drained
200    /// in, rather than going on answering with a value nothing is keeping current.
201    pub fn unwatch(&mut self, leaf: Leaf, vein: Vein) {
202        self.queue(Op::Unwatch { leaf, vein });
203    }
204
205    /// What a watched property last read, or `None` if it is not watched, has not been read yet, or
206    /// the element does not carry it.
207    ///
208    /// [`Grove::tap`] for a thread with no world to read: the same question, answered from what the
209    /// last frame published rather than from the tree. Everything read here comes from a single
210    /// frame, so two readings never disagree about which frame they are from.
211    ///
212    /// A property an element does not carry reads `None` exactly as it does at the frame's own
213    /// callsite. A watch that has never been drained reads `None` too, which is the ordinary state
214    /// for the moment between asking and the frame that answers.
215    pub fn tap(&self, leaf: Leaf, vein: Vein) -> Option<Sap> {
216        self.lock(&self.0.readings).get(&(leaf, vein)).cloned()
217    }
218
219    /// Every frame's report since this was last called, oldest first.
220    ///
221    /// The same [`Pollen`] the [`Root`](crate::Root) is handed, delivered as each frame hands it
222    /// over -- so a worker hears about a gesture on an element it grew without the app having to
223    /// relay it, and reads on its own schedule rather than the engine's.
224    ///
225    /// **Nothing is collected until the first call**, which is what arms delivery: a handle that
226    /// only writes never accumulates a report. So the first call answers with nothing, whatever has
227    /// happened, and every call after it answers with the frames since.
228    ///
229    /// What is reported and what [`tap`](Sprig::tap) reads are the two halves of one frame, arriving
230    /// together as they do at the frame's own callsite. A reading is never older than the last
231    /// report and may be one frame newer, which is the ordinary condition of reading a running
232    /// engine from beside it.
233    pub fn pollen(&self) -> Vec<Pollen> {
234        self.0.listening.store(true, Ordering::Relaxed);
235        core::mem::take(&mut *self.lock(&self.0.inbox))
236    }
237
238    /// Hands this frame's ambient state over.
239    pub(crate) fn ambient(&self, conditions: Conditions) {
240        *self.lock(&self.0.conditions) = Some(conditions);
241    }
242
243    /// Where the watched readings are kept, for the frame to bring up to date.
244    fn readings(&self) -> MutexGuard<'_, HashMap<(Leaf, Vein), Sap>> {
245        self.lock(&self.0.readings)
246    }
247
248    /// Drops a reading whose watch has ended.
249    pub(crate) fn forget(&self, leaf: Leaf, vein: Vein) {
250        self.readings().remove(&(leaf, vein));
251    }
252
253    /// Hands this frame's report to whoever is listening. Costs a clone only once someone is.
254    pub(crate) fn deliver(&self, pollen: &Pollen) {
255        if !self.0.listening.load(Ordering::Relaxed) {
256            return;
257        }
258        self.lock(&self.0.inbox).push(pollen.clone());
259    }
260
261    /// A poisoned lock means another thread panicked while holding it. What is behind these is a
262    /// value written whole rather than a structure kept in step across calls, so it is well-formed
263    /// either way and going on with it is better than propagating the panic into the event loop.
264    fn lock<'a, T>(&self, held: &'a Mutex<T>) -> MutexGuard<'a, T> {
265        held.lock().unwrap_or_else(|poisoned| poisoned.into_inner())
266    }
267}

269impl Queues for Sprig {
270    fn queue(&mut self, op: Op) {
271        self.0.queue.push(op);
272        // The frame that will drain it. An op arriving from off the frame is the one kind nothing
273        // else announces: the loop is asleep, and the platform has nothing to say about it.
274        self.0.wake.rouse();
275    }
276
277    fn allocate(&self) -> (Leaf, Growth) {
278        self.0.naming.leaf()
279    }
280
281    fn name(&self) -> Tween {
282        self.0.naming.tween()
283    }
284
285    fn group(&self) -> Sequence {
286        self.0.naming.sequence()
287    }
288
289    fn picture(&mut self) -> Plate {
290        self.0.naming.plate()
291    }
292}

294/// What is true of the tree as a whole.
295///
296/// Exactly what [`Grove`] answers about the frame rather than about an element, for a thread that
297/// cannot ask it: what a worker positioning anything needs before it can state a location in
298/// something other than hardcoded pixels.
299#[derive(Copy, Clone, PartialEq, Debug)]
300#[non_exhaustive]
301pub struct Conditions {
302    /// The visible area.
303    pub viewport: Area,
304    /// The width breakpoint in force, which every placement is read against.
305    pub layout: Layout,
306    /// Whether the viewport is vertically cramped.
307    pub short: Short,
308    /// What every [`Palette`](crate::Palette) role currently resolves to.
309    pub scheme: Scheme,
310    /// What holds focus, if anything does.
311    pub focused: Option<Leaf>,
312    /// How long the last frame took.
313    pub frame_time: Duration,
314    /// Time since the engine was built.
315    pub elapsed: Duration,
316}

318/// Every standing watch, as the frame holds them.
319///
320/// Engine-side and not shared: what is asked for crosses the boundary as an op and what is read
321/// crosses back as a reading, so the set itself never has to be locked.
322#[derive(Default)]
323pub(crate) struct Watches(HashSet<(Leaf, Vein)>);
324
325impl Watches {
326    pub(crate) fn watch(&mut self, leaf: Leaf, vein: Vein) -> bool {
327        self.0.insert((leaf, vein))
328    }
329
330    pub(crate) fn unwatch(&mut self, leaf: Leaf, vein: Vein) -> bool {
331        self.0.remove(&(leaf, vein))
332    }
333
334    pub(crate) fn is_empty(&self) -> bool {
335        self.0.is_empty()
336    }
337}

339/// Publishes what a thread with no world to read would otherwise have to ask for.
340///
341/// After settle, so what is published is what the frame ended at rather than what it passed
342/// through, and before extraction, which is the backend's business and not the boundary's.
343pub(crate) fn publish(grove: &mut Grove) {
344    grove.sprig.ambient(Conditions {
345        viewport: grove.viewport,
346        layout: grove.layout,
347        short: grove.short,
348        scheme: grove.scheme,
349        focused: grove.focus.held(),
350        frame_time: grove.clock.delta(),
351        elapsed: grove.clock.elapsed(),
352    });
353    if grove.watched.is_empty() {
354        return;
355    }
356    // Taken out so the withered ones can be dropped while the rest are read. Put back whole below,
357    // so a frame that drops none allocates nothing here.
358    let mut watched = core::mem::take(&mut grove.watched);
359    {
360        let mut readings = grove.sprig.readings();
361        watched.0.retain(|&(leaf, vein)| {
362            // A name is never handed out twice, so a watch on something that has withered is one
363            // nothing can ever answer again.
364            if !grove.tree.is_live(leaf) {
365                readings.remove(&(leaf, vein));
366                return false;
367            }
368            match grove.tap(leaf, vein) {
369                Some(sap) if readings.get(&(leaf, vein)) != Some(&sap) => {
370                    trace!(leaf = leaf.id(), vein = ?vein, "read");
371                    readings.insert((leaf, vein), sap);
372                }
373                Some(_) => {}
374                // Live, and not carrying the property. The watch stands -- it ends where every other
375                // one does, with the element or with `unwatch` -- and there is nothing to publish.
376                None => {
377                    readings.remove(&(leaf, vein));
378                }
379            }
380            true
381        });
382    }
383    grove.watched = watched;
